1"""The scan-ingest remote probe: runs on the host holding the bytes. 

2 

3Slice 17's transport seam. A witnessed capture's file lives on the 

4detector host (e.g. `tomdet`), never on the CORA host: measured 2026-08-18, 

5pulling one ~24 GB file over the deployment's 1 GbE link takes roughly 

6twice the scan cadence, so the bytes must not move. This module is the 

7other half of that decision: it runs, via SSH, ON the host holding the 

8file, and does exactly what `DataExchangeScanReader` / `PosixChecksumAdapter` 

9already do locally -- unchanged, reused, not re-implemented. Only a JSON 

10verdict (a few hundred bytes) crosses the network. 

11 

12This is a COMPOSITION, not a new layout or digest implementation: fusing 

13transport into a rewritten reader is exactly the mistake 

14`cora.data.ports.scan_reader`'s module docstring warns against ("layout" 

15and "transport" are independent axes). Here transport is handled entirely 

16by `cora.data.adapters._ssh_probe` (the client side) and this module (the 

17server side); the layout and digest code is the same class, imported 

18unchanged, and confinement is the same `resolve_confined_file_uri` both 

19local adapters already share -- one safety rule, enforced on whichever 

20host actually has the bytes. 

21 

22## Protocol 

23 

24One JSON object on stdin, one JSON object on stdout, newline-terminated. 

25No argv, no shell: the locator is untrusted (`full_file_name` comes from 

26`2bmSP2:HDF1:FullFileName_RBV`, writable by anyone with Channel Access), 

27so it must never be re-parsed by a shell. See 

28`cora.data.adapters._ssh_probe` for the client side of this contract and 

29why stdin, not argv, carries it. 

30 

31Request: 

32 

33 {"op": "describe", "locator_uri": "file://...", "allowed_roots": [...], 

34 "captured_at_source": "start_date"|"end_date"} 

35 {"op": "checksum", "locator_uri": "file://...", "allowed_roots": [...], 

36 "supply_id": "<uuid>", "max_walk_seconds": 60.0} 

37 

38Response, one line, always valid JSON, always exit 0: a malformed request 

39or an uncaught exception is a `ProbeError` verdict, not a process failure, 

40so the client only ever has one thing to parse (never raise, mirroring 

41the never-raise contract both composed adapters already hold). 

42 

43`allowed_roots` arrives FROM the request, not from a local default: this 

44process trusts the caller's confinement policy rather than defending 

45itself with one of its own. That is sound only because reaching this 

46process's stdin already requires SSH access as the deployment's own 

47service account (`scan_probe_remote_host`'s configured account); a 

48principal that could set the request's `allowed_roots` arbitrarily would 

49already need that access, at which point the confinement check is 

50defense against the WRONG kind of accident (a stale config, a copy-paste 

51error), not against an adversary who has not yet arrived here. 

52"""

88MAX_LOCATE_MATCHES = 8 

89"""Cap on the matches one `locate` verdict carries. `match_count` is 

90reported UNCAPPED beside them, so a caller can tell 2 matches from 50 

91even though it only ever sees the first few: the caller refuses 

92anything but exactly one match, and a silently truncated list would let 

93a pathological request return a response sized by the directory rather 

94than by the answer."""

97def _description_to_json(result: Description | Unreadable | Unrecognized) -> dict[str, Any]: 

98 if isinstance(result, Description): 

99 payload = asdict(result) 

100 payload["captured_at"] = result.captured_at.isoformat() if result.captured_at else None 

101 payload["kind"] = "Description" 

102 return payload 

103 if isinstance(result, Unreadable): 

104 return {"kind": "Unreadable", "reason": result.reason} 

105 return {"kind": "Unrecognized", "reason": result.reason}

109 if isinstance(result, ComputedChecksum): 

110 payload = asdict(result) 

111 payload["kind"] = "ComputedChecksum" 

112 return payload 

113 return {"kind": "Unreachable", "error_detail": result.error_detail}

116def _locate(request: dict[str, Any], *, allowed_roots: tuple[str, ...]) -> dict[str, Any]: 

117 """Find the durable copy of a file whose directory CORA cannot name. 

118 

119 At 2-BM an experiment folder is `{yyyy-mm}-{PIsurname}-{GUP}`, and 

120 CORA deliberately holds no surname (`run_experiment_identity` 

121 carries proposal, ESAF and ESAF-DOI numbers and nothing else), so 

122 the durable copy has to be found from the parts CORA does hold. The 

123 request therefore names each path segment literally except the 

124 experiment directory, which it matches by suffix. 

125 

126 No pattern is ever built from the request. Every segment is checked 

127 against `is_safe_path_segment` first, matching is `str.endswith` on 

128 entries this process itself enumerated, and the resolved match is 

129 re-confined afterwards so a symlink out of the tree is refused 

130 rather than followed. Deciding what a given match COUNT means is 

131 the caller's policy, not this process's: it reports what it found. 

132 

133 Each match carries the file's own `st_mtime` beside its path. The 

134 caller vaults that as `observed_at`, a column whose stated meaning 

135 is the SUBSTRATE's timestamp rather than CORA's clock, and only 

136 this side can read it. Sending it with the match is also what makes 

137 a retry idempotent: the vault's upsert is monotonic in 

138 `observed_at`, so a re-probe of an unchanged file writes the same 

139 value instead of a newer one. 

140 """ 

141 root = request.get("root") 

142 if not isinstance(root, str): 

143 return {"kind": "ProbeError", "detail": "malformed request: root"} 

144 if matched_storage_root(root, allowed_roots) is None: 

145 return {"kind": "ProbeError", "detail": "root is not under an allowed root"} 

146 

147 segments = {name: request.get(name) for name in ("directory_suffix", "filename")} 

148 for name, value in segments.items(): 

149 if not isinstance(value, str) or not is_safe_path_segment(value): 

150 return {"kind": "ProbeError", "detail": f"malformed request: {name}"} 

151 directory_suffix = str(segments["directory_suffix"]) 

152 filename = str(segments["filename"]) 

153 

154 raw_months = request.get("months") 

155 if not isinstance(raw_months, list) or not raw_months: 

156 return {"kind": "ProbeError", "detail": "malformed request: months"} 

157 months = cast("list[Any]", raw_months) 

158 if len(months) > MAX_LOCATE_MONTHS or not all( 

159 isinstance(month, str) and is_safe_path_segment(month) for month in months 

160 ): 

161 return {"kind": "ProbeError", "detail": "malformed request: months"} 

162 

163 subdirectory = request.get("subdirectory") 

164 if subdirectory is not None and ( 

165 not isinstance(subdirectory, str) or not is_safe_path_segment(subdirectory) 

166 ): 

167 return {"kind": "ProbeError", "detail": "malformed request: subdirectory"} 

168 

169 entries: list[Path] = [] 

170 for month in months: 

171 try: 

172 entries.extend(entry for entry in (Path(root) / str(month)).iterdir() if entry.is_dir()) 

173 except OSError: 

174 continue 

175 entries.sort() 

176 

177 matches: list[dict[str, Any]] = [] 

178 for entry in entries: 

179 if not entry.name.endswith(directory_suffix): 

180 continue 

181 candidate = entry / subdirectory / filename if subdirectory else entry / filename 

182 # Every filesystem call on `candidate` stays inside this block. 

183 # `resolve`, `is_file` and `stat` all raise `OSError` subclasses 

184 # whose `str()` embeds the filename they failed on, and this 

185 # tree's filenames sit under a directory named for a person. One 

186 # such call left outside the guard is all it takes for that 

187 # string to reach the caller through `_main`'s catch-all, which 

188 # is exactly how `is_file` leaked before. 

189 try: 

190 resolved = candidate.resolve(strict=True) 

191 if not resolved.is_file(): 

192 continue 

193 modified_at = resolved.stat().st_mtime 

194 except OSError: 

195 continue 

196 if matched_storage_root(str(resolved), allowed_roots) is None: 

197 continue 

198 matches.append({"path": str(resolved), "modified_at": modified_at}) 

199 

200 return { 

201 "kind": "Located", 

202 "matches": matches[:MAX_LOCATE_MATCHES], 

203 "match_count": len(matches), 

204 }

207async def _handle(request: dict[str, Any]) -> dict[str, Any]: 

208 op = request.get("op") 

209 allowed_roots = tuple(request.get("allowed_roots") or ()) 

210 if op == _OP_LOCATE: 

211 return _locate(request, allowed_roots=allowed_roots) 

212 

213 locator_uri = request.get("locator_uri") 

214 if not isinstance(locator_uri, str): 

215 return {"kind": "ProbeError", "detail": "malformed request: locator_uri"} 

216 

217 if op == _OP_DESCRIBE: 

218 captured_at_source = request.get("captured_at_source") or "start_date" 

219 reader = DataExchangeScanReader( 

220 allowed_roots=allowed_roots, captured_at_source=captured_at_source 

221 ) 

222 return _description_to_json(await reader.describe(locator_uri=locator_uri)) 

223 if op == _OP_CHECKSUM: 

224 raw_supply_id = request.get("supply_id") 

225 if not isinstance(raw_supply_id, str): 

226 return {"kind": "ProbeError", "detail": "malformed request: supply_id"} 

227 try: 

228 supply_id = UUID(raw_supply_id) 

229 except ValueError: 

230 return {"kind": "ProbeError", "detail": "malformed request: supply_id is not a UUID"} 

231 raw_max_walk_seconds = request.get("max_walk_seconds") 

232 if raw_max_walk_seconds is not None and ( 

233 isinstance(raw_max_walk_seconds, bool) 

234 or not isinstance(raw_max_walk_seconds, int | float) 

235 or raw_max_walk_seconds <= 0 

236 ): 

237 return {"kind": "ProbeError", "detail": "malformed request: max_walk_seconds"} 

238 computer = ( 

239 PosixChecksumAdapter(allowed_roots=allowed_roots) 

240 if raw_max_walk_seconds is None 

241 else PosixChecksumAdapter( 

242 allowed_roots=allowed_roots, max_walk_seconds=float(raw_max_walk_seconds) 

243 ) 

244 ) 

245 return _checksum_to_json( 

246 await computer.compute(locator_uri=locator_uri, supply_id=supply_id) 

247 ) 

248 # The op is NOT echoed. Every other refusal in this module is a 

249 # fixed literal so that a caller may log it, and interpolating a 

250 # request value here would make that true of all but one. 

251 return {"kind": "ProbeError", "detail": "unknown op"}

254@contextlib.contextmanager 

255def _stdout_reserved_for_the_verdict() -> Generator[TextIO]: 

256 """Give the caller the ONLY writable handle on real stdout. 

257 

258 Stdout is the protocol: the client reads the first line and parses 

259 it as the verdict, so anything else written there corrupts the 

260 answer. `describe` and `checksum` both log, and structlog writes to 

261 stdout, which puts a log line exactly where the verdict belongs. 

262 

263 Done at the file-descriptor level rather than by rebinding 

264 `sys.stdout`, and the difference is not academic. A rebind is 

265 defeated the moment anyone calls `configure_logging()`, because its 

266 `StreamHandler` captures the stream OBJECT once and keeps writing to 

267 the real descriptor afterwards. Pointing fd 1 itself at `/dev/null` 

268 catches that, plus a stray `print`, plus anything a library writes 

269 from C. What comes back is a private duplicate of the original fd, 

270 which only this function's caller holds. 

271 

272 Discarded rather than folded into stderr, which is the tempting 

273 version: the client puts nothing from stderr into its refusal text 

274 precisely so a path cannot reach a log, and sending the probe's own 

275 log lines there would undo that. Nothing is lost, since these writes 

276 were already being swallowed by the SSH pipe. 

277 """ 

278 sys.stdout.flush() 

279 saved = os.dup(_STDOUT_FD) 

280 devnull = os.open(os.devnull, os.O_WRONLY) 

281 try: 

282 os.dup2(devnull, _STDOUT_FD) 

283 os.close(devnull) 

284 with os.fdopen(saved, "w", closefd=False) as verdict_stream: 

285 yield verdict_stream 

286 finally: 

287 # Flush BEFORE restoring, so anything `print` left sitting in 

288 # `sys.stdout`'s buffer drains into `/dev/null` rather than onto 

289 # the real descriptor once it is back. Without this the guard 

290 # holds for line-buffered writers and leaks for block-buffered 

291 # ones, which is what a pipe gives you, which is what SSH is. 

292 with contextlib.suppress(ValueError): 

293 sys.stdout.flush() 

294 os.dup2(saved, _STDOUT_FD) 

295 os.close(saved)

298async def _main() -> None: 

299 line = sys.stdin.readline() 

300 # The guard itself does syscalls (`dup`, `open`, `dup2`) and they can 

301 # fail, on fd exhaustion most plausibly. Outside this try that would 

302 # exit non-zero with nothing on stdout, which the client reads as a 

303 # dead TRANSPORT rather than a bad request, and a dead transport is 

304 # the one verdict that stops a sweep without moving past the 

305 # candidate. So the module's never-raise, always-one-JSON-line 

306 # promise has to cover the guard, not just what runs inside it. 

307 try: 

308 await _respond(line) 

309 except Exception as exc: 

310 sys.stdout.write(json.dumps({"kind": "ProbeError", "detail": _unhandled(exc)})) 

311 sys.stdout.write("\n") 

312 sys.stdout.flush()

315def _unhandled(exc: BaseException) -> str: 

316 """The exception TYPE only, never `str(exc)`. 

317 

318 The paths this process walks embed a PI surname, and the exceptions 

319 most likely to reach a catch-all are `OSError` subclasses that 

320 render the filename they failed on. A class name cannot carry one, 

321 which makes the no-path property of these two arms structural 

322 rather than a promise about every call beneath them. 

323 """ 

324 return f"unhandled {type(exc).__name__}"

327async def _respond(line: str) -> None: 

328 with _stdout_reserved_for_the_verdict() as verdict_stream: 

329 try: 

330 parsed: Any = json.loads(line) 

331 if not isinstance(parsed, dict): 

332 raise TypeError("request is not a JSON object") 

333 response = await _handle(cast("dict[str, Any]", parsed)) 

334 except Exception as exc: 

335 response = {"kind": "ProbeError", "detail": _unhandled(exc)} 

336 verdict_stream.write(json.dumps(response)) 

337 verdict_stream.write("\n") 

338 verdict_stream.flush()
