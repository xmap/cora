1"""In-memory `ControlPort` adapter for unit tests and `app_env=test`. 

2 

3Mirrors `cora.infrastructure.adapters.in_memory_event_store` and 

4`cora.infrastructure.adapters.in_memory_profile_store`: dict-backed, 

5no sockets, no real IOC / Tango DS / OPC UA server. Tests inject 

6values via `set_reading` / `simulate_disconnect` / `simulate_connect`; 

7production code paths call `read` / `write` / `subscribe` against 

8the same instance and observe the injected state. 

9 

10Per [[project_test_infra]]'s 5-tier pyramid, this adapter serves the 

11unit tier. Production substrate adapters (Caproto / EpicsCa / 

12EpicsPva, plus future Tango / OPC UA) implement the same 

13`ControlPort` Protocol but speak real wire protocols. 

14 

15## Connection model 

16 

17An address is "connected" iff `set_reading` or `simulate_connect` 

18has been called for it. `read` and `write` against a disconnected 

19address raise `ControlNotConnectedError`. `simulate_disconnect` 

20marks a connected address as disconnected, drops its cached value 

21(matches real CA/PVA/OPC UA reconnect semantics), and pushes a 

22sentinel to every active subscriber so the subscribe iterator 

23raises `ControlNotConnectedError` mid-stream. 

24 

25## write semantics 

26 

27`write` accepts the same primitive value set as the production 

28adapters (`int | float | bool | str | tuple[Any, ...]`) and 

29constructs a `Reading` for the store: tuples become 

30`kind="Array"`, other primitives become `kind="Scalar"`. 

31`sampled_at` is read from the injectable `now` callable (defaults 

32to `lambda: datetime.now(tz=UTC)`); `quality` stays `"Good"`, 

33`quality_detail` stays `""`. Tests that need a specific kind 

34(`Image` / `Categorical` / `Tabular`), non-`Good` quality, or a 

35populated `quality_detail` use `set_reading` to push an explicit 

36`Reading`. 

37 

38`wait` and `timeout_s` are accepted to satisfy the Protocol but 

39ignored: in-memory has no substrate round-trip, so write-confirm / 

40timeout semantics do not apply. The contract is preserved at the 

41type level, mirroring how `InMemoryProfileStore.scrub_and_delete` 

42accepts and ignores the asyncpg `conn` parameter. 

43 

44## Subscribe semantics 

45 

46`subscribe` is a plain `def` that synchronously registers a 

47subscriber queue and returns an async iterator. The queue is 

48registered eagerly so a `set_reading` that lands between 

49`subscribe()` and the first `__anext__` still fans out (Subscriber 

50pattern in real CA / PVA brokers behaves the same: the listener is 

51hot before the first event arrives). The connect-state check fires 

52on the iterator's first `__anext__`, so a `subscribe()` against an 

53unconnected address raises `ControlNotConnectedError` through 

54`anext`, not at `subscribe()` time. Each subscriber gets its own 

55queue; pushing to an address fans out to every subscriber's queue 

56(no coalescing in this adapter; production adapters layer their 

57substrate-specific policy on top). The iterator cleans up its queue 

58on `aclose()` or on `ControlNotConnectedError` propagation. 

59 

60## aclose 

61 

62`aclose()` is a no-op for the in-memory adapter: there is no 

63substrate context to release. Provided so production code paths can 

64call `aclose()` polymorphically against any `ControlPort` 

65implementation without branching on type. 

66"""

92class _DisconnectSentinel: 

93 """Marker pushed to subscriber queues on `simulate_disconnect`. 

94 

95 Kept as a private class (not a singleton instance) so 

96 `isinstance` checks in the drain loop stay unambiguous and the 

97 sentinel cannot collide with any legitimate `Reading` a consumer 

98 might push through a future extension.

106class InMemoryControlPort: 

107 """Process-local dict adapter for `ControlPort`. 

108 

109 See module docstring for the connection model, write semantics, 

110 and subscribe semantics. Per-address state is kept in three 

111 dicts so address collisions across kinds raise type errors 

112 loudly rather than silently overlap. 

113 """ 

114 

115 def __init__(self, *, now: Callable[[], datetime] | None = None) -> None: 

116 self._values: dict[str, Reading] = {} 

117 self._connected: set[str] = set() 

118 self._subscribers: dict[str, list[asyncio.Queue[Reading | _DisconnectSentinel]]] = {} 

119 self._now: Callable[[], datetime] = now or (lambda: datetime.now(tz=UTC)) 

120 self._closed = False 

121 

122 def set_reading(self, address: str, reading: Reading) -> None: 

123 """Install `reading` as the current value of `address` and fan out. 

124 

125 Marks the address as connected if it was not already. Test 

126 entry point for seeding state before exercising production 

127 code paths. 

128 """ 

129 self._values[address] = reading 

130 self._connected.add(address) 

131 for queue in self._subscribers.get(address, []): 

132 queue.put_nowait(reading) 

133 

134 def simulate_connect(self, address: str) -> None: 

135 """Mark `address` as connected without installing a value. 

136 

137 Useful for tests that want `write` to succeed (and the 

138 resulting `Reading` to be observable via `read`) without 

139 pre-seeding a `set_reading` call. 

140 """ 

141 self._connected.add(address) 

142 

143 def simulate_disconnect(self, address: str) -> None: 

144 """Mark `address` disconnected, drop its cached value, signal subscribers. 

145 

146 Subsequent `read` / `write` raise `ControlNotConnectedError`. 

147 Each active subscriber's iterator raises 

148 `ControlNotConnectedError` through the `async for` (matching 

149 production CA / PVA / Tango / OPC UA disconnect semantics 

150 where the cached value is invalidated on disconnect and must 

151 be re-fetched after reconnect). 

152 """ 

153 self._connected.discard(address) 

154 self._values.pop(address, None) 

155 for queue in self._subscribers.get(address, []): 

156 queue.put_nowait(_DISCONNECT) 

157

168 address: str,

170 *, 

171 wait: bool = True, 

172 timeout_s: float = 30.0, 

173 ) -> None: 

174 _ = (wait, timeout_s) 

175 correlation_id = get_dispatch_correlation_id() 

176 _log.info( 

177 _DISPATCH_EVENT, 

178 address=address, 

179 operation="write",

181 status="started", 

182 ) 

183 if address not in self._connected: 

184 _log.info(

186 address=address, 

187 operation="write",

191 )

193 kind: ReadingKind = "Array" if isinstance(value, tuple) else "Scalar" 

194 reading = Reading(value=value, kind=kind, quality="Good", sampled_at=self._now()) 

195 self._values[address] = reading 

196 for queue in self._subscribers.get(address, []): 

197 queue.put_nowait(reading) 

198 _log.info( 

199 _DISPATCH_COMPLETED_EVENT, 

200 address=address, 

201 operation="write", 

202 correlation_id=str(correlation_id) if correlation_id is not None else None, 

203 status="completed", 

204 )

206 def subscribe(self, address: str) -> AsyncGenerator[Reading]: 

207 """Return type narrows the Protocol's `AsyncIterator` to `AsyncGenerator`. 

208 

209 Covariant return type lets tests close subscriptions via the 

210 iterator's `aclose()` while production callers still see the 

211 `AsyncIterator` contract through the Protocol surface. 

212 

213 Queue registration happens synchronously here so a 

214 `set_reading` landing between `subscribe()` and the first 

215 `await anext(iterator)` still fans out. The connect-state 

216 check fires on the iterator's first iteration via `_drain`, 

217 matching the Protocol's lazy-setup contract. 

218 """ 

219 queue: asyncio.Queue[Reading | _DisconnectSentinel] = asyncio.Queue() 

220 self._subscribers.setdefault(address, []).append(queue) 

221 return self._drain(address, queue) 

222 

223 async def _drain( 

224 self, 

225 address: str, 

226 queue: asyncio.Queue[Reading | _DisconnectSentinel], 

227 ) -> AsyncGenerator[Reading]: 

228 try: 

229 if address not in self._connected: 

230 raise ControlNotConnectedError(address) 

231 while True: 

232 item = await queue.get() 

233 if isinstance(item, _DisconnectSentinel): 233 ↛ 234line 233 didn't jump to line 234 because the condition on line 233 was never true

234 raise ControlNotConnectedError(address) 

235 yield item 

236 finally: 

237 subs = self._subscribers.get(address) 

238 if subs is not None and queue in subs: 

239 subs.remove(queue)

241 async def aclose(self) -> None: 

242 """No-op for the in-memory adapter; idempotent. 

243 

244 Provided so production code paths can call `aclose()` on any 

245 `ControlPort` without type-checking. The dict-backed state is 

246 not a substrate resource; it does not need explicit release. 

247 """ 

248 self._closed = True
