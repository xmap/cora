1"""Shared SSH transport for the two remote scan-ingest adapters. 

2 

3`SshDataExchangeScanReader` and `SshPosixChecksumComputer` both need the 

4same thing: run `cora.data._remote_scan_probe` on a host that holds bytes 

5CORA's own host cannot reach in time (see that module's docstring for the 

6measured reasoning), ship it a request over stdin, and read one JSON 

7verdict back over stdout. Splitting that mechanism across two copies is 

8how the two adapters would drift on the one property that actually 

9matters here -- how the untrusted locator reaches the far side -- so it 

10lives once, here, mirroring `_file_uri.py`'s reason for existing. 

11 

12## The locator is untrusted; this is the module's whole reason to exist 

13 

14`full_file_name` (the source of every locator this ever runs against) 

15comes from `2bmSP2:HDF1:FullFileName_RBV`, writable by anyone with 

16Channel Access. `run_probe` NEVER lets it reach a shell or an argv 

17element: the ssh argv is fixed at call time (host, a pinned interpreter 

18path, `-m`, the probe module name) and the request -- including the 

19locator -- travels as one JSON line over the child's stdin. `ssh 

20host cmd arg` would otherwise be re-parsed by the remote shell, turning a 

21crafted PV value into remote code execution on the detector host; stdin 

22is not re-parsed by anything. 

23 

24Two more guards run before any process is ever spawned, both refusing 

25rather than probing: 

26 

27- `_charset_ok`: the decoded path must be printable, control-character 

28 and NUL free. Cheap, and it means a malformed PV value never even 

29 reaches the network. 

30- `_prefix_allowed`: the raw (unresolved) path must start with one of the 

31 deployment's configured roots. This is NOT the authoritative 

32 confinement check -- it cannot be, since `..` and symlinks resolve on 

33 the REMOTE host's filesystem, which this process cannot see. The 

34 authoritative check is `resolve_confined_file_uri`, run inside 

35 `_remote_scan_probe.py` on the host that actually owns the path. This 

36 is only the early refusal that saves a network round trip on an 

37 obviously-wrong value. 

38 

39## Never raises 

40 

41Mirrors every port this feeds (`ScanReader.describe`, `ChecksumComputer.compute`): 

42a dead host, a timed-out probe, a malformed response, or a refused 

43locator all return a `{"kind": "ProbeError", ...}` dict, never an 

44exception. The two adapter classes translate that into the shape their 

45own port expects (`Unreadable` / `Unreachable`). 

46""" 

47 

48from __future__ import annotations 

49 

50import asyncio 

51import contextlib 

52import json 

53from dataclasses import dataclass 

54from typing import Any, cast 

55from urllib.parse import unquote, urlparse 

56 

57_PROBE_MODULE = "cora.data._remote_scan_probe" 

58 

59 

60@dataclass(frozen=True) 

61class SshProbeConfig: 

62 """Deployment-declared transport parameters for one remote host. 

63 

64 `allowed_roots` are roots as valid ON `host`, never on CORA's own 

65 filesystem; see `Settings.scan_probe_allowed_roots`. 

66 """ 

67 

68 host: str 

69 remote_python: str 

70 allowed_roots: tuple[str, ...] 

71 connect_timeout_seconds: float 

72 command_timeout_seconds: float 

73 

74 

75def raw_path_from_file_uri(uri: str) -> str | None: 

76 """Decode a `file://` URI's path with NO filesystem resolution. 

77 

78 Unlike `cora.data.adapters._file_uri.resolve_confined_file_uri`, this 

79 never calls `os.path.realpath`: that would resolve against THIS 

80 process's filesystem, which is the wrong host. Returns `None` for a 

81 non-`file` scheme or a non-local netloc, mirroring that module's own 

82 scheme rule. 

83 """ 

84 parsed = urlparse(uri) 

85 if parsed.scheme != "file" or parsed.netloc not in ("", "localhost"): 

86 return None 

87 raw_path = unquote(parsed.path) 

88 return raw_path or None 

89 

90 

91def _charset_ok(path: str) -> bool: 

92 """Printable, control-character and NUL free.""" 

93 return path.isprintable() and "\x00" not in path 

94 

95 

96def _prefix_allowed(path: str, allowed_roots: tuple[str, ...]) -> bool: 

97 """Early, non-authoritative refusal; see the module docstring.""" 

98 return any(path == root or path.startswith(root.rstrip("/") + "/") for root in allowed_roots) 

99 

100 

101def _refuse(reason: str) -> dict[str, Any]: 

102 # NEVER embeds the locator: this detail string is what 

103 # `SshDataExchangeScanReader.describe` / `SshPosixChecksumComputer.compute` 

104 # log verbatim on refusal (`reason=` / `error_detail=`), and the 

105 # locator is the same personal-data value 

106 # `run.aggregates.run.capture_path` vaults rather than logs. 

107 return {"kind": "ProbeError", "detail": f"refused before probing: {reason}"} 

108 

109 

110async def run_probe(request: dict[str, Any], *, config: SshProbeConfig) -> dict[str, Any]: 

111 """Validate, then run one request against `config.host`; never raises. 

112 

113 `request` must carry `locator_uri`; `allowed_roots` is filled in from 

114 `config` here (the caller never supplies it), so the SAME allowlist 

115 the client-side guard checked is the one the remote probe enforces. 

116 """ 

117 locator_uri = request.get("locator_uri") 

118 if not isinstance(locator_uri, str): 

119 return _refuse("no locator_uri in request") 

120 raw_path = raw_path_from_file_uri(locator_uri) 

121 if raw_path is None: 

122 return _refuse("not a recognizable file:// URI") 

123 if not _charset_ok(raw_path): 

124 return _refuse("path contains non-printable or control characters") 

125 if not _prefix_allowed(raw_path, config.allowed_roots): 

126 return _refuse("path is outside the configured allowed roots") 

127 

128 payload = {**request, "allowed_roots": list(config.allowed_roots)} 

129 # `max(1, ...)`: OpenSSH reads `ConnectTimeout=0` as "no timeout 

130 # configured here, use the default" -- silently removing the bound 

131 # for a sub-1s config value instead of enforcing a very short one. 

132 # `--` before `config.host`: operator-configured, never 

133 # attacker-reachable, but a host string starting with `-` would 

134 # otherwise be parsed by `ssh` as an option rather than a hostname. 

135 argv = [ 

136 "ssh", 

137 "-o", 

138 "BatchMode=yes", 

139 "-o", 

140 f"ConnectTimeout={max(1, int(config.connect_timeout_seconds))}", 

141 "--", 

142 config.host, 

143 config.remote_python, 

144 "-m", 

145 _PROBE_MODULE, 

146 ] 

147 try: 

148 process = await asyncio.create_subprocess_exec( 

149 *argv, 

150 stdin=asyncio.subprocess.PIPE, 

151 stdout=asyncio.subprocess.PIPE, 

152 stderr=asyncio.subprocess.PIPE, 

153 ) 

154 except OSError as exc: 

155 return {"kind": "ProbeError", "detail": f"could not launch ssh: {exc}"} 

156 

157 try: 

158 stdout, stderr = await asyncio.wait_for( 

159 process.communicate(json.dumps(payload).encode("utf-8") + b"\n"), 

160 timeout=config.command_timeout_seconds, 

161 ) 

162 except TimeoutError: 

163 process.kill() 

164 with contextlib.suppress(ProcessLookupError): 

165 await process.wait() 

166 return { 

167 "kind": "ProbeError", 

168 "detail": f"timed out after {config.command_timeout_seconds}s", 

169 } 

170 

171 if process.returncode != 0: 

172 tail = stderr.decode(errors="replace").strip()[-300:] 

173 return {"kind": "ProbeError", "detail": f"ssh exited {process.returncode}: {tail}"} 

174 

175 try: 

176 parsed: Any = json.loads(stdout.decode("utf-8").splitlines()[0]) 

177 except (IndexError, UnicodeDecodeError, json.JSONDecodeError) as exc: 

178 return {"kind": "ProbeError", "detail": f"unparseable probe response: {exc}"} 

179 if not isinstance(parsed, dict): 

180 return {"kind": "ProbeError", "detail": "probe response is not a JSON object"} 

181 return cast("dict[str, Any]", parsed) 

182 

183 

184__all__ = ["SshProbeConfig", "raw_path_from_file_uri", "run_probe"]
