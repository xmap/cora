1"""Shared helpers for the consequence-gate (Gate IV) enforcer subscribers. 

2 

3Leading-underscore module: intra-package machinery, not a subscriber itself (the 

4`test_agent_subscribers_completeness` discovery skips `_`-prefixed files). Home for 

5what `ratification_hold.py` and `ratification_release.py` both need: the pinned 

6enforcer identity guard and this gate's hold-claim helpers. 

7 

8The two subscribers, acting as the pinned RatificationEnforcer agent, turn the 

9consequence gate's refuse-and-park flow into real RunHeld / RunResumed transitions 

10on the SAME shared hold the kill-switch uses. Both write via Pattern C (load, 

11guard status in-process, authorize as their own principal, append from 

12cora.run.aggregates), because cora.agent may depend on cora.run.aggregates + 

13cora.trust.aggregates but NOT cora.run.features (the tach BC boundary). Neither 

14writes a Decision: the Ratification events (Requested/Granted/Denied) are the 

15provenance. 

16"""

36async def enforcer_standing_down(event_store: EventStore) -> bool: 

37 """True if the enforcer is not seeded yet or was operator-deactivated. 

38 

39 Same operator-deactivation revocation surface every sibling agent uses 

40 (load_actor().active): an operator disables the gate the same way they disable 

41 any agent. 

42 """ 

43 actor = await load_actor(event_store, RATIFICATION_ENFORCER_AGENT_ID) 

44 return actor is None or not actor.active 

45 

46 

47def ratification_claim_id(run_id: UUID) -> UUID: 

48 """The claim id this gate holds `run_id` under. 

49 

50 Derived, not stored, so hold and release agree without a lookup and a 

51 re-delivered hold re-derives the same claim instead of stacking a second. 

52 """ 

53 return derive_claim_id(run_id, HOLD_CAUSE_RATIFICATION) 

54 

55 

56def enforcer_holds_claim(claims: dict[UUID, str], run_id: UUID) -> bool: 

57 """True iff this gate has an active hold claim on the run.""" 

58 return ratification_claim_id(run_id) in claims 

59 

60 

61def others_still_holding(claims: dict[UUID, str], run_id: UUID) -> tuple[str, ...]: 

62 """Causes of every active claim on the run OTHER than this gate's. 

63 

64 Non-empty means the release must discharge its own claim WITHOUT resuming: 

65 the run stays Held on behalf of whoever remains. This is the check that 

66 replaced scanning backward for the latest `RunHeld` and comparing its 

67 envelope principal. That scan answered "did I place the most recent hold", 

68 which is the right question only when holds cannot overlap; because every 

69 holder used to no-op on an already-held run, a hold arriving during this 

70 gate's co-signature wait left no event at all and the scan could not see it. 

71 """ 

72 own = ratification_claim_id(run_id) 

73 return tuple(cause for claim_id, cause in claims.items() if claim_id != own)
