1"""CaptureExperimentIdentityReader: read a witnessed Run's proposal / ESAF / 

2ESAF-DOI PVs once, at the instant a capture promotes to a Run. 

3 

4Slice 14a. Mirrors `_capture_baseline_reader.py`'s ONE-READ-NOT-A-FEED 

5shape exactly: invoked exactly once per promotion by 

6`RunTranslator._promote`, right after `record_witnessed_run` 

7returns a `run_id`, alongside (not instead of) the genesis-baseline 

8read. There is no buffer, no tick, and no ongoing liveness claim. 

9 

10## Why the vault, not the event 

11 

12See `cora.run.aggregates.run.experiment_identity`'s module docstring 

13for the full argument (memory/project_witnessed_run_prelive_slices.md, 

14slice 14a). Short version: `RecordWitnessedRun` has no operator behind 

15it the way `start_run.external_refs` does, so writing these PVs onto 

16`RunStarted` would be CORA auto-harvesting a re-identifying fact (a 

17proposal number plus a timestamp, per D0) into an immutable, 

18INSERT-only event with no way back. `ESAFDOINumber` was checked and 

19traced to an internal, authenticated APS API 

20(`EsafApsDbApi.getStationEsafById`, via the upstream `dmagic` source), 

21not a DOI registration agency; unconfirmed as a genuinely resolvable 

22public identifier, so it vaults alongside the other two. 

23 

24## Two traps, both silently recording a WRONG fact rather than failing 

25 

261. Every one of these PVs defaults to the substrate literal `"Unknown"` 

27 when `dmagic` has not populated it. An unpopulated PV therefore 

28 reads as a plausible string. `resolved_experiment_identity_text` treats 

29 `"Unknown"`, and an empty string, as ABSENT and returns `None`; the 

30 caller never writes a literal "Unknown" into the vault. 

312. Nothing in the IOC populates these PVs; `dmagic` does, from APS 

32 scheduling, so a value PERSISTS ACROSS BEAMTIMES until the next 

33 sync overwrites it. If a value is stale, this reader cannot detect 

34 that -- there is no freshness heuristic to invent, per the design 

35 memo's own instruction. Each `*_observed_at` carries the substrate's 

36 own reading time (`Measurement.produced_at`), the only staleness 

37 evidence available, so a reader downstream can at least see how old 

38 a value is. Whether these PVs are reliably synced at 2-BM before a 

39 beamtime starts is a staff question, not a code question. 

40 

41## Per-PV failure posture mirrors `_capture_baseline_reader.py` 

42 

43Every exception is caught and logged, never raised into the caller: a 

44`ControlPort.read()` failure on one PV drops only that PV's reading and 

45lets the sweep continue over the rest (mirroring 

46`capture_watch_preflight.py`'s own per-PV independence), and the vault 

47write's own failure must never unwind or retry the promotion that 

48already committed (mirroring `_read_baseline`'s exact posture in 

49`_run_translator.py`). 

50 

51Unlike `CaptureBaselineReader`, none of these three values is personal 

52data, so a write failure's exception text is logged in full (no 

53`error_class`-only redaction the way `_write_capture_path` requires). 

54"""

88UNKNOWN_EXPERIMENT_IDENTITY_LITERAL = "Unknown" 

89"""The literal `dmagic` / the IOC leaves an unpopulated experiment-identity 

90PV reading. Treated as ABSENT, never as a plausible value (Trap 1). 

91Module-public: `capture_watch_preflight` imports this directly (alongside 

92`resolved_experiment_identity_text`) so its own decode verdict can distinguish 

93"substrate's own placeholder" from "genuinely empty" without a second 

94copy of the literal."""

99def resolved_experiment_identity_text(value: object) -> str | None: 

100 """Resolve one experiment-identity PV reading to a usable string, or 

101 `None` when it must be treated as absent. 

102 

103 `None` for: a non-string reading, an empty (after stripping) 

104 string, or the substrate's own `"Unknown"` placeholder literal 

105 (matched case-insensitively: the exact casing is an IOC default we 

106 have observed, not a wire-level guarantee, and a differently-cased 

107 variant is still the same "not populated" fact, not a real value). 

108 Otherwise the stripped string. Module-public: `capture_watch_preflight` 

109 imports this directly so its own decode verdict can never drift from 

110 what this reader actually accepts. 

111 """ 

112 if not isinstance(value, str): 

113 return None 

114 stripped = value.strip() 

115 if not stripped or stripped.casefold() == UNKNOWN_EXPERIMENT_IDENTITY_LITERAL.casefold(): 

116 return None 

117 return stripped

120class CaptureExperimentIdentityReader: 

121 """Reads `experiment_identity_pvs[capture_code]`'s three roles once and vaults 

122 whatever survives. 

123 

124 `experiment_identity_pvs` is code -> role -> PV, matching 

125 `Settings.capture_experiment_identity_pvs`. A code with no entry (or 

126 an empty one) makes `read` a no-op, mirroring `CaptureBaselineReader`'s 

127 own per-code optionality. A role absent from a code's declared set is 

128 simply never read; it does not stop the other two. 

129 """ 

130 

131 def __init__( 

132 self, 

133 *, 

134 deps: Kernel, 

135 control_port: ControlPort, 

136 experiment_identity_pvs: Mapping[str, Mapping[str, str]], 

137 store: ExperimentIdentityStore, 

138 ) -> None: 

139 self._deps = deps 

140 self._control_port = control_port 

141 self._experiment_identity_pvs = experiment_identity_pvs 

142 self._store = store 

143 

144 async def read(self, capture_code: str, run_id: UUID) -> None: 

145 """Read every role declared for `capture_code` CONCURRENTLY, 

146 once, and vault whatever survives as one row against `run_id`. 

147 

148 Concurrent, not sequential: mirrors `CaptureBaselineReader.read`'s 

149 reasoning exactly -- this runs inline inside 

150 `RunTranslator._promote`, on `run_translator_loop`'s single 

151 consumer path, so a slow or partially-unreachable control system 

152 must not block the loop from reacting to the next observation. 

153 

154 Never raises: every failure mode (a dead PV, an unusable 

155 reading, or the vault write itself) is caught and logged here. 

156 """ 

157 roles = self._experiment_identity_pvs.get(capture_code) 

158 if not roles: 

159 return 

160 

161 ( 

162 (proposal_number, proposal_number_observed_at), 

163 (esaf_number, esaf_number_observed_at), 

164 (esaf_doi_number, esaf_doi_number_observed_at), 

165 ) = await asyncio.gather( 

166 *(self._read_one(capture_code, role, roles.get(role)) for role in _ROLES) 

167 ) 

168 

169 if proposal_number is None and esaf_number is None and esaf_doi_number is None: 

170 _log.info( 

171 "capture_experiment_identity.nothing_to_record", 

172 capture_code=capture_code, 

173 run_id=str(run_id), 

174 ) 

175 return 

176 

177 try: 

178 await self._store.upsert( 

179 run_id=run_id, 

180 proposal_number=proposal_number, 

181 proposal_number_observed_at=proposal_number_observed_at, 

182 esaf_number=esaf_number, 

183 esaf_number_observed_at=esaf_number_observed_at, 

184 esaf_doi_number=esaf_doi_number, 

185 esaf_doi_number_observed_at=esaf_doi_number_observed_at, 

186 created_at=self._deps.clock.now(), 

187 ) 

188 except asyncio.CancelledError: 

189 raise 

190 except Exception: 

191 # Unlike `_write_capture_path`, none of these three values 

192 # is personal data, so the full exception (which could 

193 # include the failing row's values in an asyncpg DETAIL 

194 # line) is safe to log. 

195 _log.exception( 

196 "capture_experiment_identity.vault_write_failed", 

197 capture_code=capture_code, 

198 run_id=str(run_id), 

199 ) 

200 return 

201 _log.info( 

202 "capture_experiment_identity.recorded", 

203 capture_code=capture_code, 

204 run_id=str(run_id), 

205 )

207 async def _read_one( 

208 self, capture_code: str, role: str, pv: str | None 

209 ) -> tuple[str | None, datetime | None]: 

210 """One role's reading, or `(None, None)` when the role is 

211 undeclared for this code, unreachable, unusable, or resolves to 

212 `resolved_experiment_identity_text`'s absent case (Trap 1).""" 

213 if pv is None: 

214 return None, None 

215 try: 

216 reading = await self._control_port.read(pv) 

217 except asyncio.CancelledError: 

218 raise 

219 except (ControlNotConnectedError, ControlTimeoutError, ControlAccessDeniedError) as exc: 

220 _log.warning( 

221 "capture_experiment_identity.read_unreachable", 

222 capture_code=capture_code, 

223 role=role, 

224 pv=pv, 

225 detail=str(exc), 

226 ) 

227 return None, None 

228 except ControlValueCoercionError as exc: 

229 _log.warning( 

230 "capture_experiment_identity.read_uncoercible", 

231 capture_code=capture_code, 

232 role=role, 

233 pv=pv, 

234 detail=str(exc), 

235 ) 

236 return None, None 

237 except Exception: 

238 _log.exception( 

239 "capture_experiment_identity.read_failed", 

240 capture_code=capture_code, 

241 role=role, 

242 pv=pv, 

243 ) 

244 return None, None 

245 return self._to_value(capture_code, role, pv, reading)

247 def _to_value( 

248 self, capture_code: str, role: str, pv: str, reading: Measurement 

249 ) -> tuple[str | None, datetime | None]: 

250 if reading.produced_at is None: 

251 # The port's dual-clock rule forbids substituting CORA's own 

252 # clock for an absent substrate time (Trap 2: there would be 

253 # no honest staleness evidence to carry). 

254 _log.info( 

255 "capture_experiment_identity.no_substrate_time", 

256 capture_code=capture_code, 

257 role=role, 

258 pv=pv, 

259 ) 

260 return None, None 

261 value = resolved_experiment_identity_text(reading.value) 

262 if value is None: 

263 _log.info( 

264 "capture_experiment_identity.absent_reading", 

265 capture_code=capture_code, 

266 role=role, 

267 pv=pv, 

268 ) 

269 return None, None 

270 return value, reading.produced_at
