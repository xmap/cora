1"""DurableCopyRegistrar's periodic sweep: lifespan wiring. 

2 

3Mirrors `cora.api._capture_scan_ingestor.capture_scan_ingestor_lifespan` 

4phase for phase: kill-switch early return that still yields, a startup 

5grant probe, a pool-conditional candidate lookup, and a `_sweep_loop` 

6that logs and carries on rather than letting one bad tick take the 

7whole task down. 

8 

9## The driver instance is long-lived, on purpose 

10 

11`DurableDistributionDriver` carries a keyset cursor as instance state 

12(`self._cursor`). Constructing a fresh driver per tick would reset that 

13cursor to `None` on every call, reproducing exactly the starvation the 

14cursor was built to prevent (see `_durable_distribution_driver`'s own 

15module docstring): a population of permanently-stuck candidates would 

16consume the whole attempt budget every tick and the sweep would never 

17reach anything past them. The driver here is built ONCE, outside 

18`_sweep_loop`, and the loop calls `.tick()` on the same instance 

19forever. 

20 

21## Why the probe and the checksum computer are both SSH-only here 

22 

23`SshLocateProbe` is the only `LocateProbe` this codebase ships (no 

24local counterpart exists, matching the fact that at 2-BM the durable 

25copy is never on CORA's own host); the boot guard 

26(`_enforce_run_witness_recording_gate`) refuses to enable the sweep 

27without `scan_probe_remote_host` set, so by the time this module runs 

28the assertions below hold. The checksum computer is built through the 

29SAME `active_scan_transport` conditional `cora.data.wire 

30._build_scan_ingest_pair` uses for `ingest_scan`, sharing one 

31`SshProbeConfig` with the probe: both must reach the SAME host that 

32holds the bytes, or the digest silently runs against a host that never 

33had them. 

34"""

74async def _sweep_loop(driver: DurableDistributionDriver, *, interval_seconds: float) -> None: 

75 """Periodic sweep. A failed tick is logged (inside `tick()` itself, 

76 which never raises); cancellation propagates.""" 

77 while True: 

78 try: 

79 await driver.tick() 

80 except asyncio.CancelledError: 

81 raise 

82 except Exception: 

83 _log.exception("durable_distribution.tick_failed") 

84 await asyncio.sleep(interval_seconds) 

85 

86 

87@contextlib.asynccontextmanager 

88async def durable_distribution_lifespan( 

89 deps: Kernel, *, interval_seconds: float | None = None 

90) -> AsyncGenerator[None]: 

91 """Spawn the durable-distribution sweep for the duration of the 

92 context. No-op unless `settings.durable_distribution_sweep_enabled` 

93 is True (default off, so a deployment opts in explicitly). When 

94 enabled, probes the `RegisterDistribution` grant once at startup (or 

95 refuses boot in strict mode) so a missing grant is surfaced 

96 immediately rather than only at the first denied tick, mirroring 

97 every sibling watcher/writer runtime's `probe_read_grant` startup 

98 check. 

99 

100 `interval_seconds` overrides `settings.durable_distribution_sweep_tick_seconds`; 

101 tests use it to avoid a 30-second real wait, mirroring 

102 `capture_scan_ingestor_lifespan`'s identical parameter. 

103 """ 

104 if not deps.settings.durable_distribution_sweep_enabled: 

105 _log.info("durable_distribution.skipped", reason="disabled") 

106 yield 

107 return 

108 

109 await probe_read_grant( 

110 deps, 

111 agent_id=DURABLE_COPY_REGISTRAR_AGENT_ID, 

112 read_command=_READ_COMMAND, 

113 log_prefix=_LOG_PREFIX, 

114 strict=deps.settings.watcher_authz_strict, 

115 ) 

116 

117 bindings = deps.settings.capture_scan_ingestor_bindings 

118 candidate_lookup: DurableDistributionCandidateLookup = ( 

119 PostgresDurableDistributionCandidateLookup( 

120 deps.pool, 

121 durable_roots=durable_roots(bindings), 

122 durable_supply_ids=durable_supply_ids(bindings), 

123 ) 

124 if deps.pool is not None 

125 else NeverDurableDistributionCandidateLookup() 

126 ) 

127 

128 # Same transport the locator minting uses (see module docstring): 

129 # the boot guard guarantees scan_probe_remote_host is set whenever 

130 # this lifespan runs for real. 

131 host, roots = active_scan_transport(deps) 

132 remote_host = deps.settings.scan_probe_remote_host 

133 assert remote_host is not None, ( 

134 "durable_distribution_sweep_enabled=True; " 

135 "_enforce_run_witness_recording_gate guarantees scan_probe_remote_host is set" 

136 ) 

137 remote_python = deps.settings.scan_probe_remote_python 

138 assert remote_python is not None, ( 

139 "scan_probe_remote_host is set; Settings validation guarantees " 

140 "scan_probe_remote_python is too" 

141 ) 

142 config = SshProbeConfig( 

143 host=host, 

144 remote_python=remote_python, 

145 allowed_roots=roots, 

146 connect_timeout_seconds=deps.settings.scan_probe_ssh_connect_timeout_seconds, 

147 command_timeout_seconds=deps.settings.scan_probe_ssh_command_timeout_seconds, 

148 max_walk_seconds=deps.settings.posix_checksum_max_walk_seconds, 

149 ) 

150 

151 registrar = DigestingDurableCopyRegistrar( 

152 event_store=deps.event_store, 

153 authz=deps.authz, 

154 supply_lookup=deps.supply_lookup, 

155 checksum_computer=SshPosixChecksumComputer(config=config), 

156 clock=deps.clock, 

157 id_generator=deps.id_generator, 

158 ) 

159 capture_paths = ( 

160 PostgresCapturePathStore(deps.pool) if deps.pool is not None else InMemoryCapturePathStore() 

161 ) 

162 driver = DurableDistributionDriver( 

163 candidate_lookup=candidate_lookup, 

164 durable_locations=CaptureScanIngestorDurableLocationLookup(bindings), 

165 probe=SshLocateProbe(config=config), 

166 capture_paths=capture_paths, 

167 registrar=registrar, 

168 host=host, 

169 clock=deps.clock, 

170 ) 

171 

172 default_interval = deps.settings.durable_distribution_sweep_tick_seconds 

173 interval = interval_seconds if interval_seconds is not None else default_interval 

174 _log.info("durable_distribution.started", interval_seconds=interval) 

175 task = asyncio.create_task( 

176 _sweep_loop(driver, interval_seconds=interval), name="durable-distribution-sweep" 

177 ) 

178 try: 

179 yield 

180 finally: 

181 task.cancel() 

182 with contextlib.suppress(asyncio.CancelledError): 

183 await task 

184 _log.info("durable_distribution.stopped") 

185 

186
