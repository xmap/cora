1"""Recipe-expansion replay helpers for the `conduct_procedure` handler. 

2 

3Per [[project-run-procedure-replay-design]] the run-time replay path 

4locates the genesis `RecipeExpansionRecorded` provenance event in a 

5Procedure stream, extracts the pinned hash + bindings + port-version 

6tuple, then verifies a freshly-re-expanded `tuple[Step, ...]` matches 

7the recorded pins. This module collects the pure helpers; the handler 

8threads them after authz + Procedure load. 

9 

10This is the FIRST handler-tier site in CORA that reads 

11`StoredEvent.payload` directly outside a projection. Per replay-design 

12§Locks the rule-of-three threshold gates promoting the helper to a 

13shared module: when a SECOND handler (any BC) needs payload-direct 

14access, hoist `find_recipe_expansion_record` to a generic 

15`cora.infrastructure.event_payload` helper. For comparison, projections 

16also read `.payload` but at projection-fold time, not at 

17handler-orchestration time. See replay-design Anti-hook 12. 

18"""

36@dataclass(frozen=True) 

37class RecipeExpansionPins: 

38 """The replay-pinned subset of a `RecipeExpansionRecorded` payload. 

39 

40 Constructed by `pins_from_payload`. Carries only the fields the 

41 replay path needs (control flow), NOT the audit-only fields 

42 (procedure_id, recipe_id, capability_id, capability_version, 

43 step_count, occurred_at) which are read directly at the handler 

44 entry for logging. 

45 """ 

46 

47 recipe_version: str | None 

48 bindings: Mapping[str, Any] 

49 bindings_hash: str 

50 steps_hash: str 

51 expansion_port_version: str 

52 

53 

54def find_recipe_expansion_record( 

55 stored_events: Iterable[StoredEvent], 

56) -> StoredEvent | None: 

57 """Locate the `RecipeExpansionRecorded` event in a Procedure stream. 

58 

59 Scans linearly from head, returns the first match, early-exits on 

60 first hit. In well-formed Recipe-driven Procedure streams the match 

61 lands at index 1 (the second event in the genesis 2-event block 

62 emitted by `register_procedure_from_recipe`); the unit test pins 

63 this position invariant. Tail-scan is wrong: only the genesis 

64 `RecipeExpansionRecorded` defines the replay snapshot. 

65 

66 Returns `None` when no match. The caller decides whether None is 

67 expected (legacy Procedure with `recipe_id is None`) or an error 

68 (recipe-driven Procedure missing its provenance event, raised as 

69 `RecipeExpansionRecordNotFoundError` by the handler). 

70 """ 

71 for event in stored_events: 

72 if event.event_type == "RecipeExpansionRecorded": 

73 return event 

74 return None 

75 

76 

77_REQUIRED_PINS_KEYS = ( 

78 "bindings", 

79 "bindings_hash", 

80 "expansion_port_version", 

81 "steps_hash", 

82) 

83 

84 

85def pins_from_payload(procedure_id: UUID, payload: Mapping[str, Any]) -> RecipeExpansionPins: 

86 """Extract the replay-pinned subset from a `RecipeExpansionRecorded` payload. 

87 

88 Defensive: raises `RecipeExpansionRecordNotFoundError(procedure_id)` 

89 if any required key is missing (covers the corrupt-payload case 

90 distinct from missing-event case; both surface the same error 

91 family per the replay-design lock on triage simplicity). 

92 """ 

93 missing = [key for key in _REQUIRED_PINS_KEYS if key not in payload] 

94 if missing: 

95 raise RecipeExpansionRecordNotFoundError(procedure_id) 

96 return RecipeExpansionPins( 

97 recipe_version=payload.get("recipe_version"), 

98 bindings=dict(payload["bindings"]), 

99 bindings_hash=payload["bindings_hash"], 

100 steps_hash=payload["steps_hash"], 

101 expansion_port_version=payload["expansion_port_version"], 

102 ) 

103 

104 

105def verify_bindings_hash(procedure_id: UUID, pins: RecipeExpansionPins) -> None: 

106 """Verify the recorded `bindings` payload still hashes to `bindings_hash`. 

107 

108 Raises `RecipeExpansionReplayMismatchError(procedure_id, "bindings")` 

109 on mismatch. Bindings drift is input drift (the recorded payload 

110 no longer canonicalizes to its recorded hash, i.e. payload 

111 corruption); failing it BEFORE the steps check isolates the failure 

112 mode in the discriminator value, easier to triage than a downstream 

113 steps mismatch caused by upstream binding corruption. 

114 """ 

115 recomputed = hashlib.sha256(canonical_json_bytes(dict(pins.bindings))).hexdigest() 

116 if recomputed != pins.bindings_hash: 

117 raise RecipeExpansionReplayMismatchError(procedure_id, "bindings") 

118 

119 

120def verify_steps_hash( 

121 procedure_id: UUID, 

122 steps: tuple[Step, ...], 

123 pins: RecipeExpansionPins, 

124) -> None: 

125 """Verify the re-expanded steps still hash to the recorded `steps_hash`. 

126 

127 Raises `RecipeExpansionReplayMismatchError(procedure_id, "steps")` 

128 on mismatch. Steps drift is expansion-logic drift (the port 

129 produces different output for the same input than at write time); 

130 runs AFTER `verify_bindings_hash` because steps drift downstream 

131 of bindings is a confusing diagnostic. 

132 """ 

133 recomputed = hashlib.sha256(canonical_json_bytes(steps_to_wire(steps))).hexdigest() 

134 if recomputed != pins.steps_hash: 

135 raise RecipeExpansionReplayMismatchError(procedure_id, "steps") 

136 

137 

138MismatchField = Literal["bindings", "steps"]
