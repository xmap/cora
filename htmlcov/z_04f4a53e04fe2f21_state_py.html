1"""Assembly aggregate state, value objects, status enum, and domain errors. 

2 

3An `Assembly` is a content-addressed composition blueprint for a 

4reusable cluster of Assets (e.g., the MCTOptics fixture at APS 2-BM: 

5microscope + 3 objectives + camera + scintillator, wired together). 

6Declares `required_slots` (Family-typed, cardinality-annotated, 

7optionally pre-Placed) and `required_wires` (slot-keyed 4-tuples). 

8Exposes a stable `presents_as_family_id` so other aggregates 

9(Method.needed_families, Capability bindings) can treat an 

10instantiated Assembly as one typed unit at the same level as a 

11single Asset. 

12 

13See `project_assembly_aggregate_design` for the locked design memo. 

14 

15## Identity 

16 

17`id` is the opaque UUID (CORA's standard internal id) used for 

18event-store stream keying. `name` is a human-readable AssemblyName 

19(non-unique). `content_hash` is the structural fingerprint 

20(SHA-256 hex over the canonical subset 

21`{name, presents_as_family_id, required_slots, required_wires, 

22parameter_overrides_schema}`); two operators independently authoring 

23the same Assembly converge on the same hash. 

24 

25## Slot keying 

26 

27`required_slots: frozenset[TemplateSlot]` and `required_wires: 

28frozenset[TemplateWire]` BOTH key by `slot_name` (string), NOT by 

29Asset UUID. Reason: an Assembly is a template; the Assets it 

30references do not exist at template-definition time. Slot-to-asset 

31translation happens at `register_fixture` time. 

32This inverts the timing of Plan.wires validation, which has access 

33to concrete Asset.ports and so enforces direction + signal-type + 

34fan-in at write time. 

35 

36## Internal closure 

37 

38`Assembly.__post_init__` enforces that every `TemplateWire`'s 

39endpoints reference a slot present in `required_slots`. This is the 

40structural well-formedness check; direction / signal_type / fan-in 

41rules live at instantiate time when concrete Assets are bound. 

42 

43## Revision lineage 

44 

45`status` is the AssemblyStatus FSM (Defined / Versioned / 

46Deprecated). `version` is an operator-curated free-form label 

47(no SemVer enforcement). Multiple AssemblyVersioned events on the 

48same stream are permitted; each writes a fresh content_hash 

49snapshot. Mirrors CalibrationRevision's append-only revisions. 

50 

51## Drawing 

52 

53`drawing` is the optional engineering reference for the assembly 

54itself (ICMS or DOI). Excluded from the content_hash canonical 

55subset because it is operator-curatorial metadata, not structural 

56identity: two Assemblies with identical structure but different 

57drawings collide on content_hash, which is the intended semantic. 

58 

59## Bounded VOs 

60 

61`AssemblyName` is trimmed 1-200 chars (matches Family/Mount). 

62`SlotName` is trimmed 1-100 chars (matches Wire port-name bound). 

63Both raise dedicated InvalidXError classes on violation. 

64"""

438@dataclass(frozen=True) 

439class TemplateSlot: 

440 """One slot in an Assembly's composition blueprint. 

441 

442 `slot_name` is the canonical slot identity within this Assembly 

443 (string, 1-100 chars). `required_family_ids` is the non-empty set 

444 of FamilyIds any instantiated Asset must include at least one of. 

445 `cardinality` says how many Assets can fill this slot 

446 (Exactly1 / ZeroOrOne / OneOrMore / ZeroOrMore). `default_settings` 

447 and `default_placement` are optional template defaults applied 

448 at instantiation unless overridden. 

449 

450 `default_settings` validation against the intersection of all 

451 required_family_ids' settings schemas runs at define_assembly time 

452 (handler-side, not in __post_init__) because it requires loading 

453 Family records. The VO's __post_init__ only enforces structural 

454 well-formedness. 

455 """ 

456 

457 slot_name: SlotName 

458 required_family_ids: frozenset[UUID] 

459 cardinality: SlotCardinality 

460 default_settings: dict[str, Any] | None = None 

461 default_placement: Placement | None = None 

462 

463 def __post_init__(self) -> None: 

464 # Defensive: cardinality is annotated as SlotCardinality but 

465 # nothing in Python stops a caller from passing a raw string 

466 # at construction. Catching it here surfaces a typed domain 

467 # error rather than a downstream AttributeError on .value. 

468 if not isinstance(self.cardinality, SlotCardinality): # pyright: ignore[reportUnnecessaryIsInstance] 

469 raise InvalidSlotCardinalityError(str(self.cardinality)) 

470 if not self.required_family_ids: 

471 raise InvalidTemplateSlotError( 

472 f"TemplateSlot {self.slot_name.value!r} requires at least one Family" 

473 ) 

474 

475 def __hash__(self) -> int: 

476 # TemplateSlot is frozen but its `default_settings` field is a 

477 # dict (unhashable by Python default). Required because every 

478 # Assembly carries `required_slots: frozenset[TemplateSlot]`. 

479 # Hash canonicalizes the dict portion via stdlib json sort-keys 

480 # to give a stable, deterministic hash independent of insertion 

481 # order; full-record __eq__ stays dataclass-generated. 

482 settings_key = ( 

483 json.dumps(self.default_settings, sort_keys=True, separators=(",", ":")) 

484 if self.default_settings is not None 

485 else None 

486 ) 

487 return hash( 

488 ( 

489 self.slot_name, 

490 self.required_family_ids, 

491 self.cardinality, 

492 settings_key, 

493 self.default_placement, 

494 ) 

495 )

498@dataclass(frozen=True) 

499class TemplateWire: 

500 """A typed slot-to-slot connection in an Assembly's blueprint. 

501 

502 Tuple `(source_slot_name, source_port_name, target_slot_name, 

503 target_port_name)` describes one connection. The 4-tuple IS the 

504 identity; `frozenset[TemplateWire]` deduplicates on the tuple. 

505 

506 Mirrors `Wire` (the Plan-tier signal-routing VO) in shape but 

507 keys by slot_name strings rather than Asset UUIDs because an 

508 Assembly cannot reference Assets that do not exist yet at 

509 template-definition time. 

510 

511 Validation rules at instantiation time (NOT here): 

512 - source port must have `direction=OUTPUT` 

513 - target port must have `direction=INPUT` 

514 - `source_port.signal_type == target_port.signal_type` 

515 - target port is the destination of at most one Wire (fan-in 

516 forbidden); fan-out (one source to many targets) is allowed 

517 

518 `__post_init__` enforces structural shape only: each of the four 

519 string fields trims and bounds 1-100 chars, and the degenerate 

520 full-loop case (same slot AND same port on both endpoints) is 

521 rejected. Self-slot wires with different port names are allowed 

522 (PandABox LUT pattern, mirroring the Plan invariant). 

523 

524 Cross-wire closure (every slot_name must exist in the parent 

525 Assembly's `required_slots`) is enforced at the Assembly level 

526 via `WireReferencesUnknownSlotError`, not here. 

527 """ 

528 

529 source_slot_name: str 

530 source_port_name: str 

531 target_slot_name: str 

532 target_port_name: str 

533 

534 def __post_init__(self) -> None: 

535 for label, value in ( 

536 ("source_slot_name", self.source_slot_name), 

537 ("source_port_name", self.source_port_name), 

538 ("target_slot_name", self.target_slot_name), 

539 ("target_port_name", self.target_port_name), 

540 ): 

541 trimmed = value.strip() 

542 if not trimmed: 

543 raise InvalidWireSpecError(f"{label} cannot be empty after trimming") 

544 if len(trimmed) > WIRE_PORT_NAME_MAX_LENGTH: 

545 raise InvalidWireSpecError( 

546 f"{label} must be 1-{WIRE_PORT_NAME_MAX_LENGTH} chars after trimming " 

547 f"(got: {value!r})" 

548 ) 

549 object.__setattr__(self, label, trimmed) 

550 if ( 

551 self.source_slot_name == self.target_slot_name 

552 and self.source_port_name == self.target_port_name 

553 ): 

554 raise InvalidWireSpecError( 

555 f"degenerate full self-loop on slot {self.source_slot_name!r} " 

556 f"port {self.source_port_name!r}" 

557 )

560@dataclass(frozen=True) 

561class Assembly: 

562 """Aggregate root: a reusable composition blueprint. 

563 

564 `id` is the opaque UUID stream key; `name` is the human-readable 

565 AssemblyName. `presents_as_family_id` is the FamilyId the 

566 instantiated Assembly looks like to Method.needed_families and 

567 Plan binding (one Asset-shaped unit). 

568 

569 `required_slots` and `required_wires` together describe the 

570 composition: slots declare what kinds of Assets fill which roles, 

571 wires declare how those Assets connect. Both key by slot_name 

572 (not by Asset UUID) since Assets do not exist at template time. 

573 

574 `parameter_overrides_schema` is an optional JSON Schema subset 

575 declaring the shape of parameter_overrides accepted at 

576 instantiation. `drawing` is the optional engineering reference. 

577 

578 `status` transitions Defined -> Versioned (multiple times) -> 

579 Deprecated. `version` is an operator-curated label. `content_hash` 

580 is the SHA-256 hex fingerprint of the canonical subset 

581 `{name, presents_as_family_id, required_slots, required_wires, 

582 parameter_overrides_schema}` (excludes id / drawing / version / 

583 status, which are not structural identity per the design memo). 

584 

585 `__post_init__` enforces internal closure: every TemplateWire's 

586 source_slot_name and target_slot_name MUST appear in 

587 `required_slots`. Cross-aggregate references (FamilyId existence, 

588 schema validation) live in handler-side projection checks. 

589 """ 

590 

591 id: UUID 

592 name: AssemblyName 

593 presents_as_family_id: UUID 

594 required_slots: frozenset[TemplateSlot] = field(default_factory=frozenset[TemplateSlot]) 

595 required_wires: frozenset[TemplateWire] = field(default_factory=frozenset[TemplateWire]) 

596 parameter_overrides_schema: dict[str, Any] | None = None 

597 drawing: Drawing | None = None 

598 status: AssemblyStatus = AssemblyStatus.DEFINED 

599 version: str | None = None 

600 content_hash: str | None = None 

601 presents_as: frozenset[RoleId] = field(default_factory=frozenset[RoleId]) 

602 """Layer 3 sub-slice 3C: the set of global Role contracts this 

603 composed Assembly advertises (Lock 4 universal presents_as). 

604 Parallel mechanism to the scalar `presents_as_family_id`: 3C 

605 keeps the scalar (per anti-hook #6, one migration cycle), and 

606 layers `presents_as` alongside for Role-based binding via 3D's 

607 bind_plan_role role_kind path. MCTOptics-Assembly seeds 

608 `{Imager}` at scenario-fixture time. 

609 

610 NOT included in `content_subset()` -- additive orthogonal-axis 

611 field, parallel to Family.settings_schema; adding or removing a 

612 Role advertisement does not produce a structurally-distinct 

613 Assembly identity. 

614 

615 Affordance-superset check (Family.affordances >= 

616 Role.required_affordances) DEFERRED at 3C: Assembly affordances 

617 derive from the constituent Family union at register_fixture 

618 time, not Assembly template time, so the check belongs at the 

619 fixture-registration layer. Watch item logged."""

621 def __post_init__(self) -> None:

629 def content_subset(self) -> dict[str, object]: 

630 """Canonical content subset hashed into `content_hash`. 

631 

632 Pins identity per `project_content_addressed_identity_design`: 

633 `name + presents_as_family_id + required_slots + required_wires + 

634 parameter_overrides_schema`. Excluded: `id` (identity, not 

635 content), `status` and `version` (lifecycle, derived in evolver 

636 from event type and version label), `drawing` (operator- 

637 curatorial metadata per the design memo's content_hash 

638 composition lock), `content_hash` itself (cannot self-contain). 

639 

640 Delegates to `canonical_assembly_subset` so the helper used by 

641 `compute_assembly_content_hash` (raw-args path) and this 

642 method (state path) materialize the same shape. Field 

643 addition lands in ONE place; drift between the two paths 

644 becomes structurally impossible. 

645 """ 

646 return canonical_assembly_subset( 

647 name=self.name, 

648 presents_as_family_id=self.presents_as_family_id, 

649 required_slots=self.required_slots, 

650 required_wires=self.required_wires, 

651 parameter_overrides_schema=self.parameter_overrides_schema, 

652 )

655def canonical_assembly_subset( 

656 *, 

657 name: "AssemblyName | str", 

658 presents_as_family_id: UUID, 

659 required_slots: frozenset[TemplateSlot], 

660 required_wires: frozenset[TemplateWire], 

661 parameter_overrides_schema: dict[str, object] | None, 

662) -> dict[str, object]: 

663 """Materialize the canonical content subset of an Assembly. 

664 

665 Single source of truth for the structural-identity body. Called by 

666 both `Assembly.content_subset()` (state path) and 

667 `compute_assembly_content_hash` (raw-args path); the round-trip 

668 equivalence between the two is pinned in tests. 

669 

670 Slots render as a list of dicts sorted by slot_name; wires render 

671 as sorted 4-tuples-of-strings for canonical-sort determinism. 

672 UUIDs render as strings. Adding a new identity-bearing field 

673 requires editing this one function plus the content_hash 

674 differs-test corpus. 

675 """ 

676 name_value = name.value if isinstance(name, AssemblyName) else name 

677 return { 

678 "name": name_value, 

679 "presents_as_family_id": str(presents_as_family_id), 

680 "required_slots": sorted( 

681 ( 

682 { 

683 "slot_name": slot.slot_name.value, 

684 "required_family_ids": sorted(str(f) for f in slot.required_family_ids), 

685 "cardinality": slot.cardinality.value, 

686 "default_settings": slot.default_settings, 

687 "default_placement": ( 

688 canonical_placement_subset(slot.default_placement) 

689 if slot.default_placement is not None 

690 else None 

691 ), 

692 } 

693 for slot in required_slots 

694 ), 

695 key=lambda d: str(d["slot_name"]), 

696 ), 

697 "required_wires": sorted( 

698 ( 

699 wire.source_slot_name, 

700 wire.source_port_name, 

701 wire.target_slot_name, 

702 wire.target_port_name, 

703 ) 

704 for wire in required_wires 

705 ), 

706 "parameter_overrides_schema": parameter_overrides_schema, 

707 }
