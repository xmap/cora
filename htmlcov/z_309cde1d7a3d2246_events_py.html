1"""Domain events emitted by the Actor aggregate, plus the discriminated union. 

2 

3Events live in the aggregate folder (not the slice) because they are 

4intrinsic facts about the aggregate's history -- the slice just decides 

5when to emit them. The evolver dispatches on the union; new event types 

6are appended both as a class definition and to the union alias. 

7 

8Per the locked "primitives in events" convention, payloads serialize 

9to plain dicts of primitives. `to_payload` and `from_stored` are the 

10single home for the (de)serialization logic; per-slice handlers no 

11longer carry their own serializers. 

12 

13## V1/V2 dispatch for ActorRegistered (PII vault) 

14 

15`ActorRegistered` events written before the PII vault landed carry a 

16`name` field in the payload (V1). Post-vault writes emit the event with 

17the new `event_type` string `"ActorRegisteredV2"` and no `name` field; 

18display names live in the `actor_profile` table. Both `event_type` 

19arms produce the same modern `ActorRegistered` dataclass (no `name`). 

20The Marten / Axon legacy-rename precedent mirrors 

21`cora/equipment/aggregates/asset/events.py` (Capability→Family 

22rename). Backfill migration `20260523120100_backfill_actor_profile.sql` 

23copies V1 names into `actor_profile` before the new code ships, so 

24read paths find the right display name even for legacy actors. 

25""" 

26

37@dataclass(frozen=True) 

38class ActorRegistered: 

39 """A new actor was registered. 

40 

41 Carries NO PII -- display name and future PII fields live in the 

42 `actor_profile` table per the PII vault pattern. The cross-BC 

43 invariant `Agent.id == Actor.id` still holds: `register_actor` and 

44 `define_agent` both emit this event with the same `actor_id` they 

45 use to upsert the profile row. 

46 

47 `kind` discriminates `human` from `agent` from `service_account`. 

48 REQUIRED at construction: both callsites 

49 (`register_actor` in Access BC, `define_agent` in Agent BC) 

50 MUST pass it explicitly so drift between them surfaces as a pyright 

51 error rather than silently minting an agent-kind Actor through the 

52 human path. See [[project_agent_bc_design]] P0-4 cleanup pass. 

53 

54 New writes emit `event_type = "ActorRegisteredV2"`; legacy V1 

55 writes (carrying `name` in the payload) keep their `event_type = 

56 "ActorRegistered"` string and replay via the legacy arm in 

57 `from_stored`, which drops the `name` on rebuild (the backfill 

58 migration copied legacy names into actor_profile already). 

59 """ 

60 

61 actor_id: UUID 

62 occurred_at: datetime 

63 kind: ActorKind

66@dataclass(frozen=True) 

67class ActorDeactivated: 

68 """An existing actor was deactivated. The actor remains in the 

69 system but `Actor.active` flips to False.""" 

70 

71 actor_id: UUID 

72 occurred_at: datetime

75@dataclass(frozen=True) 

76class ActorProfileForgotten: 

77 """Audit event for PII erasure (GDPR / PIPL / LGPD / CCPA "right to 

78 be forgotten"). 

79 

80 Carries NO PII — only `actor_id` + the timestamp when the 

81 erasure landed. Emitted by the `forget_actor` slice in the same 

82 Postgres transaction that scrubs + deletes the corresponding 

83 `actor_profile` row, so the event log carries a permanent audit 

84 trail of "this actor's PII was erased on this date" without 

85 re-introducing any identifying data. Aggregate state is 

86 unchanged by this event; the event exists purely for downstream 

87 subscribers (the projection swaps the cached `name` for the 

88 tombstone literal). 

89 """ 

90 

91 actor_id: UUID 

92 occurred_at: datetime 

93

95# Discriminated union of every event the Actor aggregate emits. Add new 

96# event classes above and extend this alias when new slices land. 

97ActorEvent = ActorRegistered | ActorDeactivated | ActorProfileForgotten

100def event_type_name(event: ActorEvent) -> str: 

101 """Discriminator string written into StoredEvent.event_type. 

102 

103 `ActorRegistered` writes use the V2 string; the V1 string lives 

104 only in `from_stored` for legacy replay. `ActorDeactivated` has 

105 one shape forever. 

106 """ 

107 if isinstance(event, ActorRegistered): 

108 return "ActorRegisteredV2" 

109 return type(event).__name__

112def to_payload(event: ActorEvent) -> dict[str, Any]: 

113 """Serialize an Actor event to a JSON-friendly dict for jsonb storage. 

114 

115 Primitives only: UUIDs become strings, datetimes become ISO-8601 strings. 

116 No PII -- display name lives in actor_profile table. 

117 """ 

118 match event: 

119 case ActorRegistered(actor_id=actor_id, occurred_at=occurred_at, kind=kind): 

120 return { 

121 "actor_id": str(actor_id), 

122 "occurred_at": occurred_at.isoformat(), 

123 "kind": kind.value, 

124 } 

125 case ActorDeactivated(actor_id=actor_id, occurred_at=occurred_at): 

126 return { 

127 "actor_id": str(actor_id), 

128 "occurred_at": occurred_at.isoformat(), 

129 } 

130 case ActorProfileForgotten(actor_id=actor_id, occurred_at=occurred_at): 

131 return { 

132 "actor_id": str(actor_id), 

133 "occurred_at": occurred_at.isoformat(), 

134 } 

135 case _: # pragma: no cover # exhaustiveness guard 

136 assert_never(event)

139def from_stored(stored: StoredEvent) -> ActorEvent: 

140 """Rebuild an Actor event from a StoredEvent loaded from the event store. 

141 

142 Dispatches on `stored.event_type`; raises ValueError on unknown 

143 discriminators so a stream contaminated with foreign event types 

144 fails loud rather than silently being dropped by the evolver. 

145 

146 `ActorRegistered` has two `event_type` strings (Marten / Axon 

147 legacy-rename pattern; mirrors 

148 `cora/equipment/aggregates/asset/events.py` Capability→Family): 

149 

150 - `"ActorRegisteredV2"`: post-PII-vault writes; payload has no 

151 `name` field. 

152 - `"ActorRegistered"` (legacy V1): pre-vault writes carry a 

153 `name` field in the payload. The arm drops the name on rebuild 

154 (the backfill migration copied legacy names into actor_profile 

155 before the new code went live). 

156 

157 Both arms produce the same modern `ActorRegistered` dataclass. 

158 """ 

159 payload = stored.payload 

160 match stored.event_type: 

161 case "ActorRegisteredV2": 

162 return deserialize_or_raise( 

163 "ActorRegisteredV2", 

164 lambda: ActorRegistered( 

165 actor_id=UUID(payload["actor_id"]), 

166 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

167 kind=ActorKind(payload["kind"]), 

168 ), 

169 extra=(ValueError,), 

170 ) 

171 case "ActorRegistered": 

172 # Legacy V1: payload still carries `name`. Drop it on 

173 # rebuild; the backfill migration copied the legacy name 

174 # into actor_profile before this arm started replaying. 

175 # `kind` may also be absent on the oldest legacy payloads; 

176 # fall back to HUMAN. 

177 return deserialize_or_raise( 

178 "ActorRegistered", 

179 lambda: ActorRegistered( 

180 actor_id=UUID(payload["actor_id"]), 

181 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

182 kind=ActorKind(payload.get("kind", ActorKind.HUMAN.value)), 

183 ), 

184 extra=(ValueError,), 

185 message_suffix=" (V1)", 

186 ) 

187 case "ActorDeactivated": 

188 return deserialize_or_raise( 

189 "ActorDeactivated", 

190 lambda: ActorDeactivated( 

191 actor_id=UUID(payload["actor_id"]), 

192 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

193 ), 

194 extra=(ValueError,), 

195 ) 

196 case "ActorProfileForgotten": 

197 return deserialize_or_raise( 

198 "ActorProfileForgotten", 

199 lambda: ActorProfileForgotten( 

200 actor_id=UUID(payload["actor_id"]), 

201 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

202 ), 

203 extra=(ValueError,), 

204 ) 

205 case _: 

206 msg = f"Unknown ActorEvent event_type: {stored.event_type!r}" 

207 raise ValueError(msg)
