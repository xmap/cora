1"""SurfaceActiveVisitProjection: folds Visit control events into the 

2`proj_trust_surface_active_visit` read model -- the "who drives this 

3Surface right now?" projection. 

4 

5Subscribed events: 

6 - VisitSurfaceControlTaken -> 2-statement transaction: 

7 (1) UPDATE prior holder's released_at, then 

8 (2) INSERT new holder row. 

9 - VisitSurfaceControlReleased -> UPDATE the open row's released_at. 

10 

11Take-control must atomically mark the prior holder released + INSERT 

12the new holder row so the invariant "at most one row per surface with 

13released_at IS NULL" is preserved without a partial unique index. The 

14`async with conn.transaction()` block guarantees atomicity at the 

15projection-worker tier (savepoint within the outer worker transaction). 

16""" 

17 

18# pyright: reportUnknownMemberType=false, reportUnknownVariableType=false, reportUnknownArgumentType=false 

19 

20from dataclasses import dataclass 

21from datetime import datetime 

22from uuid import UUID 

23 

24from asyncpg import Pool 

25 

26from cora.infrastructure.ports.event_store import StoredEvent 

27from cora.infrastructure.projection.handler import ConnectionLike 

28 

29_SUBSCRIBED: frozenset[str] = frozenset({"VisitSurfaceControlTaken", "VisitSurfaceControlReleased"}) 

30 

31# Statement 1: mark whatever row is currently open on this Surface as 

32# released at the same instant. The `since_at < $2` predicate is load- 

33# bearing for replay safety: a stale replay of an OLDER Took event 

34# must NOT close a NEWER open row. Without it, sequence 

35# Took(A)@t1 -> Took(B)@t2 followed by a replay of Took(A)@t1 would 

36# stomp B's row with released_at=t1 (i.e., before B took control). 

37# Replay-idempotent: second replay of Took(A)@t1 finds either zero 

38# open rows or only rows with since_at >= $2 and is a no-op. 

39_TAKE_CONTROL_UPDATE_PRIOR_SQL = """ 

40UPDATE proj_trust_surface_active_visit 

41SET released_at = $2, updated_at = now() 

42WHERE surface_id = $1 AND released_at IS NULL AND since_at < $2 

43""" 

44 

45# Statement 2: INSERT the new holder. PK includes since_at so re-applying 

46# the same event finds an existing row and the ON CONFLICT clause makes it 

47# a no-op. 

48_TAKE_CONTROL_INSERT_NEW_SQL = """ 

49INSERT INTO proj_trust_surface_active_visit 

50 (surface_id, visit_id, since_at, released_at) 

51VALUES ($1, $2, $3, NULL) 

52ON CONFLICT (surface_id, visit_id, since_at) DO NOTHING 

53""" 

54 

55# VisitSurfaceControlReleased: UPDATE the open row's released_at. 

56# Naturally idempotent: second replay finds zero rows with released_at IS NULL. 

57_RELEASE_CONTROL_SQL = """ 

58UPDATE proj_trust_surface_active_visit 

59SET released_at = $3, updated_at = now() 

60WHERE surface_id = $1 AND visit_id = $2 AND released_at IS NULL 

61""" 

62 

63# Read-side: who currently holds this Surface? At most one row with 

64# released_at IS NULL per surface (invariant maintained by the take-control 

65# 2-statement transaction above). ORDER BY since_at DESC + LIMIT 1 is 

66# defensive: if a duplicate replay ever produced two open rows, the most 

67# recent wins; the partial index makes this a fast index-only scan. 

68_READ_ACTIVE_HOLDER_SQL = """ 

69SELECT visit_id, since_at 

70FROM proj_trust_surface_active_visit 

71WHERE surface_id = $1 AND released_at IS NULL 

72ORDER BY since_at DESC 

73LIMIT 1 

74"""

85async def load_surface_active_visit(pool: Pool, surface_id: UUID) -> SurfaceActiveVisit | None: 

86 """Return the current controlling Visit for a Surface, or None if free. 

87 

88 Pool MUST be live; None-check belongs to caller. Returns None when no 

89 open row exists for the surface (free Surface) or when the projection 

90 has not yet drained the genesis event for the controlling Visit. 

91 """ 

92 row = await pool.fetchrow(_READ_ACTIVE_HOLDER_SQL, surface_id) 

93 if row is None: 

94 return None 

95 return SurfaceActiveVisit(visit_id=row["visit_id"], since_at=row["since_at"])

98class SurfaceActiveVisitProjection: 

99 """Maintains the `proj_trust_surface_active_visit` read model.""" 

100 

101 name = "proj_trust_surface_active_visit" 

102 subscribed_event_types = _SUBSCRIBED 

103 

104 async def apply( 

105 self, 

106 event: StoredEvent, 

107 conn: ConnectionLike, 

108 ) -> None: 

109 if event.event_type not in _SUBSCRIBED: 

110 return 

111 

112 payload = event.payload 

113 surface_id = UUID(payload["surface_id"]) 

114 visit_id = UUID(payload["visit_id"]) 

115 occurred_at = datetime.fromisoformat(payload["occurred_at"]) 

116 

117 match event.event_type: 

118 case "VisitSurfaceControlTaken": 

119 async with conn.transaction(): 

120 await conn.execute(_TAKE_CONTROL_UPDATE_PRIOR_SQL, surface_id, occurred_at) 

121 await conn.execute( 

122 _TAKE_CONTROL_INSERT_NEW_SQL, surface_id, visit_id, occurred_at 

123 ) 

124 case "VisitSurfaceControlReleased": 

125 await conn.execute(_RELEASE_CONTROL_SQL, surface_id, visit_id, occurred_at) 

126 case _: # pragma: no cover # _SUBSCRIBED gate above prevents reaching here 

127 pass
