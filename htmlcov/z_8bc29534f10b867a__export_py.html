1"""Walk the event store and follow every logbook envelope into its entries. 

2 

3Per `project_record_export_v3.md` F0/F2: a single stream query over the 

4whole `events` table, ordered by `(transaction_id, position)`, bounded by 

5one `pg_snapshot_xmin(pg_current_snapshot())` watermark captured up front 

6(not re-evaluated per row, unlike the projection worker's catch-up query, 

7which this SQL otherwise mirrors) so the export sees one consistent 

8snapshot. On each row whose `event_type` names a `*LogbookOpened` class 

9(`registered_envelope_classes()`, from Step 1's `_registry`), `kind` and 

10`logbook_id` come straight out of the already-decoded `payload` dict and 

11resolve through Step 1's `resolve()` to the matching entries-tier reader. 

12 

13Deliberately does not write bundle files: that is Step 3/4's job, once 

14hashing and the manifest exist. 

15 

16`entries_run_feed_heartbeats` (S5a), `entries_run_capture_probes` 

17(S5b), and `entries_enclosure_permit_probes` (S5c) have no envelope, so 

18none can be reached from the envelope-driven walk below. Their rows are 

19pulled separately, once, by a whole-table unscoped read each after the 

20walk finishes, via `_registry.EntriesTableSpec.unscoped_reader`. That 

21read shares this function's caller's `REPEATABLE READ` snapshot (opened 

22in `_shell.py`, never here), same as every envelope-driven entries read 

23above it, but the entries tier as a whole remains snapshot-bounded 

24while the streams tier is xmin-bounded: an unscoped row can have been 

25written by a transaction that committed above the watermark. Not 

26closable without a `transaction_id` column on the entries tables; 

27stated, not fixed, here. 

28 

29Also times each per-kind read (`ExportedRecord.read_seconds_by_logbook_kind`, 

30S5d), because this is the one place with a seam onto each individual 

31`spec.reader` / `spec.unscoped_reader` call; a caller outside this 

32module has no way to time a kind's read without re-issuing it a second 

33time against a different snapshot. 

34 

35`capture_source_row_count_by_logbook_kind` (S2b) lives here too, alongside `capture_watermark`: 

36a second, independent count per registered kind (an unscoped `count(*)` on 

37each `spec.table`, sharing no predicate with any reader above) that 

38`_manifest.py` compares against what this walk actually put in 

39`ExportedRecord.logbooks`. See that function's own docstring for why 

40independence is not the same shape for the six envelope-driven kinds as it 

41is for the three unscoped ones. 

42"""

62_WATERMARK_SQL = "SELECT pg_snapshot_xmin(pg_current_snapshot())::text" 

63 

64# Same column list as postgres_event_store.py's _LOAD_SQL and 

65# projection/worker.py's _ADVANCE_SQL. transaction_id is xid8; asyncpg has 

66# no output codec for it (same gap those two modules work around), so it 

67# is cast to text and aliased back onto its own name. 

68# 

69# ORDER BY is qualified `events.transaction_id`, not the bare column name: 

70# a bare reference resolves to the OUTPUT alias (per Postgres's documented 

71# "ORDER BY prefers an output column name over an input column of the same 

72# name" rule), which is the ::text expression above, so an unqualified 

73# ORDER BY sorts LEXICOGRAPHICALLY rather than numerically. That silently 

74# reorders any two rows whose transaction_id crosses a digit-count boundary 

75# (e.g. "10001137" sorts before "9995093"), corrupting the one ordering 

76# `fold` depends on for every consumer downstream of this query. Measured 

77# on arcturus 2026-08-25 (cora.api.record_fidelity_check caught it): a Run 

78# whose RunStarted and RunCompleted straddled the 7-to-8-digit xid boundary 

79# exported with RunCompleted first, so it could not refold. Only a 

80# table-qualified reference can name the input column once an output alias 

81# shares its name; renaming the alias instead would change the bundle's 

82# on-disk `transaction_id` key, a wider and unnecessary blast radius. 

83_STREAM_SQL = """ 

84SELECT position, event_id, stream_type, stream_id, version, event_type, 

85 schema_version, payload, metadata, correlation_id, causation_id, 

86 principal_id, occurred_at, recorded_at, 

87 signature, signature_kid, signature_version, 

88 transaction_id::text AS transaction_id 

89FROM events 

90WHERE transaction_id < $1::xid8 

91ORDER BY events.transaction_id, position 

92"""

105@dataclass(frozen=True, slots=True) 

106class ExportedRecord: 

107 """One export's two tiers, rendered but not yet hashed or redacted. 

108 

109 `streams` is every `events` row below the captured watermark, in 

110 `(transaction_id, position)` order. `logbooks` groups every 

111 entries-tier row pulled via an envelope by `kind`, in each kind's own 

112 registry order-by order; rows are UNFOLDED (one dict per row), never 

113 aggregated. 

114 

115 `watermark` is the SAME xmin value `capture_watermark` produced and 

116 the stream query above was bounded by, carried on the record itself 

117 so `build_manifest` can read it here rather than take it as an 

118 independent parameter: a caller wanting "the watermark this export 

119 used" for the manifest had no way to obtain it other than by calling 

120 `capture_watermark` a SECOND time, which returns a different snapshot 

121 than the one the rows were actually bounded by. Defaults to 0 for 

122 hand-built test fixtures that do not exercise watermark plumbing; 

123 every real export sets it from the same call `export_record` used. 

124 

125 `read_seconds_by_logbook_kind` is the wall-clock time spent inside 

126 `spec.reader` / `spec.unscoped_reader` calls for each kind, summed 

127 across every envelope occurrence for an envelope-driven kind (one 

128 call each time its `*LogbookOpened` envelope appears on the stream) 

129 or the one call for an unscoped kind. Times the read alone, not 

130 `render_row`. Added for S5d (`project_record_completeness_design.md`): 

131 the operator command needs a per-kind timing to make the first real 

132 export against a populated database a measurement rather than a 

133 fourth deferral argument. A kind absent from this dict was never 

134 read this export -- either it is `untraversed`, or it is an 

135 envelope-driven kind whose envelope never occurred, which the S4 

136 membership decision already treats as a genuine zero rather than a 

137 coverage gap. Defaults to an empty dict for the same hand-built- 

138 fixture reason `watermark` defaults to 0. 

139 """ 

140 

141 streams: tuple[dict[str, object], ...] 

142 logbooks: dict[str, tuple[dict[str, object], ...]] 

143 watermark: int = 0 

144 read_seconds_by_logbook_kind: dict[str, float] = field(default_factory=dict)

147async def capture_source_row_count_by_logbook_kind(conn: asyncpg.Connection) -> dict[str, int]: 

148 """One independent count per registered kind, for S2b's `source_row_count`. 

149 

150 Delegates to each spec's own `count_reader` (`_registry.py`'s 

151 `_make_count_reader`, `SELECT count(*) FROM <table>`, no predicate) so 

152 the SQL itself lives beside `reader` / `unscoped_reader` rather than 

153 being rebuilt here. Must run inside the SAME `REPEATABLE READ READ 

154 ONLY` transaction as `export_record`, so both counts and the traversal 

155 describe the same snapshot. Unlike `capture_watermark`, which 

156 `export_record` calls itself (its own docstring: "`export_record` is 

157 the sole caller"), this function has THREE callers today -- 

158 `_shell.py`'s `export_bundle` and `record_bundle_export.py`'s 

159 `export_record_bundles` each call it directly, back to back with 

160 `export_record`, inside the transaction they both open; neither calls 

161 it FROM `export_record`. Deliberately not folded into `export_record` 

162 / carried on `ExportedRecord` the way `watermark` and 

163 `read_seconds_by_logbook_kind` are: doing so would make this dict a 

164 mandatory field on every hand-built `ExportedRecord` test fixture 

165 across this package (`_hashing.py`, `_redaction.py`, `_bundle.py`'s 

166 tests build dozens, none caring about row counts), and giving it a 

167 default would reopen exactly the "coverage field switched off by a 

168 default" hole this design exists to close for `build_manifest`'s own 

169 required `source_row_count_by_logbook_kind` parameter. The residual 

170 this leaves is real and stated rather than hidden: nothing 

171 structurally prevents a future caller from pairing this dict with an 

172 `ExportedRecord` from a DIFFERENT snapshot the way `write_bundle`'s 

173 `ManifestRecordMismatchError` prevents a mismatched record/manifest 

174 pair; today's two production callers avoid it by construction (one 

175 `conn`, one transaction, both calls before it closes), not by a type 

176 that makes the mistake impossible. 

177 

178 Per `project_independent_check_principle.md`, this is the SECOND side of 

179 the check `_manifest.py`'s `_extent_by_logbook_kind` performs against 

180 `exported_row_count`. Independence is NOT uniform across the registry: 

181 

182 - For the six envelope-scoped kinds, this query shares no predicate at 

183 all with `spec.reader` (which filters on `logbook_id`). A row whose 

184 envelope never reached the stream walk still shows up here, which is 

185 precisely the omission-at-origin signal this design exists to raise. 

186 - For the three unscoped kinds (`heartbeat`, `capture_probe`, 

187 `permit_probe`), `spec.unscoped_reader` already runs the identical 

188 `SELECT * FROM <table>` shape, in the same snapshot. This count cannot 

189 diverge from a correct fetch of that table: it is not independent in 

190 the "different rows" sense for these three. What it still catches is a 

191 row lost or duplicated between the fetch and `_manifest.py`'s render- 

192 stage tally (`exported_row_count = len(record.logbooks[kind])`), since 

193 this query never goes through `render_row` or the `logbooks` dict at 

194 all. State this plainly rather than implying uniform independence; see 

195 `tests/integration/test_record_export_row_count_independence_postgres.py` 

196 for the proof (and the live-DB negative result) on each side, and 

197 `test_manifest.py`'s render-stage-loss unit tests for the fetch-vs- 

198 render axis. 

199 

200 Two limits worth stating in band rather than leaving implicit (the 

201 design memo's own "Adversaries, and what is actually caught" table 

202 names these as NOT CAUGHT by any form of this mechanism): 

203 

204 - Rows deleted from a table before this count runs are invisible to 

205 BOTH sides alike; a divergence proves an omission, agreement never 

206 proves nothing was deleted upstream of the whole export. 

207 - A concurrent, unrelated long-lived transaction elsewhere in the 

208 database can pin `pg_snapshot_xmin` below the actual point this 

209 transaction's own snapshot was taken, so `_STREAM_SQL`'s explicit 

210 `transaction_id < watermark` filter can exclude an envelope whose 

211 committing transaction the REPEATABLE READ snapshot itself already 

212 considers visible -- the same streams-xmin-vs-entries-snapshot 

213 asymmetry `_export.py`'s own module docstring already names for the 

214 three unscoped kinds, but reachable here for an envelope-scoped kind 

215 too, because this count has no equivalent xmin bound (the entries 

216 tables carry no `transaction_id` column to bound it by). A divergence 

217 caused by this is real -- the rows genuinely are not in `logbooks` -- 

218 but it is not evidence of a code defect in the traversal; a fresh 

219 retry captures a fresh watermark against a fresh snapshot and, absent 

220 an actual bug, will not reproduce it. A divergence that survives a 

221 retry is the actual signal to investigate. Closing this class fully 

222 needs the `transaction_id` column already named as unbuilt in the 

223 design memo's watch items; not done here. 

224 

225 Valid ONLY because `export_record` is a whole-database export with no run 

226 or tenant filter (`_export.py`'s own module docstring): the same scoping 

227 caveat `_make_unscoped_reader` already carries for its own read. 

228 """ 

229 counts: dict[str, int] = {} 

230 for spec in all_specs(): 

231 counts[spec.kind] = await spec.count_reader(conn) 

232 return counts

235async def capture_watermark(conn: asyncpg.Connection) -> int: 

236 """The xmin bound for one consistent export snapshot. 

237 

238 Returns a plain `int`: asyncpg has no xid8 codec on either side, so 

239 (mirroring `projection/worker.py`'s bookmark parameter, the existing 

240 precedent for binding a value against an `xid8` column) the value is 

241 cast to `text` on the way out here and to `int` on the way back in, 

242 then bound as `$1::xid8` by the caller -- never compared as a string. 

243 

244 `export_record` is the sole caller: it binds this value into the 

245 stream query AND carries it on the returned `ExportedRecord`, so 

246 there is exactly one snapshot per export and one place that reads it 

247 back. 

248 """ 

249 value = await conn.fetchval(_WATERMARK_SQL) 

250 assert value is not None, "pg_snapshot_xmin(pg_current_snapshot()) returned NULL" 

251 return int(value)

254async def export_record(conn: asyncpg.Connection) -> ExportedRecord: 

255 """Walk the whole event store and its envelope-linked logbooks. 

256 

257 After the envelope-driven walk, also pulls every kind whose registry 

258 spec declares an `unscoped_reader` (`heartbeat` since S5a, 

259 `capture_probe` since S5b, `permit_probe` since S5c) with one 

260 whole-table read each, keyed onto `logbooks` the same as an 

261 envelope-driven kind. Done once, after the walk, so a kind can never 

262 be read twice even if a future spec somehow declared both a 

263 `envelope_class` and an `unscoped_reader`. 

264 

265 Raises `EmptyExportError` on zero stream rows, 

266 `cora.infrastructure.record_export.UnknownStreamTypeError` on a 

267 `stream_type` outside the declared closed set (refuses immediately; 

268 never skips), and `UnknownLogbookKindError` on an envelope's `kind` 

269 with no registry entry. 

270 """ 

271 watermark = await capture_watermark(conn) 

272 rows = await conn.fetch(_STREAM_SQL, watermark) 

273 if not rows: 

274 raise EmptyExportError( 

275 "export_record found zero events.rows below the captured " 

276 f"watermark ({watermark!r}); an empty record is never a " 

277 "valid export." 

278 ) 

279 

280 envelope_classes = registered_envelope_classes() 

281 streams: list[dict[str, object]] = [] 

282 logbooks: dict[str, list[dict[str, object]]] = {} 

283 read_seconds: dict[str, float] = {} 

284 

285 for row in rows: 

286 ensure_stream_type_known(row["stream_type"]) 

287 streams.append(render_row(row)) 

288 

289 if row["event_type"] not in envelope_classes: 

290 continue 

291 payload = row["payload"] 

292 kind = payload["kind"] 

293 logbook_id = UUID(payload["logbook_id"]) 

294 spec = resolve(kind) 

295 started = time.perf_counter() 

296 entries = await spec.reader(conn, logbook_id) 

297 read_seconds[kind] = read_seconds.get(kind, 0.0) + (time.perf_counter() - started) 

298 logbooks.setdefault(kind, []).extend(render_row(entry) for entry in entries) 

299 

300 for spec in all_specs(): 

301 if spec.unscoped_reader is None: 

302 continue 

303 started = time.perf_counter() 

304 entries = await spec.unscoped_reader(conn) 

305 read_seconds[spec.kind] = read_seconds.get(spec.kind, 0.0) + (time.perf_counter() - started) 

306 logbooks.setdefault(spec.kind, []).extend(render_row(entry) for entry in entries) 

307 

308 return ExportedRecord( 

309 streams=tuple(streams), 

310 logbooks={kind: tuple(entries) for kind, entries in logbooks.items()}, 

311 watermark=watermark, 

312 read_seconds_by_logbook_kind=read_seconds, 

313 )
