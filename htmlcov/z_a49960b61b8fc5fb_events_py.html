1"""Domain events emitted by the Run aggregate, plus the discriminated union. 

2 

3Mirrors the locked event-module shape: event classes, discriminated 

4union, `event_type_name`, `to_payload`, `from_stored`. The 

5persistence-envelope construction (`NewEvent`) lives at 

6`cora.infrastructure.event_envelope.to_new_event`. 

7 

8Lifecycle events: 

9 - `RunStarted` — genesis. Payload + lifecycle context for the 

10 other transitions to depend on. 

11 - `RunCompleted` — happy-path terminal (Running → Completed). 

12 Payload is `run_id` + `occurred_at` only; substantive run 

13 summary (frame_count, duration, final detector positions, etc.) 

14 will land when DAQ-channel integration arrives. Per the 

15 fold-cost principles, the completion event SHOULD eventually 

16 carry summary state so consumers don't have to re-fold per-step 

17 history just to ask "what happened in Run X?" — but the 

18 substantive shape only crystallizes once the observation-channel 

19 infrastructure exists to source it. 

20 - `RunAborted` — emergency-exit terminal (Running | Held → Aborted). 

21 Payload carries `run_id` + free-form `reason: str` (1-500 chars) 

22 + `occurred_at`. Reason is stored as primitive string today; 

23 future-additive structured taxonomy is documented at 

24 `InvalidRunAbortReasonError` along with its re-evaluation 

25 triggers. 

26 - `RunHeld` — pause transition (Running → Held). Payload is 

27 `run_id` + `occurred_at` only. No reason field — matches PackML 

28 / Bluesky precedent (Hold / pause carries no domain reason in 

29 either standard). Holds are routine (alignment, brief beam 

30 dropout, operator break); a reason field would be friction 

31 without audit value at this layer. Future-additive on the same 

32 triggers as RunAborted's reason if vocabulary / Decision BC 

33 integration / compliance demand crystallize. 

34 - `RunResumed` — resume transition (Held → Running). Payload is 

35 `run_id` + `occurred_at` only. Resume is just permission to 

36 proceed; no reason field. 

37 - `RunStopped` — controlled-exit terminal (Running | Held → Stopped). 

38 Payload carries `run_id` + free-form `reason: str` (1-500 chars) 

39 + `occurred_at`. Mirrors RunAborted shape — controlled-but-early 

40 terminal exit deserves audit explanation. Distinct from Aborted: 

41 Stopped data is valid up to the stop point; Aborted data is 

42 flagged as potentially invalid (PackML + Bluesky semantic 

43 distinction at the lifecycle layer; observation-channel cleanup 

44 semantics materialize on the observation channel). 

45 

46Hold ⇄ Resume is a bidirectional cycle with unlimited repeats; the 

47event stream may interleave [RunStarted, RunHeld, RunResumed, RunHeld, 

48RunResumed, RunCompleted] with arbitrary cycle counts. The fold 

49preserves only the latest status; per-cycle audit lives in the 

50event stream itself. 

51 

52`RunTruncated` — cleanup terminal (Running | Held → Truncated). 

53Payload carries `run_id` + free-form `reason: str` (1-500 chars) 

54+ optional `interrupted_at: datetime | None` (operator's best guess 

55at when the actual interruption occurred, separate from 

56`occurred_at` which is when truncation was processed) + `occurred_at`. 

57Mirrors RunStopped's reason shape; adds the interrupted_at field 

58because Truncated is uniquely retroactive among the terminals (the 

59Run was already de-facto over before the operator could mark it). 

60 

61Observation events (per-frame triggers, motor positions) do NOT live 

62on the main Run stream; they're observation-channel territory. When 

63the Run aggregate adds logbooks, the truncate_run decider extends to 

64emit RunLogbookClosed events for each open logbook before 

65RunTruncated (gate-review L4). 

66 

67## Payload conventions 

68 

69`plan_id` and `subject_id` carry as primitive UUIDs (or null for 

70subject_id in calibration runs). Eventual-consistency stance: 

71neither is verified at the persistence layer — handler pre-loads 

72both before reaching the decider. 

73 

74Status is NOT carried in event payloads — the event type itself 

75encodes the state change. Same precedent as PracticeDefined / 

76MethodDefined / FamilyDefined / SubjectMounted / 

77ActorDeactivated / PlanDefined. 

78"""

549def to_payload(event: RunEvent) -> dict[str, Any]: 

550 """Serialize a Run event to a JSON-friendly dict for jsonb storage. 

551 

552 Primitives only: UUIDs become strings, datetimes become ISO-8601 

553 strings. `subject_id` serializes as null when None (dark-field / 

554 calibration runs). 

555 """ 

556 match event: 

557 case RunStarted( 

558 run_id=run_id, 

559 name=name, 

560 plan_id=plan_id, 

561 subject_id=subject_id, 

562 raid=raid, 

563 override_parameters=override_parameters, 

564 effective_parameters=effective_parameters, 

565 trigger_source=trigger_source, 

566 external_refs=external_refs, 

567 acknowledged_cautions=acknowledged_cautions, 

568 campaign_id=campaign_id, 

569 decided_by_decision_id=decided_by_decision_id, 

570 pinned_calibration_ids=pinned_calibration_ids, 

571 occurred_at=occurred_at, 

572 ): 

573 return { 

574 "run_id": str(run_id), 

575 "name": name,

725def from_stored(stored: StoredEvent) -> RunEvent: 

726 """Rebuild a Run event from a StoredEvent loaded from the event store. 

727 

728 Dispatches on `stored.event_type`; raises ValueError on unknown 

729 discriminators so a stream contaminated with foreign event types 

730 fails loud rather than silently being dropped by the evolver. 

731 """ 

732 payload = stored.payload 

733 match stored.event_type: 

734 case "RunStarted": 

735 

736 def _build_run_started() -> RunStarted: 

737 raw_subject = payload["subject_id"] 

738 # Forward-compat additive evolution: `raid`, 

739 # `override_parameters` / `effective_parameters` / 

740 # `trigger_source`, `external_refs`, 

741 # `acknowledged_cautions`, `campaign_id`, 

742 # `decided_by_decision_id` (Decision-to-Run linkage), 

743 # `pinned_calibration_ids` (Calibration AsShot anchor) 

744 # were all added additively. Each .get(...) returns 

745 # the field's default when the key isn't in the jsonb 

746 # payload, so legacy streams replay without an upcaster. 

747 raw_campaign_id = payload.get("campaign_id") 

748 raw_decided_by = payload.get("decided_by_decision_id") 

749 return RunStarted( 

750 run_id=UUID(payload["run_id"]), 

751 name=payload["name"], 

752 plan_id=UUID(payload["plan_id"]), 

753 subject_id=UUID(raw_subject) if raw_subject is not None else None, 

754 raid=payload.get("raid"), 

755 override_parameters=payload.get("override_parameters", {}), 

756 effective_parameters=payload.get("effective_parameters", {}), 

757 trigger_source=payload.get("trigger_source"), 

758 external_refs=tuple(payload.get("external_refs", [])), 

759 acknowledged_cautions=tuple( 

760 CautionAcknowledgement( 

761 caution_id=UUID(ack["caution_id"]), 

762 target_kind=ack["target_kind"], 

763 target_id=UUID(ack["target_id"]), 

764 category=ack["category"], 

765 severity=ack["severity"], 

766 text_excerpt=ack["text_excerpt"], 

767 workaround_excerpt=ack["workaround_excerpt"], 

768 ) 

769 for ack in payload.get("acknowledged_cautions", []) 

770 ), 

771 campaign_id=UUID(raw_campaign_id) if raw_campaign_id is not None else None, 

772 decided_by_decision_id=UUID(raw_decided_by) 

773 if raw_decided_by is not None 

774 else None, 

775 pinned_calibration_ids=tuple( 

776 UUID(p) for p in payload.get("pinned_calibration_ids", []) 

777 ), 

778 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

779 ) 

780 

781 return deserialize_or_raise("RunStarted", _build_run_started) 

782 case "RunHeld": 

783 return deserialize_or_raise( 

784 "RunHeld", 

785 lambda: RunHeld( 

786 run_id=UUID(payload["run_id"]), 

787 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

788 ), 

789 ) 

790 case "RunResumed": 

791 return deserialize_or_raise( 

792 "RunResumed", 

793 lambda: RunResumed( 

794 run_id=UUID(payload["run_id"]), 

795 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

796 ), 

797 ) 

798 case "RunCompleted": 

799 return deserialize_or_raise( 

800 "RunCompleted", 

801 lambda: RunCompleted( 

802 run_id=UUID(payload["run_id"]), 

803 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

804 ), 

805 ) 

806 case "RunAborted": 

807 

808 def _build_run_aborted() -> RunAborted: 

809 # `decided_by_decision_id` optional. Forward-compat 

810 # additive evolution: pre-existing streams replay without the 

811 # key via `.get(..., None)`. 

812 raw_decided_by_abort = payload.get("decided_by_decision_id") 

813 return RunAborted( 

814 run_id=UUID(payload["run_id"]), 

815 reason=payload["reason"], 

816 decided_by_decision_id=( 

817 UUID(raw_decided_by_abort) if raw_decided_by_abort is not None else None 

818 ), 

819 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

820 ) 

821 

822 return deserialize_or_raise("RunAborted", _build_run_aborted) 

823 case "RunStopped": 

824 return deserialize_or_raise( 

825 "RunStopped", 

826 lambda: RunStopped( 

827 run_id=UUID(payload["run_id"]), 

828 reason=payload["reason"], 

829 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

830 ), 

831 ) 

832 case "RunTruncated": 

833 

834 def _build_run_truncated() -> RunTruncated: 

835 raw_interrupted_at = payload["interrupted_at"] 

836 return RunTruncated( 

837 run_id=UUID(payload["run_id"]), 

838 reason=payload["reason"], 

839 interrupted_at=( 

840 datetime.fromisoformat(raw_interrupted_at) 

841 if raw_interrupted_at is not None 

842 else None 

843 ), 

844 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

845 ) 

846 

847 return deserialize_or_raise("RunTruncated", _build_run_truncated) 

848 case "RunAdjusted": 

849 

850 def _build_run_adjusted() -> RunAdjusted: 

851 # `decided_by_decision_id` is optional on the 

852 # event payload. Forward-compat additive: synthetic / future 

853 # callers omitting the key (None semantically) deserialize 

854 # as decided_by_decision_id=None. `parameters_patch` and 

855 # `effective_parameters` are always carried (never optional). 

856 # `adjusted_by` is REQUIRED per the fold-symmetry rule 

857 # (pre-pilot rename; legacy streams pre-date the field but 

858 # dev DBs are wiped on migration run). 

859 raw_decision_id = payload.get("decided_by_decision_id") 

860 return RunAdjusted( 

861 run_id=UUID(payload["run_id"]), 

862 parameters_patch=payload["parameters_patch"], 

863 effective_parameters=payload["effective_parameters"], 

864 reason=payload["reason"], 

865 adjusted_by=ActorId(UUID(payload["adjusted_by"])), 

866 decided_by_decision_id=( 

867 UUID(raw_decision_id) if raw_decision_id is not None else None 

868 ), 

869 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

870 ) 

871 

872 return deserialize_or_raise("RunAdjusted", _build_run_adjusted) 

873 case "RunObservationLogbookOpened": 

874 return deserialize_or_raise( 

875 "RunObservationLogbookOpened", 

876 lambda: RunObservationLogbookOpened( 

877 run_id=UUID(payload["run_id"]), 

878 logbook_id=UUID(payload["logbook_id"]), 

879 kind=payload["kind"], 

880 schema=LogbookSchema.from_dict(payload["schema"]), 

881 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

882 ), 

883 ) 

884 case "RunAddedToCampaign": 

885 return deserialize_or_raise( 

886 "RunAddedToCampaign", 

887 lambda: RunAddedToCampaign( 

888 run_id=UUID(payload["run_id"]), 

889 campaign_id=UUID(payload["campaign_id"]), 

890 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

891 ), 

892 ) 

893 case "RunRemovedFromCampaign": 

894 return deserialize_or_raise( 

895 "RunRemovedFromCampaign", 

896 lambda: RunRemovedFromCampaign( 

897 run_id=UUID(payload["run_id"]), 

898 campaign_id=UUID(payload["campaign_id"]), 

899 reason=payload["reason"], 

900 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

901 ), 

902 ) 

903 case "DecisionDebriefRequested": 

904 return deserialize_or_raise( 

905 "DecisionDebriefRequested", 

906 lambda: DecisionDebriefRequested( 

907 run_id=UUID(payload["run_id"]), 

908 debriefer_agent_id=UUID(payload["debriefer_agent_id"]), 

909 terminal_event_id=UUID(payload["terminal_event_id"]), 

910 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

911 ), 

912 ) 

913 case _: 

914 msg = f"Unknown RunEvent event_type: {stored.event_type!r}" 

915 raise ValueError(msg)
