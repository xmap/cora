1"""Domain events emitted by the Attestation aggregate plus the discriminated union. 

2 

3Mirrors the locked event-module shape: event classes, discriminated union, 

4``event_type_name``, ``to_payload``, ``from_stored``. The persistence-envelope 

5construction (``NewEvent``) lives at 

6``cora.infrastructure.event_envelope.to_new_event``. 

7 

8## This module ships ONE event today 

9 

10 - ``AttestationRecorded`` (genesis-and-terminal): a recorded fact 

11 about a Dataset (and optionally a Distribution byte-copy). Status 

12 is implicit (``Recorded``); the evolver sets it. 

13 

14## Future events NAMED but not locked here 

15 

16 - ``AttestationSuperseded`` (Recorded -> Superseded; future W3 if 

17 operators demand a corrective event). 

18 

19## Payload conventions 

20 

21- UUIDs serialize as strings; ``distribution_id`` may serialize as 

22 ``None`` (kind ``ConformsToValidated``). 

23- ``kind`` and ``outcome`` serialize as the StrEnum value bare-string. 

24- ``evidence`` serializes as a nested object whose shape is 

25 discriminated by the sibling ``kind`` field (NOT by a duplicated 

26 discriminator inside the sub-object). Today only the 

27 ``ChecksumVerified`` shape is concrete: ``{"algorithm": str, 

28 "value": str | None, "verifier_supply_id": str, "verifier_kind": 

29 str, "error_detail": str | None}``. 

30- ``occurred_at`` is the payload-time key per CORA convention; the 

31 in-memory state field is ``attested_at`` (mirrors Calibration 

32 ``occurred_at`` payload to ``established_at`` state precedent). 

33- Status is NOT carried in the payload; the event type encodes it 

34 (``AttestationRecorded`` -> ``RECORDED``). 

35"""

55@dataclass(frozen=True) 

56class AttestationRecorded: 

57 """A new Attestation fact was recorded. 

58 

59 Status is implicit (``Recorded``); the evolver sets it. The 

60 cross-aggregate references (``dataset_id`` always; ``distribution_id`` 

61 when kind is byte-level) are eventual-consistency primitives; the 

62 handler pre-loads each on the write path, the evolver does NOT 

63 re-verify on fold. 

64 

65 Per CONTRIBUTING.md "Primitives in event payloads": every field 

66 here is a primitive (str, int, UUID, datetime, dict-of-primitives). 

67 The discriminated VO is reconstructed by the evolver on fold; the 

68 decider unwraps it before constructing the event. 

69 

70 Fold-symmetry attribution per [[project_fold_symmetry_design]]: 

71 ``attested_by: ActorId`` carries the envelope ``principal_id`` of 

72 the record-slice caller (``SYSTEM_PRINCIPAL_ID`` for verifier-port 

73 driven attestations; operator ``ActorId`` for operator-initiated 

74 re-verification). 

75 """ 

76 

77 attestation_id: UUID 

78 dataset_id: UUID 

79 distribution_id: UUID | None 

80 kind: str 

81 outcome: str 

82 evidence: dict[str, Any] 

83 occurred_at: datetime 

84 attested_by: ActorId 

85

87#: Discriminated union over Attestation events. Has one arm today; 

88#: future slices extend additively (W3 ``AttestationSuperseded`` only 

89#: if operator demand fires). 

90AttestationEvent = AttestationRecorded

98def to_payload(event: AttestationEvent) -> dict[str, Any]: 

99 """Serialize an Attestation event to a JSON-friendly dict for jsonb storage.""" 

100 match event: 

101 case AttestationRecorded( 

102 attestation_id=attestation_id, 

103 dataset_id=dataset_id, 

104 distribution_id=distribution_id, 

105 kind=kind, 

106 outcome=outcome, 

107 evidence=evidence, 

108 occurred_at=occurred_at, 

109 attested_by=attested_by, 

110 ): 

111 return { 

112 "attestation_id": str(attestation_id), 

113 "dataset_id": str(dataset_id), 

114 "distribution_id": (str(distribution_id) if distribution_id is not None else None), 

115 "kind": kind, 

116 "outcome": outcome, 

117 "evidence": dict(evidence), 

118 "occurred_at": occurred_at.isoformat(), 

119 "attested_by": str(attested_by), 

120 } 

121 case _: # pragma: no cover # exhaustiveness guard for future arms 

122 assert_never(event)

125def from_stored(stored: StoredEvent) -> AttestationEvent: 

126 """Rebuild an Attestation event from a ``StoredEvent`` loaded from the event store. 

127 

128 Dispatches on ``stored.event_type``; raises ``ValueError`` on 

129 unknown discriminators so a stream contaminated with foreign event 

130 types fails loud rather than silently being dropped by the 

131 evolver. Each per-event builder is wrapped in 

132 ``deserialize_or_raise`` to surface malformed payloads as 

133 ``MalformedAttestationRecorded`` per 

134 [[project_from_stored_wrap_convention]]. 

135 """ 

136 payload = stored.payload 

137 match stored.event_type: 

138 case "AttestationRecorded": 

139 

140 def _build_recorded() -> AttestationRecorded: 

141 raw_distribution = payload["distribution_id"] 

142 kind_str = payload["kind"] 

143 outcome_str = payload["outcome"] 

144 # Defensive enum-membership re-checks (the wire payload is 

145 # bare-str; legacy or hand-crafted payloads with stale 

146 # values must fail loud at fold rather than poison state). 

147 try: 

148 AttestationKind(kind_str) 

149 except ValueError as exc: 

150 raise InvalidAttestationKindError(kind_str) from exc 

151 try: 

152 AttestationOutcome(outcome_str) 

153 except ValueError as exc: 

154 raise InvalidAttestationOutcomeError(outcome_str) from exc 

155 return AttestationRecorded( 

156 attestation_id=UUID(payload["attestation_id"]), 

157 dataset_id=UUID(payload["dataset_id"]), 

158 distribution_id=( 

159 UUID(raw_distribution) if raw_distribution is not None else None 

160 ), 

161 kind=kind_str, 

162 outcome=outcome_str, 

163 evidence=dict(payload["evidence"]), 

164 occurred_at=datetime.fromisoformat(payload["occurred_at"]), 

165 attested_by=ActorId(UUID(payload["attested_by"])), 

166 ) 

167 

168 return deserialize_or_raise( 

169 "AttestationRecorded", 

170 _build_recorded, 

171 extra=( 

172 ValueError, 

173 InvalidAttestationKindError, 

174 InvalidAttestationOutcomeError, 

175 InvalidAttestationEvidenceError, 

176 ), 

177 ) 

178 case _: 

179 msg = f"Unknown AttestationEvent event_type: {stored.event_type!r}" 

180 raise ValueError(msg)

183def build_checksum_verified_evidence_payload( 

184 evidence: ChecksumVerifiedEvidence, 

185) -> dict[str, Any]: 

186 """Serialize a ChecksumVerifiedEvidence VO into its on-disk shape. 

187 

188 On disk the evidence carries a single ``value`` keyed by the 

189 sibling ``outcome`` discriminator: ``computed_checksum`` on 

190 Match/Mismatch, ``None`` on Unreachable. ``expected_checksum`` is 

191 NOT serialized (the canonical value lives on the Distribution 

192 row; storing it on the Attestation would create drift potential 

193 across re-checks). 

194 """ 

195 payload: dict[str, Any] = { 

196 "algorithm": evidence.algorithm, 

197 "value": evidence.computed_checksum, 

198 "verifier_supply_id": str(evidence.verifier_supply_id), 

199 "verifier_kind": evidence.verifier_kind, 

200 } 

201 if evidence.error_detail is not None: 

202 payload["error_detail"] = evidence.error_detail 

203 return payload
