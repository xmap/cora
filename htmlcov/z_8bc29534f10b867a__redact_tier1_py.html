1"""Redact `events` rows: fixed columns plus disposition-table-driven payload. 

2 

3Per `project_record_export_build_brief.md` step 6 and 

4`project_record_export_v3.md` F5. 

5 

6Two DIFFERENT failure modes, easy to conflate (the brief's own step-6 

7notes and its terser Rejections list read ambiguously against each 

8other on this point; this resolves it the way that keeps schema 

9evolution possible): 

10 

11- `event_type` absent from `_dispositions.DISPOSITIONS` entirely -> 

12 ABORT (`UnknownEventTypeError`). Step 0's generator is exhaustive 

13 over every currently-declared event class; a stream carrying a type 

14 the table has never heard of means the table is stale relative to 

15 the code, which is a build problem, not a per-row one. 

16- A payload KEY present on a row but absent from 

17 `DISPOSITIONS[event_type]`'s own field list -> DROP (omit the key). 

18 This is what makes schema evolution survivable: an OLDER 

19 `schema_version`'s row can carry a field the CURRENT dataclass no 

20 longer declares (removed in a later version), and dropping it rather 

21 than aborting the whole export is the graceful path. This is also 

22 exactly what "a bare str on a NEW event drops with nobody editing a 

23 list" is really about at the OTHER end: Step 0's generator 

24 auto-classifies every new bare `str` field as `drop:text` the moment 

25 it is generated, so the field already has a rule (drop) with no 

26 manual list-editing required; it is not the "missing key" case. 

27"""

56class UnknownEventTypeError(LookupError): 

57 """`event_type` has no entry in `_dispositions.DISPOSITIONS` at all. 

58 

59 Refuses loudly: the disposition table is exhaustive over every 

60 currently-declared event class, so this means the table is stale 

61 relative to the code, not that this one row should be skipped. 

62 """ 

63 

64 def __init__(self, event_type: str) -> None: 

65 super().__init__( 

66 f"event_type {event_type!r} has no entry in DISPOSITIONS; the " 

67 "table is stale relative to the code (run `make record-dispositions`)." 

68 ) 

69 self.event_type = event_type

72def _apply_field_disposition(disposition: Any, value: Any, *, token_map: TokenMap) -> Any: 

73 if isinstance(disposition, dict): 

74 # `None` on a dict-shaped (nested-VO or fixed-tuple) disposition is 

75 # the field's own declared absence (e.g. `SafetyEnvelopeVerdict | 

76 # None = None`, `CaptureProgressSnapshot | None = None`), not a 

77 # malformed value: preserve it as an explicit null, matching the 

78 # "present-as-null, not omit-when-None" convention every optional 

79 # scalar field on these same events already gets (see 

80 # `RunCompleted.observed_at`'s `to_payload` comment). Falling 

81 # through to the OMITTED branch below would silently drop the key 

82 # for the COMMON case (a Conducted Run's `safety_envelope_verdict`, 

83 # a driven or operator-terminal's `capture_progress_snapshot`), 

84 # making "this stream predates the field" and "this stream has the 

85 # field, unset" indistinguishable in the exported record. 

86 if value is None: 

87 return None 

88 if "[]" in disposition: 88 ↛ 90line 88 didn't jump to line 90 because the condition on line 88 was never true

89 # Fixed-length heterogeneous tuple: one disposition per position. 

90 per_position = disposition["[]"] 

91 if not isinstance(value, (list, tuple)): 

92 return OMITTED 

93 return [ 

94 _apply_field_disposition(pos_disposition, item, token_map=token_map) 

95 for pos_disposition, item in zip(per_position, value, strict=True) 

96 ] 

97 # A recursed value object: apply this same per-key logic one level down. 

98 if not isinstance(value, dict): 

99 return OMITTED 

100 result: dict[str, Any] = {} 

101 for key, sub_disposition in disposition.items(): 

102 if key not in value: 102 ↛ 103line 102 didn't jump to line 103 because the condition on line 102 was never true

103 continue 

104 redacted = _apply_field_disposition(sub_disposition, value[key], token_map=token_map) 

105 if redacted is not OMITTED: 

106 result[key] = redacted 

107 return result 

108 

109 if disposition == "by-value": 109 ↛ 110line 109 didn't jump to line 110 because the condition on line 109 was never true

110 return apply_leaf_rule(value, token_map=token_map) 

111 if disposition.startswith("keep:"): 

112 return value 

113 if disposition == "token:uuid": 

114 # A UUID-collection field (e.g. `target_asset_ids`) carries 

115 # `token:uuid` too, per Step 0's census: the disposition names 

116 # the ELEMENT type, not the field's own cardinality. 

117 if isinstance(value, (list, tuple)): 

118 return [token_map.token_uuid(item) for item in value] 

119 return token_map.token_uuid(value) 

120 if disposition.startswith("drop:"): 120 ↛ 122line 120 didn't jump to line 122 because the condition on line 120 was always true

121 return OMITTED 

122 return OMITTED

125def redact_tier1_payload( 

126 event_type: str, 

127 payload: dict[str, Any], 

128 *, 

129 token_map: TokenMap, 

130 fired_fields: dict[str, set[str]] | None = None, 

131) -> dict[str, Any]: 

132 """Redact one event's `payload`, iterating the STORED payload's own 

133 keys (never the disposition table's), per F5's fail-closed property. 

134 

135 `fired_fields`, when given, records which of `DISPOSITIONS[event_type]`'s 

136 DECLARED field keys were actually present on this row -- the tier-1 

137 completeness twin to tier-2's `fired_pointers`. A field can be 

138 declared but never fire within one export's event types if every row 

139 of that type in this export happens to come from an older 

140 `schema_version` that predates the field; `redact_record` uses this 

141 to report the fact on the manifest, mirroring 

142 `Manifest.unfired_tier2_clearances`. 

143 """ 

144 if event_type not in DISPOSITIONS: 

145 raise UnknownEventTypeError(event_type) 

146 field_dispositions = DISPOSITIONS[event_type] 

147 

148 result: dict[str, Any] = {} 

149 for key, value in payload.items(): 

150 disposition = field_dispositions.get(key) 

151 if disposition is None: 

152 continue # known event type, unlisted field: schema-evolution DROP 

153 if fired_fields is not None: 

154 fired_fields.setdefault(event_type, set()).add(key) 

155 redacted = _apply_field_disposition(disposition, value, token_map=token_map) 

156 if redacted is not OMITTED: 

157 result[key] = redacted 

158 return result

161class Tier1Redactor: 

162 """Stateful per-export redactor for `events` rows. 

163 

164 Must be fed rows in the SAME order `export_record` produced them 

165 (`(transaction_id, position)`): `position`/`transaction_id` 

166 re-indexing depends on that order, and `version` re-indexing depends 

167 on rows for one `stream_id` arriving in stream order (already true 

168 of the exporter's global order). 

169 """ 

170 

171 def __init__(self, token_map: TokenMap) -> None: 

172 self._token_map = token_map 

173 self._next_position = 1 

174 self._next_version_by_stream: dict[str, int] = {} 

175 self._transaction_id_index: dict[str, int] = {} 

176 self._next_transaction_id = 1 

177 self._fired_fields: dict[str, set[str]] = {} 

178 

179 @property 

180 def fired_fields(self) -> dict[str, frozenset[str]]: 

181 """Per event type redacted so far, the declared field keys that 

182 actually appeared on at least one row. A copy; callers cannot 

183 mutate the accumulator this instance still writes to.""" 

184 return {event_type: frozenset(keys) for event_type, keys in self._fired_fields.items()} 

185 

186 def _dense_version(self, raw_stream_id: str) -> int: 

187 version = self._next_version_by_stream.get(raw_stream_id, 1) 

188 self._next_version_by_stream[raw_stream_id] = version + 1 

189 return version 

190 

191 def _dense_transaction_id(self, raw_transaction_id: object) -> int: 

192 key = str(raw_transaction_id) 

193 if key not in self._transaction_id_index: 

194 self._transaction_id_index[key] = self._next_transaction_id 

195 self._next_transaction_id += 1 

196 return self._transaction_id_index[key] 

197 

198 def redact_row(self, row: dict[str, Any]) -> dict[str, Any]: 

199 raw_stream_id = row["stream_id"] 

200 redacted: dict[str, Any] = { 

201 "position": self._next_position, 

202 "version": self._dense_version(raw_stream_id), 

203 "transaction_id": self._dense_transaction_id(row["transaction_id"]), 

204 } 

205 self._next_position += 1 

206 

207 for column in FIXED_KEEP_COLUMNS: 

208 redacted[column] = row[column] 

209 for column in FIXED_TOKEN_COLUMNS: 

210 redacted[column] = self._token_map.token_uuid(row[column]) 

211 # FIXED_DROP_COLUMNS (metadata, signature*) intentionally absent. 

212 

213 redacted["payload"] = redact_tier1_payload( 

214 row["event_type"], 

215 row["payload"], 

216 token_map=self._token_map, 

217 fired_fields=self._fired_fields, 

218 ) 

219 return redacted
