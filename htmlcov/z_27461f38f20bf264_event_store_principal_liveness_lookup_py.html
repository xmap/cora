1"""EventStorePrincipalLivenessLookup: resolve liveness from the Actor stream. 

2 

3Reads through `load_actor`, the same fold every seeded agent already 

4uses to check its own operator-deactivation switch 

5(`agent/subscribers/_ratification_shared.py`). Using the identical 

6surface for the gate is the point: one fact, one reader, so a human and 

7an agent cannot drift apart on it. 

8 

9## Why the event store and not `proj_access_actor_summary` 

10 

11The projection carries a `status` column and would be one indexed row 

12instead of a fold, which is cheaper. It is refused anyway, because a 

13projection lags: a principal deactivated a moment ago would keep 

14reading `active` for as long as the worker is behind, and the whole 

15value of this lookup is that turning someone off takes effect. A 

16revocation surface that is eventually consistent is a revocation 

17surface with a window in it. 

18 

19The cost is bounded in a way it would not be for a larger aggregate: 

20an Actor stream is `ActorRegistered` plus at most a handful of 

21deactivate / reactivate / forget events, so the fold is a short read of 

22a stream that is small by construction. 

23""" 

24 

25from uuid import UUID 

26 

27from cora.access.aggregates.actor import load_actor 

28from cora.infrastructure.ports.event_store import EventStore 

29from cora.infrastructure.ports.principal_liveness_lookup import PrincipalLiveness 

30 

31 

32class EventStorePrincipalLivenessLookup: 

33 """Resolve `PrincipalLiveness` by folding the principal's Actor stream.""" 

34 

35 def __init__(self, event_store: EventStore) -> None: 

36 self._event_store = event_store 

37 

38 async def liveness_of(self, principal_id: UUID) -> PrincipalLiveness: 

39 """Fold the Actor stream for `principal_id` into a liveness value. 

40 

41 A principal id IS an Actor id: `register_actor` mints the Actor 

42 whose id humans authenticate as, and `define_agent` writes the 

43 agent's Actor at the same id in one transaction. So no mapping 

44 step is needed here, and a missing stream means the principal 

45 was never registered rather than that the lookup failed. 

46 """ 

47 actor = await load_actor(self._event_store, principal_id) 

48 if actor is None: 

49 return PrincipalLiveness.UNREGISTERED 

50 return PrincipalLiveness.ACTIVE if actor.active else PrincipalLiveness.DEACTIVATED 

51 

52 

53__all__ = ["EventStorePrincipalLivenessLookup"]
