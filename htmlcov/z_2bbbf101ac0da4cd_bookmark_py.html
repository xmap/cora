1"""Bookmark read/write SQL helpers. 

2 

3The worker's advance loop calls these inside its transaction: 

4 

5 1. `read_bookmark(conn, name)` -> `(last_transaction_id, last_position)` 

6 locks the bookmark row FOR UPDATE so concurrent workers (when 

7 deferred multi-worker arrives) won't double-advance. 

8 2. `write_bookmark(conn, name, ...)` updates the bookmark and 

9 resets the failure-tracking columns; commits with the same 

10 transaction as the projection writes for atomic at-least-once 

11 delivery. 

12 3. `write_bookmark_failure(pool, name, error_message)` updates ONLY 

13 the failure columns in a SEPARATE small transaction (NOT the 

14 advance-loop transaction, which is rolling back). The bookmark 

15 position itself stays put so retry semantics are preserved. 

16 

17`projection_bookmarks` carries four observability columns 

18(`last_event_recorded_at`, `last_error_at`, `last_error_message`, 

19`consecutive_failures`). Success path resets failure columns to NULL 

20+ counter to 0 atomically with the position advance; failure path 

21increments the counter and records the error message in its own 

22small transaction. Pre-positions for the full read-side observability 

23surface (admin endpoint + lag sampler + OTel) which stays deferred 

24until the trigger fires. 

25 

26Bookmark rows are created by per-projection migrations (`INSERT INTO 

27projection_bookmarks (name) VALUES (...) ON CONFLICT DO NOTHING`), 

28not by the worker — registering a projection without its migration 

29having landed will fail loudly at first advance, which is the 

30behavior we want. 

31""" 

32 

33# pyright: reportUnknownMemberType=false, reportUnknownVariableType=false, reportUnknownArgumentType=false

35from datetime import datetime 

36 

37import asyncpg 

38 

39from cora.infrastructure.projection.handler import ConnectionLike

41# Bound the bookmark UPDATE size. Operators see error class + first 

42# part of message; full traceback goes to OTel spans when the full 

43# observability surface ships. 

44_ERROR_MESSAGE_MAX_CHARS = 500 

45 

46_READ_BOOKMARK_SQL = """ 

47SELECT last_transaction_id::text AS last_tx, last_position 

48FROM projection_bookmarks 

49WHERE name = $1 

50FOR UPDATE 

51""" 

52# `last_transaction_id::text` for the same reason as `events`: asyncpg 

53# 0.31 has no built-in xid8 OUTPUT codec, so we cast and parse to int. 

54 

55_WRITE_BOOKMARK_SQL = """ 

56UPDATE projection_bookmarks 

57SET last_transaction_id = $2::xid8, 

58 last_position = $3, 

59 last_event_recorded_at = $4, 

60 last_error_at = NULL, 

61 last_error_message = NULL, 

62 consecutive_failures = 0, 

63 updated_at = now() 

64WHERE name = $1 

65""" 

66 

67_WRITE_BOOKMARK_FAILURE_SQL = """ 

68UPDATE projection_bookmarks 

69SET last_error_at = now(), 

70 last_error_message = $2, 

71 consecutive_failures = consecutive_failures + 1, 

72 updated_at = now() 

73WHERE name = $1 

74"""

77class MissingBookmarkError(Exception): 

78 """No bookmark row exists for the given projection name. Indicates 

79 the projection's migration didn't land (or its `INSERT INTO 

80 projection_bookmarks` statement was missed).""" 

81 

82 def __init__(self, name: str) -> None: 

83 super().__init__( 

84 f"No projection_bookmarks row for {name!r}. The projection's " 

85 "migration must include an `INSERT INTO projection_bookmarks " 

86 "(name) VALUES (...) ON CONFLICT DO NOTHING` statement." 

87 ) 

88 self.name = name

91async def read_bookmark( 

92 conn: ConnectionLike, 

93 name: str, 

94) -> tuple[int, int]: 

95 """Read `(last_transaction_id, last_position)` for the projection. 

96 

97 Locks the row FOR UPDATE; caller is responsible for the surrounding 

98 transaction. Raises `MissingBookmarkError` if the row doesn't exist. 

99 """ 

100 row = await conn.fetchrow(_READ_BOOKMARK_SQL, name) 

101 if row is None: 101 ↛ 102line 101 didn't jump to line 102 because the condition on line 101 was never true

102 raise MissingBookmarkError(name) 

103 return int(row["last_tx"]), int(row["last_position"])

106async def write_bookmark( 

107 conn: ConnectionLike, 

108 name: str, 

109 *, 

110 last_transaction_id: int, 

111 last_position: int, 

112 last_event_recorded_at: datetime, 

113) -> None: 

114 """Update the bookmark to the new (last_tx, last_pos) cursor and 

115 reset failure-tracking columns. 

116 

117 Called inside the advance-loop transaction; the position advance 

118 + failure-reset commit together with the projection writes for 

119 atomic at-least-once delivery + observability consistency. 

120 """ 

121 await conn.execute( 

122 _WRITE_BOOKMARK_SQL, 

123 name, 

124 last_transaction_id, 

125 last_position, 

126 last_event_recorded_at, 

127 ) 

128 

129 

130async def write_bookmark_failure( 

131 pool: asyncpg.Pool, 

132 name: str, 

133 *, 

134 error_message: str, 

135) -> None: 

136 """Record an advance-loop failure in a SEPARATE small transaction. 

137 

138 Observability hook called from the advance-loop exception 

139 handler AFTER the failed batch has rolled back. Increments 

140 `consecutive_failures` and records `last_error_at` + 

141 `last_error_message`. Does NOT touch `last_transaction_id` or 

142 `last_position`, so retry semantics are preserved by leaving 

143 the cursor where it was. 

144 

145 Truncates `error_message` to 500 chars to bound bookmark UPDATE 

146 size. If this UPDATE itself fails, the worker swallows that loss 

147 silently: we already failed once; the operator's signal is the 

148 log + (eventually) OTel span on the original failure. 

149 """ 

150 truncated = error_message[:_ERROR_MESSAGE_MAX_CHARS] 

151 async with pool.acquire() as conn: 

152 await conn.execute(_WRITE_BOOKMARK_FAILURE_SQL, name, truncated) 

153
