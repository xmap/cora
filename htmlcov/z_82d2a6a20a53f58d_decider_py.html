1"""Pure decider for the `BindPlanRole` command. 

2 

3Mirrors the cross-aggregate-read pattern from `add_plan_wire`. The 

4context object is constructed by the handler from `load_method` + 

5`load_asset` (plus, for the 3D role_kind path, `RoleLookup.lookup` 

6and a `FamilyLookup.lookup` batch over `asset.family_ids`); tests 

7pass it directly to bypass the I/O. Fail-fast order is the same as 

8the Invariants list below. 

9 

10## Bifurcation (Layer 3 sub-slice 3D) 

11 

12`RoleRequirement` carries either `role_kind` (federation-portable; 

13Layer 3) OR `family_id` (anatomical escape hatch; slice-1) per the 

14XOR invariant enforced by the VO. The satisfaction check splits: 

15 

16 - family_id path: existing slice-1 check 

17 `role.family_id in asset.family_ids` -> PlanRoleFamilyMismatchError 

18 - role_kind path: ANY-single-family disjunction per Lock 17, OR-ed 

19 with the composed-Assembly branch. For each family_id in 

20 `asset.family_ids` the handler must have loaded a 

21 `FamilyLookupResult`; the decider walks the dict and accepts 

22 iff AT LEAST ONE Family both (a) declares `role_kind` in its 

23 `presents_as` AND (b) has `affordances` superset 

24 `Role.required_affordances`. None-on-lookup raises 

25 `PlanRoleFamilyNotResolvableError`. If no Family satisfies but 

26 the Asset carries `fixture_id` AND the handler-loaded 

27 `AssemblyLookupResult.presents_as` contains `role_kind`, the 

28 Assembly satisfaction path accepts the binding (Assembly 

29 affordance-superset is enforced at register_fixture time, not 

30 at template time per 3C state docstring). If neither path 

31 succeeds, `PlanRoleAssetCannotPresentError` fires. 

32 

33Invariants: 

34 - State must not be None -> PlanNotFoundError 

35 - Plan must be in Defined status -> PlanCannotMutateRoleBindingsError 

36 - command.asset_id must be in state.asset_ids -> 

37 PlanRoleAssetNotBoundError 

38 - role_name must not already be in state.role_bindings (strict-not- 

39 idempotent) -> PlanRoleAlreadyBoundError 

40 - Method must be loaded via state.method_id -> MethodNotFoundError 

41 - role_name must match a RoleRequirement on method.required_roles 

42 -> PlanRoleNameNotDeclaredError 

43 - Asset must be loaded -> AssetNotFoundError 

44 - Family-path satisfaction OR role_kind-path satisfaction (see above) 

45 - Port coverage + wire-graph consistency (unchanged from slice-2) 

46"""

71def decide( 

72 state: Plan | None, 

73 command: BindPlanRole, 

74 *, 

75 context: BindPlanRoleContext, 

76 now: datetime, 

77) -> list[PlanRoleBound]: 

78 """Decide the events produced by binding a role to an Asset. 

79 

80 Invariants: 

81 - State must not be None -> PlanNotFoundError 

82 - Plan status must be Defined -> PlanCannotMutateRoleBindingsError 

83 - asset_id must be in state.asset_ids -> PlanRoleAssetNotBoundError 

84 - role_name must not already be bound (strict-not-idempotent) -> 

85 PlanRoleAlreadyBoundError 

86 - Method must be loaded -> MethodNotFoundError 

87 - role_name must match a RoleRequirement on method.required_roles 

88 -> PlanRoleNameNotDeclaredError 

89 - Asset must be loaded -> AssetNotFoundError 

90 - Family-path (slice-1): asset.family_ids must include 

91 role.family_id -> PlanRoleFamilyMismatchError 

92 - Role-kind path (3D): every family_id in asset.family_ids 

93 must resolve via FamilyLookup -> 

94 PlanRoleFamilyNotResolvableError; at least one of those 

95 Families must declare role_kind in presents_as AND have 

96 affordances superset role.required_affordances (Lock 17 

97 ANY-single-family disjunction), OR (if the Asset is in a 

98 Fixture) the loaded Assembly's presents_as must contain 

99 role_kind -> 

100 PlanRoleAssetCannotPresentError 

101 - Asset.ports must cover every role.required_ports triple 

102 exactly -> PlanRolePortCoverageNotSatisfiedError 

103 - No existing wire's endpoint port may claim a role's 

104 required_port at an Asset other than the candidate bound 

105 Asset -> PlanWireRoleEndpointMismatchError 

106 """ 

107 if state is None: 

108 raise PlanNotFoundError(command.plan_id) 

109 

110 if state.status is not PlanStatus.DEFINED: 

111 raise PlanCannotMutateRoleBindingsError(state.id, state.status) 

112 

113 if command.asset_id not in state.asset_ids: 

114 raise PlanRoleAssetNotBoundError(state.id, command.role_name, command.asset_id) 

115 

116 if any(b.role_name == command.role_name for b in state.role_bindings): 

117 raise PlanRoleAlreadyBoundError(state.id, command.role_name) 

118 

119 method = context.method 

120 if method is None: 

121 # Unlike `add_plan_wire`, this decider CANNOT proceed without a 

122 # Method: the role_name must be validated against 

123 # method.required_roles, and there is no fallback semantic that 

124 # could justify silently binding to an unknown role. The 

125 # asymmetry with add_plan_wire (which silently skips the role 

126 # check when method is None) is principled and documented in 

127 # the add_plan_wire decider. `state.method_id or command.plan_id` 

128 # surfaces the better sentinel: prefer the genuine method_id when 

129 # the Plan declares one, otherwise echo plan_id so the operator 

130 # sees the offending Plan. 

131 raise MethodNotFoundError(state.method_id or command.plan_id) 

132 

133 matching_role: RoleRequirement | None = None 

134 for role in method.required_roles: 

135 if role.role_name == command.role_name: 

136 matching_role = role 

137 break 

138 if matching_role is None: 

139 raise PlanRoleNameNotDeclaredError(state.id, method.id, command.role_name) 

140 

141 asset = context.asset 

142 if asset is None: 

143 raise AssetNotFoundError(command.asset_id) 

144 

145 # Bifurcated satisfaction check per memo Lock 17. 

146 if matching_role.role_kind is not None: 

147 # 3D role_kind path: ANY-single-family disjunction. Walk 

148 # asset.family_ids, lookup each in context.family_lookups, 

149 # accept iff AT LEAST ONE Family advertises role_kind in 

150 # presents_as AND has affordances superset 

151 # role.required_affordances. 

152 # Handler contract: role_lookup_result MUST be populated when 

153 # matching_role.role_kind is set. Mirror the symmetric XOR 

154 # guard at line 178 below: assert (not a domain error) so a 

155 # buggy handler / test fails loud rather than silently mis- 

156 # binding. Operators never see this; only callers that 

157 # bypass the handler can trigger it. 

158 assert context.role_lookup_result is not None, ( 

159 "handler contract violation: role_lookup_result must be populated " 

160 f"when matching_role.role_kind is set (role_kind={matching_role.role_kind})" 

161 ) 

162 role_lookup = context.role_lookup_result 

163 required_affordances = role_lookup.required_affordances 

164 satisfied_by: bool = False 

165 for family_id in asset.family_ids: 

166 family_row = context.family_lookups.get(family_id) 

167 if family_row is None: 

168 raise PlanRoleFamilyNotResolvableError( 

169 state.id, 

170 command.role_name, 

171 command.asset_id, 

172 family_id, 

173 ) 

174 if matching_role.role_kind in family_row.presents_as and ( 

175 required_affordances <= family_row.affordances 

176 ): 

177 satisfied_by = True 

178 break 

179 # Assembly satisfaction OR-branch: when no Family satisfied 

180 # and the Asset is part of a materialized Assembly (via 

181 # fixture_id), accept the binding iff the loaded Assembly's 

182 # presents_as contains role_kind. Affordance-superset is NOT 

183 # checked here -- Assembly affordances derive from the 

184 # constituent Family union at register_fixture time per 

185 # the 3C state.py docstring, so the check belongs at the 

186 # fixture-registration layer, not the bind-time layer. 

187 if not satisfied_by and asset.fixture_id is not None: 

188 assembly_row = context.assembly_lookup_result 

189 if assembly_row is not None and matching_role.role_kind in assembly_row.presents_as: 

190 satisfied_by = True 

191 if not satisfied_by: 

192 raise PlanRoleAssetCannotPresentError( 

193 state.id, 

194 command.role_name, 

195 command.asset_id, 

196 matching_role.role_kind, 

197 asset.family_ids, 

198 ) 

199 else: 

200 # Slice-1 family_id path: anatomical escape hatch unchanged. 

201 # XOR invariant guarantees family_id is non-None here. 

202 assert matching_role.family_id is not None, ( 

203 "RoleRequirement.__post_init__ invariant violated: neither role_kind nor family_id set" 

204 ) 

205 if matching_role.family_id not in asset.family_ids: 

206 raise PlanRoleFamilyMismatchError( 

207 state.id, 

208 command.role_name, 

209 command.asset_id, 

210 matching_role.family_id, 

211 asset.family_ids, 

212 ) 

213 

214 # Port coverage: each required (port_name, direction, signal_type) 

215 # triple must match an existing Asset.port triple exactly. 

216 asset_port_triples = {(p.name, p.direction, p.signal_type) for p in asset.ports} 

217 missing: list[tuple[str, str, str]] = [] 

218 for required in matching_role.required_ports: 

219 triple = (required.port_name, required.direction, required.signal_type) 

220 if triple not in asset_port_triples: 

221 missing.append((triple[0], triple[1].value, triple[2])) 

222 if missing: 

223 raise PlanRolePortCoverageNotSatisfiedError( 

224 state.id, 

225 command.role_name, 

226 command.asset_id, 

227 missing, 

228 ) 

229 

230 # Wire-graph consistency: scan existing wires for any endpoint 

231 # that already claims this role's port (by name + side) at a 

232 # different Asset than the candidate binding. Without this scan, 

233 # the wire-then-bind temporal ordering (operator adds a wire 

234 # before binding the role) and the unbind-rebind ordering 

235 # (operator unbinds a role while wires still reference its 

236 # port, then rebinds to a different Asset) would silently 

237 # produce role-table-vs-wire-graph divergence. The add_plan_wire 

238 # decider already enforces the bind-then-wire ordering; this 

239 # branch closes the bind ordering symmetrically. Mirrors the 

240 # add_plan_wire role-endpoint check exactly: OUTPUT required_ports 

241 # constrain wire SOURCE endpoints, INPUT required_ports constrain 

242 # wire TARGET endpoints. 

243 for required in matching_role.required_ports: 

244 if required.direction is PortDirection.OUTPUT: 

245 for wire in state.wires: 

246 if ( 

247 wire.source_port_name == required.port_name 

248 and wire.source_asset_id != command.asset_id 

249 ): 

250 raise PlanWireRoleEndpointMismatchError( 

251 state.id, 

252 wire, 

253 command.role_name, 

254 "source", 

255 command.asset_id, 

256 wire.source_asset_id, 

257 ) 

258 if required.direction is PortDirection.INPUT: 

259 for wire in state.wires: 

260 if ( 260 ↛ 259line 260 didn't jump to line 259 because the condition on line 260 was always true

261 wire.target_port_name == required.port_name 

262 and wire.target_asset_id != command.asset_id 

263 ): 

264 raise PlanWireRoleEndpointMismatchError( 

265 state.id, 

266 wire, 

267 command.role_name, 

268 "target", 

269 command.asset_id, 

270 wire.target_asset_id, 

271 ) 

272 

273 return [ 

274 PlanRoleBound( 

275 plan_id=state.id, 

276 role_name=command.role_name.value, 

277 asset_id=command.asset_id, 

278 occurred_at=now, 

279 ) 

280 ]
