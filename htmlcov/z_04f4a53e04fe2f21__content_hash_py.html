1"""Content-hash helper for the Assembly aggregate. 

2 

3Wraps the shared `cora.shared.content_hash.compute_content_hash` 

4pipeline with the Assembly-specific payload_type and the canonical 

5subset materialized by `canonical_assembly_subset`. 

6 

7Used at two sites: `define_assembly` and `version_assembly`. Both 

8fire the same hash for the same structural content; the 

9payload_type is aggregate-level 

10(`application/vnd.cora.assembly+json`), NOT event-level, so the 

11hash is stable across "define" and "version" snapshots of the same 

12canonical subset. 

13 

14Per `project_content_addressed_identity_design`: SHA-256 hex of 

15DSSE PAE-wrapped canonical JSON. Per 

16`project_canonicalization_research`: stdlib json sort-keys + NFC + 

17set-to-sorted-list, all delegated to the shared 

18`canonical_body_bytes`. 

19 

20Lives inside the Assembly aggregate dir 

21(`equipment/aggregates/assembly/_content_hash.py`) because it is 

22exclusively consumed by Assembly's own slices (define + version). 

23Unlike `_drawing.py` / `_placement.py` at the parent `aggregates/` 

24level, which are genuinely shared by multiple aggregates, this 

25helper has no cross-aggregate consumers. The future Fixture-tier 

26content hash (`project_fixture_configuration_hash_followup`) would 

27be a sibling inside `aggregates/fixture/`, not a reuse of this 

28module. 

29""" 

30

43# Aggregate-level payload_type: stable across define / version events 

44# so two snapshots of the same canonical subset produce the same hash. 

45ASSEMBLY_PAYLOAD_TYPE = "application/vnd.cora.assembly+json" 

46 

47# Per-event payload_types for signing the events themselves (not the 

48# canonical-subset hash). Kept here to avoid scattering the strings 

49# across slice deciders that emit these events. 

50ASSEMBLY_DEFINED_PAYLOAD_TYPE = event_type_to_payload_type("AssemblyDefined") 

51ASSEMBLY_VERSIONED_PAYLOAD_TYPE = event_type_to_payload_type("AssemblyVersioned") 

52ASSEMBLY_DEPRECATED_PAYLOAD_TYPE = event_type_to_payload_type("AssemblyDeprecated")

55def compute_assembly_content_hash( 

56 name: AssemblyName | str, 

57 presents_as: frozenset[RoleId], 

58 required_slots: frozenset[TemplateSlot], 

59 required_wires: frozenset[TemplateWire], 

60 parameter_overrides_schema: dict[str, object] | None, 

61 *, 

62 required_sub_assemblies: frozenset[SubAssemblyLink] = frozenset(), 

63) -> str: 

64 """Compute SHA-256 hex over an Assembly's canonical content subset. 

65 

66 Accepts either an AssemblyName VO or a raw string for `name` so 

67 callers building from operator-supplied input do not need to 

68 construct the VO twice (the AssemblyDefined decider validates 

69 once via the VO, then passes the VO here for hashing). 

70 

71 Round-trip equivalence with 

72 `compute_assembly_content_hash_from_state(Assembly(...))` is 

73 pinned in tests; both paths funnel through 

74 `canonical_assembly_subset` so structural drift between them is 

75 impossible. 

76 """ 

77 body = canonical_assembly_subset( 

78 name=name, 

79 presents_as=presents_as, 

80 required_slots=required_slots, 

81 required_wires=required_wires, 

82 required_sub_assemblies=required_sub_assemblies, 

83 parameter_overrides_schema=parameter_overrides_schema, 

84 ) 

85 return compute_content_hash(ASSEMBLY_PAYLOAD_TYPE, body)

88def compute_assembly_content_hash_from_state(state: Assembly) -> str: 

89 """Compute the content_hash for a fully-constructed Assembly state. 

90 

91 Convenience wrapper: extracts the canonical subset via 

92 `state.content_subset()` (which itself delegates to 

93 `canonical_assembly_subset`) and feeds it to the shared 

94 `compute_content_hash` pipeline under `ASSEMBLY_PAYLOAD_TYPE`. 

95 

96 Used in tests to verify that the state-method path and the 

97 explicit-args path produce identical hashes, and may be useful 

98 in future re-hash flows (e.g., a one-shot fitness that recomputes 

99 all live Assembly hashes and asserts they match the stored 

100 content_hash). 

101 """ 

102 return compute_content_hash(ASSEMBLY_PAYLOAD_TYPE, state.content_subset())
