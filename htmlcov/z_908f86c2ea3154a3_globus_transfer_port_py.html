1"""GlobusTransferPort: production `TransferPort` over the Globus Transfer service. 

2 

3The first real substrate behind CORA's `TransferPort`. It takes an 

4already-authorized Globus `TransferClient` by injection (the OAuth2 dance is a 

5composition-root concern, not the adapter's), translates a `TransferRequest` 

6into a Globus `TransferData` submission, and maps Globus task status back into 

7CORA's `TransferState`. CORA owns the seam; this adapter is an ACL that speaks 

8Globus on one side and CORA vocabulary on the other. 

9 

10The Globus client is synchronous (it speaks HTTP via `requests`), so every 

11call is run in a worker thread (`asyncio.to_thread`) to keep the async port 

12contract: the event loop is never blocked on a network round-trip. 

13 

14## Status mapping 

15 

16Globus reports exactly four task statuses; CORA's `TransferState` adds the 

17Pending/Cancelled framing the engine needs: 

18 

19- `ACTIVE` -> `Active` 

20- `INACTIVE` -> `Suspended` (Globus only enters INACTIVE on credential 

21 expiry, which is exactly CORA's intervention-required state) 

22- `SUCCEEDED` -> `Succeeded` 

23- `FAILED` -> `Failed` 

24 

25A partial move has no native Globus terminal: a sync that skips unchanged 

26files still ends `SUCCEEDED`, and a genuine partial ends `FAILED` carrying 

27`subtasks_failed > 0` alongside `files_transferred > 0`. That is exactly the 

28counts-on-`TransferProgress` partial signal the port models, so no enum value 

29is needed. A caller-issued `cancel` lands the Globus task in `FAILED`; Globus 

30does not distinguish it from a fault on the status alone, which the port 

31explicitly allows ("Failed on a substrate that does not distinguish the two"). 

32 

33## Error mapping 

34 

35Globus transport failures (`NetworkError` and its connection / timeout 

36subclasses) become `TransferEndpointUnreachableError`: the substrate did not 

37answer. Globus API errors (`GlobusAPIError`) are dispatched on `http_status`: 

38401 / 403 -> `TransferAccessDeniedError`, 5xx -> `TransferEndpointUnreachableError`, 

39and everything else the substrate refused (400 / 404 / 409 / 429 / ...) -> 

40`TransferRejectedError`. This adapter does not raise `TransferTimeoutError` 

41(a transport timeout folds into unreachable, since the adapter sets no await 

42ceiling of its own) nor `TransferIntegrityError` (a verify-on-arrival mismatch 

43surfaces as a `Failed` observation carrying the substrate's fatal-error text, 

44not an exception); both stay available for adapters that need them. 

45 

46## Not run against live Globus 

47 

48The adapter is unit-tested against a fake `TransferClient`; it has not been 

49exercised against a real Globus endpoint (that needs credentials and two live 

50collections). Treat live behaviour as unverified until a sanity run happens. 

51"""

70SyncLevel = Literal["exists", "size", "mtime", "checksum"] 

71"""Globus sync levels in ascending strictness. `skip_unchanged` maps to one of 

72these; `checksum` is the default because CORA's "do not re-move identical 

73bytes" intent is a byte-identity claim, which only the checksum level 

74guarantees (the cheaper levels compare size or mtime)."""

121def _split_location(location: str) -> tuple[str, str]: 

122 """Split a CORA `endpoint:path` location into (endpoint_id, path). 

123 

124 Raises `TransferRejectedError` on a malformed location before any network 

125 call, so an authoring error fails fast rather than reaching Globus. 

126 """ 

127 endpoint, separator, path = location.partition(":") 

128 if not separator or not endpoint or not path: 

129 msg = f"location {location!r} is not in 'endpoint:path' form" 

130 raise TransferRejectedError(msg) 

131 return endpoint, path

134def _classify_api_error( 

135 http_status: int, code: str | None, message: str, *, endpoint: str 

136) -> Exception: 

137 """Map a Globus API error's HTTP status to a CORA transfer error. 

138 

139 Pure (no Globus types), so the mapping is unit-tested directly with plain 

140 status codes. 401 / 403 are access denials; 5xx are the substrate failing 

141 to serve (unreachable); everything else is the substrate refusing this 

142 particular request. 

143 """ 

144 if http_status in (401, 403): 

145 return TransferAccessDeniedError(endpoint) 

146 if http_status in (500, 502, 503, 504): 

147 return TransferEndpointUnreachableError(endpoint) 

148 reason = f"{code or http_status}: {message}" if message else f"{code or http_status}" 

149 return TransferRejectedError(reason)

152def _map_task_status(response: _TransferResponse) -> TransferProgress: 

153 """Translate a Globus get_task response into a CORA `TransferProgress`.""" 

154 status = response["status"] 

155 state = _STATUS_TO_STATE.get(status, TransferState.ACTIVE) 

156 detail: str | None = None 

157 if state is TransferState.FAILED: 

158 fatal = response.get("fatal_error") 

159 if fatal: 

160 detail = fatal.get("description") or fatal.get("code") 

161 elif state is TransferState.SUSPENDED: 

162 detail = response.get("nice_status") 

163 return TransferProgress( 

164 state=state, 

165 bytes_moved=response.get("bytes_transferred", 0), 

166 files_total=response.get("files"), 

167 files_moved=response.get("files_transferred", 0), 

168 files_skipped=response.get("files_skipped", 0), 

169 files_failed=response.get("subtasks_failed", 0), 

170 detail=detail, 

171 )

174class GlobusTransferPort: 

175 """`TransferPort` backed by an injected Globus `TransferClient`. 

176 

177 Construct with an already-authorized client; the adapter never builds the 

178 authorizer. `skip_unchanged_sync_level` chooses how a `skip_unchanged` 

179 request is realized (default `checksum`, the byte-identity guarantee). 

180 """ 

181 

182 def __init__( 

183 self, 

184 client: GlobusTransferClient, 

185 *, 

186 skip_unchanged_sync_level: SyncLevel = "checksum", 

187 ) -> None: 

188 self._client = client 

189 self._skip_unchanged_sync_level: SyncLevel = skip_unchanged_sync_level 

190 

191 async def begin(self, request: TransferRequest) -> TransferHandle: 

192 source_endpoint, source_path = _split_location(request.source) 

193 destination_endpoint, destination_path = _split_location(request.destination) 

194 sync_level: SyncLevel | MissingType = ( 

195 self._skip_unchanged_sync_level if request.skip_unchanged else MISSING 

196 ) 

197 data = TransferData( 

198 source_endpoint=source_endpoint, 

199 destination_endpoint=destination_endpoint, 

200 label=request.label if request.label is not None else MISSING, 

201 submission_id=request.idempotency_key 

202 if request.idempotency_key is not None 

203 else MISSING, 

204 sync_level=sync_level, 

205 verify_checksum=request.verify_on_arrival, 

206 ) 

207 data.add_item(source_path, destination_path, recursive=request.recursive) 

208 context = f"{request.source} -> {request.destination}" 

209 response = await self._invoke(self._client.submit_transfer, data, context=context) 

210 return TransferHandle(str(response["task_id"])) 

211 

212 async def observe(self, handle: TransferHandle) -> TransferProgress: 

213 response = await self._invoke(self._client.get_task, str(handle), context=str(handle)) 

214 return _map_task_status(response) 

215 

216 async def cancel(self, handle: TransferHandle) -> None: 

217 await self._invoke(self._client.cancel_task, str(handle), context=str(handle)) 

218 

219 async def aclose(self) -> None: 

220 """No-op: the injected client's lifecycle is the composition root's.""" 

221 

222 async def _invoke(self, call: Any, *args: Any, context: str) -> _TransferResponse: 

223 """Run a synchronous Globus call off the event loop and map its failures.""" 

224 try: 

225 return await asyncio.to_thread(call, *args) 

226 except NetworkError as exc: 

227 raise TransferEndpointUnreachableError(context) from exc 

228 except GlobusAPIError as exc: 

229 raise _classify_api_error( 

230 getattr(exc, "http_status", 0), 

231 getattr(exc, "code", None), 

232 getattr(exc, "message", "") or str(exc), 

233 endpoint=context, 

234 ) from exc
