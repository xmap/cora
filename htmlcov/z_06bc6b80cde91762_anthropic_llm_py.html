1"""Anthropic implementation of `LLM`. 

2 

3Lives under `cora.agent.adapters` per the cross-BC 

4adapter-ownership convention (Safety BC owns 

5`PostgresClearanceLookup`, Caution BC owns `PostgresCautionLookup`, 

6Agent BC owns this). 

7 

8## Structured output via tool-use 

9 

10The Anthropic API exposes structured output through the 

11tool-use-as-structured-output convention: define a single tool 

12whose `input_schema` IS the desired output schema, set 

13`tool_choice` to force that tool, then parse the resulting 

14`tool_use` block's `input` field as the output. 

15 

16This convention has been stable since Claude 3.5 Sonnet (mid-2024) 

17and survives across model upgrades. The adapter generates the 

18synthetic tool name `cora_structured_output` (a single name across 

19all calls so the cache layer for the tools block remains stable; 

20the `input_schema` differs per call but the tool definition's 

21position in the request is identical). 

22 

23## Cache breakpoints 

24 

25`LLMContentBlock.cache` markers are translated to 

26`cache_control={"type": "ephemeral", "ttl": ...}` on the 

27corresponding `TextBlockParam`. Anthropic accepts at most 4 

28cache-marked blocks per request across system + tools + messages; 

29the adapter raises `LLMInvalidRequestError` BEFORE the API call 

30when this is exceeded so misconfigured prompts fail loudly. 

31 

32When any breakpoint requests the `"1h"` TTL, the adapter sets the 

33`anthropic-beta: extended-cache-ttl-2025-04-11` header (the 1h tier 

34is gated behind this beta flag). 

35 

36## Retries 

37 

38The Anthropic SDK ships with `max_retries=2` exponential backoff 

39out of the box (0.5s to 8s on 408/409/429/5xx + connection 

40errors). The adapter uses this default unchanged per the design 

41memo lock; setting `max_retries=2` explicitly here makes the value 

42auditable from the adapter without grepping SDK source. The 

4310-minute request timeout is a defensive ceiling: RunDebriefer 

44generation typically completes in 5-15 seconds; 600s catches 

45catastrophic provider stalls without paying for a forever-hung 

46subscriber call. 

47 

48## Error translation 

49 

50After the SDK exhausts inner retries, any remaining SDK error is 

51translated to the `LLM` taxonomy so consumers depend only on 

52the port-level error classes. Mapping: 

53 

54 - `AuthenticationError` (401 / 403) -> `LLMAuthenticationError` 

55 - `RateLimitError` (429) -> `LLMRateLimitError` 

56 - `APITimeoutError` -> `LLMTimeoutError` 

57 - `BadRequestError` (400) -> `LLMInvalidRequestError` 

58 - `InternalServerError` (5xx) -> `LLMServerError` 

59 - `APIConnectionError` -> `LLMServerError` (network-flavored) 

60 - Any other `APIStatusError` -> `LLMServerError` (defensive default) 

61 - Tool-use missing / schema mismatch on response -> `LLMSchemaValidationError` 

62"""

95# Single stable tool name across every call so the tools-layer 

96# cache breakpoint stays warm. The schema differs per call (lives 

97# in `input_schema`); the name does not. 

98_STRUCTURED_OUTPUT_TOOL_NAME = "cora_structured_output" 

99 

100# Anthropic's per-request limit on cache_control markers. Exceeding 

101# it is rejected by the API; we validate client-side so adapters 

102# fail loudly with a useful message instead of cryptic 400. 

103_MAX_CACHE_BREAKPOINTS = 4 

104 

105# Default request-level timeout for one LLM call (seconds). Catches 

106# catastrophic stalls without exceeding the design's 10-minute 

107# ceiling. Adjustable via constructor for tests / future tuning. 

108_DEFAULT_REQUEST_TIMEOUT_SECONDS = 600.0 

109 

110# Anthropic SDK default retries; pinned here for auditability per 

111# the design memo (max_retries=2, exponential 0.5s-8s). 

112_DEFAULT_MAX_RETRIES = 2 

113 

114# Beta header required when any cache breakpoint requests the 1h 

115# TTL. The 5m TTL is the default tier and doesn't need a header. 

116_EXTENDED_CACHE_TTL_BETA = "extended-cache-ttl-2025-04-11"

121class AnthropicLLM: 

122 """Production `LLM` implementation backed by `anthropic.AsyncAnthropic`. 

123 

124 Constructor accepts `api_key` directly (read from `Settings.anthropic_api_key` 

125 by the Kernel factory) rather than implicitly via the SDK's env-var 

126 lookup. Centralising the credential at the composition root keeps 

127 sensitive material off the global environment for the rest of the 

128 process and gives tests a clean injection point. 

129 

130 Optionally accepts an explicit `client: AsyncAnthropic` for tests 

131 that want to wire a mock or a recorded fixture; production wiring 

132 leaves it `None` and the adapter constructs its own. 

133 """ 

134 

135 def __init__( 

136 self, 

137 *, 

138 api_key: str, 

139 max_retries: int = _DEFAULT_MAX_RETRIES, 

140 request_timeout_seconds: float = _DEFAULT_REQUEST_TIMEOUT_SECONDS, 

141 client: anthropic.AsyncAnthropic | None = None, 

142 ) -> None: 

143 self._client = client or anthropic.AsyncAnthropic( 

144 api_key=api_key, 

145 max_retries=max_retries, 

146 timeout=request_timeout_seconds, 

147 ) 

148 

149 async def aclose(self) -> None: 

150 """Release the underlying httpx connection pool. 

151 

152 Wired into `Kernel.Teardown` from `cora.api.main` so the 

153 FastAPI lifespan closes the SDK client at shutdown. Without 

154 this the underlying `httpx.AsyncClient` leaks its connection 

155 pool on every process exit (a watch item flagged at gate 

156 review). 

157 """ 

158 await self._client.close() 

159 

160 async def chat(self, request: LLMChatRequest) -> LLMResponse: 

161 cache_count = _count_cache_breakpoints(request) 

162 if cache_count > _MAX_CACHE_BREAKPOINTS: 

163 msg = ( 

164 f"LLMChatRequest has {cache_count} cache breakpoints; " 

165 f"Anthropic accepts at most {_MAX_CACHE_BREAKPOINTS}. " 

166 "Merge layers or drop a breakpoint." 

167 ) 

168 raise LLMInvalidRequestError(msg) 

169 

170 system_blocks: list[TextBlockParam] = [ 

171 _to_text_block_param(block.text, block.cache) for block in request.system.blocks 

172 ] 

173 user_blocks: list[TextBlockParam] = [ 

174 _to_text_block_param(request.user_message.text, request.user_message.cache) 

175 ] 

176 

177 synthetic_tool = ToolParam( 

178 name=_STRUCTURED_OUTPUT_TOOL_NAME, 

179 description=( 

180 "Emit the structured Decision payload. Always call this tool exactly " 

181 "once; do not return free-form prose. Every field in the input_schema " 

182 "is required unless marked optional." 

183 ), 

184 input_schema=cast("dict[str, object]", dict(request.structured_output_schema)), 

185 ) 

186 tool_choice: ToolChoiceToolParam = { 

187 "type": "tool", 

188 "name": _STRUCTURED_OUTPUT_TOOL_NAME, 

189 } 

190 

191 extra_headers = _maybe_extended_cache_header(request) 

192 model_id = _resolve_model_id(request.model_ref) 

193 

194 with _tracer.start_as_current_span("llm.chat") as span: 

195 try: 

196 message = await self._client.messages.create( 

197 model=model_id, 

198 max_tokens=request.max_output_tokens, 

199 system=system_blocks, 

200 messages=[{"role": "user", "content": user_blocks}], 

201 tools=[synthetic_tool], 

202 tool_choice=tool_choice, 

203 extra_headers=extra_headers, 

204 ) 

205 except anthropic.AuthenticationError as exc: 

206 raise LLMAuthenticationError(str(exc)) from exc 

207 except anthropic.RateLimitError as exc: 

208 raise LLMRateLimitError(str(exc)) from exc 

209 except anthropic.APITimeoutError as exc: 

210 raise LLMTimeoutError(str(exc)) from exc 

211 except anthropic.BadRequestError as exc: 

212 raise LLMInvalidRequestError(str(exc)) from exc 

213 except anthropic.InternalServerError as exc: 

214 raise LLMServerError(str(exc)) from exc 

215 except anthropic.APIConnectionError as exc: 

216 raise LLMServerError(f"network error: {exc}") from exc 

217 except anthropic.APIStatusError as exc: 

218 # Defensive default for any APIStatusError subclass not 

219 # named above. Mapping to LLMServerError treats it as 

220 # retryable; the alternative (silently surfacing the SDK 

221 # exception) would skip the retry layer. 

222 raise LLMServerError(str(exc)) from exc 

223 

224 parsed = _extract_structured_output(message) 

225 raw_text = _extract_raw_text(message) 

226 usage = _to_llm_usage(message.usage) 

227 response_model_id = message.model 

228 stop_reason = message.stop_reason or "end_turn" 

229 

230 # `record_llm_call` returns the computed USD cost; the 

231 # adapter intentionally discards it here. The cost is 

232 # already persisted on the `cora.agent.llm.cost.usd` 

233 # histogram (where dashboards consume it), and surfacing 

234 # it on `LLMResponse` would force every consumer to think 

235 # about pricing semantics that are an observability 

236 # concern, not a domain concern. Tests that want the 

237 # cost value call `compute_cost_usd` directly. 

238 record_llm_call( 

239 span, 

240 system="anthropic", 

241 request_model_ref=request.model_ref, 

242 response_model_id=response_model_id, 

243 usage=usage, 

244 stop_reason=stop_reason, 

245 max_tokens=request.max_output_tokens, 

246 ) 

247 

248 return LLMResponse( 

249 parsed=parsed, 

250 raw_text=raw_text, 

251 usage=usage, 

252 stop_reason=stop_reason, 

253 model_id=response_model_id, 

254 )

257def _count_cache_breakpoints(request: LLMChatRequest) -> int: 

258 """Sum cache markers across system + user blocks. 

259 

260 The adapter does NOT include the synthetic structured-output 

261 tool in the count because it never carries `cache_control` here 

262 (the schema differs per call so caching the tool definition 

263 would be meaningless). Tools-layer caching for stable tools 

264 (such as a future RecipeScreener) lands when an additive `tools` 

265 field appears on `LLMChatRequest`. 

266 """ 

267 count = 0 

268 for block in request.system.blocks: 

269 if block.cache is not None: 

270 count += 1 

271 if request.user_message.cache is not None: 

272 count += 1 

273 return count

276def _to_text_block_param( 

277 text: str, 

278 cache: CacheBreakpoint | None, 

279) -> TextBlockParam: 

280 if cache is None: 

281 return {"type": "text", "text": text} 

282 cache_control = CacheControlEphemeralParam(type="ephemeral", ttl=cache.ttl) 

283 return {"type": "text", "text": text, "cache_control": cache_control}

286def _maybe_extended_cache_header(request: LLMChatRequest) -> dict[str, str]: 

287 """Return the beta header dict when any breakpoint asks for 1h TTL. 

288 

289 Empty dict otherwise (`AsyncAnthropic.messages.create(extra_headers={})` 

290 is a no-op; we always pass the kwarg so the SDK doesn't trip on 

291 `None` in some edge path). 

292 """ 

293 needs_1h = any( 

294 block.cache is not None and block.cache.ttl == "1h" for block in request.system.blocks 

295 ) or (request.user_message.cache is not None and request.user_message.cache.ttl == "1h") 

296 if needs_1h: 

297 return {"anthropic-beta": _EXTENDED_CACHE_TTL_BETA} 

298 return {}

301def _resolve_model_id(model_ref: ModelRef) -> str: 

302 """Resolve `ModelRef` to the API-level model identifier. 

303 

304 Anthropic accepts either `claude-opus-4-7` (latest stable) or 

305 a pinned snapshot like `claude-opus-4-7-20260301`. We compose 

306 `<model>-<snapshot_pin>` when a pin is set; otherwise the bare 

307 model name selects "latest stable for this model family" per 

308 Anthropic's convention. 

309 """ 

310 if model_ref.snapshot_pin is None: 

311 return model_ref.model 

312 return f"{model_ref.model}-{model_ref.snapshot_pin}"

315def _extract_structured_output(message: anthropic.types.Message) -> dict[str, object]: 

316 """Find the synthetic-tool input block; raise if absent or wrong-named. 

317 

318 The SDK's typing guarantees `ToolUseBlock.input` is `dict[str, object]` 

319 (Anthropic JSON-decodes server-side). We trust the type at the 

320 structural level and validate presence-by-name, not shape. 

321 """ 

322 for block in message.content: 

323 if isinstance(block, ToolUseBlock) and block.name == _STRUCTURED_OUTPUT_TOOL_NAME: 

324 return block.input 

325 msg = ( 

326 f"response had no tool_use block named {_STRUCTURED_OUTPUT_TOOL_NAME!r}; " 

327 f"stop_reason={message.stop_reason}, content_block_types=" 

328 f"{[b.type for b in message.content]}" 

329 ) 

330 raise LLMSchemaValidationError(msg)

333def _extract_raw_text(message: anthropic.types.Message) -> str: 

334 """Concatenate every TextBlock in the response into a single string. 

335 

336 Tool-use-structured-output responses often have zero TextBlocks 

337 (the model went straight to the tool call). Returning `""` in 

338 that case is intentional: the structured output is in `parsed`, 

339 not in raw_text. 

340 """ 

341 parts: list[str] = [] 

342 for block in message.content: 

343 if block.type == "text": 

344 parts.append(block.text) 

345 return "".join(parts)

348def _to_llm_usage(usage: anthropic.types.Usage) -> LLMUsage: 

349 """Map Anthropic's Usage shape to the port-level LLMUsage. 

350 

351 Cache fields default to 0 in `Usage` when caching wasn't used 

352 or the provider didn't return them; the dataclass coerces None 

353 to 0 defensively. 

354 """ 

355 return LLMUsage( 

356 input_tokens=usage.input_tokens, 

357 output_tokens=usage.output_tokens, 

358 cache_creation_input_tokens=usage.cache_creation_input_tokens or 0, 

359 cache_read_input_tokens=usage.cache_read_input_tokens or 0, 

360 )
