1"""MCP tool for the `get_method` query slice. 

2 

3Surfaces the same handler the REST route uses. Returns a structured 

4MethodOutput on hit. On miss raises an exception that FastMCP wraps 

5as `isError: true` with a text diagnostic — matches the REST 404 

6behaviour in MCP's error idiom (LLM consumers get a clear "method 

7not found" message rather than null structuredContent they have to 

8interpret). 

9""" 

10 

11from collections.abc import Callable 

12from datetime import datetime 

13from typing import Annotated, Any 

14from uuid import UUID 

15 

16from mcp.server.fastmcp import Context, FastMCP 

17from pydantic import BaseModel, Field 

18 

19from cora.infrastructure.mcp_principal import get_mcp_principal_id 

20from cora.infrastructure.observability import current_correlation_id 

21from cora.infrastructure.routing import get_mcp_surface_id 

22from cora.recipe.aggregates.method import METHOD_NAME_MAX_LENGTH 

23from cora.recipe.features.get_method.handler import Handler 

24from cora.recipe.features.get_method.query import GetMethod 

25

27class MethodOutput(BaseModel): 

28 """Structured output of the `get_method` MCP tool. 

29 

30 `created_at` / `versioned_at` / `deprecated_at` mirror the REST 

31 `MethodResponse` (Path C): sourced from the 

32 `proj_recipe_method_summary` projection. Null semantics: read 

33 together with `status`: a populated `status` with a null 

34 timestamp means the projection has not yet folded that lifecycle 

35 event (transient eventual-consistency window), never a missing 

36 transition. A not-found Method raises (MCP `isError: true`) 

37 rather than returning null timestamps. 

38 """ 

39 

40 id: UUID 

41 name: str = Field(..., max_length=METHOD_NAME_MAX_LENGTH) 

42 needed_family_ids: list[UUID] 

43 needed_supplies: list[str] 

44 status: str 

45 version: str | None 

46 # compute classification, read from folded state (mirrors the REST 

47 # MethodResponse). execution_pattern is the enum string value 

48 # (Batch / Iterative / Streaming) or null; the two claims default False. 

49 execution_pattern: str | None 

50 monotone_quality: bool 

51 resumable_from_checkpoint: bool 

52 created_at: datetime | None = None 

53 versioned_at: datetime | None = None 

54 deprecated_at: datetime | None = None 

55 

56 

57def register(mcp: FastMCP, *, get_handler: Callable[[], Handler]) -> None: 

58 """Register the `get_method` tool on the given MCP server.""" 

59 

60 @mcp.tool( 

61 name="get_method", 

62 description="Read the current state of an existing method by id.", 

63 ) 

64 async def get_method_tool( # pyright: ignore[reportUnusedFunction] 

65 ctx: Context[Any, Any, Any], 

66 method_id: Annotated[ 

67 UUID, 

68 Field(description="Target method's id."), 

69 ], 

70 ) -> MethodOutput: 

71 handler = get_handler() 

72 view = await handler( 

73 GetMethod(method_id=method_id), 

74 principal_id=get_mcp_principal_id(ctx), 

75 correlation_id=current_correlation_id(), 

76 surface_id=get_mcp_surface_id(), 

77 ) 

78 if view is None: 

79 msg = f"Method {method_id} not found" 

80 raise ValueError(msg) 

81 method = view.method 

82 timestamps = view.timestamps 

83 return MethodOutput( 

84 id=method.id, 

85 name=method.name.value, 

86 needed_family_ids=sorted(method.needed_family_ids, key=str), 

87 needed_supplies=sorted(method.needed_supplies), 

88 status=method.status.value, 

89 version=method.version, 

90 execution_pattern=( 

91 method.execution_pattern.value if method.execution_pattern is not None else None 

92 ), 

93 monotone_quality=method.monotone_quality, 

94 resumable_from_checkpoint=method.resumable_from_checkpoint, 

95 created_at=timestamps.created_at if timestamps is not None else None, 

96 versioned_at=timestamps.versioned_at if timestamps is not None else None, 

97 deprecated_at=timestamps.deprecated_at if timestamps is not None else None, 

98 )
