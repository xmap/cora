1"""Settings-loadable shape for `CaptureScanIngestor`'s per-capture-code bindings. 

2 

3Mirrors the layering used for `ControlPortRoute` and 

4`IdentityProviderConfig`: the typed config models live here in 

5`cora.infrastructure` so `Settings` can validate the env var at 

6startup, while the code that consumes the validated shape 

7(`cora.api._capture_scan_ingestor`) lives in the composition root. 

8 

9A capture code binds one producing Asset plus one or more locations 

10(vault rows) the finished file may land on, each naming the Supply and 

11access protocol CORA should record for `IngestScan`. See 

12`run.aggregates.run.capture_path`'s module docstring for why a Run may 

13hold more than one vault row (one per storage location it was observed 

14under) and `CapturePathStore.upsert`'s own docstring for why that store 

15is "idempotent per LOCATION, not per Run". 

16 

17## Env var shape 

18 

19`Settings.capture_scan_ingestor_bindings` reads from 

20CAPTURE_SCAN_INGESTOR_BINDINGS as JSON, keyed by capture code: 

21 

22 CAPTURE_SCAN_INGESTOR_BINDINGS='{ 

23 "2bmb-tomoscan": { 

24 "producing_asset_id": "0c5e...-camera-asset-uuid", 

25 "camera_channel_name": "camera_selected", 

26 "expected_camera_label": "Camera Selected 0", 

27 "locations": { 

28 "/local1/2BM": { 

29 "supply_id": "b2a1...-storage-supply-uuid", 

30 "access_protocol": "POSIX" 

31 }, 

32 "/gdata/dm/2BM": { 

33 "supply_id": "77f0...-storage-supply-uuid", 

34 "access_protocol": "NFS", 

35 "durable": true, 

36 "subdirectory": "data" 

37 } 

38 } 

39 } 

40 }' 

41 

42A code absent from this map is never auto-ingested, mirroring every 

43other per-code table's optionality. 

44 

45## `camera_channel_name` + `expected_camera_label`: the producing-Asset cross-check 

46 

47Both optional, both-or-neither (see `_validate_camera_expectation`). Named 

48separately from `producing_asset_id` rather than replacing it with a 

49per-camera map: `producing_asset_id` stays the one fixed Asset this code 

50mints Datasets under, and these two fields let `CaptureScanIngestor` REFUSE 

51a candidate rather than auto-route to a different Asset when the recorded 

52evidence disagrees, per this deployment's own preference for a loud 

53refusal over a guessed correction. 

54 

55`camera_channel_name` names a `capture_baseline_pvs` channel (recorded once 

56per Run, at promotion, by the already-shipped `CaptureBaselineReader`) that 

57must ALSO be configured under `CAPTURE_BASELINE_PVS` for this to do 

58anything; `expected_camera_label` is the resolved label (`Observation. 

59categorical_value`) that channel is expected to carry when this code's 

60`producing_asset_id` is correct. See `cora.api._capture_scan_ingestor`'s 

61module docstring for why this reads each candidate's OWN recorded 

62observation rather than polling the camera-selection PV live. 

63"""

75# Local mirror of `cora.data.aggregates.distribution.state.AccessProtocol`. 

76# Not imported directly: `cora.infrastructure` may not depend on 

77# `cora.data` (`tach check`), and the Data BC's own `IngestScan` decider 

78# already validates the value it receives, so this mirror only needs to 

79# keep an operator's typo from surfacing at the first sweep tick instead 

80# of at boot. `tests/architecture/test_capture_scan_ingestor_access_protocol_sync.py` 

81# pins the value set against the real enum so the two cannot silently drift. 

82_AccessProtocolLiteral = Literal["HTTPS", "Globus", "S3", "POSIX", "NFS", "OAI_PMH"]

85class CaptureScanIngestorLocation(BaseModel): 

86 """One location a capture code's file may reach: which Supply holds it, over what protocol. 

87 

88 `durable` lives here, on the location, rather than in a separate 

89 top-level setting listing durable roots. A separate list would be a 

90 second place to keep in sync with this one, and the two would 

91 drift; the fact that a tier is the durable one belongs with the 

92 tier it describes. 

93 """ 

94 

95 model_config = ConfigDict(extra="forbid") 

96 

97 supply_id: UUID = Field( 

98 description="The Distribution Supply `IngestScan` should record as reading this location." 

99 ) 

100 access_protocol: _AccessProtocolLiteral = Field( 

101 description=( 

102 "The transport family `IngestScan` should record for a " 

103 "Dataset minted from this location." 

104 ) 

105 ) 

106 durable: bool = Field( 

107 default=False, 

108 description=( 

109 "Whether this is the durable copy: the one the sweep should " 

110 "find and register as a second Distribution on the same " 

111 "Dataset once the acquisition-tier copy risks being purged. " 

112 "At most one location per capture code may set this." 

113 ), 

114 ) 

115 subdirectory: str | None = Field( 

116 default=None, 

117 description=( 

118 "The path segment below the experiment folder where files " 

119 "at this location live, such as APS Data Management's " 

120 "`data` directory under DMagic's layout. None when files " 

121 "sit directly in the experiment folder, which is the " 

122 "acquisition tier's shape and most locations'." 

123 ), 

124 ) 

125 

126 @field_validator("subdirectory") 

127 @classmethod 

128 def _validate_subdirectory(cls, value: str | None) -> str | None: 

129 """Refuse anything that is not one safe path segment. 

130 

131 Uses the same `is_safe_path_segment` rule both ends of the scan 

132 probe already apply to `subdirectory`, rather than a second 

133 rule that could disagree with it. 

134 """ 

135 if value is not None and not is_safe_path_segment(value): 

136 msg = f"subdirectory {value!r} is not one safe path segment." 

137 raise ValueError(msg) 

138 return value

141class CaptureScanIngestorBinding(BaseModel): 

142 """What `IngestScan` needs for one capture code: the producing Asset, plus its locations.""" 

143 

144 model_config = ConfigDict(extra="forbid") 

145 

146 producing_asset_id: UUID = Field( 

147 description="The Asset that produced files under this capture code." 

148 ) 

149 locations: dict[str, CaptureScanIngestorLocation] = Field( 

150 description=( 

151 "One location per storage root the finished file may land on, " 

152 "keyed by the root itself (normalized at validation time)." 

153 ) 

154 ) 

155 camera_channel_name: str | None = Field( 

156 default=None, 

157 description=( 

158 "The `capture_baseline_pvs` channel name this code's operator " 

159 "declared for the beamline's camera-selection readback PV (e.g. " 

160 "2-BM's `2bm:MCTOptics:CameraSelected`, conventionally under the " 

161 "channel name `camera_selected`). Set together with " 

162 "`expected_camera_label` to have `CaptureScanIngestor` refuse a " 

163 "candidate whose recorded camera-selection observation disagrees " 

164 "with (or is missing for) the camera `producing_asset_id` " 

165 "assumes, rather than silently attributing every scan under " 

166 "this code to one fixed Asset regardless of which camera " 

167 "actually produced it. None (the default) leaves this " 

168 "cross-check off, matching this binding's other optional fields." 

169 ), 

170 ) 

171 expected_camera_label: str | None = Field( 

172 default=None, 

173 description=( 

174 "The resolved camera-selection label (the `Observation." 

175 "categorical_value` `CaptureBaselineReader` records under " 

176 "`camera_channel_name`) that `producing_asset_id` assumes. " 

177 "Compared against each candidate's OWN recorded observation for " 

178 "its `run_id`, never against a live PV read, since a candidate " 

179 "may be backlogged: see `cora.api._capture_scan_ingestor`'s " 

180 "module docstring for why the check must be per-run." 

181 ), 

182 )

184 @model_validator(mode="after") 

185 def _validate_camera_expectation(self) -> "CaptureScanIngestorBinding": 

186 """`camera_channel_name` and `expected_camera_label` are both-or-neither. 

187 

188 One set without the other is a config mistake, not a valid partial 

189 state: a channel name with no expected label has nothing to compare 

190 against, and an expected label with no channel name has nowhere to 

191 read the observation from. Neither half can silently no-op the 

192 other's intent. 

193 """ 

194 has_channel = self.camera_channel_name is not None 

195 has_label = self.expected_camera_label is not None 

196 if has_channel != has_label: 

197 msg = ( 

198 "camera_channel_name and expected_camera_label must be set " 

199 "together or not at all. Got " 

200 f"camera_channel_name={self.camera_channel_name!r}, " 

201 f"expected_camera_label={self.expected_camera_label!r}." 

202 ) 

203 raise ValueError(msg) 

204 if has_channel and (not self.camera_channel_name or not self.expected_camera_label): 

205 msg = "camera_channel_name and expected_camera_label must both be non-empty when set." 

206 raise ValueError(msg) 

207 return self

209 @field_validator("locations") 

210 @classmethod 

211 def _validate_locations( 

212 cls, value: dict[str, CaptureScanIngestorLocation] 

213 ) -> dict[str, CaptureScanIngestorLocation]: 

214 """Refuse an empty map, and normalize + dedupe root keys. 

215 

216 A relative or root-collapsing key is meaningless to 

217 `cora.api._capture_scan_ingestor`'s join against the vault's own 

218 `root` column. Keys are stored NORMALIZED so a deployment 

219 writing a trailing slash still matches the vault's normalized 

220 column; see `cora.shared.storage_root`'s module docstring for 

221 why one un-normalized caller makes every lookup miss. Two keys 

222 that normalize to the same root raise rather than silently 

223 collapsing to whichever one iteration visited last, since that 

224 would drop a Supply/protocol pairing an operator wrote on 

225 purpose with no signal that it happened. 

226 

227 At most one location may be marked `durable`: two would leave 

228 the sweep no way to choose between them, and failing here beats 

229 discovering the ambiguity on the first sweep tick. Zero is 

230 fine; it means this code has no durable tier configured yet and 

231 the sweep simply skips it. 

232 """ 

233 if not value: 

234 msg = "locations is empty. A binding with no location can never ingest anything." 

235 raise ValueError(msg) 

236 normalized: dict[str, CaptureScanIngestorLocation] = {} 

237 original_spelling: dict[str, str] = {} 

238 for root, location in value.items(): 

239 require_nonempty_absolute_root(root, label="locations key") 

240 key = normalize_storage_root(root) 

241 if key in normalized: 

242 msg = ( 

243 f"locations keys {original_spelling[key]!r} and {root!r} " 

244 f"both normalize to {key!r}. Each location must name a " 

245 "distinct storage root." 

246 ) 

247 raise ValueError(msg) 

248 normalized[key] = location 

249 original_spelling[key] = root 

250 durable_keys = [key for key, location in normalized.items() if location.durable] 

251 if len(durable_keys) > 1: 

252 msg = ( 

253 f"locations marks {len(durable_keys)} roots durable: " 

254 f"{sorted(durable_keys)!r}. At most one location per capture " 

255 "code may be durable, or the sweep has no way to choose." 

256 ) 

257 raise ValueError(msg) 

258 return normalized

261def durable_roots(bindings: Mapping[str, CaptureScanIngestorBinding]) -> frozenset[str]: 

262 """The normalized storage roots marked durable, across every capture code. 

263 

264 Derived from `bindings` rather than read from a separate Settings 

265 field, since deriving it is what keeps it from drifting out of sync 

266 with the per-location flag it is computed from. Empty when nothing 

267 is marked durable, which the caller treats as the sweep being off. 

268 """ 

269 return frozenset( 

270 root 

271 for binding in bindings.values() 

272 for root, location in binding.locations.items() 

273 if location.durable 

274 )

277def durable_supply_ids(bindings: Mapping[str, CaptureScanIngestorBinding]) -> frozenset[UUID]: 

278 """The Supply ids of the locations marked durable, across every capture code. 

279 

280 Empty under the same condition as `durable_roots`, and derived the 

281 same way: from `bindings` directly, never from a Settings field 

282 that could fall out of step with it. 

283 """ 

284 return frozenset( 

285 location.supply_id 

286 for binding in bindings.values() 

287 for location in binding.locations.values() 

288 if location.durable

292@dataclass(frozen=True) 

293class DurableLocationBinding: 

294 """The one durable location configured for a capture code. 

295 

296 Structurally satisfies 

297 `cora.api._durable_distribution_driver.DurableLocationBinding` (same 

298 four properties, by name) without importing it: `cora.infrastructure` 

299 may not depend on `cora.api` per `tach.toml`, and duck typing is 

300 exactly what that Protocol asks a caller to provide. 

301 """ 

302 

303 root: str 

304 supply_id: UUID 

305 access_protocol: str 

306 subdirectory: str | None

309class CaptureScanIngestorDurableLocationLookup: 

310 """`DurableLocationLookup` read directly off the validated binding 

311 config; no derived index kept alongside it to fall out of sync. 

312 

313 `CaptureScanIngestorBinding._validate_locations` already enforces at 

314 most one durable location per capture code, so the scan below always 

315 finds zero or one match. 

316 """ 

317 

318 def __init__(self, bindings: Mapping[str, CaptureScanIngestorBinding]) -> None: 

319 self._bindings = bindings 

320 

321 def durable_location_for(self, capture_code: str) -> DurableLocationBinding | None: 

322 binding = self._bindings.get(capture_code) 

323 if binding is None: 

324 return None 

325 for root, location in binding.locations.items(): 

326 if location.durable: 

327 return DurableLocationBinding( 

328 root=root, 

329 supply_id=location.supply_id, 

330 access_protocol=location.access_protocol, 

331 subdirectory=location.subdirectory, 

332 ) 

333 return None
