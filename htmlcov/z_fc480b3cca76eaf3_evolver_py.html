1"""Evolver: replay events to reconstruct Asset state. 

2 

3Mirror of the other aggregate evolvers. The terminal `assert_never` 

4case forces pyright (and the runtime) to error if a new event type 

5is added to `AssetEvent` without a matching match arm here. 

6 

7Lifecycle mapping per event type: 

8 - `AssetRegistered` -> COMMISSIONED (genesis) 

9 - `AssetActivated` -> ACTIVE 

10 - `AssetDecommissioned` -> DECOMMISSIONED (3-source guard at decider) 

11 - `AssetRelocated` -> (lifecycle UNCHANGED; mutates parent_id only) 

12 - `AssetMaintenanceEntered` -> MAINTENANCE 

13 - `AssetRestoredFromMaintenance` -> ACTIVE 

14 - `AssetFamilyAdded` -> (lifecycle UNCHANGED; inserts into families frozenset) 

15 - `AssetFamilyRemoved` -> (lifecycle UNCHANGED; removes from families frozenset) 

16 - `AssetDegraded` -> (lifecycle UNCHANGED; condition -> DEGRADED) 

17 - `AssetFaulted` -> (lifecycle UNCHANGED; condition -> FAULTED) 

18 - `AssetRestored` -> (lifecycle UNCHANGED; condition -> NOMINAL) 

19 - `AssetSettingsUpdated` -> (lifecycle UNCHANGED; settings -> event.settings) 

20 - `AssetPortAdded` -> (lifecycle UNCHANGED; inserts AssetPort into ports frozenset) 

21 - `AssetPortRemoved` -> (lifecycle UNCHANGED; removes AssetPort matching name) 

22 

23The lifecycle mapping is hardcoded per match arm — the event type 

24IS the lifecycle-change indicator (no lifecycle field in event 

25payloads). Same precedent as Subject / Family. Condition 

26mapping follows the same pattern (event type encodes the target 

27condition; no condition field in payload). 

28 

29`level` IS reconstructed from the payload of AssetRegistered (set 

30at registration, never changes; payload-carried by design — see 

31events.py docstring). `parent_id` IS reconstructed from 

32AssetRegistered's payload AND mutated by AssetRelocated's 

33`to_parent_id` field. `families` defaults to empty frozenset on 

34AssetRegistered (additive-state pattern; existing AssetRegistered 

35events from before 5f-1 fold cleanly without an upcaster) and is 

36mutated incrementally by `AssetFamilyAdded` / 

37`AssetFamilyRemoved`. 

38 

39**Critical invariant**: every transition arm MUST carry 

40`families` AND `condition` AND `settings` AND `ports` through 

41from prior state. Constructing `Asset(id=..., name=..., level=..., 

42parent_id=..., lifecycle=...)` without explicitly passing them 

43would silently WIPE the fields to their defaults (empty frozenset / 

44NOMINAL / empty dict / empty frozenset). 5f-1 added `families` 

45with a default solely for additive-state forward compatibility on 

46genesis events; 5g-b added `condition`, 5g-c added `settings`, 5h 

47added `ports` likewise. Transition arms must explicitly carry all 

48four. Pinned by `test_evolve_<transition>_preserves_capabilities`, 

49`test_evolve_<transition>_preserves_condition`, 

50`test_evolve_<transition>_preserves_settings`, and 

51`test_evolve_<transition>_preserves_ports` for each transition. 

52 

53Transition events applied to empty state raise ValueError: they 

54can never appear before `AssetRegistered` in a well-formed stream. 

55The `require_state` helper keeps the per-arm bodies short 

56(precedent locked by Subject's evolver in 4c). 

57"""

91def evolve(state: Asset | None, event: AssetEvent) -> Asset: 

92 """Apply one event to the current state.""" 

93 match event: 

94 case AssetRegistered(asset_id=asset_id, name=name, level=level, parent_id=parent_id): 

95 _ = state # AssetRegistered is the genesis event; prior state ignored 

96 return Asset( 

97 id=asset_id, 

98 name=AssetName(name), 

99 level=AssetLevel(level), 

100 parent_id=parent_id, 

101 lifecycle=AssetLifecycle.COMMISSIONED, 

102 # families defaults to empty frozenset; condition 

103 # defaults to NOMINAL. Additive-state pattern: both 

104 # default-via-state so pre-5f-1 / pre-5g-b streams fold 

105 # cleanly without an upcaster. 

106 ) 

107 case AssetActivated(): 

108 prior = require_state(state, "AssetActivated") 

109 return Asset( 

110 id=prior.id, 

111 name=prior.name, 

112 level=prior.level, 

113 parent_id=prior.parent_id, 

114 lifecycle=AssetLifecycle.ACTIVE, 

115 condition=prior.condition, 

116 families=prior.families, 

117 settings=prior.settings, 

118 ports=prior.ports, 

119 ) 

120 case AssetDecommissioned(): 

121 prior = require_state(state, "AssetDecommissioned") 

122 return Asset( 

123 id=prior.id, 

124 name=prior.name, 

125 level=prior.level, 

126 parent_id=prior.parent_id, 

127 lifecycle=AssetLifecycle.DECOMMISSIONED, 

128 condition=prior.condition, 

129 families=prior.families, 

130 settings=prior.settings, 

131 ports=prior.ports, 

132 ) 

133 case AssetRelocated(to_parent_id=to_parent_id): 

134 # Hierarchy mutation: only parent_id changes; lifecycle / level 

135 # / name / families / condition / settings carry over from 

136 # prior state. The from_parent_id and reason fields in the 

137 # event aren't read here (audit metadata; prior state's 

138 # parent_id is the source of truth for the read path). 

139 prior = require_state(state, "AssetRelocated") 

140 return Asset( 

141 id=prior.id, 

142 name=prior.name, 

143 level=prior.level, 

144 parent_id=to_parent_id, 

145 lifecycle=prior.lifecycle, 

146 condition=prior.condition, 

147 families=prior.families, 

148 settings=prior.settings, 

149 ports=prior.ports, 

150 ) 

151 case AssetMaintenanceEntered(): 

152 prior = require_state(state, "AssetMaintenanceEntered") 

153 return Asset( 

154 id=prior.id, 

155 name=prior.name, 

156 level=prior.level, 

157 parent_id=prior.parent_id, 

158 lifecycle=AssetLifecycle.MAINTENANCE, 

159 condition=prior.condition, 

160 families=prior.families, 

161 settings=prior.settings, 

162 ports=prior.ports, 

163 ) 

164 case AssetRestoredFromMaintenance(): 

165 prior = require_state(state, "AssetRestoredFromMaintenance") 

166 return Asset( 

167 id=prior.id, 

168 name=prior.name, 

169 level=prior.level, 

170 parent_id=prior.parent_id, 

171 lifecycle=AssetLifecycle.ACTIVE, 

172 condition=prior.condition, 

173 families=prior.families, 

174 settings=prior.settings, 

175 ports=prior.ports, 

176 ) 

177 case AssetFamilyAdded(family_id=family_id): 

178 # Family mutation: only `families` changes; everything 

179 # else carries over. Frozenset semantics: adding an already- 

180 # present id is a no-op AT THE EVOLVER LAYER (the decider's 

181 # strict-not-idempotent guard enforces "must not already be 

182 # present" at command time). 

183 prior = require_state(state, "AssetFamilyAdded") 

184 return Asset( 

185 id=prior.id, 

186 name=prior.name, 

187 level=prior.level, 

188 parent_id=prior.parent_id, 

189 lifecycle=prior.lifecycle, 

190 condition=prior.condition, 

191 families=prior.families | {family_id}, 

192 settings=prior.settings, 

193 ports=prior.ports, 

194 ) 

195 case AssetFamilyRemoved(family_id=family_id): 

196 # Mirror of AssetFamilyAdded. Frozenset difference is a 

197 # no-op when the id isn't present (decider enforces 

198 # presence at command time). Settings are NOT auto-purged 

199 # when a Family is removed (5g-c lock: preserve orphans; 

200 # the update_asset_settings slice's null-write is the 

201 # cleanup path). 

202 prior = require_state(state, "AssetFamilyRemoved") 

203 return Asset( 

204 id=prior.id, 

205 name=prior.name, 

206 level=prior.level, 

207 parent_id=prior.parent_id, 

208 lifecycle=prior.lifecycle, 

209 condition=prior.condition, 

210 families=prior.families - {family_id}, 

211 settings=prior.settings, 

212 ports=prior.ports, 

213 ) 

214 case AssetDegraded(): 

215 # Condition mutation: only `condition` changes; everything 

216 # else carries over. The reason field in the event is audit 

217 # metadata, not folded into state. Decider's no-op-on-unchanged 

218 # guard prevents redundant transitions; if one slips through 

219 # (concurrent writers), this arm idempotently sets DEGRADED. 

220 prior = require_state(state, "AssetDegraded") 

221 return Asset( 

222 id=prior.id, 

223 name=prior.name, 

224 level=prior.level, 

225 parent_id=prior.parent_id, 

226 lifecycle=prior.lifecycle, 

227 condition=AssetCondition.DEGRADED, 

228 families=prior.families, 

229 settings=prior.settings, 

230 ports=prior.ports, 

231 ) 

232 case AssetFaulted(): 

233 prior = require_state(state, "AssetFaulted") 

234 return Asset( 

235 id=prior.id, 

236 name=prior.name, 

237 level=prior.level, 

238 parent_id=prior.parent_id, 

239 lifecycle=prior.lifecycle, 

240 condition=AssetCondition.FAULTED, 

241 families=prior.families, 

242 settings=prior.settings, 

243 ports=prior.ports, 

244 ) 

245 case AssetRestored(): 

246 prior = require_state(state, "AssetRestored") 

247 return Asset( 

248 id=prior.id, 

249 name=prior.name, 

250 level=prior.level, 

251 parent_id=prior.parent_id, 

252 lifecycle=prior.lifecycle, 

253 condition=AssetCondition.NOMINAL, 

254 families=prior.families, 

255 settings=prior.settings, 

256 ports=prior.ports, 

257 ) 

258 case AssetSettingsUpdated(settings=settings): 

259 # Settings mutation: only `settings` changes. Event payload 

260 # carries the FULL post-merge dict (5g-c lock), so this arm 

261 # simply replaces. Validation already happened at the handler 

262 # boundary before append; an event in the stream is by 

263 # definition validated. 

264 prior = require_state(state, "AssetSettingsUpdated") 

265 return Asset( 

266 id=prior.id, 

267 name=prior.name, 

268 level=prior.level, 

269 parent_id=prior.parent_id, 

270 lifecycle=prior.lifecycle, 

271 condition=prior.condition, 

272 families=prior.families, 

273 settings=settings, 

274 ports=prior.ports, 

275 ) 

276 case AssetPortAdded( 

277 port_name=port_name, 

278 direction=direction, 

279 signal_type=signal_type, 

280 ): 

281 # Ports mutation: only `ports` changes; everything else 

282 # carries over. Frozenset semantics: adding the exact 

283 # AssetPort tuple twice is a no-op AT THE EVOLVER LAYER; 

284 # the decider's strict-not-idempotent guard enforces "no 

285 # port with this name already exists" at command time 

286 # (which is stricter than tuple equality). 

287 prior = require_state(state, "AssetPortAdded") 

288 new_port = AssetPort( 

289 name=port_name, 

290 direction=PortDirection(direction), 

291 signal_type=signal_type, 

292 ) 

293 return Asset( 

294 id=prior.id, 

295 name=prior.name, 

296 level=prior.level, 

297 parent_id=prior.parent_id, 

298 lifecycle=prior.lifecycle, 

299 condition=prior.condition, 

300 families=prior.families, 

301 settings=prior.settings, 

302 ports=prior.ports | {new_port}, 

303 ) 

304 case AssetPortRemoved(port_name=port_name): 

305 # Mirror of AssetPortAdded. Removes the port whose `name` 

306 # matches; AssetPort tuple equality alone wouldn't suffice 

307 # because we may not know the direction/signal_type at the 

308 # call site. Use a comprehension to filter by name. The 

309 # decider enforces "port with this name MUST exist" at 

310 # command time; the evolver-level filter is forgiving (no 

311 # error if name not found, since by-design the only 

312 # producer is the decider that already validated). 

313 prior = require_state(state, "AssetPortRemoved") 

314 return Asset( 

315 id=prior.id, 

316 name=prior.name, 

317 level=prior.level, 

318 parent_id=prior.parent_id, 

319 lifecycle=prior.lifecycle, 

320 condition=prior.condition, 

321 families=prior.families, 

322 settings=prior.settings, 

323 ports=frozenset(p for p in prior.ports if p.name != port_name), 

324 ) 

325 case _: # pragma: no cover # exhaustiveness guard 

326 assert_never(event)

329def fold(events: Sequence[AssetEvent]) -> Asset | None: 

330 """Replay a stream of events from the empty initial state.""" 

331 state: Asset | None = None 

332 for event in events: 

333 state = evolve(state, event) 

334 return state
