29# Bootstrap Policy id. Bound to (conduit=nil, surface=HTTP). Production 

30# deployments set `TRUST_POLICY_ID=00000000-0000-0000-0000-000000000002` 

31# to point at it; the seed permits SYSTEM_PRINCIPAL_ID to call 

32# {DefinePolicy, RegisterActor} so operators can register a real admin 

33# Actor and promote a real admin Policy. Seeded by 

34# 20260519200000_seed_default_surfaces_and_v2_policy.sql. 

35# 

36# The earlier nil-surface bootstrap policy (...0001) is retired: its 

37# evaluate-time nil-as-wildcard fold was removed, so it now strict-denies 

38# every real-surface call. Deployments must use the surface-bound id 

39# below. 

40SYSTEM_BOOTSTRAP_POLICY_ID = UUID("00000000-0000-0000-0000-000000000002")

42# Default Surfaces. The first three are seeded by 

43# `20260519200000_seed_default_surfaces_and_v2_policy.sql`; 

44# `SYSTEM_IN_PROCESS_SURFACE_ID` by a later migration (see 

45# `cora.infrastructure.schema_version.EXPECTED_SCHEMA_VERSION` for the

55async def verify_bootstrap_seed_present(deps: Kernel) -> None: 

56 """Fail-fast at lifespan start when the configured bootstrap seed 

57 stream — or its dependencies — is missing. 

58 

59 When `trust_policy_id == SYSTEM_BOOTSTRAP_POLICY_ID`, verify the 

60 bootstrap policy stream and the 3 seeded Surface streams exist, and 

61 that the policy folded to the HTTP Surface. The policy references the 

62 HTTP Surface; without the Surface streams, evaluate would deny on a 

63 phantom surface_id and the verdict audit log would silently skip 

64 entries (partial-fail mitigation). Otherwise no-op: custom operator 

65 policies are the operator's responsibility to verify. 

66 

67 The retired nil-surface bootstrap policy (...0001) is not handled 

68 here. Its evaluate-time wildcard fold was removed, so it strict-denies 

69 every real-surface call; a deployment still pointed at it would be 

70 locked out. Point TRUST_POLICY_ID at the surface-bound id. 

71 """ 

72 settings = deps.settings 

73 

74 if settings.trust_policy_id != SYSTEM_BOOTSTRAP_POLICY_ID: 

75 return 

76 

77 policy = await load_policy(deps.event_store, SYSTEM_BOOTSTRAP_POLICY_ID) 

78 if policy is None: 

79 msg = ( 

80 f"Configured trust_policy_id={SYSTEM_BOOTSTRAP_POLICY_ID} " 

81 "(bootstrap policy) but the seed stream is missing from the " 

82 "event store. Re-run `make migrate-apply` — the seed migration " 

83 "20260519200000_seed_default_surfaces_and_v2_policy.sql is " 

84 "idempotent (ON CONFLICT DO NOTHING) and safe to re-apply. " 

85 "See memory/project_conduit_injection_design.md." 

86 ) 

87 raise RuntimeError(msg) 

88 

89 # The bootstrap policy binds to SYSTEM_HTTP_SURFACE_ID; without all 3 

90 # seeded Surfaces present, the audit / authz substrate is broken 

91 # (partial-fail). 

92 for surface_id in ( 

93 SYSTEM_HTTP_SURFACE_ID, 

94 SYSTEM_MCP_STDIO_SURFACE_ID, 

95 SYSTEM_MCP_STREAMABLE_HTTP_SURFACE_ID, 

96 ): 

97 surface = await load_surface(deps.event_store, surface_id) 

98 if surface is None: 

99 msg = ( 

100 f"trust_policy_id={SYSTEM_BOOTSTRAP_POLICY_ID} is configured " 

101 f"but seeded Surface {surface_id} is missing from the event " 

102 "store — the policy references the HTTP Surface and the audit " 

103 "path expects all 3. Re-run `make migrate-apply`; the seed " 

104 "migration is idempotent." 

105 ) 

106 raise RuntimeError(msg) 

107 

108 # Assert the bootstrap policy's surface_id binding is correct. A 

109 # typo'd / mis-wired seed (folded surface_id != HTTP) would silently 

110 # deny every request, since evaluate strict-matches the surface. 

111 if policy.surface_id != SYSTEM_HTTP_SURFACE_ID: 

112 msg = ( 

113 f"trust_policy_id={SYSTEM_BOOTSTRAP_POLICY_ID} was loaded but " 

114 f"folded with surface_id={policy.surface_id} instead of the " 

115 f"expected SYSTEM_HTTP_SURFACE_ID ({SYSTEM_HTTP_SURFACE_ID}). The " 

116 "seed migration may have been mutated post-seed, or a non-seed " 

117 "PolicyDefined event was appended to this stream. Investigate the " 

118 "event log." 

119 ) 

120 raise RuntimeError(msg)

123async def warn_if_verdict_log_dormant(deps: Kernel) -> None: 

124 """Warn (loudly, once at boot) when authz is ENFORCED but the 

125 per-Conduit Verdict audit log cannot populate. 

126 

127 `TrustAuthorize` writes a Verdict row per decision only when the 

128 conduit a command flows through has an open verdict logbook. Checks 

129 the EFFECTIVE conduit — `settings.trust_conduit_id` when an operator 

130 has opted in, the nil sentinel otherwise (every handler still passes 

131 nil; `TrustAuthorize` resolves it to the configured conduit only when 

132 the caller passed nil, per `authorize.py`'s `_effective_conduit_id`). 

133 An operator who has run `verify_local_conduit_seed_present` and 

134 pointed `trust_conduit_id` at a seeded, logbook-open Conduit sees no 

135 warning here, because there is nothing dormant to report. 

136 

137 Non-fatal by design: a known-limitation notice, not a misconfig. When 

138 `trust_policy_id` is unset (AllowAll) there are no decisions to record 

139 and no warning is emitted. 

140 """ 

141 settings = deps.settings 

142 if settings.trust_policy_id is None: 

143 return 

144 

145 effective_conduit_id = settings.trust_conduit_id or NIL_SENTINEL_ID 

146 conduit = await load_conduit(deps.event_store, effective_conduit_id) 

147 if conduit is not None and conduit.logbooks.get(LOGBOOK_KIND_VERDICT) is not None: 

148 return 

149 

150 _log.warning( 

151 "trust_authorize.verdict_log_dormant", 

152 trust_policy_id=str(settings.trust_policy_id), 

153 conduit_id=str(effective_conduit_id), 

154 detail=( 

155 "Authorization is ENFORCED but the per-Conduit Verdict audit log " 

156 "will NOT populate: commands traverse the nil-sentinel conduit, " 

157 "which has no open verdict logbook. Set `trust_conduit_id` to " 

158 "SYSTEM_LOCAL_CONDUIT_ID (seeded by " 

159 "20260831140000_seed_local_zone_conduit_verdict_logbook.sql) to " 

160 "populate entries_conduit_verdicts. Authz decisions are still " 

161 "recorded in structured logs (trust_authorize.allow / " 

162 "trust_authorize.deny) and OTel spans. See memory " 

163 "project_authorization_envelope_design (watch item 6) + " 

164 "project_conduit_injection_design." 

165 ), 

166 )

169async def verify_local_conduit_seed_present(deps: Kernel) -> None: 

170 """Fail-fast at lifespan start when `trust_conduit_id` is configured 

171 but the Conduit it names cannot actually populate the verdict log. 

172 

173 Same failure this whole area keeps re-learning: asking for an audit 

174 control and silently not getting one. Two ways that happens here — 

175 the Conduit stream is missing, or it exists with no open `verdict` 

176 logbook — and both are checked. No-op when `trust_conduit_id` is 

177 unset (today's default; every deployment behaves exactly as before). 

178 """ 

179 settings = deps.settings 

180 if settings.trust_conduit_id is None: 

181 return 

182 

183 conduit = await load_conduit(deps.event_store, settings.trust_conduit_id) 

184 if conduit is None: 

185 hint = ( 

186 "Re-run `make migrate-apply` — " 

187 "20260831140000_seed_local_zone_conduit_verdict_logbook.sql is " 

188 "idempotent (ON CONFLICT DO NOTHING) and safe to re-apply." 

189 if settings.trust_conduit_id == SYSTEM_LOCAL_CONDUIT_ID 

190 else "A custom Conduit must be defined via `POST /conduits` first." 

191 ) 

192 msg = ( 

193 f"trust_conduit_id={settings.trust_conduit_id} is configured but " 

194 f"no Conduit stream exists at that id. {hint}" 

195 ) 

196 raise RuntimeError(msg) 

197 

198 if conduit.logbooks.get(LOGBOOK_KIND_VERDICT) is None: 

199 msg = ( 

200 f"trust_conduit_id={settings.trust_conduit_id} is configured and " 

201 "the Conduit exists, but it has no open verdict logbook, so " 

202 "TrustAuthorize would resolve to this conduit and still write " 

203 "nothing to entries_conduit_verdicts. `define_conduit` opens one " 

204 "automatically for a Conduit created through the API; a hand-" 

205 "seeded Conduit must include a ConduitLogbookOpened(kind=" 

206 f"{LOGBOOK_KIND_VERDICT!r}) event." 

207 ) 

208 raise RuntimeError(msg)

211async def verify_local_conduit_matches_policy(deps: Kernel) -> None: 

212 """Fail-fast when `trust_conduit_id` and `trust_policy_id` are both 

213 set but the configured Policy governs a different Conduit. 

214 

215 `Policy.evaluate` checks conduit before surface or principal 

216 (`aggregates/policy/state.py`), so a mismatch here denies EVERY 

217 command the instant `trust_conduit_id` starts being resolved — not a 

218 narrowing of what the policy permits, a silent lockout of the whole 

219 deployment. Existing nil-bound bootstrap/operator policies keep 

220 working unmigrated as long as this stays unset; this guard is what 

221 makes leaving them unmigrated safe rather than an oversight. 

222 """ 

223 settings = deps.settings 

224 if settings.trust_conduit_id is None or settings.trust_policy_id is None: 

225 return 

226 

227 policy = await load_policy(deps.event_store, settings.trust_policy_id) 

228 if policy is None: 

229 # verify_bootstrap_seed_present (or an operator's own responsibility 

230 # for a custom policy) already covers a missing policy stream. 

231 return 

232 

233 if policy.conduit_id != settings.trust_conduit_id: 

234 msg = ( 

235 f"trust_conduit_id={settings.trust_conduit_id} is configured but " 

236 f"trust_policy_id={settings.trust_policy_id} governs conduit " 

237 f"{policy.conduit_id}, a different one. Every command would be " 

238 "denied at the conduit check, before principal or command are " 

239 "even consulted. Re-define the policy bound to " 

240 f"{settings.trust_conduit_id}, or point trust_conduit_id at " 

241 f"{policy.conduit_id} instead." 

242 ) 

243 raise RuntimeError(msg)

246async def verify_in_process_policy_matches_surface(deps: Kernel) -> None: 

247 """Fail-fast when `trust_in_process_policy_id` is configured but the 

248 Policy it names does not actually govern the in-process door. 

249 

250 A policy configured for the backdoor but bound to the wrong Surface — 

251 or, once `trust_conduit_id` is also set, the wrong Conduit — would 

252 still be reachable from `_effective_policy_id` at every in-process 

253 call, but `evaluate` would strict-deny every one of them at the 

254 surface (or conduit) check, indistinguishable from having no backdoor 

255 policy configured at all. This is the same "looks wired, governs 

256 nothing" failure `verify_local_conduit_matches_policy` closes for the 

257 conduit knob, applied to the second policy slot. 

258 """ 

259 settings = deps.settings 

260 if settings.trust_in_process_policy_id is None: 

261 return 

262 

263 policy = await load_policy(deps.event_store, settings.trust_in_process_policy_id) 

264 if policy is None: 

265 # A missing policy stream is the operator's responsibility to 

266 # define (this policy has no bootstrap seed); not this guard's 

267 # concern, same reasoning as verify_local_conduit_matches_policy. 

268 return 

269 

270 if policy.surface_id != SYSTEM_IN_PROCESS_SURFACE_ID: 

271 msg = ( 

272 f"trust_in_process_policy_id={settings.trust_in_process_policy_id} " 

273 f"is configured but governs surface {policy.surface_id}, not " 

274 f"SYSTEM_IN_PROCESS_SURFACE_ID ({SYSTEM_IN_PROCESS_SURFACE_ID}). " 

275 "Every in-process call would be denied at the surface check " 

276 "instead of reaching this policy's principal/command rules. " 

277 "Re-define the policy bound to the in-process Surface, or point " 

278 "trust_in_process_policy_id at one that is." 

279 ) 

280 raise RuntimeError(msg) 

281 

282 if settings.trust_conduit_id is not None and policy.conduit_id != settings.trust_conduit_id: 

283 msg = ( 

284 f"trust_conduit_id={settings.trust_conduit_id} is configured but " 

285 f"trust_in_process_policy_id={settings.trust_in_process_policy_id} " 

286 f"governs conduit {policy.conduit_id}, a different one. Every " 

287 "in-process command would be denied at the conduit check. " 

288 "Re-define the backdoor policy bound to " 

289 f"{settings.trust_conduit_id}, or point trust_conduit_id at " 

290 f"{policy.conduit_id} instead." 

291 ) 

292 raise RuntimeError(msg)
