28# deployments set `TRUST_POLICY_ID=00000000-0000-0000-0000-000000000002` 

29# to point at it; the seed permits SYSTEM_PRINCIPAL_ID to call

50async def verify_bootstrap_seed_present(deps: Kernel) -> None: 

51 """Fail-fast at lifespan start when the configured bootstrap seed 

52 stream — or its dependencies — is missing. 

53 

54 When `trust_policy_id == SYSTEM_BOOTSTRAP_POLICY_ID`, verify the 

55 bootstrap policy stream and the 3 seeded Surface streams exist, and 

56 that the policy folded to the HTTP Surface. The policy references the 

57 HTTP Surface; without the Surface streams, evaluate would deny on a 

58 phantom surface_id and the verdict audit log would silently skip 

59 entries (partial-fail mitigation). Otherwise no-op: custom operator 

60 policies are the operator's responsibility to verify. 

61 

62 The retired nil-surface bootstrap policy (...0001) is not handled 

63 here. Its evaluate-time wildcard fold was removed, so it strict-denies 

64 every real-surface call; a deployment still pointed at it would be 

65 locked out. Point TRUST_POLICY_ID at the surface-bound id. 

66 """ 

67 settings = deps.settings 

68 

69 if settings.trust_policy_id != SYSTEM_BOOTSTRAP_POLICY_ID: 

70 return 

71 

72 policy = await load_policy(deps.event_store, SYSTEM_BOOTSTRAP_POLICY_ID) 

73 if policy is None: 

74 msg = ( 

75 f"Configured trust_policy_id={SYSTEM_BOOTSTRAP_POLICY_ID} " 

76 "(bootstrap policy) but the seed stream is missing from the " 

77 "event store. Re-run `make migrate-apply` — the seed migration " 

78 "20260519200000_seed_default_surfaces_and_v2_policy.sql is " 

79 "idempotent (ON CONFLICT DO NOTHING) and safe to re-apply. " 

80 "See memory/project_conduit_injection_design.md." 

81 ) 

82 raise RuntimeError(msg) 

83 

84 # The bootstrap policy binds to SYSTEM_HTTP_SURFACE_ID; without all 3 

85 # seeded Surfaces present, the audit / authz substrate is broken 

86 # (partial-fail). 

87 for surface_id in ( 

88 SYSTEM_HTTP_SURFACE_ID, 

89 SYSTEM_MCP_STDIO_SURFACE_ID, 

90 SYSTEM_MCP_STREAMABLE_HTTP_SURFACE_ID, 

91 ): 

92 surface = await load_surface(deps.event_store, surface_id) 

93 if surface is None: 

94 msg = ( 

95 f"trust_policy_id={SYSTEM_BOOTSTRAP_POLICY_ID} is configured " 

96 f"but seeded Surface {surface_id} is missing from the event " 

97 "store — the policy references the HTTP Surface and the audit " 

98 "path expects all 3. Re-run `make migrate-apply`; the seed " 

99 "migration is idempotent." 

100 ) 

101 raise RuntimeError(msg) 

102 

103 # Assert the bootstrap policy's surface_id binding is correct. A 

104 # typo'd / mis-wired seed (folded surface_id != HTTP) would silently 

105 # deny every request, since evaluate strict-matches the surface. 

106 if policy.surface_id != SYSTEM_HTTP_SURFACE_ID: 

107 msg = ( 

108 f"trust_policy_id={SYSTEM_BOOTSTRAP_POLICY_ID} was loaded but " 

109 f"folded with surface_id={policy.surface_id} instead of the " 

110 f"expected SYSTEM_HTTP_SURFACE_ID ({SYSTEM_HTTP_SURFACE_ID}). The " 

111 "seed migration may have been mutated post-seed, or a non-seed " 

112 "PolicyDefined event was appended to this stream. Investigate the " 

113 "event log." 

114 ) 

115 raise RuntimeError(msg)

118async def warn_if_verdict_log_dormant(deps: Kernel) -> None: 

119 """Warn (loudly, once at boot) when authz is ENFORCED but the 

120 per-Conduit Verdict audit log cannot populate. 

121 

122 `TrustAuthorize` writes a Verdict row per decision only when the 

123 conduit a command flows through has an open verdict logbook. Handlers 

124 currently route every command through the nil-sentinel conduit 

125 (surface / conduit injection is not wired yet), and no verdict 

126 logbook is seeded there, so the audit log silently stays empty even 

127 with enforcement on. Surfacing it here turns a silent gap into a 

128 boot-time heads-up rather than a discovery during a compliance audit. 

129 Authz decisions are still captured in structured logs 

130 (`trust_authorize.allow` / `trust_authorize.deny`) and OTel spans. 

131 

132 Non-fatal by design: a known-limitation notice, not a misconfig. When 

133 `trust_policy_id` is unset (AllowAll) there are no decisions to record 

134 and no warning is emitted. 

135 """ 

136 settings = deps.settings 

137 if settings.trust_policy_id is None: 

138 return 

139 

140 conduit = await load_conduit(deps.event_store, NIL_SENTINEL_ID) 

141 if conduit is not None and conduit.logbooks.get(LOGBOOK_KIND_VERDICT) is not None: 

142 return 

143 

144 _log.warning( 

145 "trust_authorize.verdict_log_dormant", 

146 trust_policy_id=str(settings.trust_policy_id), 

147 conduit_id=str(NIL_SENTINEL_ID), 

148 detail=( 

149 "Authorization is ENFORCED but the per-Conduit Verdict audit log " 

150 "will NOT populate: handlers route through the nil-sentinel " 

151 "conduit, which has no open verdict logbook (conduit injection is " 

152 "not wired yet). Authz decisions are still recorded in structured " 

153 "logs (trust_authorize.allow / trust_authorize.deny) and OTel " 

154 "spans. See memory project_authorization_envelope_design (watch " 

155 "item 6) + project_conduit_injection_design." 

156 ), 

157 )
