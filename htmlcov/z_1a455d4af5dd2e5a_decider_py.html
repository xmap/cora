1"""Pure decider for the `RegisterDataset` command. 

2 

3Pure function: given the (always None) Dataset state, a 

4`RegisterDataset` command, and a pre-loaded 

5`DatasetRegistrationContext`, returns the events to append. No 

6I/O, no awaits, no side effects. 

7 

8`now` and `new_id` are injected by the application handler from 

9the Clock and IdGenerator ports. 

10 

11## Cross-aggregate validation (gate-review Q2 lock B + 7b status check) 

12 

13Existence-only checks for Run + Subject; existence + non-Discarded 

14status check for derived_from sources. The decider trusts the 

15handler's loads. Specifically: 

16 

17 - If `command.producing_run_id` is set, `context.producing_run` 

18 must be non-None (handler raises `ProducingRunNotFoundError` 

19 upstream if the Run doesn't exist; this branch validates the 

20 handler's contract was honoured). 

21 - If `command.subject_id` is set, `context.subject` must be 

22 non-None (same shape). 

23 - For each id in `command.derived_from`, `context.derived_from` 

24 must contain it AND its status must NOT be `Discarded` (we 

25 don't allow new lineage edges into bytes that no longer exist). 

26 

27No status checks for Run / Subject (Datasets register against any 

28Run state, against any Subject state). Discarded-status check only 

29applies to derived_from lineage sources (7b tightening). 

30 

31Per the precedent set by `start_run`'s decider, defensive guards 

32on context.* alignment are assertions of the handler's contract, 

33not user-facing validations. If the handler is correct, they 

34never fire. 

35 

36## VO trim semantics 

37 

38Field VOs (`DatasetName`, `DatasetUri`, `DatasetChecksum`, 

39`DatasetEncoding`) handle their own trimming + validation in 

40`__post_init__`; the on-the-wire payload carries the trimmed 

41values (so the persisted event payload is canonical). 

42""" 

43 

44from datetime import datetime 

45from uuid import UUID 

46 

47from cora.data.aggregates.dataset import ( 

48 DatasetAlreadyExistsError, 

49 DatasetChecksum, 

50 DatasetEncoding, 

51 DatasetName, 

52 DatasetRegistered, 

53 DatasetStatus, 

54 DatasetUri, 

55 DerivedFromDatasetsDiscardedError, 

56 DerivedFromDatasetsNotFoundError, 

57 LinkedSubjectNotFoundError, 

58 ProducingProcedureNotFoundError, 

59 ProducingProcedureNotTerminalError, 

60 ProducingRunNotFoundError, 

61 validate_byte_size, 

62 validate_derived_from, 

63 validate_used_calibration_ids, 

64) 

65from cora.data.aggregates.dataset.state import Dataset 

66from cora.data.features.register_dataset.command import RegisterDataset 

67from cora.data.features.register_dataset.context import DatasetRegistrationContext 

68from cora.shared.identity import ActorId 

69 

70 

71def decide( 

72 state: Dataset | None, 

73 command: RegisterDataset, 

74 *, 

75 context: DatasetRegistrationContext, 

76 now: datetime, 

77 new_id: UUID, 

78 registered_by: ActorId, 

79) -> list[DatasetRegistered]: 

80 """Decide the events produced by registering a new Dataset. 

81 

82 Invariants: 

83 - State must be None (genesis-only) 

84 -> DatasetAlreadyExistsError 

85 - Name must be valid -> InvalidDatasetNameError 

86 (via DatasetName VO) 

87 - URI must be valid -> InvalidDatasetUriError (via DatasetUri VO) 

88 - Checksum must be valid -> InvalidDatasetChecksumError 

89 (via DatasetChecksum VO) 

90 - byte_size must be valid -> InvalidDatasetByteSizeError 

91 (via validate_byte_size) 

92 - Encoding (media_type + conforms_to) must be valid 

93 -> InvalidDatasetEncodingError (via DatasetEncoding VO) 

94 - derived_from must satisfy cardinality + UUID shape 

95 -> InvalidDerivedFromError (via validate_derived_from) 

96 - used_calibration_ids must satisfy cardinality 

97 -> InvalidUsedCalibrationsError (via validate_used_calibration_ids) 

98 - When producing_run_id is set, the Run must exist 

99 -> ProducingRunNotFoundError 

100 - When producing_procedure_id is set, the Procedure must exist 

101 -> ProducingProcedureNotFoundError, and must be terminal 

102 -> ProducingProcedureNotTerminalError (its terminal actuation_kind 

103 is derived into producing_actuation_kind, the promote-gate carrier) 

104 - When subject_id is set, the Subject must exist 

105 -> LinkedSubjectNotFoundError 

106 - All derived_from Datasets must exist 

107 -> DerivedFromDatasetsNotFoundError 

108 - No derived_from Dataset may be Discarded 

109 -> DerivedFromDatasetsDiscardedError 

110 """ 

111 if state is not None: 

112 raise DatasetAlreadyExistsError(state.id) 

113 

114 # Field-level validation via VOs (trims + bounded-length checks 

115 # + encoding-specific invariants). Each VO raises its specific 

116 # error class on failure, mapped to HTTP 400. 

117 name = DatasetName(command.name) 

118 uri = DatasetUri(command.uri) 

119 checksum = DatasetChecksum( 

120 algorithm=command.checksum_algorithm, 

121 value=command.checksum_value, 

122 ) 

123 byte_size = validate_byte_size(command.byte_size) 

124 encoding = DatasetEncoding( 

125 media_type=command.media_type, 

126 conforms_to=command.conforms_to, 

127 ) 

128 derived_from = validate_derived_from(command.derived_from) 

129 # cardinality-only check on the AsShot citation set. 

130 # NO cross-BC existence check (revision-cited atomic-ID model; 

131 # eventual-consistency stance per [[project_calibration_design]] 

132 # anti-hook #3; mirrors Run.pinned_calibration_ids decider-time 

133 # treatment exactly). 

134 used_calibration_ids = validate_used_calibration_ids(command.used_calibration_ids) 

135 

136 # Cross-aggregate checks (existence-only per Q2 lock B). 

137 # The handler's pre-loads either populate context.* or raise 

138 # the not-found error before we get here; these branches are 

139 # the decider-level statement of the contract. 

140 if command.producing_run_id is not None and context.producing_run is None: 

141 raise ProducingRunNotFoundError(command.producing_run_id) 

142 if command.producing_procedure_id is not None and context.producing_procedure is None: 

143 raise ProducingProcedureNotFoundError(command.producing_procedure_id) 

144 # The producing Procedure must be terminal so its actuation kind is final 

145 # at snapshot time (option A of the item-6 None-tightening; see 

146 # [[project_actuation_kind_stage1_design]]). A still-Running Procedure has 

147 # no resolved kind, and a stale-None snapshot would later be wrongly 

148 # blocked at promote. 

149 if ( 

150 context.producing_procedure is not None 

151 and not context.producing_procedure.status.is_terminal 

152 ): 

153 raise ProducingProcedureNotTerminalError( 

154 context.producing_procedure.id, 

155 current_status=context.producing_procedure.status.value, 

156 ) 

157 if command.subject_id is not None and context.subject is None: 

158 raise LinkedSubjectNotFoundError(command.subject_id) 

159 missing_derived = sorted( 

160 (d for d in derived_from if d not in context.derived_from), 

161 key=str, 

162 ) 

163 if missing_derived: 

164 raise DerivedFromDatasetsNotFoundError(missing_derived) 

165 # context.derived_from is built ONLY from command.derived_from 

166 # (see register_dataset/handler.py); every key here is in 

167 # derived_from by construction. We don't re-filter on `d in 

168 # derived_from`. 

169 discarded_derived = sorted( 

170 ( 

171 d 

172 for d, loaded in context.derived_from.items() 

173 if loaded.status is DatasetStatus.DISCARDED 

174 ), 

175 key=str, 

176 ) 

177 if discarded_derived: 

178 raise DerivedFromDatasetsDiscardedError(discarded_derived) 

179 

180 # capture producing Run's terminal status at registration 

181 # (per non-determinism principle: capture, don't recompute). The 

182 # Run is in some end state by the time a Dataset references it 

183 # — production scenario expects Completed, but the Run may be 

184 # Running / Held / Aborted / Stopped / Truncated at register time 

185 # because in-situ measurements register Datasets while the Run 

186 # is still active. The captured value powers promote_dataset's 

187 # Run-must-be-Completed guard later. 

188 producing_run_end_state: str | None = ( 

189 context.producing_run.status.value if context.producing_run is not None else None 

190 ) 

191 

192 # derive the actuation-kind provenance from the producing Procedure's 

193 # terminal state (per non-determinism principle: capture, don't recompute; 

194 # mirrors producing_run_end_state). The Conductor recorded the observed 

195 # kind on the Procedure terminal event; here it is snapshotted onto the 

196 # Dataset, where promote_dataset's guard blocks Simulated / Hybrid. None 

197 # when no producing Procedure (external / non-conducted Datasets) or when 

198 # the conduct observed nothing. Server-derived, never caller-asserted. 

199 producing_actuation_kind: str | None = ( 

200 context.producing_procedure.actuation_kind 

201 if context.producing_procedure is not None 

202 else None 

203 ) 

204 

205 return [ 

206 DatasetRegistered( 

207 dataset_id=new_id, 

208 name=name.value, 

209 uri=uri.value, 

210 checksum_algorithm=checksum.algorithm, 

211 checksum_value=checksum.value, 

212 byte_size=byte_size, 

213 media_type=encoding.media_type, 

214 conforms_to=encoding.conforms_to, 

215 producing_run_id=command.producing_run_id, 

216 producing_procedure_id=command.producing_procedure_id, 

217 subject_id=command.subject_id, 

218 derived_from=derived_from, 

219 occurred_at=now, 

220 registered_by=registered_by, 

221 producing_run_end_state=producing_run_end_state, 

222 # raw ActuationKind value derived server-side from the producing 

223 # Procedure (None for non-conducted registrations); the promote 

224 # gate blocks Simulated / Hybrid. 

225 producing_actuation_kind=producing_actuation_kind, 

226 # intent defaults to "Trial" on the dataclass; promotion is a 

227 # separate explicit slice (promote_dataset). 

228 # sort before emit so the event-payload bytes are 

229 # deterministic (matches Run.pinned_calibration_ids 

230 # decider-time treatment + the derived_from sorted-list 

231 # precedent on the same payload). 

232 used_calibration_ids=tuple(sorted(used_calibration_ids)), 

233 ) 

234 ]
