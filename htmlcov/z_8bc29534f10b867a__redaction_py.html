1"""Redact a whole `ExportedRecord`: the fail-closed switch plus both tiers. 

2 

3Per `project_record_export_v3.md` F5's fail-closed switch: the exporter 

4refuses to produce a bundle marked publishable unless a redaction 

5profile is supplied and its hash matches. `redact_record` computes 

6`hash_redaction_profile()` itself and compares against the caller- 

7supplied `expected_redaction_profile_hash` BEFORE touching a single row, 

8so a stale or substituted disposition table aborts rather than silently 

9redacting under the wrong rules. 

10"""

33class RedactionProfileMismatchError(RuntimeError): 

34 """`expected_redaction_profile_hash` does not match the disposition 

35 table this checkout would actually redact with. 

36 

37 Refuses rather than redacting under an unverified or stale table: 

38 the exact failure mode the fail-closed switch exists to catch. 

39 """ 

40 

41 def __init__(self, expected: str, actual: str) -> None: 

42 super().__init__( 

43 f"redaction profile mismatch: expected {expected!r}, this checkout's " 

44 f"disposition table hashes to {actual!r}. Refusing to redact." 

45 ) 

46 self.expected = expected 

47 self.actual = actual

59@dataclass(frozen=True, slots=True) 

60class RedactionResult: 

61 """`redacted_record` is safe to hash as the published record. 

62 `token_map` is an artifact of THIS export, retained separately under 

63 H1's obligation; it must never be shipped or fed into anything 

64 hashed as the published record. 

65 

66 `unfired_tier2_clearances` names every declared jsonb clearance 

67 (`kind`, `column`, `pointer`) that never matched a row in a kind 

68 THIS export actually carried. It is a completeness fact about the 

69 export, not a safety finding: tier 2 is an allowlist, so a 

70 clearance not firing means a field was published less often than 

71 permitted, never more. Callers pass it to `build_manifest` so a 

72 reader can see, from the artifact itself, which parts of the 

73 redaction profile this particular export was too narrow to 

74 exercise. 

75 

76 `unfired_tier1_fields` is the same completeness fact one tier up: 

77 every `(event_type, field)` pair DECLARED in `DISPOSITIONS` for an 

78 event type THIS export carried, whose field never actually appeared 

79 on any row of that type. Unlike tier 2's hand-curated allowlist, 

80 tier 1 is exhaustively generated by Step 0's generator, so most of 

81 the time every declared field fires (a payload key is present, even 

82 as `null`, on every schema version that still declares it); the case 

83 this catches is an older `schema_version` row whose payload predates 

84 a field the CURRENT dataclass declares. This is a narrowness caveat, 

85 not a leak: a field that never fired was never published either way. 

86 A rule that is dead in EVERY export (one whose declared key can never 

87 match anything the store actually writes, as opposed to one export's 

88 narrow coverage) is a different problem -- a mismatch between what 

89 the generator resolved from the dataclass and what `to_payload` 

90 actually serializes -- and needs a build-time check comparing the two 

91 directly; this per-export field cannot see that case and does not 

92 claim to. 

93 """ 

94 

95 redacted_record: RedactedRecord 

96 token_map: TokenMap

101def redact_record( 

102 record: ExportedRecord, *, expected_redaction_profile_hash: str 

103) -> RedactionResult: 

104 """Redact both tiers of `record` under one shared `TokenMap`. 

105 

106 Raises `RedactionProfileMismatchError` before touching any row if 

107 the hash does not match; `UnknownEventTypeError` if a stream row's 

108 `event_type` has no entry in the disposition table at all. Does NOT 

109 raise over an unfired tier-2 clearance; see 

110 `RedactionResult.unfired_tier2_clearances` and 

111 `unfired_clearances`'s own docstring for why an earlier version of 

112 this function did and was wrong to. 

113 """ 

114 actual_hash = hash_redaction_profile() 

115 if expected_redaction_profile_hash != actual_hash: 

116 raise RedactionProfileMismatchError(expected_redaction_profile_hash, actual_hash) 

117 

118 token_map = TokenMap() 

119 

120 tier1 = Tier1Redactor(token_map) 

121 redacted_streams = tuple(tier1.redact_row(row) for row in record.streams) 

122 

123 fired_pointers: dict[tuple[str, str], set[str]] = {} 

124 redacted_logbooks = { 

125 kind: tuple( 

126 redact_tier2_row(kind, row, token_map=token_map, fired_pointers=fired_pointers) 

127 for row in rows 

128 ) 

129 for kind, rows in record.logbooks.items() 

130 } 

131 unfired_tier2 = unfired_clearances(fired_pointers, kinds_present=frozenset(record.logbooks)) 

132 

133 event_types_present = {str(row["event_type"]) for row in record.streams} 

134 fired_fields = tier1.fired_fields 

135 unfired_tier1 = frozenset( 

136 (event_type, field) 

137 for event_type in event_types_present 

138 # `tier1.redact_row` above already raised UnknownEventTypeError for 

139 # any event_type not in DISPOSITIONS, so every member of 

140 # event_types_present is guaranteed present here. 

141 for field in DISPOSITIONS[event_type] 

142 if field not in fired_fields.get(event_type, frozenset()) 

143 ) 

144 

145 return RedactionResult( 

146 redacted_record=RedactedRecord(streams=redacted_streams, logbooks=redacted_logbooks), 

147 token_map=token_map, 

148 unfired_tier2_clearances=unfired_tier2, 

149 unfired_tier1_fields=unfired_tier1, 

150 )
