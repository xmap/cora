1"""RFC 9728 Protected Resource Metadata endpoint. 

2 

3MCP spec 2025-11-25 (basic/authorization) mandates that OAuth-2.1 

4resource servers expose a Protected Resource Metadata document at 

5`/.well-known/oauth-protected-resource`. Clients GET this to 

6discover which authorization servers issue tokens for which 

7resources (per-Surface audience identifiers), what scopes are 

8supported, and what token formats are accepted. 

9 

10CORA's RFC 9728 document is generated from 

11`Settings.identity_providers` + the 3 SYSTEM Surface audience 

12identifiers — no DB, no auth, just config-driven JSON. 

13 

14The 401 `WWW-Authenticate` header that the edge middleware returns 

15includes `resource_metadata="https://<deployment>/.well-known/oauth-protected-resource"` 

16so clients can fetch this document programmatically on first-401 

17and discover where to obtain a fresh token. 

18 

19## What's NOT here 

20 

21 - The `authorization_servers_metadata` linked documents — those 

22 live at the IdP's `/.well-known/oauth-authorization-server`, 

23 which IS the IdP's responsibility per RFC 8414. CORA's RFC 9728 

24 only POINTS to them via `authorization_servers: [<issuer URL>, 

25 ...]`. 

26 - Per-tenant fan-out (different Surface IDs per tenant) — deferred 

27 per WI10 of the design lock. 

28"""

43def build_protected_resource_metadata( 

44 *, 

45 resource: str, 

46 identity_providers: list[IdentityProviderConfig], 

47 surface_audiences: dict[str, str | None], 

48) -> dict[str, Any]: 

49 """Build the RFC 9728 document body. 

50 

51 `resource` is the resource server's stable identifier — for 

52 CORA, the deployment's base URL (for example, 

53 `https://aps-2bm.cora.example`). Surfaces are exposed as 

54 distinct sub-resources via `resource_documentation` / 

55 `signed_metadata` extensions; the per-Surface audience strings 

56 live in `aud_values_supported` so clients know which to request 

57 when calling each Surface. 

58 

59 `identity_providers` is the `Settings.identity_providers` list. 

60 Issuers are exposed via the standard `authorization_servers` 

61 array; clients dereference each to its `/.well-known/oauth- 

62 authorization-server` (RFC 8414). 

63 

64 `surface_audiences` maps the human-readable Surface name 

65 (`"http"`, `"mcp_stdio"`, `"mcp_streamable_http"`) to the 

66 configured audience string (None if the deployment hasn't 

67 registered any IdP for that Surface). 

68 """ 

69 issuers = sorted({str(idp.issuer) for idp in identity_providers}) 

70 aud_values = [v for v in surface_audiences.values() if v] 

71 document: dict[str, Any] = { 

72 "resource": resource, 

73 "authorization_servers": issuers, 

74 "bearer_methods_supported": ["header"], 

75 } 

76 if aud_values: 

77 document["aud_values_supported"] = sorted(set(aud_values)) 

78 # Per-Surface audience map as a non-standard but namespaced CORA 

79 # extension. The `x-cora-` prefix was considered first; per 

80 # RFC 6648 (2012) the `X-`/`x-` convention is deprecated across 

81 # IETF protocols and RFC 9728 itself doesn't reserve `x-` keys. 

82 # Reverse- 

83 # DNS namespacing (`io.cora.surface_audiences`) matches the JSON- 

84 # Schema / OAuth-2.0 extension conventions and avoids both the 

85 # deprecated prefix and accidental collision with future RFC 9728 

86 # registered keys. 

87 document["io.cora.surface_audiences"] = { 

88 name: aud for name, aud in surface_audiences.items() if aud is not None 

89 } 

90 return document

93def register_protected_resource_metadata_route(app: FastAPI) -> None: 

94 """Mount `GET /.well-known/oauth-protected-resource` on the app. 

95 

96 The handler reads `app.state.deps.settings.identity_providers` 

97 at request time so re-deploys with a new identity-provider list 

98 take effect on next request (no app restart needed for config 

99 rotation — provided the lifespan re-loaded Settings, which it 

100 doesn't today; this is a forward-looking handler shape). 

101 """ 

102 

103 @app.get( 

104 "/.well-known/oauth-protected-resource", 

105 tags=["edge-auth"], 

106 summary="RFC 9728 Protected Resource Metadata", 

107 ) 

108 async def protected_resource_metadata( # pyright: ignore[reportUnusedFunction] 

109 request: Request, 

110 ) -> JSONResponse: 

111 settings = request.app.state.deps.settings 

112 # Build a per-Surface audience map by inverting the 

113 # identity_providers list. When multiple IdPs declare the 

114 # same Surface, the audience strings must agree (it's the 

115 # SAME resource URL) so the values collapse cleanly. 

116 surface_name_by_uuid = { 

117 SYSTEM_HTTP_SURFACE_ID: "http", 

118 SYSTEM_MCP_STDIO_SURFACE_ID: "mcp_stdio", 

119 SYSTEM_MCP_STREAMABLE_HTTP_SURFACE_ID: "mcp_streamable_http", 

120 } 

121 surface_audiences: dict[str, str | None] = { 

122 name: None for name in surface_name_by_uuid.values() 

123 } 

124 for idp in settings.identity_providers: 

125 for surface_uuid, aud_str in idp.audiences.items(): 

126 name = surface_name_by_uuid.get(surface_uuid) 

127 if name is not None: 127 ↛ 125line 127 didn't jump to line 125 because the condition on line 127 was always true

128 surface_audiences[name] = aud_str 

129 

130 # Resource identifier: scheme + host of the inbound request, 

131 # no path. Per RFC 9728 §3.1 the resource value SHOULD be the 

132 # canonical URL of the resource server. Honor standard reverse- 

133 # proxy headers (X-Forwarded-Proto + X-Forwarded-Host) because 

134 # production CORA always sits behind one (Cloudflare / nginx / 

135 # IAP) — without this, the `resource` field reads 

136 # `http://internal-pod-name:8000` instead of the public URL 

137 # and clients can't discover the auth flow correctly. 

138 # Gate-review test#6 + security F4. 

139 scheme = request.headers.get("x-forwarded-proto", request.url.scheme) 

140 host = request.headers.get("x-forwarded-host", request.url.netloc) 

141 resource = f"{scheme}://{host}" 

142 

143 document = build_protected_resource_metadata( 

144 resource=resource, 

145 identity_providers=list(settings.identity_providers), 

146 surface_audiences=surface_audiences, 

147 ) 

148 # Cache-Control: clients SHOULD cache for the audience- 

149 # rotation window; 5 minutes balances "operator rotates IdPs" 

150 # vs "100 MCP clients hammering /.well-known/" on cold start. 

151 return JSONResponse( 

152 content=document, 

153 headers={"Cache-Control": "public, max-age=300"}, 

154 )
