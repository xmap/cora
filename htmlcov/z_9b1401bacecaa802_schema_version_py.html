1"""Refuse to serve a database whose shape this build does not expect. 

2 

3Migrations are applied out of band (`make migrate-apply`), never by the 

4app: `apps/api/Dockerfile` says so, and forward-only migrations mean the 

5image carries no way back. That separation is right, and it leaves one 

6window open. Applying migrations and starting the process are two 

7independent acts, so nothing except sequencing keeps them in step, and a 

8restore breaks the sequence by construction: the database returns to the 

9shape it had when the backup was taken while the image stays current. 

10 

11`scripts/restore_drill.py` already reads the applied version out of 

12Atlas's own bookkeeping to prove a restore brought the schema back with 

13the data. Its docstring is candid that the drill migrates before it backs 

14up, so it cannot construct the stale-schema case, and that the only guard 

15for that case is an operator remembering to run `make migrate-status` 

16afterwards. This module is that guard as a mechanism instead. 

17 

18## Why refusing is the proportionate response 

19 

20An event store is append-only at the role level 

21(`project_immutability_guarantee`), so events written against the wrong 

22schema are not rows to correct later, they are history. The failure is 

23also not reliably loud: a mismatch that DROPS a constraint added by a 

24later migration leaves every write succeeding and admits exactly the 

25records the constraint existed to reject. Crashing on a missing column is 

26the lucky outcome. Refusing to start converts both into a five-second 

27message. 

28 

29## Direction decides the remedy, so it decides the message 

30 

31Behind and ahead are not symmetric, and a check that only reported 

32"mismatch" would send an operator down the wrong path half the time. 

33Behind is fixed by applying migrations. Ahead cannot be fixed that way at 

34all, because forward-only means there is nothing to apply, so the only 

35move is to run the image that matches. Each case names its own remedy. 

36"""

52SchemaPosture = Literal["matched", "degraded"] 

53"""Whether the applied schema is the one this build expects. 

54 

55`degraded` is only reachable through an explicit operator override and 

56means the process booted against a mismatched schema with writes 

57refused. There is no posture for "mismatched and writing"; that state is 

58what this module exists to prevent. 

59""" 

60 

61 

62@dataclass(frozen=True) 

63class SchemaCheck: 

64 """The outcome of a boot-time schema check. 

65 

66 Carries both versions, not just the posture, because the degraded 

67 path needs them downstream: `ReadOnlyEventStore` names both sides in 

68 its refusal, and a caller who hits that refusal is usually not the 

69 operator who set the override. 

70 """ 

71 

72 posture: SchemaPosture 

73 applied: str 

74 expected: str 

75 

76 

77EXPECTED_SCHEMA_VERSION: Final = "20260820120000" 

78"""The newest migration this build was written against. 

79 

80Hand-maintained, and deliberately not derived at runtime: the image does 

81not ship `infra/atlas/migrations` (see the Dockerfile), so there is 

82nothing on disk to read. `tests/architecture/test_schema_version_pin.py` 

83fails the build when this drifts from the newest tracked migration, which 

84puts the cost of forgetting on CI rather than on a beamline. 

85""" 

86 

87_VERSION_PATTERN: Final = re.compile(r"^\d{14}$") 

88"""Atlas's `YYYYMMDDHHMMSS`. Fixed width is what makes the ordering 

89comparison below a plain string compare; the architecture test pins the 

90format so that stays true.""" 

91 

92_APPLIED_VERSION_QUERY: Final = ( 

93 "SELECT coalesce(max(version), '') FROM atlas_schema_revisions.atlas_schema_revisions" 

94)

97class SchemaVersionError(RuntimeError): 

98 """The database's shape is not the one this build expects. 

99 

100 Subclasses carry the remedy. Callers catch this base: catching a 

101 subclass would silently stop covering a case when another is added. 

102 """ 

103 

104 

105class SchemaAbsentError(SchemaVersionError): 

106 """No migrations have been applied at all.""" 

107 

108 def __init__(self, expected: str) -> None: 

109 super().__init__( 

110 f"This database has no schema. CORA needs migration {expected}.\n" 

111 f"\n" 

112 f"Nothing has been applied here yet, so this is a new database " 

113 f"rather than a mismatched one.\n" 

114 f"\n" 

115 f"To fix: make migrate-apply" 

116 ) 

117 self.expected = expected 

118 

119 

120class SchemaBehindError(SchemaVersionError): 

121 """The database predates this build.""" 

122 

123 def __init__(self, applied: str, expected: str) -> None: 

124 super().__init__( 

125 f"CORA will not start: the database is older than this build.\n" 

126 f"\n" 

127 f" database is at {applied}\n" 

128 f" this build needs {expected}\n" 

129 f"\n" 

130 f"This is the normal state after restoring a backup, because a " 

131 f"restore returns the schema to its shape on the day the backup " 

132 f"was taken while this image stays current.\n" 

133 f"\n" 

134 f"To fix: make migrate-apply" 

135 ) 

136 self.applied = applied 

137 self.expected = expected 

138 

139 

140class SchemaAheadError(SchemaVersionError): 

141 """The database postdates this build, which migrating cannot fix.""" 

142 

143 def __init__(self, applied: str, expected: str) -> None: 

144 super().__init__( 

145 f"CORA will not start: the database is newer than this build.\n" 

146 f"\n" 

147 f" database is at {applied}\n" 

148 f" this build needs {expected}\n" 

149 f"\n" 

150 f"Applying migrations will NOT fix this. CORA's migrations are " 

151 f"forward-only, so there is nothing to apply and no way back. " 

152 f"Something started an older image than the one this database " 

153 f"has been migrated for.\n" 

154 f"\n" 

155 f"To fix: run the image built for {applied} or later." 

156 ) 

157 self.applied = applied 

158 self.expected = expected

161async def read_applied_version(pool: asyncpg.Pool) -> str | None: 

162 """Return the newest applied migration, or None if none have been. 

163 

164 None covers both shapes a never-migrated database takes: Atlas's 

165 schema absent entirely, and the table absent within it. Neither is an 

166 error here, because "brand new database" and "wrong database" want 

167 different words and the caller is the one that knows which it wants. 

168 

169 Every other failure propagates. A pool that cannot answer this query 

170 cannot serve traffic either, and swallowing that into a None would 

171 report a connectivity outage as a fresh install. 

172 """ 

173 try: 

174 applied: object = await pool.fetchval(_APPLIED_VERSION_QUERY) 

175 except (asyncpg.UndefinedTableError, asyncpg.InvalidSchemaNameError): 

176 return None 

177 return str(applied) if applied else None 

178

180def compare_versions(applied: str | None, expected: str) -> None: 

181 """Raise unless `applied` is the expected schema. Pure, so the unit 

182 tier can drive every branch without a database. 

183 

184 Deliberately reports no step count. Counting how far behind a 

185 database is would need the full list of versions, which the image 

186 does not carry (it ships no migrations directory), and baking 160 

187 strings into source to earn one sentence is a poor trade. The two 

188 timestamps are legible on their own: an operator can see at a glance 

189 whether they restored last Tuesday or last quarter. 

190 """ 

191 if applied is None: 

192 raise SchemaAbsentError(expected) 

193 if applied == expected: 

194 return 

195 if applied > expected: 

196 raise SchemaAheadError(applied, expected) 

197 raise SchemaBehindError(applied, expected)

200async def verify_schema_version( 

201 pool: asyncpg.Pool, 

202 *, 

203 allow_mismatch: bool = False, 

204 expected: str = EXPECTED_SCHEMA_VERSION, 

205) -> SchemaCheck: 

206 """Check the applied schema and report the resulting posture. 

207 

208 `allow_mismatch` downgrades a version mismatch from a refusal to a 

209 read-only boot, so a restored database can be inspected without 

210 risking the append-only history. It deliberately does NOT cover 

211 `SchemaAbsentError`: the override exists to read a database that has 

212 the wrong schema, and one with no schema has nothing to read. Letting 

213 it through would turn "I want to look at last week's data" into a 

214 process that boots against an empty database and reports it as fine. 

215 """ 

216 applied = await read_applied_version(pool) 

217 try: 

218 compare_versions(applied, expected) 

219 except SchemaAbsentError: 

220 raise 

221 except SchemaVersionError: 

222 if not allow_mismatch: 

223 raise 

224 # `applied` is never None here: the None case raised 

225 # SchemaAbsentError above and re-raised past this arm. 

226 return SchemaCheck("degraded", applied=applied or "", expected=expected) 

227 return SchemaCheck("matched", applied=expected, expected=expected)

230def parse_versions(migration_files: Sequence[Path]) -> tuple[str, ...]: 

231 """Version strings from migration filenames, in applied order. 

232 

233 Shared by the architecture test that pins `EXPECTED_SCHEMA_VERSION` 

234 and by the test fixture that records what Atlas would have recorded, 

235 so the two cannot disagree about what a version is. 

236 """ 

237 return tuple(sorted(path.name.split("_", 1)[0] for path in migration_files))

239 

240def is_well_formed(version: str) -> bool: 

241 """Whether a version is Atlas's fixed-width timestamp. 

242 

243 Fixed width is load-bearing: `compare_versions` orders with a string 

244 compare, which is only correct while every version has the same 

245 number of digits. 

246 """ 

247 return _VERSION_PATTERN.match(version) is not None
