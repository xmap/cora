1"""Pure decider for the `regenerate_run_debrief` slice. 

2 

3Composes the `DecisionRegistered` event for an on-demand RunDebriefer 

4invocation. Pure function: given the (always-None) Decision state, the 

5`RegenerateRunDebrief` command, and a handler-built 

6`RegenerateRunDebriefContext`, returns the event(s) to append. No I/O, 

7no awaits, no clock / id-generator calls. 

8 

9Invariants: 

10 - State is always None (genesis event, fresh Decision stream). 

11 - `context.actor.active` is True (handler raised 

12 AgentDeactivatedError otherwise). 

13 - `command.parent_decision_id` (when set) points at a real Decision 

14 in the same Run AND with `context = "RunDebrief"` (handler raised 

15 DecisionParentNotFoundError / DecisionParentRunMismatchError / 

16 DecisionParentAgentMismatchError otherwise). 

17 - `context.choice` is a valid `DecisionChoice` string (open-string VO 

18 raises `InvalidDecisionChoiceError` otherwise). 

19 - `context.reasoning` is non-empty + within length bound 

20 (`validate_reasoning` raises otherwise). 

21 - `context.confidence` is in [0.0, 1.0] when not None 

22 (`validate_confidence` raises otherwise). 

23 

24## Why a separate decider 

25 

26The cross-BC slice-contract test 

27(`tests/architecture/test_slice_contract.py`) requires `decider.py` 

28for every command slice. The LLM-response-to-`DecisionRegistered` 

29mapping is genuinely pure and benefits from extraction: future 

30on-demand agent slices and a Pattern B subscriber would both consume 

31the same shape. 

32 

33The Pattern A subscriber (`cora.agent.subscribers.run_debriefer`) 

34currently inlines an equivalent composition. Rule-of-three trigger 

35to hoist this decider out of the slice and into a shared module 

36fires when EITHER a second on-demand agent slice ships OR the 

37subscriber's composition is refactored to call through this decider. 

38""" 

39 

40from datetime import datetime 

41from typing import Any 

42from uuid import UUID 

43 

44from cora.agent.features.regenerate_run_debrief.command import RegenerateRunDebrief 

45from cora.agent.features.regenerate_run_debrief.context import RegenerateRunDebriefContext 

46from cora.agent.prompts import RUN_DEBRIEF_PROMPT_TEMPLATE_ID 

47from cora.decision.aggregates.decision import ( 

48 DECISION_CONTEXT_RUN_DEBRIEF, 

49 Decision, 

50 DecisionChoice, 

51 DecisionConfidenceSource, 

52 DecisionContext, 

53 DecisionRegistered, 

54 DecisionRule, 

55 validate_confidence, 

56 validate_inputs, 

57 validate_reasoning, 

58) 

59from cora.shared.identity import ActorId 

60 

61_DECISION_RULE = "agent:RunDebriefer:v1" 

62 

63 

64def decide( 

65 state: Decision | None, 

66 command: RegenerateRunDebrief, 

67 *, 

68 context: RegenerateRunDebriefContext, 

69 now: datetime, 

70 new_id: UUID, 

71) -> list[DecisionRegistered]: 

72 """Compose the on-demand RunDebriefer `DecisionRegistered` event. 

73 

74 `state` is always None (fresh Decision stream; the handler doesn't 

75 load Decision state). Accepted as a parameter for canonical 

76 signature parity with every other create-style decider. 

77 

78 `context.choice` is constrained at the projection layer (and by the 

79 LLM's structured output schema) to the closed 6-value set, but the 

80 Decision BC's open-string `DecisionChoice` VO is used here so the 

81 decider stays vocabulary-agnostic. 

82 

83 `context.extra_inputs` merges into the base inputs after 

84 the base keys are set; collisions on `run_id` / `trigger` / 

85 `prompt_template_id` are silently overwritten by the extra dict 

86 (intentional: callers supplying these keys override on purpose, 

87 eg. the DebriefDeferred path's `failure_error_class`). 

88 """ 

89 _ = state # always None for genesis; signature parity only. 

90 decision_choice = DecisionChoice(context.choice) 

91 decision_context = DecisionContext(DECISION_CONTEXT_RUN_DEBRIEF) 

92 rule = DecisionRule(_DECISION_RULE) 

93 base_inputs: dict[str, Any] = { 

94 "run_id": str(command.run_id), 

95 "trigger": "on-demand", 

96 "prompt_template_id": str(RUN_DEBRIEF_PROMPT_TEMPLATE_ID), 

97 } 

98 if context.extra_inputs: 

99 base_inputs.update(context.extra_inputs) 

100 inputs = validate_inputs(base_inputs) 

101 validated_reasoning = validate_reasoning(context.reasoning) 

102 validated_confidence = validate_confidence(context.confidence) 

103 

104 return [ 

105 DecisionRegistered( 

106 decision_id=new_id, 

107 decided_by=ActorId(context.actor.id), 

108 context=decision_context.value, 

109 choice=decision_choice.value, 

110 parent_id=command.parent_decision_id, 

111 override_kind=None, 

112 rule=rule.value, 

113 reasoning=validated_reasoning, 

114 confidence=validated_confidence, 

115 confidence_source=DecisionConfidenceSource.SELF_REPORTED, 

116 alternatives=(), 

117 inputs=inputs, 

118 reasoning_signature=None, 

119 occurred_at=now, 

120 ) 

121 ] 

122 

123 

124__all__ = ["decide"]
