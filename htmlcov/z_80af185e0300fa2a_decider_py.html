1"""Pure decider for the `StopRun` command. 

2 

3Multi-source controlled-exit terminal: `Running | Held -> Stopped`. 

4Symmetric source set with abort_run — operator-initiated controlled 

5exits don't require an active state, only any non-terminal state. 

6Stopping any terminal Run (Completed | Aborted | Stopped) raises; 

7re-stopping a `Stopped` Run raises (strict-not-idempotent). 

8 

9## Consequence gate (Gate IV) 

10 

11`StopRun` is in the declared consequence class 

12(`cora.shared.consequence.COMMANDS_REQUIRING_RATIFICATION`): a deliberate, 

13irreversible early termination requires a second, independent principal's 

14co-signature. The handler pre-loads coverage via the `ConsequenceLookup` port and 

15passes `ratification_covered` here (the pure decider stays I/O-free). When the 

16command is in-scope AND coverage is absent, the gate refuses with 

17`RunRequiresRatificationError` BEFORE the status/transition checks: admission is 

18the outer precondition. Kind-blind (the decider never reads actor kind). On that 

19refusal the operator requests a ratification, which parks the run in the shared 

20hold pending the co-sign (see the RatificationHoldSubscriber). 

21 

22`reason` validation goes through the `RunStopReason` VO (which 

23calls the shared `validate_bounded_text` helper). The on-the-wire payload 

24in `RunStopped.reason` carries the trimmed string. 

25 

26Invariants: 

27 - Consequence-classed command without Granted coverage 

28 -> RunRequiresRatificationError 

29 - State must not be None -> RunNotFoundError 

30 - command.reason must be 1-500 chars after trimming 

31 -> InvalidRunStopReasonError 

32 - State.status must be in {Running, Held} 

33 -> RunCannotStopError(current_status=...) 

34""" 

35 

36from datetime import datetime 

37 

38from cora.run.aggregates.run import ( 

39 Run, 

40 RunCannotStopError, 

41 RunNotFoundError, 

42 RunRequiresRatificationError, 

43 RunStatus, 

44 RunStopped, 

45 RunStopReason, 

46) 

47from cora.run.features.stop_run.command import StopRun 

48from cora.shared.consequence import requires_ratification 

49 

50_STOPPABLE_STATUSES: tuple[RunStatus, ...] = (RunStatus.RUNNING, RunStatus.HELD) 

51 

52_COMMAND_NAME = "StopRun" 

53 

54 

55def decide( 

56 state: Run | None, 

57 command: StopRun, 

58 *, 

59 now: datetime, 

60 ratification_covered: bool, 

61) -> list[RunStopped]: 

62 """Decide the events produced by stopping a Run. 

63 

64 `ratification_covered` is the consequence-gate coverage fact pre-loaded by the 

65 handler from the `ConsequenceLookup` port: True iff a Granted Ratification 

66 covers `(run_id, StopRun)`. The gate is checked first (admission precondition) 

67 so an un-co-signed stop is refused before any transition logic runs. 

68 """ 

69 if requires_ratification(_COMMAND_NAME) and not ratification_covered: 

70 raise RunRequiresRatificationError(command.run_id, _COMMAND_NAME) 

71 if state is None: 

72 raise RunNotFoundError(command.run_id) 

73 reason = RunStopReason(command.reason) 

74 if state.status not in _STOPPABLE_STATUSES: 

75 raise RunCannotStopError(state.id, current_status=state.status) 

76 return [ 

77 RunStopped( 

78 run_id=state.id, 

79 reason=reason.value, 

80 decided_by_decision_id=command.decided_by_decision_id, 

81 occurred_at=now, 

82 ) 

83 ]
