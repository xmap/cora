1"""Hash an exported record with `cora.shared.content_hash`, no exclusions. 

2 

3Per `project_record_export_v3.md` F6: one canonicalization, `content_hash`'s 

4profile, for both serialization and hashing. `canonical_json` (used 

5elsewhere in the codebase) is a DIFFERENT recipe; mixing the two means the 

6committed file is not the bytes the hash covers. `cora.shared` is exactly 

7what `tach.toml` allows `cora.infrastructure` to depend on, so this import 

8is the dependency the layering rule exists to permit. 

9 

10No exclusions: every rendered stream row and every rendered logbook row 

11goes into the hashed body. Re-exporting the same database twice is 

12measured stable (`test_hashing.py`'s integration test); a regenerated 

13record hashing differently across genuinely different content is content 

14addressing working, not a defect, and cross-record comparison is the 

15shape layer's job, not this module's. 

16 

17Payload types follow the scheme 

18`application/vnd.cora.<event-type>+json` (`content_hash.py`'s own 

19convention) and the three names `project_record_export_v3.md`'s "Naming" 

20section already picked: `record`, `record-streams`, `record-logbooks`. 

21That section calls `record+json` the manifest's type; nothing writes a 

22manifest yet (step 4), so there is no live collision today. Used here for 

23the whole-bundle hash because `{streams, logbooks}` together *is* the 

24record the manifest will describe -- if step 4 wants manifest.json to 

25carry its own distinct wrapping type when it starts writing bytes to 

26disk, that is a decision for whoever builds it, made with an actual 

27manifest body in hand rather than guessed at here. 

28""" 

29

39RECORD_PAYLOAD_TYPE = "application/vnd.cora.record+json" 

40STREAMS_PAYLOAD_TYPE = "application/vnd.cora.record-streams+json" 

41LOGBOOKS_PAYLOAD_TYPE = "application/vnd.cora.record-logbooks+json" 

42REDACTION_PROFILE_PAYLOAD_TYPE = "application/vnd.cora.record-redaction-profile+json"

45def hash_streams(streams: tuple[dict[str, object], ...]) -> str: 

46 """SHA-256 content hash over the streams tier alone, in its own order.""" 

47 return compute_content_hash(STREAMS_PAYLOAD_TYPE, list(streams)) 

48 

49 

50def hash_logbooks(logbooks: dict[str, tuple[dict[str, object], ...]]) -> str: 

51 """SHA-256 content hash over the logbooks tier alone. 

52 

53 Kind keys are NFC-normalized and sort themselves via 

54 `json.dumps(sort_keys=True)`; each kind's row order is preserved 

55 (the registry's own order-by), never re-sorted. 

56 """ 

57 body = {kind: list(rows) for kind, rows in logbooks.items()} 

58 return compute_content_hash(LOGBOOKS_PAYLOAD_TYPE, body) 

59 

60 

61def hash_record(record: ExportedRecord) -> str: 

62 """SHA-256 content hash over the whole bundle, both tiers, no exclusions. 

63 

64 This is THE record hash: per F2, it covers everything `export_record` 

65 produced. Re-running the export against an unchanged database 

66 reproduces this value exactly; any single differing byte anywhere in 

67 either tier changes it. 

68 """ 

69 body = { 

70 "streams": list(record.streams), 

71 "logbooks": {kind: list(rows) for kind, rows in record.logbooks.items()}, 

72 } 

73 return compute_content_hash(RECORD_PAYLOAD_TYPE, body)

76def hash_redaction_profile() -> str: 

77 """SHA-256 content hash over every table that decides what a 

78 published record discloses: tier 1's generated `DISPOSITIONS` AND 

79 tier 2's hand-authored `TIER2_DISPOSITIONS` / 

80 `TIER2_JSONB_CLEARED_POINTERS` / `TIER2_JSONB_DROPPED_COLUMNS`. 

81 

82 This IS the redaction profile hash (H2). Step 7's security re-review 

83 found the tier-2 tables missing from this hash: the fail-closed 

84 switch (`redact_record`'s `expected_redaction_profile_hash` check) 

85 was fail-closed for tier 1 only, silently blind to a tier-2 table 

86 edit that weakened a disposition (e.g. `conduit_verdicts.reason` 

87 `DROP` -> `KEEP`) or dropped a jsonb clearance restriction. Both 

88 tiers must be in H2, or "the hash matches" does not mean what 

89 `RedactionProfileMismatchError`'s docstring claims it means. 

90 

91 Tuple-keyed dicts (`TIER2_JSONB_CLEARED_POINTERS` / 

92 `TIER2_JSONB_DROPPED_COLUMNS`) are flattened to `"kind/column"` 

93 string keys before hashing rather than relying on 

94 `compute_content_hash`'s `str(key)` fallback for non-string Mapping 

95 keys, which would hash a Python tuple's `repr()` instead of a 

96 reviewable string. 

97 

98 Regenerating tier 1's table via `make record-dispositions` after a 

99 real event-model change, or hand-editing tier 2's tables, is 

100 expected to change this value; `test_record_dispositions_drift.py` 

101 guards tier 1's generator output specifically, and 

102 `test_redact_tier2.py`'s live-schema drift test guards tier 2's 

103 column coverage, but only THIS hash is what a caller's 

104 `expected_redaction_profile_hash` actually pins. 

105 """ 

106 body = { 

107 "tier1": DISPOSITIONS, 

108 "tier2_dispositions": TIER2_DISPOSITIONS, 

109 "tier2_jsonb_cleared_pointers": { 

110 f"{kind}/{column}": sorted(pointers) 

111 for (kind, column), pointers in TIER2_JSONB_CLEARED_POINTERS.items() 

112 }, 

113 "tier2_jsonb_dropped_columns": sorted( 

114 f"{kind}/{column}" for kind, column in TIER2_JSONB_DROPPED_COLUMNS 

115 ), 

116 } 

117 return compute_content_hash(REDACTION_PROFILE_PAYLOAD_TYPE, body)
