1"""Local (facility-hosted) implementation of `LLM`. 

2 

3Serves an open model over an OpenAI-compatible endpoint (vLLM, Ollama, 

4llama.cpp) as the paper's "build" path, and meters the GPU time each 

5call consumes so a facility has visibility into shadow cost and 

6capacity. Lives under `cora.agent.adapters` per the cross-BC 

7adapter-ownership convention, beside `AnthropicLLM`, and fills the 

8local-model slot the `LLM` port anticipates. 

9 

10## Instrument first, model later 

11 

12The adapter talks to a `LocalCompletionBackend`, not directly to an 

13engine, so the serving-and-metering seam is built and tested against a 

14deterministic `StubLocalBackend` before a real engine is wired in, which 

15the design deliberately sequences last. Any OpenAI-compatible engine 

16becomes one more backend; the adapter and its meter integration do not 

17change. 

18 

19## Metering 

20 

21Every `chat` opens the shared `OccupancyShareMeter` on the call's device 

22before the backend runs and closes it after, so concurrent calls sharing 

23one GPU are attributed their occupancy share, not their full wall-clock 

24each (see `cora.infrastructure.observability.gpu_accounting`). The 

25measured GPU-seconds are handed to an optional `on_measure` sink for 

26observability, and also carried on the returned `LLMResponse.gpu_seconds` 

27so a producer can persist a durable copy alongside the call's other 

28provenance; neither ever touches the budget gate. In the locked design 

29in-house serving is metered-free by default, so this is visibility, not 

30a debit. Elapsed time is read from an injected `MonotonicClock`, never 

31wall-clock, so a duration is never corrupted by an NTP correction and 

32tests reproduce exactly. 

33 

34## Structured output 

35 

36Consumers require a schema-valid `parsed` Decision. A capable local 

37server produces it via guided decoding / JSON-schema mode; the backend 

38returns `parsed=None` when it could not, and the adapter raises 

39`LLMSchemaValidationError` (the outer retry layer then defers), the same 

40contract `AnthropicLLM` presents from forced tool-use. 

41 

42## Provider guard and telemetry 

43 

44`chat` refuses a request whose `model_ref.provider` is not `"local"`, 

45mirroring `AnthropicLLM`'s guard: cost resolves from the Agent's 

46declared `(provider, model)` while the serving route comes from 

47`LLM_PROVIDER` configuration, so a mismatched pair would silently 

48misattribute spend between the bought and built arms this adapter 

49exists to compare. Unlike `AnthropicLLM`, the provider name here is a 

50fixed constant rather than an injected constructor argument: no 

51gateway composes `LocalLLM` the way `ArgoLLM` composes `AnthropicLLM`, 

52so there is nothing yet that would need to override it. 

53 

54`chat` also emits the same OpenTelemetry GenAI signals `AnthropicLLM` 

55does: `track_in_flight_call` around the backend call for the 

56concurrent-calls counter, an `llm.chat` span, and `record_llm_call` on 

57the success path only. The GPU occupancy meter (see Metering above) 

58still opens and closes around every call unchanged; the telemetry 

59context nests inside it. 

60"""

140class LocalLLM: 

141 """`LLM` adapter for a facility-hosted open model, with GPU metering. 

142 

143 Holds one `OccupancyShareMeter` shared across concurrent `chat` calls 

144 so their overlapping device time is divided honestly. `device_id` 

145 names the served model's device (one model on one device for now; 

146 multi-device assignment is a later slice). `on_measure`, if given, 

147 receives each call's `GpuUsageRecord` for observability. 

148 """ 

149 

150 def __init__( 

151 self, 

152 *, 

153 backend: LocalCompletionBackend, 

154 monotonic_clock: MonotonicClock, 

155 meter: OccupancyShareMeter | None = None, 

156 device_id: str = "gpu0", 

157 on_measure: Callable[[GpuUsageRecord], None] | None = None, 

158 ) -> None: 

159 self._backend = backend 

160 self._clock = monotonic_clock 

161 self._meter = meter if meter is not None else OccupancyShareMeter() 

162 self._device_id = device_id 

163 self._on_measure = on_measure 

164 self._call_seq = 0 

165 

166 async def chat(self, request: LLMChatRequest) -> LLMResponse: 

167 if request.model_ref.provider != _PROVIDER_NAME: 

168 msg = ( 

169 f"model_ref names provider {request.model_ref.provider!r} but this " 

170 f"adapter serves {_PROVIDER_NAME!r}. Cost resolves from the " 

171 "model_ref while the serving route comes from configuration, so a " 

172 "call served by one and priced as the other misattributes spend " 

173 "with nothing to show for it. Price the catalog entry as " 

174 f"{_PROVIDER_NAME!r}, or select the adapter that matches it." 

175 ) 

176 raise LLMInvalidRequestError(msg) 

177 

178 self._call_seq += 1 

179 call_id = f"local-{self._call_seq}" 

180 model = request.model_ref.model 

181 gpu_seconds: float 

182 self._meter.open(call_id, device_id=self._device_id, at_s=self._clock.now()) 

183 try: 

184 with ( 

185 track_in_flight_call(request.model_ref), 

186 _tracer.start_as_current_span("llm.chat") as span, 

187 ): 

188 completion = await self._backend.complete(request) 

189 if completion.parsed is None: 

190 msg = ( 

191 f"local server returned no schema-valid structured " 

192 f"output for model {model!r}" 

193 ) 

194 raise LLMSchemaValidationError(msg) 

195 parsed = completion.parsed 

196 

197 # record_llm_call returns the computed USD cost for telemetry 

198 # only; the durable spend ledger is written by the caller 

199 # from the returned LLMResponse's usage, not from here. 

200 record_llm_call( 

201 span, 

202 provider_name=_PROVIDER_NAME, 

203 request_model_ref=request.model_ref, 

204 response_model_id=completion.model_id, 

205 usage=completion.usage, 

206 stop_reason=completion.stop_reason, 

207 max_tokens=request.max_output_tokens, 

208 ) 

209 finally: 

210 # gpu_seconds is only known once the call (success or failure) 

211 # has fully elapsed, which is one step later than the `return` 

212 # below would otherwise let it happen: the meter closes here in 

213 # every case, including the exception paths above, so it must 

214 # be read from `finally`, not folded into the `try` body's own 

215 # return statement. 

216 gpu_seconds = self._meter.close(call_id, at_s=self._clock.now()) 

217 if self._on_measure is not None: 

218 self._on_measure( 

219 GpuUsageRecord( 

220 call_id=call_id, 

221 device_id=self._device_id, 

222 model=model, 

223 gpu_seconds=gpu_seconds, 

224 ) 

225 ) 

226 

227 return LLMResponse( 

228 parsed=parsed, 

229 raw_text=completion.raw_text, 

230 usage=completion.usage, 

231 stop_reason=completion.stop_reason, 

232 model_id=completion.model_id, 

233 response_id=completion.response_id, 

234 gpu_seconds=gpu_seconds, 

235 )

246class StubLocalBackend: 

247 """Deterministic backend: canned completions that consume set GPU time. 

248 

249 Each `complete` pops the next `StubCompletion` and advances the 

250 injected `FakeMonotonicClock` by its `gpu_seconds`, so a solo call is 

251 measured at exactly that duration. It lets the whole serving path run 

252 without a real GPU (the instrument-first slice). Overlapping-call 

253 concurrency is exercised with a gated backend in tests, not here. 

254 """ 

255 

256 def __init__(self, clock: FakeMonotonicClock, completions: list[StubCompletion]) -> None: 

257 self._clock = clock 

258 self._queue = list(completions) 

259 

260 async def complete(self, request: LLMChatRequest) -> LocalCompletion: 

261 if not self._queue: 261 ↛ 262line 261 didn't jump to line 262 because the condition on line 261 was never true

262 msg = "StubLocalBackend queue exhausted; enqueue more completions" 

263 raise LLMServerError(msg) 

264 head = self._queue.pop(0) 

265 self._clock.advance(head.gpu_seconds) 

266 return head.completion
