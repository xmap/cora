1"""Matching a filesystem path against a facility's configured storage roots. 

2 

3Lives in `cora.shared` (`depends_on = []`) rather than in either BC 

4because three layers need the SAME answer and would otherwise each 

5derive it: `cora.api`'s `RunWitnessRecorder` records which root a 

6capture path was observed under, `cora.data`'s locator embeds that root 

7in an indirect locator, and `cora.run`'s vault persists and keys on it. 

8The Run aggregate is restricted to `cora.infrastructure` + `cora.shared`, 

9so a home in either BC would put the definition of `root` out of reach 

10of the store that keys on it, which is exactly how a recorded location 

11and a minted location drift apart into a locator that resolves to 

12nothing. 

13 

14Normalization belongs here for the same reason. A trailing slash in a 

15configured root is a legal, plausible thing for an operator to type, and 

16if one caller strips it while another does not, the two produce 

17different strings for the same directory and every lookup misses. 

18""" 

19 

20from __future__ import annotations 

21 

22 

23def normalize_storage_root(root: str) -> str: 

24 """The canonical form of a configured storage root. 

25 

26 Trailing slashes are stripped so `/local1/2BM/` and `/local1/2BM` 

27 are one root, not two. A root of `/` normalizes to the empty 

28 string, which callers must reject rather than store: the vault's 

29 own CHECK constraint forbids it. 

30 """ 

31 return root.rstrip("/") 

32 

33 

34def path_is_under_root(path: str, root: str) -> bool: 

35 """Whether `path` is `root` itself or lies beneath it. 

36 

37 Compares whole path segments, so `/local1/2BM` does NOT match 

38 `/local1/2BMX/scan.h5`: a plain `startswith` would treat a sibling 

39 directory whose name merely shares a prefix as being inside the 

40 allowlisted tier. 

41 """ 

42 normalized = normalize_storage_root(root) 

43 return path == normalized or path.startswith(normalized + "/") 

44 

45 

46def require_nonempty_absolute_root(root: str, *, label: str) -> None: 

47 """Raise `ValueError` unless `root` is absolute and does not 

48 normalize to the empty string. 

49 

50 Shared by three callers that enforce this same shape rule but then 

51 diverge: `Settings._validate_posix_checksum_roots` and 

52 `_validate_scan_probe_allowed_roots` call this then store `root` AS 

53 TYPED, normalizing only at comparison time via `path_is_under_root`; 

54 `CaptureScanIngestorBinding`'s locations validator calls this then 

55 stores `normalize_storage_root(root)` itself, since that field is 

56 later looked up by exact key rather than compared path-by-path, so 

57 it must be normalization-stable at rest, not just at compare time. 

58 `label` names the field and, where the caller has one, the entry 

59 itself (for example `"posix_checksum_roots entry"`), so the raised 

60 message reads naturally as `f"{label} {root!r} ..."`. 

61 """ 

62 if not root.startswith("/"): 

63 msg = f"{label} {root!r} is not an absolute path." 

64 raise ValueError(msg) 

65 if not normalize_storage_root(root): 

66 msg = ( 

67 f"{label} {root!r} normalizes to the empty string. A root must " 

68 "name a facility-level storage tier, not the filesystem root itself." 

69 ) 

70 raise ValueError(msg) 

71 

72 

73def matched_storage_root(path: str, roots: tuple[str, ...]) -> str | None: 

74 """The first root in `roots` that `path` falls under, normalized. 

75 

76 Returns `None` when none match, which every caller treats as a 

77 refusal rather than a default: a fabricated root would be embedded 

78 verbatim in a locator and recorded verbatim in the vault, and both 

79 are worse than declining. 

80 

81 The return is normalized so that the recorder, the locator and the 

82 vault all hold byte-identical strings for one directory regardless 

83 of how the deployment spelled it in settings. 

84 """ 

85 for root in roots: 

86 if path_is_under_root(path, root): 

87 return normalize_storage_root(root) 

88 return None
