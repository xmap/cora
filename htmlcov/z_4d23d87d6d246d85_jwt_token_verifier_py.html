1"""`JwtTokenVerifier` — local JWKS-based JWT verification. 

2 

3Implements `cora.infrastructure.ports.TokenVerifier` for IdPs that 

4issue self-describing JWT access tokens — the industry default: 

5Microsoft Entra, Google, Auth0, Okta, AWS Cognito, Helmholtz AAI, 

6ORCID-via-OIDC, and any IdP following RFC 9068 ("OAuth 2.0 Access 

7Tokens in JWT"). 

8 

9## Library choice 

10 

11PyJWT (`pyjwt[crypto]` dep) is the one library dependency for 

12edge-auth; rationale captured in `memory/project_edge_auth_design.md` §2. 

13The `[crypto]` extra pulls `cryptography` for RS256/ES256 — every 

14real-world IdP uses one of these two. CORA does NOT roll its own 

15signature verify or JOSE parsing; that's the library's whole job. 

16 

17## Per-issuer one `PyJWKClient` 

18 

19PyJWT's `PyJWKClient` caches the JWKS in-memory and refetches on 

20kid-miss. We instantiate one per registered issuer at startup and 

21keep them alive for the process lifetime — repeated `.get_signing_key_from_jwt(token)` 

22calls hit the cache, and a token signed with a fresh key triggers 

23one refetch then resumes from cache. 

24 

25Cache TTL is governed by PyJWT's internal `lifespan` (default 5 min); 

26we leave it at the default. The kid-miss refetch is the real 

27safety net per the Cloudflare discipline (OAuth corpus survey). 

28 

29## What `verify()` checks (in order) 

30 

31 1. Parse token + extract unverified header (`alg`, `kid`, `typ`). 

32 2. Reject if `alg` not in the configured whitelist (RFC 9068 §4 

33 specifically calls out `alg=none` rejection; we go further and 

34 pin a small whitelist per-issuer). 

35 3. Fetch the signing key for `kid` from the JWKS (cached + on-miss-refetch). 

36 4. PyJWT verifies signature + standard claims (`exp`, `nbf`, `iat`, 

37 `iss`, `aud`) with explicit `algorithms`, `audience`, `issuer` 

38 options. NO `options={"verify_signature": False}` anywhere. 

39 5. Map `sub` claim to a `principal_id` via the injected 

40 `SubjectMapper` (the Access BC owns the IdP-subject → Actor 

41 mapping; the verifier doesn't reach into it directly). 

42 6. Return `VerifiedPrincipal(principal_id, sub, iss, kind, scopes)`. 

43 

44Any failure raises `InvalidTokenError` with a specific reason code 

45so route-layer logs distinguish modes (bad_signature vs expired vs 

46wrong_audience). 

47 

48## What we do NOT check 

49 

50- Token **revocation**. JWT-AT has no native revocation (RFC 9068 §5, 

51 RFC 6819 §5.2.1). Mitigation = short token TTL (5-15 min, 

52 passive-revocation pattern per RFC 6819 §5.1.5.3). When a 

53 must-revoke-immediately need lands, that subject moves to 

54 introspection-required (WI5). 

55- Token **scope→capability** mapping. Edge-auth ships without 

56 scope-aware authorization; scopes flow into `VerifiedPrincipal.scopes` 

57 for future use (WI6). 

58"""

77class JwtTokenVerifier: 

78 """RFC 9068 JWT access-token verifier for a single IdP. 

79 

80 One instance per registered issuer. The `IdentityProviderRegistry` 

81 holds the dict of `iss → JwtTokenVerifier` and routes by token's `iss` 

82 claim. 

83 """ 

84 

85 def __init__( 

86 self, 

87 *, 

88 issuer: str, 

89 jwks_url: str, 

90 audience_for_surface: dict[UUID, str], 

91 subject_mapper: SubjectMapper, 

92 allowed_algorithms: list[str], 

93 principal_kind: PrincipalKind = "human", 

94 allow_insecure_jwks_url: bool = False, 

95 ) -> None: 

96 """Construct a verifier bound to one IdP issuer. 

97 

98 `issuer` — expected `iss` claim on every token (exact match, 

99 no trailing-slash leniency). 

100 `jwks_url` — `https://<idp>/.well-known/jwks.json` or 

101 equivalent; `PyJWKClient` fetches + caches keys from here. 

102 `audience_for_surface` — Surface UUID → audience string the 

103 IdP signs into the token's `aud` claim for that Surface. 

104 `subject_mapper` — async callable resolving (iss, sub) to 

105 (principal_id, kind). See SubjectMapper docstring. 

106 `allowed_algorithms` — explicit whitelist. RFC 9068 §4 

107 rejects `alg=none`; we go further with a small list. Typical: 

108 `["RS256", "ES256"]`. Pinning the list per-issuer matches 

109 the IdP's actual capabilities + locks out algorithm- 

110 confusion attacks at the verifier layer. 

111 `principal_kind` — defaults `"human"`. Per-IdP override for 

112 deployments where the entire IdP issues only service-account 

113 tokens (for example a CI-only IdP). 

114 

115 `allow_insecure_jwks_url` — production MUST be False (default). 

116 Test/dev fixtures using `http://127.0.0.1:...` JWKS endpoints 

117 opt in by passing True. Without HTTPS, an attacker who MITMs 

118 the JWKS fetch owns all signature verification for this issuer 

119 (gate-review F2). Localhost is implicitly safe but explicit 

120 opt-in > implicit allow. 

121 """ 

122 if not allowed_algorithms: 

123 msg = ( 

124 f"JwtTokenVerifier for issuer={issuer!r} requires a non-empty " 

125 "allowed_algorithms whitelist (no alg=none). " 

126 "Explicit > implicit." 

127 ) 

128 raise ValueError(msg) 

129 # Strip + lowercase so " None ", "NONE", "noNe" are all caught. 

130 if "none" in (a.strip().lower() for a in allowed_algorithms): 

131 msg = ( 

132 f"JwtTokenVerifier for issuer={issuer!r}: " 

133 "allowed_algorithms must not include 'none'." 

134 ) 

135 raise ValueError(msg) 

136 if not jwks_url.startswith("https://") and not allow_insecure_jwks_url: 

137 msg = ( 

138 f"JwtTokenVerifier for issuer={issuer!r}: jwks_url must be HTTPS " 

139 f"(got scheme={jwks_url.split(':')[0]!r}). Pass " 

140 "allow_insecure_jwks_url=True only for test/dev fixtures " 

141 "(gate-review F2: HTTP JWKS is MITM-exploitable)." 

142 ) 

143 raise ValueError(msg) 

144 self._issuer = issuer 

145 self._audience_for_surface = audience_for_surface 

146 self._subject_mapper = subject_mapper 

147 self._algorithms = list(allowed_algorithms) 

148 self._principal_kind = principal_kind 

149 self._jwks_client = PyJWKClient(jwks_url) 

150 

151 @property 

152 def issuer(self) -> str: 

153 return self._issuer 

154 

155 async def verify( 

156 self, 

157 token: str, 

158 *, 

159 expected_audience: UUID, 

160 ) -> VerifiedPrincipal: 

161 expected_aud_str = self._audience_for_surface.get(expected_audience) 

162 if expected_aud_str is None: 

163 raise InvalidTokenError( 

164 "wrong_audience", 

165 f"no aud configured for surface={expected_audience}", 

166 ) 

167 

168 try: 

169 signing_key = self._jwks_client.get_signing_key_from_jwt(token) 

170 except jwt.PyJWKClientError as exc: 

171 raise InvalidTokenError("bad_signature", f"jwks lookup failed: {exc}") from exc 

172 except jwt.DecodeError as exc: 

173 raise InvalidTokenError("malformed", str(exc)) from exc 

174 

175 try: 

176 claims = jwt.decode( 

177 token, 

178 signing_key.key, 

179 algorithms=self._algorithms, 

180 audience=expected_aud_str, 

181 issuer=self._issuer, 

182 options={"require": ["exp", "iss", "aud", "sub"]}, 

183 ) 

184 except jwt.ExpiredSignatureError as exc: 

185 raise InvalidTokenError("expired", str(exc)) from exc 

186 except jwt.ImmatureSignatureError as exc: 

187 raise InvalidTokenError("not_yet_valid", str(exc)) from exc 

188 except jwt.InvalidAudienceError as exc: 

189 raise InvalidTokenError("wrong_audience", str(exc)) from exc 

190 except jwt.InvalidIssuerError as exc: 

191 raise InvalidTokenError("wrong_issuer", str(exc)) from exc 

192 except jwt.InvalidAlgorithmError as exc: 

193 raise InvalidTokenError("unsupported_algorithm", str(exc)) from exc 

194 except jwt.InvalidSignatureError as exc: 

195 raise InvalidTokenError("bad_signature", str(exc)) from exc 

196 except jwt.MissingRequiredClaimError as exc: 

197 raise InvalidTokenError("malformed", str(exc)) from exc 

198 except jwt.InvalidTokenError as exc: 

199 raise InvalidTokenError("malformed", str(exc)) from exc 

200 

201 subject = str(claims["sub"]) 

202 principal_id, kind = await safe_map_subject(self._subject_mapper, self._issuer, subject) 

203 scopes = _parse_scopes_claim(claims.get("scope") or claims.get("scp")) 

204 return VerifiedPrincipal( 

205 principal_id=principal_id, 

206 subject=subject, 

207 issuer=self._issuer, 

208 kind=kind or self._principal_kind, 

209 scopes=scopes, 

210 )

213def _parse_scopes_claim(raw: object) -> frozenset[str]: 

214 """Normalize the OAuth `scope` (RFC 6749 §3.3) / `scp` (Microsoft) 

215 claim shape to `frozenset[str]`. 

216 

217 Accepts: space-separated string (standard), list of strings 

218 (Microsoft Entra v2 sometimes), None / missing / other shape → 

219 empty frozenset. 

220 """ 

221 if isinstance(raw, str): 

222 return frozenset(raw.split()) 

223 if isinstance(raw, list): 

224 items: list[str] = [str(item) for item in raw] # type: ignore[misc] 

225 return frozenset(items) 

226 return frozenset()

229async def safe_map_subject( 

230 mapper: SubjectMapper, issuer: str, subject: str 

231) -> tuple[UUID, PrincipalKind]: 

232 """Call the SubjectMapper with defense-in-depth (gate-review F4 + F5). 

233 

234 Wraps any exception from the mapper as 

235 `InvalidTokenError("unknown_subject", ...)` so route-layer logs 

236 distinguish "the subject isn't registered" from generic 500s. 

237 Rejects nil-UUID returns (which would silently escalate to 

238 SYSTEM_PRINCIPAL_ID) and invalid `kind` values (which would 

239 silently degrade `service_account` → `human` via the 

240 `kind or principal_kind` fallback). 

241 

242 Shared helper used by both JWT and Introspection adapters. 

243 """ 

244 try: 

245 principal_id, kind = await mapper(issuer, subject) 

246 except InvalidTokenError: 

247 raise 

248 except Exception as exc: 

249 raise InvalidTokenError("unknown_subject", f"subject mapper raised: {exc}") from exc 

250 if principal_id == _NIL_SENTINEL_ID: 

251 raise InvalidTokenError( 

252 "unknown_subject", 

253 "subject mapper returned nil sentinel (would escalate to SYSTEM)", 

254 ) 

255 if kind not in _VALID_KINDS: 

256 raise InvalidTokenError( 

257 "malformed", 

258 f"subject mapper returned kind={kind!r}; expected one of {sorted(_VALID_KINDS)}", 

259 ) 

260 return principal_id, kind
