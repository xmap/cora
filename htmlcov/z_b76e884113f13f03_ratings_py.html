1"""DecisionRatingsProjection: folds `DecisionRated` events into the 

2`proj_decision_ratings` read model. 

3 

4Subscribed events: 

5 - DecisionRated -> UPSERT one row per (decision_id, rated_by) 

6 

7Latest-per-actor-wins implementation: ON CONFLICT (decision_id, 

8rated_by) DO UPDATE WHERE EXCLUDED.rated_at > rated_at. 

9The WHERE guard makes the apply() out-of-order-replay safe: if a 

10projection rebuild lands an older rating after a newer one, the 

11older one does not overwrite (defensive; production at-least-once 

12delivery is in-order per stream, but rebuild-from-scratch crosses 

13streams via the projection-worker advance loop). 

14 

15`confidence_at_rating` is captured at write time on the 

16`DecisionRated` event payload (gate-review cross-BC P2-4: the 

17earlier projection-side denorm pattern would have read 

18`proj_decision_summary` at apply() time, which races under 

19rebuild when the summary has not yet caught up with the rated 

20Decision). The projection simply forwards the captured value into 

21its column; null when the rated Decision had no confidence value. 

22 

23`apply()` IS idempotent per the at-least-once delivery contract: 

24re-applying the same DecisionRated event runs the same UPSERT, 

25which Postgres handles correctly (latest-rated_at predicate 

26short-circuits when EXCLUDED.rated_at <= existing rated_at). 

27""" 

28 

29# pyright: reportUnknownMemberType=false, reportUnknownVariableType=false, reportUnknownArgumentType=false 

30 

31from datetime import datetime 

32from uuid import UUID 

33 

34from cora.infrastructure.ports.event_store import StoredEvent 

35from cora.infrastructure.projection.handler import ConnectionLike 

36 

37_UPSERT_RATING_SQL = """ 

38INSERT INTO proj_decision_ratings 

39 (decision_id, rated_by, rating, comment, rated_at, confidence_at_rating) 

40VALUES ($1, $2, $3, $4, $5, $6) 

41ON CONFLICT (decision_id, rated_by) DO UPDATE 

42 SET rating = EXCLUDED.rating, 

43 comment = EXCLUDED.comment, 

44 rated_at = EXCLUDED.rated_at, 

45 confidence_at_rating = EXCLUDED.confidence_at_rating 

46 WHERE EXCLUDED.rated_at > proj_decision_ratings.rated_at 

47""" 

48 

49 

50class DecisionRatingsProjection: 

51 """Maintains the `proj_decision_ratings` read model.""" 

52 

53 name = "proj_decision_ratings" 

54 subscribed_event_types = frozenset({"DecisionRated"}) 

55 

56 async def apply( 

57 self, 

58 event: StoredEvent, 

59 conn: ConnectionLike, 

60 ) -> None: 

61 if event.event_type != "DecisionRated": 

62 return 

63 payload = event.payload 

64 decision_id = UUID(payload["decision_id"]) 

65 rated_by = UUID(payload["rated_by"]) 

66 rating = payload["rating"] 

67 comment = payload.get("comment") 

68 rated_at = datetime.fromisoformat(payload["rated_at"]) 

69 # Captured by the handler at write time from 

70 # `Decision.state.confidence`; null when the rated Decision 

71 # had no confidence value. No cross-projection read; the 

72 # payload is the single source of truth. 

73 confidence = payload.get("confidence_at_rating") 

74 

75 await conn.execute( 

76 _UPSERT_RATING_SQL, 

77 decision_id, 

78 rated_by, 

79 rating, 

80 comment, 

81 rated_at, 

82 confidence, 

83 ) 

84
