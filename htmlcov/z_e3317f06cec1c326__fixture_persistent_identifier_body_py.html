1"""Shared Pydantic wire-format for the `assign_fixture_persistent_id` slice. 

2 

3Mirrors `_asset_persistent_identifier_body` at the BC root. Per Section 

46.5 of [[project-fixture-pidinst-design]], slice F at the Fixture tier 

5does NOT carry a domain-VO wire mirror (the `PersistentIdentifier` VO 

6is never parsed at the wire; it is server-minted inside the handler). 

7This module instead carries the request + response wire surface for the 

8single POST endpoint: 

9 

10 - `AssignFixturePersistentIdRequest`: `(scheme, suffix | None)` 

11 operator intent. The handler resolves the suffix into a full 

12 `PersistentIdentifier` via the configured `PersistentIdentifierMinter` port. 

13 - `AssignFixturePersistentIdResponse`: `(scheme, value)` echoed back 

14 so the operator learns the server-minted identifier without a 

15 follow-up GET (Section 6.5 deviation from the empty-201 convention 

16 for Fixture mutations). 

17 

18Reuses `PersistentIdentifierScheme` + `PERSISTENT_IDENTIFIER_VALUE_MAX_LENGTH` 

19from the Asset module per Locks 2-4 of the design memo (the VO + enum 

20ship at the Asset tier and the Fixture tier imports them unchanged). 

21""" 

22 

23from pydantic import BaseModel, Field 

24 

25from cora.shared.identifier import ( 

26 PERSISTENT_IDENTIFIER_VALUE_MAX_LENGTH, 

27 PersistentIdentifierScheme, 

28) 

29 

30 

31class AssignFixturePersistentIdRequest(BaseModel): 

32 """Body for `POST /fixtures/{fixture_id}/assign-persistent-identifier`. 

33 

34 `scheme` selects the PID scheme; v1 supports DOI and HANDLE. 

35 `suffix` is the optional operator-supplied local part; when absent 

36 the configured `PersistentIdentifierMinter` adapter auto-generates one. No `value` 

37 field per the server-mint posture. 

38 """ 

39 

40 scheme: PersistentIdentifierScheme = Field( 

41 ..., 

42 description="Closed PIDINST Property 1 scheme: DOI or HANDLE.", 

43 ) 

44 suffix: str | None = Field( 

45 None, 

46 min_length=1, 

47 max_length=PERSISTENT_IDENTIFIER_VALUE_MAX_LENGTH, 

48 description=( 

49 "Optional operator-supplied local part. When absent, the " 

50 "configured PersistentIdentifierMinter adapter auto-generates the suffix." 

51 ), 

52 ) 

53 

54 

55class AssignFixturePersistentIdResponse(BaseModel): 

56 """Response body for `POST /fixtures/{fixture_id}/assign-persistent-identifier`. 

57
