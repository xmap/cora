1"""Data Exchange (``file://`` HDF5) adapter implementing ``ScanReader``. 

2 

3Reads the facts out of one tomoscan-written Data Exchange scan file: 

4frame roles from the ``/exchange`` datasets, commanded counts from the 

5``/process/acquisition`` groups the areaDetector layout writes, the 

6acquisition timestamp, and a stat snapshot for the live-file guard. 

7Format vocabulary (every dataset path below) dies here; only roles, 

8counts, angles, and timestamps cross the port. 

9 

10## The read set, and who authors it 

11 

12The file has two authors. tomoscan writes ``/exchange`` (the image 

13stacks, and ``theta`` in a post-capture append); the areaDetector 

14layout XML configured in the detector IOC writes ``/process`` and 

15``/defaults``. The commanded-geometry datasets 

16(``rotation/num_angles``, the flat and dark mode-and-count groups) are 

17written on clean file close, so a crashed file lacks them and the 

18commanded counts honestly read None; ``start_date`` is written on file 

19open, so a crashed file keeps its timestamp. Flat and dark expected 

20counts use tomoscan's own arithmetic: the configured count times the 

21number of phases the mode names (Start or End contribute one each, 

22Both contributes two, None contributes zero). 

23 

24## Which timestamp is the acquisition time 

25 

26Not a layout question, a deployment one, so the caller declares it and 

27this adapter reads what it is told. The layout offers two, and at 2-BM 

28one of them is wrong: measured across six consecutive scan files, 

29``start_date`` is the PREVIOUS scan's ``end_date`` every time, while 

30``end_date`` matches the file's own close to within five seconds. The 

31areaDetector timestamp attribute refreshes only while frames flow, so 

32a file opened for scan N carries the value left over from scan N-1's 

33last frame, and the value written on close is the current one. 

34 

35The correctness of the timestamp PV is not the issue and reading it 

36live returns the right instant; the staleness is in the attribute 

37cache between scans. A deployment whose writer does not have that 

38defect keeps naming ``start_date``, which is why this is a per- 

39deployment declaration rather than a change of default. 

40 

41## Locking, and why it is not optional 

42 

43The file is opened read-only with HDF5 file locking disabled. Two 

44independent reasons: HDF5's flock is unreliable-to-fatal on network 

45filesystems, which is exactly where the analysis tier lives; and a 

46reader HOLDING a shared lock can make tomoscan's own post-capture 

47``add_theta`` append reopen fail, a failure tomoscan swallows, so 

48observation would silently break the pipeline it observes. SWMR is 

49not attempted: the writer never opts in, and the reader must stay 

50safe against files still being written, which is what the 

51stat-snapshot guard and the never-raise contract are for. 

52 

53libhdf5 is not thread-safe and h5py serializes all API calls behind 

54one lock, so each describe runs as a single open-read-close unit in a 

55worker thread; reads serialize per process and never block the event 

56loop. 

57 

58## Structural completeness is not finality 

59 

60``/exchange/theta`` present means post-processing COMPLETED. Its 

61absence means only that post-processing has not completed: not yet 

62run, failed permanently (tomoscan wraps ``add_theta`` in a bare 

63except and proceeds), or a scan that produced zero projections. 

64Deciding whether the file will still CHANGE is the finality axis, a 

65deployment protocol that lives outside this adapter. 

66"""

113class DataExchangeScanReader: 

114 """``ScanReader`` over tomoscan-written Data Exchange HDF5 files.""" 

115 

116 kind = "DataExchange"

118 def __init__( 

119 self, 

120 *, 

121 allowed_roots: tuple[str, ...], 

122 captured_at_source: str = "start_date", 

123 ) -> None: 

124 # Same canonicalisation as PosixChecksumAdapter: roots resolve 

125 # once so containment compares realpath-to-realpath. 

126 self._allowed_roots = tuple(os.path.realpath(root) for root in allowed_roots) 

127 if captured_at_source not in _CAPTURED_AT_SOURCES: 

128 raise ValueError( 

129 f"captured_at_source {captured_at_source!r} is not a timestamp this " 

130 f"layout offers; choose one of {sorted(_CAPTURED_AT_SOURCES)}." 

131 ) 

132 self._captured_at_source = captured_at_source 

133 self._captured_at_path = _CAPTURED_AT_SOURCES[captured_at_source] 

134 

135 async def describe(self, *, locator_uri: str) -> ScanReadResult: 

136 try: 

137 return await asyncio.to_thread(self._describe_sync, locator_uri) 

138 except Exception as exc: 

139 # h5py surfaces malformed files as a zoo of exception types 

140 # (OSError, KeyError, RuntimeError from libhdf5); the port 

141 # contract is that a bad FILE is a result, not a 500. 

142 _log.warning( 

143 "data_exchange_scan_reader.read_failed", 

144 locator_uri=locator_uri, 

145 error=type(exc).__name__, 

146 ) 

147 return Unreadable(reason=f"read failed: {type(exc).__name__}") 

148 

149 def _describe_sync(self, locator_uri: str) -> ScanReadResult: 

150 resolved = resolve_confined_file_uri(locator_uri, self._allowed_roots) 

151 if isinstance(resolved, Refused): 

152 _log.warning( 

153 "data_exchange_scan_reader.refused", 

154 locator_uri=locator_uri, 

155 reason=resolved.reason, 

156 ) 

157 return Unreadable(reason=resolved.reason) 

158 real_path = resolved.real_path 

159 

160 # Snapshot BEFORE the structural read; the digest pass re-stats 

161 # after its walk and the ingest handler refuses on a difference. 

162 try: 

163 stat = os.stat(real_path) 

164 except OSError as exc: 

165 return Unreadable(reason=f"stat failed: {type(exc).__name__}") 

166 

167 try: 

168 handle = h5py.File(real_path, "r", locking=False) 

169 except OSError as exc: 

170 return Unreadable(reason=f"not readable as HDF5: {type(exc).__name__}") 

171 

172 with handle: 

173 data = _dataset(handle, _DATA) 

174 if data is None: 

175 # The one dataset the Data Exchange spec makes mandatory. 

176 # Anything else absent is a count of zero, not a verdict. 

177 return Unrecognized(reason="no /exchange/data dataset") 

178 

179 projection_count = _frame_count(data) 

180 flat_count = _optional_frame_count(handle, _DATA_WHITE) 

181 dark_count = _optional_frame_count(handle, _DATA_DARK) 

182 

183 projection_angles = _angles(handle, _THETA) 

184 flat_angles = _angles(handle, _THETA_WHITE) 

185 dark_angles = _angles(handle, _THETA_DARK) 

186 

187 commanded_projections = _scalar_int(handle, _NUM_ANGLES) 

188 commanded_flats = _phase_count( 

189 _scalar_int(handle, _NUM_FLAT_FIELDS), 

190 _scalar_str(handle, _FLAT_FIELD_MODE), 

191 ) 

192 commanded_darks = _phase_count( 

193 _scalar_int(handle, _NUM_DARK_FIELDS), 

194 _scalar_str(handle, _DARK_FIELD_MODE), 

195 ) 

196 

197 captured_at_raw = _scalar_str(handle, self._captured_at_path) 

198 

199 dropped: int | None = None 

200 if commanded_projections is not None: 

201 shortfall = commanded_projections - projection_count 

202 # A negative shortfall means the source contradicts itself 

203 # (more captured than commanded); report unknowable rather 

204 # than a fabricated negative. 

205 dropped = shortfall if shortfall >= 0 else None 

206 

207 return Description( 

208 media_type=_MEDIA_TYPE, 

209 structurally_complete=projection_angles is not None, 

210 projection_count=projection_count, 

211 flat_count=flat_count, 

212 dark_count=dark_count, 

213 # Data Exchange has no per-frame invalid marking; 0 means 

214 # none-marked, not all-validated (NXtomo image_key=3 is the 

215 # role's producing format). 

216 invalid_count=0, 

217 commanded_projection_count=commanded_projections, 

218 commanded_flat_count=commanded_flats, 

219 commanded_dark_count=commanded_darks, 

220 dropped_frame_count=dropped, 

221 projection_angles_deg=projection_angles, 

222 flat_angles_deg=flat_angles, 

223 dark_angles_deg=dark_angles, 

224 captured_at=_parse_aware(captured_at_raw), 

225 captured_at_raw=captured_at_raw, 

226 captured_at_source=self._captured_at_source, 

227 byte_size=stat.st_size, 

228 mtime_ns=stat.st_mtime_ns, 

229 )

247def _angles(handle: h5py.File, path: str) -> tuple[float, ...] | None: 

248 dataset = _dataset(handle, path) 

249 if dataset is None: 

250 return None 

251 try: 

252 values = dataset[()] 

253 return tuple(float(value) for value in values) 

254 except (TypeError, ValueError, OSError): 

255 # A scalar or unreadable theta is not an angle list; report the 

256 # dataset as absent rather than fabricate a shape for it. For 

257 # /exchange/theta that also reads as structurally incomplete, 

258 # which is the honest verdict for a malformed one. 

259 return None

262def _scalar_int(handle: h5py.File, path: str) -> int | None: 

263 dataset = _dataset(handle, path) 

264 if dataset is None: 

265 return None 

266 try: 

267 value = dataset[()] 

268 except OSError: 

269 return None 

270 if isinstance(value, bytes): 

271 value = value.decode("utf-8", errors="replace") 

272 parsed = _as_int(value) 

273 if parsed is not None: 

274 return parsed 

275 # Same shape `_scalar_str` already handles: 2-BM writes every 

276 # /process/acquisition integer as a 1-element array, and NumPy 2 

277 # refuses int() on anything that is not 0-dimensional. Measured on 

278 # a real scan file: num_angles is (1,) int32, so reading only the 

279 # 0-d form left every commanded count None and made the shortfall 

280 # check unable to fire at all. A longer array is NOT a scalar 

281 # written oddly, so it stays unreadable rather than being reported 

282 # as its own first element. 

283 try: 

284 if len(value) != 1: # pyright: ignore[reportArgumentType] 

285 return None 

286 first = value[0] # pyright: ignore[reportIndexIssue] 

287 except (TypeError, IndexError, KeyError): 

288 return None 

289 if isinstance(first, bytes): 

290 first = first.decode("utf-8", errors="replace") 

291 return _as_int(first)

302 dataset = _dataset(handle, path) 

303 if dataset is None: 

304 return None 

305 try: 

306 value = dataset[()] 

307 except OSError: 

308 return None 

309 if isinstance(value, bytes): 

310 return value.decode("utf-8", errors="replace") 

311 if isinstance(value, str): 311 ↛ 312line 311 didn't jump to line 312 because the condition on line 311 was never true

312 return value

314 try: 

315 first = value[0] 

316 except (TypeError, IndexError, KeyError): 

317 return None 

318 if isinstance(first, bytes): 318 ↛ 320line 318 didn't jump to line 320 because the condition on line 318 was always true

319 return first.decode("utf-8", errors="replace") 

320 return str(first) if isinstance(first, str) else None

323def _phase_count(configured: int | None, mode: str | None) -> int | None: 

324 """Expected-count arithmetic per tomoscan: configured count times 

325 the phases the mode names. Unknown mode strings read as unknowable, 

326 not zero, so a new tomoscan mode cannot silently claim nothing was 

327 commanded.""" 

328 if configured is None or mode is None: 

329 return None 

330 if mode not in ("Start", "End", "Both", "None"): 

331 return None 

332 phases = (mode in ("Start", "Both")) + (mode in ("End", "Both")) 

333 return configured * phases

336def _parse_aware(raw: str | None) -> datetime | None: 

337 """Parse a timestamp string ONLY when it is timezone-aware. 

338 

339 A naive string stays unparsed: assigning it a zone would 

340 manufacture provenance, and the site's timezone rule is an open 

341 staff question that lands here once answered. 

342 """ 

343 if raw is None: 

344 return None 

345 try: 

346 parsed = datetime.fromisoformat(raw.strip()) 

347 except ValueError: 

348 return None 

349 return parsed if parsed.tzinfo is not None else None
