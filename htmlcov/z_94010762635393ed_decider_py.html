1"""Pure decider for the `ResumeRun` command. 

2 

3Resume transition: `Held -> Running`, discharging ONE hold claim, and 

4only when it is the last one active. The inverse of hold. 

5 

6## Three outcomes, not one 

7 

8A Run can be held by several concerns at once, so "release my hold" and 

9"resume the Run" stopped being the same act. The caller's own claim is 

10derived from (run_id, cause), and: 

11 

12 - own claim is the ONLY active claim -> `RunResumed(released_claim_id)`, 

13 Held → Running. The Run resumes because nothing is holding it. 

14 - own claim is active, others remain -> `HoldClaimReleased(claim_id)`, 

15 status unchanged. This concern has stopped holding; the Run stays Held 

16 on behalf of whoever else is. 

17 - own claim is NOT active -> `RunHoldClaimsRemainError`. 

18 Something else is holding this Run and the caller never placed a hold, 

19 so resuming would clear a hold it does not own. This is the branch that 

20 makes the reproduced fault impossible: a settled co-signature cannot 

21 resume a Run the kill-switch is holding. 

22 

23Emitting `RunResumed` only in the first case is what makes the transition 

24to `Running` mean "no concern is holding this Run" rather than "whoever 

25spoke last is done". 

26 

27Invariants: 

28 - State must not be None -> RunNotFoundError 

29 - State.status must be Held -> RunCannotResumeError(current_status=...) 

30 - `cause` must be in HOLD_CAUSES -> ValueError 

31 - Caller's claim must be active, else other claims block 

32 -> RunHoldClaimsRemainError(blocking_causes=...) 

33 

34## Legacy holds 

35 

36A hold placed before holds carried claims has no recorded owner, so no derived 

37claim id matches it, which would have made every Run held at the moment this 

38shipped permanently unresumable. Such a hold folds to `LEGACY_CLAIM_ID`, the 

39operator's resume owns it in addition to its own claim, and clearing it emits a 

40bare `RunResumed` that clears the hold outright: exactly the one-bit semantics 

41it was placed under. 

42""" 

43 

44from datetime import datetime 

45 

46from cora.run.aggregates.run import ( 

47 HOLD_CAUSE_OPERATOR, 

48 HOLD_CAUSES, 

49 LEGACY_CLAIM_ID, 

50 HoldClaimReleased, 

51 Run, 

52 RunCannotResumeError, 

53 RunHoldClaimsRemainError, 

54 RunNotFoundError, 

55 RunResumed, 

56 RunStatus, 

57 derive_claim_id, 

58) 

59from cora.run.features.resume_run.command import ResumeRun 

60 

61_RESUMABLE_STATUSES: tuple[RunStatus, ...] = (RunStatus.HELD,) 

62 

63 

64def decide( 

65 state: Run | None, 

66 command: ResumeRun, 

67 *, 

68 now: datetime, 

69) -> list[RunResumed | HoldClaimReleased]: 

70 """Decide the events produced by releasing this caller's hold on a Run.""" 

71 if state is None: 

72 raise RunNotFoundError(command.run_id) 

73 if state.status not in _RESUMABLE_STATUSES: 

74 raise RunCannotResumeError(state.id, current_status=state.status) 

75 if command.cause not in HOLD_CAUSES: 75 ↛ 76line 75 didn't jump to line 76 because the condition on line 75 was never true

76 raise ValueError( 

77 f"Unknown hold cause {command.cause!r}; expected one of {sorted(HOLD_CAUSES)}" 

78 ) 

79 claim_id = derive_claim_id(state.id, command.cause) 

80 active = tuple(active_id for active_id, _ in state.hold_claims) 

81 if not active: 

82 # Held with no active claim. Not reachable from a well-formed stream (a 

83 # `RunHeld` always yields at least the legacy claim), but if it happens 

84 # the safety property is satisfied (no concern is holding this Run), so 

85 # resume rather than wedge it. 

86 return [ 

87 RunResumed( 

88 run_id=state.id, 

89 decided_by_decision_id=command.decided_by_decision_id, 

90 released_claim_id=None, 

91 occurred_at=now, 

92 ) 

93 ] 

94 # A hold placed BEFORE holds were cause-scoped has no recorded owner, so no 

95 # derived claim id can match it and it would otherwise be unresumable: 

96 # every Run held at the moment this shipped. The operator is the authority 

97 # that could always clear such a hold, so the operator's resume owns the 

98 # legacy claim in addition to its own. 

99 owned = {claim_id} 

100 if command.cause == HOLD_CAUSE_OPERATOR: 

101 owned.add(LEGACY_CLAIM_ID) 

102 held_by_caller = tuple(cid for cid in active if cid in owned) 

103 if not held_by_caller: 

104 # Held, but not by us. Refuse and name who is, so the caller learns 

105 # which concern to address instead of only that it was refused. 

106 raise RunHoldClaimsRemainError( 

107 state.id, 

108 blocking_causes=tuple(cause for _, cause in state.hold_claims), 

109 ) 

110 # Legacy one-bit hold: clearing it means clearing the hold outright, which is 

111 # what a bare `RunResumed` does at the fold. 

112 if set(active) <= {LEGACY_CLAIM_ID}: 

113 return [ 

114 RunResumed( 

115 run_id=state.id, 

116 decided_by_decision_id=command.decided_by_decision_id, 

117 released_claim_id=None, 

118 occurred_at=now, 

119 ) 

120 ] 

121 claim_id = held_by_caller[0] 

122 if len(active) > 1: 

123 # Others still hold it: discharge our claim without moving status. 

124 return [ 

125 HoldClaimReleased( 

126 run_id=state.id, 

127 claim_id=claim_id, 

128 cause=command.cause, 

129 decided_by_decision_id=command.decided_by_decision_id, 

130 occurred_at=now, 

131 ) 

132 ] 

133 return [ 

134 RunResumed( 

135 run_id=state.id, 

136 decided_by_decision_id=command.decided_by_decision_id, 

137 released_claim_id=claim_id, 

138 occurred_at=now, 

139 ) 

140 ]
