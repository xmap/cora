1"""Per-call GPU-time attribution under concurrent serving. 

2 

3The built serving path runs an open model on facility GPUs. In the 

4locked design in-house serving is metered-free by default (the debit, 

5when a facility toggles it on, stays token-priced), so this module is 

6VISIBILITY, not the debit: it reports how much GPU time each call 

7actually consumed, so a facility can watch shadow cost and capacity and 

8decide when in-house volume is worth revisiting. It never feeds the 

9budget gate. 

10 

11The measurement problem it solves is real. Under continuous batching 

12many calls occupy one device at once, so attributing each call its full 

13wall-clock duration double-counts: four calls that share a GPU for one 

14second would report four GPU-seconds for one second of hardware. 

15 

16This module attributes the shared time honestly. At every instant the 

17GPU-second is divided among the calls in flight, weighted; each call 

18accrues its share. The scheme is processor-sharing, the same fair 

19division long used for CPU-second accounting, and it carries one 

20exactness guarantee that makes the measurement trustworthy: 

21 

22 For each device, the per-call shares sum to the device's busy 

23 wall-time (the measure of the union of the call intervals). 

24 

25So the reported GPU-seconds never exceed what the hardware ran, and no 

26call's share exceeds its own duration. That per-call ceiling is a 

27property of the measure that keeps the shadow-cost figure honest (and 

28would license a solo GPU-hour pre-estimate only IF a facility ever chose 

29to debit GPU-hours, which the locked design defers). 

30 

31## Equal versus weighted division 

32 

33The default divides each shared instant equally. This is not only 

34convenient: during the decode phase of continuous-batched inference 

35every active sequence advances by about one token per forward step, so 

36the instantaneous work of the in-flight calls is roughly equal. A 

37facility that prefers a different work proxy (prompt length, to charge 

38the prefill-heavy call more) sets a per-call `weight`; the exactness 

39guarantee holds for any positive weights, because each shared instant 

40distributes its full length however it is split. 

41 

42## Two forms 

43 

44`attribute_occupancy_share` is a batch pure function over a whole set of 

45intervals: the reconciliation oracle, and the deterministic target the 

46property tests pin the online form against. `OccupancyShareMeter` is the 

47online accumulator the serving adapter drives: it finalizes a call's 

48share the instant the call ends, which is exact because every call that 

49overlapped it has by then started, and any still in flight covers the 

50whole elapsed window up to that instant. 

51 

52Time is supplied by the caller (monotonic seconds from an injected 

53clock); this module never reads a clock itself, so both forms are pure 

54functions of their inputs and reproduce identically in a test, in 

55production, and in an after-the-fact reconciliation. Turning the 

56attributed GPU-seconds into an indicative shadow cost is a facility / 

57observability concern, not this module's, and the locked design adds no 

58GPU-hour branch to the cost path: here the unit is seconds throughout. 

59"""

113def attribute_occupancy_share(intervals: Iterable[ServedInterval]) -> dict[str, float]: 

114 """Attribute per-call GPU-seconds by weighted processor-sharing. 

115 

116 Returns a mapping from `call_id` to the GPU-seconds it is charged, 

117 with every supplied `call_id` present (a zero-duration call maps to 

118 `0.0`). Calls on different devices never share, so the computation 

119 runs per device: for each device the timeline is cut at every start 

120 and end, and each resulting slice's length is divided among the 

121 calls live across the whole slice in proportion to their weights. 

122 

123 Per device the returned charges sum to the union of the intervals 

124 (the device's busy wall-time), to within floating-point error. Two 

125 intervals that share a `call_id` (a retried call, say) accumulate 

126 into one total. 

127 """ 

128 by_device: dict[str, list[ServedInterval]] = {} 

129 charges: dict[str, float] = {} 

130 for interval in intervals: 

131 by_device.setdefault(interval.device_id, []).append(interval) 

132 charges.setdefault(interval.call_id, 0.0) 

133 

134 for device_intervals in by_device.values(): 

135 _accrue_device(device_intervals, charges) 

136 return charges

139def _accrue_device(intervals: list[ServedInterval], charges: dict[str, float]) -> None: 

140 """Sweep one device's timeline, crediting each slice to its live calls. 

141 

142 Boundaries are every distinct start and end, so within a slice no 

143 call opens or closes: a call is live across the slice exactly when it 

144 starts at or before the slice's left edge and ends at or after its 

145 right edge. 

146 """ 

147 boundaries = sorted({iv.start_s for iv in intervals} | {iv.end_s for iv in intervals}) 

148 for left, right in itertools.pairwise(boundaries): 

149 span = right - left 

150 if span <= 0.0: 150 ↛ 151line 150 didn't jump to line 151 because the condition on line 150 was never true

151 continue 

152 live = [iv for iv in intervals if iv.start_s <= left and iv.end_s >= right] 

153 total_weight = math.fsum(iv.weight for iv in live) 

154 if total_weight <= 0.0: 

155 continue 

156 for iv in live: 

157 charges[iv.call_id] += span * iv.weight / total_weight

167class OccupancyShareMeter: 

168 """Online weighted processor-sharing accumulator for served calls. 

169 

170 The serving adapter calls `open` when a request begins occupying a 

171 device and `close` when it releases it, passing monotonic seconds 

172 from its injected clock; `close` returns the call's final GPU-second 

173 charge. The results match `attribute_occupancy_share` over the same 

174 events exactly (a property test pins this). 

175 

176 Concurrency is per device: `open` and `close` advance only the 

177 device the call is on, so calls on different devices never dilute 

178 each other. The meter holds only the currently-live calls, so its 

179 memory is bounded by peak concurrency, not by total calls served. 

180 """ 

181 

182 def __init__(self) -> None: 

183 self._live: dict[str, _LiveCall] = {} 

184 self._device_clock: dict[str, float] = {} 

185 self._device_weight: dict[str, float] = {} 

186 

187 def open(self, call_id: str, *, device_id: str, at_s: float, weight: float = 1.0) -> None: 

188 """Record that `call_id` begins occupying `device_id` at `at_s`.""" 

189 _require_finite("at_s", at_s) 

190 _require_positive_finite_weight(weight) 

191 if call_id in self._live: 

192 msg = f"call {call_id!r} is already open on this meter" 

193 raise ValueError(msg) 

194 self._advance(device_id, at_s) 

195 self._live[call_id] = _LiveCall(device_id=device_id, weight=weight, accrued_s=0.0) 

196 self._device_weight[device_id] = self._device_weight.get(device_id, 0.0) + weight 

197 

198 def close(self, call_id: str, *, at_s: float) -> float: 

199 """Finalize `call_id` at `at_s` and return its GPU-second charge.""" 

200 _require_finite("at_s", at_s) 

201 live = self._live.get(call_id) 

202 if live is None: 

203 msg = f"call {call_id!r} is not open on this meter" 

204 raise ValueError(msg) 

205 self._advance(live.device_id, at_s) 

206 del self._live[call_id] 

207 self._device_weight[live.device_id] -= live.weight 

208 return live.accrued_s 

209 

210 def _advance(self, device_id: str, to_s: float) -> None: 

211 """Credit the elapsed slice since this device's last event to its live calls.""" 

212 last = self._device_clock.get(device_id) 

213 if last is None: 

214 self._device_clock[device_id] = to_s 

215 return 

216 if to_s < last: 

217 msg = f"device {device_id!r} clock went backward: {last} -> {to_s}" 

218 raise ValueError(msg) 

219 self._device_clock[device_id] = to_s 

220 elapsed = to_s - last 

221 if elapsed == 0.0: 

222 return 

223 total_weight = self._device_weight.get(device_id, 0.0) 

224 if total_weight <= 0.0: 

225 return 

226 for live in self._live.values(): 

227 if live.device_id == device_id: 

228 live.accrued_s += elapsed * live.weight / total_weight
