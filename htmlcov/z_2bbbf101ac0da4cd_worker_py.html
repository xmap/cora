1"""ProjectionWorker: advances every registered Projection along the event stream. 

2 

3One asyncio task per projection inside an `asyncio.TaskGroup`. Each 

4task runs an independent advance loop: 

5 

6 1. Read bookmark (FOR UPDATE). 

7 2. Run the canonical Khyst+Dudycz advance query: events with 

8 `(transaction_id, position) > bookmark` AND `transaction_id < 

9 pg_snapshot_xmin(...)` (in-flight exclusion) AND 

10 `event_type = ANY(subscribed_types)`, ORDER BY (transaction_id, 

11 position), LIMIT batch_size. 

12 3. Call `apply(event, conn)` per row. 

13 4. Update bookmark to the last processed (transaction_id, position). 

14 5. Commit. If anything raised, the entire batch rolls back and the 

15 bookmark stays put — at-least-once delivery on retry. 

16 

17When a batch returns zero events, the loop sleeps via the WakeupSource 

18(LISTEN/NOTIFY or poll). Errors in the loop trigger exponential 

19backoff (1s -> 2s -> ... cap 60s) with a loud log; the loop never 

20silently skips an event. 

21""" 

22

44# Khyst + Dudycz canonical advance query. Comparison is lexicographic 

45# `(transaction_id, position) > (last_tx, last_pos)`; in-flight 

46# exclusion uses `transaction_id < pg_snapshot_xmin(...)`. Uses the 

47# `events_advance_idx` (transaction_id, position) added in migration 

48# 20260512240000. 

49_ADVANCE_SQL = """ 

50SELECT position, event_id, stream_type, stream_id, version, event_type, 

51 schema_version, payload, metadata, correlation_id, causation_id, 

52 principal_id, occurred_at, recorded_at, 

53 transaction_id::text AS transaction_id_text 

54FROM events 

55WHERE (transaction_id, position) > ($1::xid8, $2) 

56 AND transaction_id < pg_snapshot_xmin(pg_current_snapshot()) 

57 AND event_type = ANY($3::text[]) 

58ORDER BY transaction_id ASC, position ASC 

59LIMIT $4 

60"""

62_BACKOFF_BASE_SECONDS = 1.0 

63_BACKOFF_MAX_SECONDS = 60.0

66def _row_to_stored_event(row: Any) -> StoredEvent: 

67 """Mirror of postgres.event_store._row_to_event for the advance query. 

68 Duplicated rather than imported because the advance query has its 

69 own SELECT shape (transaction_id alias) that differs from the 

70 stream-load query. 

71 

72 Both projections of the SELECT must surface every column on 

73 StoredEvent, otherwise subscribers see stale `None` values for 

74 fields that exist on the row but were dropped here. Keep the 

75 column lists in lock-step with `postgres.event_store._LOAD_SQL` 

76 when adding new envelope fields. 

77 """ 

78 return StoredEvent( 

79 position=int(row["position"]), 

80 event_id=row["event_id"], 

81 stream_type=str(row["stream_type"]), 

82 stream_id=row["stream_id"], 

83 version=int(row["version"]), 

84 event_type=str(row["event_type"]), 

85 schema_version=int(row["schema_version"]), 

86 payload=row["payload"], 

87 metadata=row["metadata"], 

88 correlation_id=row["correlation_id"], 

89 causation_id=row["causation_id"], 

90 occurred_at=row["occurred_at"], 

91 recorded_at=row["recorded_at"], 

92 transaction_id=int(row["transaction_id_text"]), 

93 principal_id=row["principal_id"], 

94 )

97async def advance_subscriber_once( 

98 pool: asyncpg.Pool, 

99 subscriber: Subscriber, 

100 *, 

101 batch_size: int = DEFAULT_BATCH_SIZE, 

102) -> int: 

103 """Advance one Subscriber by at most `batch_size` events. Returns 

104 the number of events processed (0 if the bookmark is already at 

105 the head of the visible-and-committed event stream). 

106 

107 The whole batch — read-bookmark + apply-all + write-bookmark — 

108 runs in a single transaction so at-least-once delivery has clean 

109 semantics: either everything in the batch advances together, or 

110 nothing does. 

111 

112 Callers from the worker pass the per-subscriber batch_size 

113 (`getattr(subscriber, "batch_size", DEFAULT_BATCH_SIZE)`); callers 

114 from the drain helper pass an explicit value to bound test cost. 

115 """ 

116 subscribed = sorted(subscriber.subscribed_event_types) 

117 async with pool.acquire() as conn, conn.transaction(): 

118 last_tx, last_pos = await read_bookmark(conn, subscriber.name) 

119 rows = await conn.fetch( 

120 _ADVANCE_SQL, 

121 last_tx, 

122 last_pos, 

123 subscribed, 

124 batch_size, 

125 ) 

126 if not rows: 

127 return 0 

128 events = [_row_to_stored_event(row) for row in rows] 

129 for event in events: 

130 await subscriber.apply(event, conn) 

131 last = events[-1] 

132 await write_bookmark( 

133 conn, 

134 subscriber.name, 

135 last_transaction_id=last.transaction_id, 

136 last_position=last.position, 

137 last_event_recorded_at=last.recorded_at, 

138 ) 

139 return len(events)

142class ProjectionWorker: 

143 """Spawns one advance loop per registered projection.""" 

144 

145 def __init__( 

146 self, 

147 pool: asyncpg.Pool, 

148 registry: ProjectionRegistry, 

149 wakeup: WakeupSource, 

150 *, 

151 poll_interval_seconds: float = 5.0, 

152 batch_size: int = DEFAULT_BATCH_SIZE, 

153 ) -> None: 

154 self._pool = pool 

155 self._registry = registry 

156 self._wakeup = wakeup 

157 self._poll_interval_seconds = poll_interval_seconds 

158 # Per-subscriber default. Each Subscriber may declare its own 

159 # `batch_size` attribute (Reactions set this to 1); the loop 

160 # below reads `subscriber.batch_size` via getattr so the 

161 # worker-level value is the fallback only. 

162 self._batch_size = batch_size 

163 

164 async def run(self) -> None: 

165 """Run advance loops until cancelled. Per-projection failures 

166 stay inside their own loop (caught + backed-off) so one 

167 misbehaving projection cannot bring the whole worker down. 

168 

169 Empty-registry case is gated by `projection_worker_lifespan` 

170 which short-circuits before constructing the worker, so 

171 `TaskGroup` always has at least one task here. 

172 """ 

173 async with asyncio.TaskGroup() as tg: 

174 for projection in self._registry: 

175 tg.create_task( 

176 self._advance_loop(projection), 

177 name=f"projection-advance:{projection.name}", 

178 ) 

179 

180 async def _advance_loop(self, projection: Subscriber) -> None: 

181 """Single projection's advance loop. Runs forever until 

182 cancelled.""" 

183 backoff = _BACKOFF_BASE_SECONDS 

184 # Per-subscriber batch_size: Projections inherit the worker 

185 # default (100), Reactions override to 1 to avoid holding a 

186 # pool connection across a slow LLM call. Read once at loop 

187 # start since the class-level attribute is immutable per 

188 # subscriber instance. 

189 effective_batch_size = getattr(projection, "batch_size", self._batch_size) 

190 while True: 

191 try: 

192 processed = await advance_subscriber_once( 

193 self._pool, 

194 projection, 

195 batch_size=effective_batch_size, 

196 ) 

197 backoff = _BACKOFF_BASE_SECONDS # reset on success 

198 if processed > 0: 

199 _log.debug( 

200 "projection.advance.batch", 

201 projection=projection.name, 

202 events=processed, 

203 ) 

204 else: 

205 await self._wakeup.wait(self._poll_interval_seconds) 

206 except asyncio.CancelledError: 

207 raise 

208 except Exception as exc: 

209 _log.exception( 

210 "projection.advance.error", 

211 projection=projection.name, 

212 backoff_seconds=backoff, 

213 ) 

214 # persist failure state in a separate 

215 # small transaction so the operator can see "this 

216 # projection has failed N times since last success 

217 # with this error" without log-spelunking. If THIS 

218 # write itself fails we swallow it; we already logged 

219 # the original error and don't want failure-recording 

220 # to itself crash the loop. 

221 with contextlib.suppress(Exception): 

222 await write_bookmark_failure( 

223 self._pool, 

224 projection.name, 

225 error_message=str(exc), 

226 ) 

227 await asyncio.sleep(backoff) 

228 backoff = min(backoff * 2, _BACKOFF_MAX_SECONDS) 

229 

230 

231__all__ = ["ProjectionWorker", "advance_subscriber_once"]
