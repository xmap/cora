1"""Operation BC factory: materialise a `DecidePort` from a config. 

2 

3`build_decide_port(config)` returns a `DecidePort` for the configured 

4decider substrate: 

5 

6 - `InMemoryDecidePort` (no config / `in_memory` substrate; the default + 

7 test convenience; replays seeded advice then advises Stop) 

8 - `GridWalkDecidePort` (`grid_walk` substrate; a deterministic, stateless 

9 grid/sweep decider over the SteeringSpace, no external optimizer) 

10 

11Two in-CORA arms today, mirroring `build_compute_port`: a later external 

12optimizer adapter (gpcam) lands as an additive arm here, and a routing 

13registry is earned only when that arrives, exactly as ControlPort earned its 

14registry from a third substrate and ComputePort deferred its. 

15 

16## Lifecycle 

17 

18The returned port owns the lifecycle of any decider resource it holds: 

19`aclose()` is a no-op for the in-CORA deciders and would release a model 

20client / optimizer subprocess for a real external adapter. The caller 

21`aclose()`s it at teardown without branching on type. 

22""" 

23 

24from __future__ import annotations 

25 

26from dataclasses import dataclass 

27from typing import TYPE_CHECKING, Literal 

28 

29from cora.operation.adapters.grid_walk_decide_port import GridWalkDecidePort 

30from cora.operation.adapters.in_memory_decide_port import InMemoryDecidePort 

31 

32if TYPE_CHECKING: 

33 from cora.operation.ports.decide_port import DecidePort 

34 

35DecideSubstrate = Literal["in_memory", "grid_walk"] 

36"""Closed set of decider substrates with a shipped adapter. 

37 

38`in_memory` is the deterministic fake; `grid_walk` is the in-CORA grid/sweep 

39decider. A future external optimizer (such as `gpcam`) lands as an additive 

40arm here plus the registry this factory does not yet build. 

41""" 

42 

43 

44@dataclass(frozen=True) 

45class DecidePortConfig: 

46 """Deployment config for the DecidePort substrate. 

47 

48 `substrate` selects the adapter. `points_per_axis` is the grid-walk 

49 resolution for continuous axes (ignored by the in-memory fake and by 

50 axes that carry explicit choices). A full route table is deferred to the 

51 external optimizer, mirroring `ComputePortConfig`. 

52 """ 

53 

54 substrate: DecideSubstrate = "in_memory" 

55 points_per_axis: int = 5 

56 

57 

58def build_decide_port(config: DecidePortConfig | None = None) -> DecidePort: 

59 """Materialise the DecidePort the conduct loop talks to. 

60 

61 None or the `in_memory` substrate returns an `InMemoryDecidePort` (the 

62 default + test convenience). `grid_walk` returns a `GridWalkDecidePort` 

63 at the configured resolution. A future external optimizer adds its arm 

64 here as it is earned, exactly as `build_compute_port` grew its 

65 `local_process` arm. 

66 """ 

67 resolved = config if config is not None else DecidePortConfig() 

68 if resolved.substrate == "in_memory": 

69 return InMemoryDecidePort() 

70 if resolved.substrate == "grid_walk": 

71 return GridWalkDecidePort(points_per_axis=resolved.points_per_axis) 

72 raise ValueError( # pragma: no cover 

73 f"unsupported decide substrate: {resolved.substrate!r}" 

74 ) 

75 

76 

77__all__ = ["DecidePortConfig", "DecideSubstrate", "build_decide_port"]
