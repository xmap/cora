1"""Application handler for the `conduct_procedure` slice. 

2 

3Thin orchestrator that delegates to `Conductor.conduct()`. The 

4handler's job is the application-layer concerns the Conductor 

5itself does not own: command-level authorization (the per-step 

6`append_activities` calls already authz internally, but the 

7ConductProcedure entry point gates the entire invocation), envelope 

8threading, recipe-replay re-expansion when the Procedure was created 

9via `register_procedure_from_recipe`, and result conversion from 

10`ConductorResult` to the slice's `ConductProcedureResult` contract. 

11 

12## Why no `_decider` 

13 

14Unlike CQRS slices that compute events from state, `conduct_procedure` 

15records no new events on the Procedure stream directly: the wrapped 

16`start_procedure` / `append_activities` / `complete_procedure` / 

17`abort_procedure` handlers are the things that write. The slice is 

18an orchestration entry point, NOT an aggregate-state-mutating 

19decider. Therefore no `decider.py`, no `context.py`. 

20 

21## Recipe-driven re-expansion 

22 

23When the loaded Procedure has `recipe_id is not None`, the handler 

24treats it as recipe-driven and runs the five-step replay gate 

25specified by [[project-run-procedure-replay-design]]: 

26forbid-non-empty-caller-steps -> find_recipe_expansion_record -> 

27pins_from_payload -> port-version strict-equals guard -> 

28load_recipe_at_version -> verify_bindings_hash -> expand -> 

29verify_steps_hash -> hand fresh steps to Conductor. Legacy 

30Procedures (`recipe_id is None`) hand `command.steps` to the 

31Conductor unchanged. 

32 

33## Authorization scope 

34 

35`ConductProcedure` is authz-checked as a distinct command. The wrapped 

36handlers (start / append / complete / abort) each authz internally 

37with their OWN command names; an operator authorized to call 

38`ConductProcedure` is NOT automatically authorized for each of those 

39individually. That's correct: `ConductProcedure` is the 

40operator-friendly entry; the underlying per-FSM-transition 

41authorization is what the policy engine actually evaluates at each 

42call site. 

43"""

66 

67_log = get_logger(__name__) 

68 

69 

70class Handler(Protocol): 

71 """Callable interface every conduct_procedure handler implements.""" 

72 

73 async def __call__( 

74 self, 

75 command: ConductProcedure, 

76 *, 

77 principal_id: UUID, 

78 correlation_id: UUID, 

79 causation_id: UUID | None = None, 

80 surface_id: UUID = NIL_SENTINEL_ID, 

81 ) -> ConductProcedureResult: ... 

82 

83 

84def bind( 

85 deps: Kernel, 

86 *, 

87 conductor: Conductor, 

88 expansion_port: RecipeExpander, 

89) -> Handler: 

90 """Build a conduct_procedure handler closed over deps + Conductor + port. 

91 

92 `conductor` is BC-internal: wire_operation constructs it from 

93 the bound FSM handlers + ControlPort + Kernel infra ports. 

94 `expansion_port` is the same instance wired for 

95 `register_procedure_from_recipe`; replay reads its `version` 

96 attribute and calls `expand` against the pinned bindings. The 

97 `event_store` is read via `deps.event_store` at the 

98 `load_procedure_with_events` call site (no separate kwarg). 

99 """ 

100 

101 async def handler( 

102 command: ConductProcedure, 

103 *, 

104 principal_id: UUID, 

105 correlation_id: UUID, 

106 causation_id: UUID | None = None, 

107 surface_id: UUID = NIL_SENTINEL_ID, 

108 ) -> ConductProcedureResult: 

109 _log.info( 

110 "conduct_procedure.start", 

111 command_name=_COMMAND_NAME, 

112 procedure_id=str(command.procedure_id), 

113 step_count=len(command.steps), 

114 principal_id=str(principal_id), 

115 correlation_id=str(correlation_id), 

116 causation_id=str(causation_id) if causation_id is not None else None, 

117 ) 

118 

119 authz = await deps.authz.authorize( 

120 principal_id=principal_id, 

121 command_name=_COMMAND_NAME, 

122 conduit_id=NIL_SENTINEL_ID, 

123 surface_id=surface_id, 

124 ) 

125 if isinstance(authz, Deny): 

126 _log.info( 

127 "conduct_procedure.denied", 

128 command_name=_COMMAND_NAME, 

129 procedure_id=str(command.procedure_id), 

130 principal_id=str(principal_id), 

131 correlation_id=str(correlation_id), 

132 reason=authz.reason, 

133 ) 

134 raise UnauthorizedError(authz.reason) 

135 

136 procedure, stored_events = await load_procedure_with_events( 

137 deps.event_store, command.procedure_id 

138 ) 

139 if procedure is None: 

140 raise ProcedureNotFoundError(command.procedure_id) 

141 

142 steps, closing_steps = await resolve_and_pin_conduct_steps( 

143 deps, 

144 command_name=_COMMAND_NAME, 

145 procedure=procedure, 

146 stored_events=stored_events, 

147 caller_steps=command.steps, 

148 expansion_port=expansion_port, 

149 principal_id=principal_id, 

150 correlation_id=correlation_id, 

151 causation_id=causation_id, 

152 ) 

153 

154 result = await conductor.conduct( 

155 procedure_id=command.procedure_id, 

156 principal_id=principal_id, 

157 correlation_id=correlation_id, 

158 causation_id=causation_id, 

159 surface_id=surface_id, 

160 steps=steps, 

161 closing_steps=closing_steps, 

162 ) 

163 

164 _log.info( 

165 "conduct_procedure.success", 

166 command_name=_COMMAND_NAME, 

167 procedure_id=str(command.procedure_id), 

168 completed_count=result.completed_count, 

169 succeeded=result.succeeded, 

170 failure_class=(result.failure.error_class if result.failure is not None else None), 

171 ) 

172 

173 return ConductProcedureResult( 

174 procedure_id=result.procedure_id, 

175 completed_count=result.completed_count, 

176 succeeded=result.succeeded, 

177 failure=result.failure, 

178 actuation_kind=( 

179 result.actuation_kind.value if result.actuation_kind is not None else None 

180 ), 

181 measurements=result.measurements, 

182 artifacts=result.artifacts, 

183 outputs=result.outputs, 

184 substrate_writes=result.substrate_writes, 

185 closing_failures=result.closing_failures, 

186 ) 

187 

188 return handler
