1"""Read repository for the Assembly aggregate. 

2 

3`load_assembly(event_store, assembly_id) -> Assembly | None` mirrors 

4`load_mount` / `load_frame` / `load_asset`. Used by update-style 

5commands (`version_assembly`, `deprecate_assembly`, 

6`register_fixture`) that need to load + fold before deciding. 

7 

8`resolve_sub_assembly_pins` is the shared cross-aggregate check that 

9the `define_assembly` and `version_assembly` handlers run over a 

10command's `required_sub_assemblies`: it loads each referenced child 

11Assembly once and classifies every problem the authoring path must 

12reject before a parent blueprint is written. It returns a 

13`SubAssemblyResolution` so the classifications stay named rather than 

14positional as the set of checks grows. 

15""" 

16 

17import asyncio 

18from dataclasses import dataclass, field 

19from uuid import UUID 

20 

21from cora.equipment.aggregates.assembly.events import from_stored 

22from cora.equipment.aggregates.assembly.evolver import fold 

23from cora.equipment.aggregates.assembly.state import Assembly, SubAssemblyLink 

24from cora.infrastructure.ports import EventStore 

25 

26_STREAM_TYPE = "Assembly" 

27 

28 

29@dataclass(frozen=True) 

30class SubAssemblyResolution: 

31 """Cross-aggregate classification of a command's sub-assembly refs. 

32 

33 Every field is computed from the SAME single load of each 

34 referenced child, so the authoring path (define / version) rejects 

35 the same shapes `register_fixture` rejects, but at authoring time 

36 where the operator can still act. 

37 

38 - `missing_ids`: sub_assembly_ids that do not resolve to a 

39 defined Assembly. 

40 - `hash_mismatches`: `(sub_assembly_id, pinned, current)` for 

41 refs whose pinned content_hash has drifted from the loaded 

42 child's current content_hash (snapshot drift; the parent must 

43 re-pin via a fresh define / version). 

44 - `too_deep_ids`: sub_assembly_ids of children that THEMSELVES 

45 declare `required_sub_assemblies`. One composing level is 

46 supported, so a child-of-a-child makes the parent 

47 un-instantiable; rejecting at authoring time keeps "a defined 

48 Assembly is always instantiable" a real invariant and, since a 

49 non-leaf child is refused, closes the A->B->A indirect-cycle 

50 hole for the two-node case. 

51 - `leaf_slot_collisions`: leaf slot_names that appear in more 

52 than one composed blueprint once the parent's own leaf slots 

53 and every leaf child's leaf slots are merged into the single 

54 flat namespace `register_fixture` materializes. Surfacing here 

55 means the operator does not discover the clash only at the end 

56 of the install-then-register choreography. 

57 """ 

58 

59 missing_ids: frozenset[UUID] = field(default_factory=frozenset[UUID]) 

60 hash_mismatches: frozenset[tuple[UUID, str, str | None]] = field( 

61 default_factory=frozenset[tuple[UUID, str, str | None]] 

62 ) 

63 too_deep_ids: frozenset[UUID] = field(default_factory=frozenset[UUID]) 

64 leaf_slot_collisions: frozenset[str] = field(default_factory=frozenset[str]) 

65

67async def load_assembly(event_store: EventStore, assembly_id: UUID) -> Assembly | None: 

68 """Load and fold an Assembly's event stream into current state.""" 

69 stored, _version = await event_store.load(_STREAM_TYPE, assembly_id) 

70 events = [from_stored(s) for s in stored] 

71 return fold(events)

74async def resolve_sub_assembly_pins( 

75 event_store: EventStore, 

76 refs: frozenset[SubAssemblyLink], 

77 *, 

78 parent_slot_names: frozenset[str] = frozenset(), 

79) -> SubAssemblyResolution: 

80 """Classify a command's sub-assembly references against the store. 

81 

82 Each distinct sub_assembly_id is loaded once (concurrently); every 

83 ref is then classified into the four fields of 

84 `SubAssemblyResolution`. The decider raises the matching 

85 sorted-first error from each set, so the handler stays free of 

86 domain-error decisions. 

87 

88 `parent_slot_names` is the parent command's own leaf slot_names; it 

89 seeds the flat namespace used to detect `leaf_slot_collisions` 

90 across the parent and its (leaf) children, mirroring the union that 

91 `register_fixture` builds at materialization time. Missing and 

92 too-deep children are excluded from the collision scan because they 

93 are rejected on their own grounds first. 

94 """ 

95 unique_ids = {ref.sub_assembly_id for ref in refs} 

96 if not unique_ids: 

97 return SubAssemblyResolution() 

98 ordered = sorted(unique_ids, key=str) 

99 loaded = await asyncio.gather(*(load_assembly(event_store, sub_id) for sub_id in ordered)) 

100 by_id = dict(zip(ordered, loaded, strict=True)) 

101 

102 missing = frozenset(sub_id for sub_id in unique_ids if by_id[sub_id] is None) 

103 too_deep = frozenset( 

104 sub_id 

105 for sub_id in unique_ids 

106 if (child := by_id[sub_id]) is not None and child.required_sub_assemblies 

107 ) 

108 

109 mismatches: set[tuple[UUID, str, str | None]] = set() 

110 for ref in refs: 

111 child = by_id[ref.sub_assembly_id] 

112 if child is not None and child.content_hash != ref.content_hash: 

113 mismatches.add((ref.sub_assembly_id, ref.content_hash, child.content_hash)) 

114 

115 # Replay register_fixture's flat-namespace union: seed with the 

116 # parent's own leaf slots, then fold in each leaf child's leaf 

117 # slots, recording any name that lands on a name already seen. Only 

118 # leaf children participate; missing / too-deep children are 

119 # rejected before the materialization union would ever form. 

120 seen_names = set(parent_slot_names) 

121 collisions: set[str] = set() 

122 for sub_id in ordered: 

123 child = by_id[sub_id] 

124 if child is None or child.required_sub_assemblies: 

125 continue 

126 for slot in sorted(child.required_slots, key=lambda s: s.slot_name.value): 

127 name = slot.slot_name.value 

128 if name in seen_names: 

129 collisions.add(name) 

130 else: 

131 seen_names.add(name) 

132 

133 return SubAssemblyResolution( 

134 missing_ids=missing, 

135 hash_mismatches=frozenset(mismatches), 

136 too_deep_ids=too_deep, 

137 leaf_slot_collisions=frozenset(collisions), 

138 )
