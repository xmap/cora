1"""BC-application-layer errors for the Agent BC. 

2 

3These errors are raised by application handlers (not domain logic) 

4and mapped to HTTP / MCP responses by the BC's exception handlers in 

5`cora/agent/routes.py`. 

6 

7Domain errors (raised by aggregates / deciders) live with their 

8aggregate at `aggregates/agent/state.py`. 

9 

10Distinct class from each other BC's `UnauthorizedError`: each BC 

11owns its own application-error namespace so an Agent 403 is 

12distinguishable from other BCs' 403s in logs / aggregator filters 

13(documented in CONTRIBUTING.md "BC-application-layer errors"). 

14"""

75class DecisionNotEmittedByCautionDrafterError(Exception): 

76 """Provenance check failed: the Decision's actor is not a CautionDrafter agent. 

77 

78 Closes the authority-to-emit gap: a Decision's 

79 `context=CautionProposal` is necessary but not sufficient. The 

80 promote slice must also 

81 verify the producing actor is a registered Agent of kind 

82 `CautionDrafter`. Without this check, any actor able to write 

83 a `DecisionRegistered` envelope with `context=CautionProposal` 

84 could have arbitrary Caution events promoted into the Caution 

85 stream (W3C PROV-O / FDA CDS Criterion 3 pattern: receiver 

86 re-validates attribution against a producer registry). 

87 

88 Raised by the `promote_caution_proposal` handler after loading 

89 the source Decision but before passing it to the pure decider. 

90 Maps to HTTP 403. 

91 """ 

92 

93 def __init__( 

94 self, 

95 *, 

96 decision_id: UUID, 

97 actor_id: UUID, 

98 observed_kind: str | None, 

99 ) -> None: 

100 observed = f"kind={observed_kind!r}" if observed_kind else "not a registered Agent" 

101 super().__init__( 

102 f"Decision {decision_id} actor_id {actor_id} is {observed}; " 

103 "promote_caution_proposal requires the Decision to have been " 

104 "emitted by an Agent of kind 'CautionDrafter'" 

105 ) 

106 self.decision_id = decision_id 

107 self.actor_id = actor_id 

108 self.observed_kind = observed_kind
