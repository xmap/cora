1"""ByteSigner port: substrate-neutral signature production over canonical bytes. 

2 

3`ByteSigner` is the sibling to `Canonicalizer`. The two are 

4paired by `adapter_version` (a v1 ByteSigner signs over v1 

5canonicalized bytes); cross-version pairing is rejected at the 

6port boundary via `CanonicalizationVersionMismatchError`. 

7 

8The v1 adapter Ed25519-signs over `CanonicalizedBytes.bytes_` and 

9narrows `KeyHandle` to a `JwksKid` carrying the JWKS kid string. 

10Future arms can narrow `KeyHandle` differently (an X.509 chain 

11reference for Sigstore-keyless; a COSE_Key thumbprint for 

12COSE_Sign1) without changing the port surface. 

13 

14`SigningTrustContext` carries the policy under which signature 

15verification runs (trusted keys, algorithm allowlist, validity 

16window, expected payload type). It is sibling to Memo 1's 

17`FederationTrustContext` which carries federation-tier policy 

18(`allowed_credentials`, `abi_tier_floor`, `required_receipt_kinds`); 

19the two are at different tiers and have distinct shapes. 

20 

21Level note: `ByteSigner` is the low-level primitive that signs raw 

22canonical bytes under a key handle. The sibling `Signer` port at 

23`cora.infrastructure.ports.signer` is the higher-level 

24event-provenance signer (it canonicalizes an event payload via the 

25content-hash pipeline and returns the signature triple the event row 

26persists). They are distinct levels, not duplicates. 

27""" 

28

36KeyHandle = Any 

37"""Opaque adapter-specific key reference. 

38 

39Typed as `Any` at the port boundary; each adapter narrows internally. 

40v1 uses `JwksKid(kid: str)`; future Sigstore arm uses an X.509 chain 

41reference; future COSE arm uses a COSE_Key thumbprint or CBOR bstr kid. 

42""" 

43

45@dataclass(frozen=True, slots=True) 

46class Signature: 

47 """Raw signature bytes plus the adapter version that produced them. 

48 

49 Parallel to `CanonicalizedBytes` so `adapter_version` is the single 

50 dispatch key on either side. `bytes_` is the raw signature output 

51 (64 bytes for v1 Ed25519; variable for future arms). `signed_at` 

52 captures the wall-clock moment the signature was produced and is 

53 bound into the signature payload metadata, not the signed bytes. 

54 """ 

55 

56 bytes_: bytes 

57 adapter_version: str 

58 key_handle: KeyHandle 

59 signed_at: datetime

62@dataclass(frozen=True, slots=True) 

63class SigningTrustContext: 

64 """Policy under which a signature verification runs. 

65 

66 `trusted_signing_keys` is the frozenset of key handles the 

67 verifier will accept (membership-tested against the 

68 `Signature.key_handle` field). `algorithm_allowlist` rejects 

69 legacy or downgrade algorithms before any crypto runs. 

70 `expected_payload_type` is the URI the canonicalized bytes MUST 

71 have been bound to. `validity_window` is an optional pair 

72 `(not_before, not_after)`; `None` skips the temporal check. 

73 """ 

74 

75 trusted_signing_keys: frozenset[KeyHandle] 

76 algorithm_allowlist: frozenset[str] 

77 expected_payload_type: str 

78 validity_window: tuple[datetime, datetime] | None

81@dataclass(frozen=True, slots=True) 

82class SignatureVerification: 

83 """Closed-enum verdict plus opaque detail string. 

84 

85 Mirrors ControlPort's `Quality` + `quality_detail` pattern: the 

86 verdict drives downstream policy, the detail string is for 

87 forensics and never parsed by callers. `Unverifiable` is distinct 

88 from `Invalid`: `Invalid` means "the math rejected the signature"; 

89 `Unverifiable` means "the verifier could not run the math at all" 

90 (key not resolvable, algorithm not implemented, transient backend 

91 outage). 

92 """ 

93 

94 verdict: Literal["Valid", "Invalid", "Unverifiable"] 

95 detail: str = ""

98@runtime_checkable 

99class ByteSigner(Protocol): 

100 """Sign canonicalized bytes under a key handle; verify signatures. 

101 

102 Both methods are async because production adapters are network- 

103 bound (Fulcio short-lived cert issuance, KMS sign, transparency- 

104 log inclusion). In-process adapters wrap their sync work as async. 

105 """ 

106 

107 @property 

108 def adapter_version(self) -> str: ... 

109 

110 async def sign( 

111 self, 

112 canonicalized: CanonicalizedBytes, 

113 key_handle: KeyHandle, 

114 ) -> Signature: ... 

115 

116 async def verify( 

117 self, 

118 canonicalized: CanonicalizedBytes, 

119 signature: Signature, 

120 signing_trust_context: SigningTrustContext, 

121 ) -> SignatureVerification: ...

124class SigningKeyNotFoundError(Exception): 

125 """Adapter cannot resolve the supplied `key_handle`. 

126 

127 Distinct from `SignatureInvalidError` so the operator can tell 

128 "key gone" from "key present but signature does not verify." 

129 Surfaces when JWKS rotation is mid-flight, the kid is not in 

130 the current JWKS, or an X.509 chain does not anchor against 

131 pinned roots. 

132 """ 

133 

134 def __init__(self, key_handle: KeyHandle, adapter_version: str) -> None: 

135 super().__init__( 

136 f"Signing key not found for key_handle={key_handle!r} " 

137 f"under adapter_version={adapter_version!r}" 

138 ) 

139 self.key_handle = key_handle 

140 self.adapter_version = adapter_version

143class SignatureInvalidError(Exception): 

144 """Signature bytes verified against canonicalized bytes and rejected. 

145 

146 Reason carries the adapter-specific failure mode: Ed25519 verify 

147 returned False; X.509 leaf expired at sign time; SAN extension 

148 mismatch on a Sigstore Fulcio cert. 

149 """ 

150 

151 def __init__(self, adapter_version: str, reason: str) -> None: 

152 super().__init__(f"Signature invalid under adapter_version={adapter_version!r}: {reason}") 

153 self.adapter_version = adapter_version 

154 self.reason = reason

157class UnsupportedSigningAlgorithmError(Exception): 

158 """The `KeyHandle` references an algorithm the adapter does not implement. 

159 

160 Raised before any crypto runs. Used by the algorithm-allowlist 

161 gate inside `ByteSigner.verify` so legacy algorithms cannot 

162 downgrade a verification. 

163 """ 

164 

165 def __init__(self, requested_algorithm: str, adapter_version: str) -> None: 

166 super().__init__( 

167 f"Signing algorithm {requested_algorithm!r} not supported " 

168 f"by adapter_version={adapter_version!r}" 

169 ) 

170 self.requested_algorithm = requested_algorithm 

171 self.adapter_version = adapter_version

174class CanonicalizationVersionMismatchError(Exception): 

175 """Cross-version signing rejected at the port boundary. 

176 

177 A v1 ByteSigner refuses to sign over v2 CanonicalizedBytes and 

178 vice versa: pairing is mandatory per the 

179 `signing_version == canonicalization_version` invariant. The 

180 architecture-fitness suite asserts this row-by-row on every 

181 signed event. 

182 """ 

183 

184 def __init__(self, canonicalized_version: str, signing_version: str) -> None: 

185 super().__init__( 

186 f"Canonicalization/signing version mismatch: " 

187 f"canonicalized={canonicalized_version!r} signing={signing_version!r}" 

188 ) 

189 self.canonicalized_version = canonicalized_version 

190 self.signing_version = signing_version

193def algorithms_intersection(requested: Iterable[str], allowlist: frozenset[str]) -> frozenset[str]: 

194 """Return the intersection of requested algorithms with an allowlist. 

195 

196 Helper that adapters reuse when narrowing a candidate algorithm 

197 set against `SigningTrustContext.algorithm_allowlist`. Returns 

198 the allowed subset; an empty result means the caller must raise 

199 `UnsupportedSigningAlgorithmError` against the first requested 

200 entry. 

201 """ 

202 return frozenset(requested) & allowlist
