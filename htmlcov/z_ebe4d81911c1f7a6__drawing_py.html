1"""Drawing value object: a typed reference to an engineering drawing. 

2 

3A `Drawing` is a structured pointer to a document in some external 

4document-management system, not the document itself. Engineering 

5drawings travel as a triple of `(system, number, revision)` per ISO 

67200 title-block convention, CFIHOS handover specs, and every PLM / 

7EDMS in the corpus (Teamcenter Dataset, Documentum, ICMS, ...). 

8 

9Storing the system separately from the number means CORA can hold 

10references across facilities without assuming a single document 

11registry: APS uses ICMS; MAX IV will use something else; PIDINST 

12mints DOIs. 

13 

14Revision is optional. `revision = None` means "resolves to latest" 

15(ISO 7200 / DOI / EDMS default-resolution semantics). When present, 

16the Drawing references a specific snapshot. Whether a facility's 

17adapter actually exposes latest-resolution is a per-adapter contract 

18(Watch item: ICMS latest-resolution semantics need adapter 

19confirmation before the first revision-less Drawing is registered). 

20 

21`Drawing` is a closed-shape frozen dataclass. NOT 

22`json_schema_validation` (that pattern is for declarer-owns-schema / 

23carrier-owns-dict); validation lives in field shape + enum 

24membership + `validate_bounded_text` from 

25`cora.shared.bounded_text` (the 22nd hoisting site for the 

26trim + length-check + raise pattern; AssetPort precedent uses 

27per-field error classes). 

28 

29`DrawingSystem` is an extensible StrEnum (registry of accepted 

30systems enforced per-facility at the adapter boundary, NOT in the 

31VO). 

32 

33Used as an optional field on both `Mount` (where the slot is in the 

34beamline) and `Asset` (what the specimen was built to). The two are 

35different documents; do NOT collapse them. 

36 

37Promote to a full `Document` BC only when drawings need their own 

38lifecycle (review status, supersedes-chain, ECO propagation per 

39MIL-HDBK-61A 5.4 / CMII, ACL). 

40"""

51class InvalidDrawingError(ValueError): 

52 """Base class for Drawing VO validation failures. 

53 

54 Tests catch the specific subclasses (`InvalidDrawingNumberError`, 

55 `InvalidDrawingRevisionError`) when they need to distinguish the 

56 failing field. The route layer's `_handle_validation_error` 

57 catches the base class and surfaces `str(exc)` into the 400 

58 body, so the formatted message is the user-facing surface; 

59 subclass-specific routing is not required today. 

60 

61 Lives in this module (an aggregate-scoped private VO module) so 

62 aggregate state.py / events.py can import without violating the 

63 aggregates-don't-depend-on-BC-root tach layering rule. 

64 """ 

65

67class InvalidDrawingNumberError(InvalidDrawingError): 

68 """`Drawing.number` failed bounded-text validation. 

69 

70 Raised by `validate_bounded_text` when the trimmed number is 

71 empty / whitespace-only or exceeds `DRAWING_NUMBER_MAX_LENGTH`. 

72 `value` carries the original untrimmed input for diagnostics 

73 (mirrors the `InvalidAssetPortNameError` shape). 

74 """ 

75 

76 def __init__(self, value: str) -> None: 

77 super().__init__( 

78 f"Invalid Drawing number (must be 1-{DRAWING_NUMBER_MAX_LENGTH} " 

79 f"chars after trimming, got: {value!r})" 

80 ) 

81 self.value = value 

82 

83 

84class InvalidDrawingRevisionError(InvalidDrawingError): 

85 """`Drawing.revision` failed bounded-text validation. 

86 

87 Raised by `validate_bounded_text` when the revision is present 

88 but trimmed empty or exceeds `DRAWING_REVISION_MAX_LENGTH`. To 

89 indicate 'latest' (resolves at the adapter), pass `revision=None` 

90 instead of an empty string. `value` carries the original 

91 untrimmed input. 

92 """ 

93 

94 def __init__(self, value: str) -> None: 

95 super().__init__( 

96 f"Invalid Drawing revision (must be 1-{DRAWING_REVISION_MAX_LENGTH} " 

97 f"chars after trimming, or None for 'latest'; got: {value!r})" 

98 ) 

99 self.value = value

102class DrawingSystem(StrEnum): 

103 """The external document-management system a Drawing lives in. 

104 

105 Closed StrEnum: v1 ships with three systems. Adding a new value 

106 is a paired change: extend this enum AND ship a migration that 

107 drops + re-adds the CHECK constraint on `drawing_system` in any 

108 projection that materializes drawings (today: `proj_equipment_ 

109 asset_summary`). Whether a given system value is actually wired 

110 to an adapter is enforced per-facility at the adapter boundary, 

111 NOT here. 

112 

113 - `ICMS`: APS Information Content Management System (Documentum- 

114 backed). 

115 - `EDMS`: generic Engineering Document Management System 

116 (CERN EDMS and similar). 

117 - `DOI`: DataCite Digital Object Identifier (used by PIDINST 

118 for instrument-level identifiers). 

119 """ 

120 

121 ICMS = "ICMS" 

122 EDMS = "EDMS" 

123 DOI = "DOI"

126@dataclass(frozen=True) 

127class Drawing: 

128 """A typed reference to an engineering drawing in an external system. 

129 

130 Deviation from Identifier VO: 3-tuple (system, number, revision?) 

131 with optional revision qualifier. 

132 

133 `system` says which document registry; `number` is the document 

134 identifier within that registry; `revision` is the optional 

135 snapshot pin (None means "latest"). 

136 

137 Equality and hash are structural across the three fields, so two 

138 Drawing instances referring to the same `(system, number, revision)` 

139 triple collapse in a set. 

140 """ 

141 

142 system: DrawingSystem 

143 number: str 

144 revision: str | None = None 

145 

146 def __post_init__(self) -> None: 

147 trimmed_number = validate_bounded_text( 

148 self.number, 

149 max_length=DRAWING_NUMBER_MAX_LENGTH, 

150 error_class=InvalidDrawingNumberError, 

151 ) 

152 object.__setattr__(self, "number", trimmed_number) 

153 

154 if self.revision is not None: 

155 trimmed_revision = validate_bounded_text( 

156 self.revision, 

157 max_length=DRAWING_REVISION_MAX_LENGTH, 

158 error_class=InvalidDrawingRevisionError, 

159 ) 

160 object.__setattr__(self, "revision", trimmed_revision) 

161

163def drawing_to_payload(drawing: Drawing) -> dict[str, object]: 

164 """Serialize a Drawing VO to a JSON-friendly dict for jsonb storage. 

165 

166 Canonical codec shared by every aggregate event that carries a 

167 Drawing (Mount, Asset, Assembly). Sibling rebuilder is 

168 `drawing_from_payload`. Hoisted to the VO module to close the 

169 duplication anti-hook flagged in `project_mount_frame_design` 

170 Watch items. 

171 """ 

172 return { 

173 "system": drawing.system.value, 

174 "number": drawing.number, 

175 "revision": drawing.revision, 

176 } 

177 

178 

179def drawing_from_payload(payload: dict[str, object]) -> Drawing: 

180 """Reconstruct a Drawing VO from its JSON payload. 

181 

182 Sibling of `drawing_to_payload`. Round-trips losslessly with it. 

183 Re-runs Drawing.__post_init__ validation. 

184 """ 

185 return Drawing( 

186 system=DrawingSystem(str(payload["system"])), 

187 number=str(payload["number"]), 

188 revision=(str(payload["revision"]) if payload.get("revision") is not None else None), 

189 )
