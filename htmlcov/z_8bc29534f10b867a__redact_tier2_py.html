382def redact_tier2_row( 

383 kind: str, 

384 row: dict[str, Any], 

385 *, 

386 token_map: TokenMap, 

387 fired_pointers: dict[tuple[str, str], set[str]], 

388) -> dict[str, Any]: 

389 """Redact one entries row of the given registry `kind`.""" 

390 column_dispositions = TIER2_DISPOSITIONS[kind] 

391 result: dict[str, Any] = {} 

392 for column, value in row.items(): 

393 disposition = column_dispositions.get(column) 

394 if disposition is None: 

395 # An entries column not enumerated above. Fail closed rather 

396 # than silently pass it through: this is this file's own 

397 # drift-detection surface (see test_redact_tier2.py). 

398 continue 

399 if disposition == KEEP: 

400 result[column] = value 

401 elif disposition == TOKEN: 

402 result[column] = token_map.token_uuid(value) 

403 elif disposition == DROP: 

404 continue 

405 elif disposition == BY_VALUE: 405 ↛ 392line 405 didn't jump to line 392 because the condition on line 405 was always true

406 if (kind, column) in TIER2_JSONB_DROPPED_COLUMNS: 

407 continue 

408 cleared = TIER2_JSONB_CLEARED_POINTERS.get((kind, column), frozenset()) 

409 fired = fired_pointers.setdefault((kind, column), set()) 

410 redacted = apply_leaf_rule( 

411 value, token_map=token_map, cleared_pointers=cleared, fired_pointers=fired 

412 ) 

413 if redacted is not OMITTED: 413 ↛ 392line 413 didn't jump to line 392 because the condition on line 413 was always true

414 result[column] = redacted 

415 return result

418def unfired_clearances( 

419 fired_pointers: dict[tuple[str, str], set[str]], *, kinds_present: frozenset[str] 

420) -> frozenset[tuple[str, str, str]]: 

421 """Every declared tier-2 jsonb clearance that never matched a row, 

422 scoped to kinds actually present in this export. 

423 

424 CORRECTED 2026-08-12. An earlier version of this function 

425 (`ensure_all_clearances_fired`) raised here, reasoning from F5's 

426 Rejections list: "an unfired rule inside a type that was exported is 

427 a leak-shaped gap." That reasoning is right for a DENYLIST, where a 

428 rule that fails to fire means the thing it should have hidden got 

429 published. It is backwards for tier 2's ALLOWLIST: a clearance that 

430 never fires means a field was published LESS often than the profile 

431 permits, never more. There is no mechanism by which that leaks 

432 anything, so treating it as fatal was importing a denylist-shaped 

433 fear into an allowlist-shaped mechanism. 

434 

435 The practical failure this produced: `activity/payload`'s cleared 

436 pointers live on different step kinds (setpoint, action, check, 

437 compute), several of them optional or failure-arm-only, so no small 

438 export -- including a first rehearsal bundle -- reliably fires every 

439 one. The export would abort with an error reading like a broken 

440 disposition table rather than "this export was too narrow to 

441 exercise every clearance." 

442 

443 Callers now record the result on the manifest 

444 (`Manifest.unfired_tier2_clearances`) instead of treating it as a 

445 reason to refuse. The genuine worry this WAS reaching for --a 

446 misspelled pointer in `TIER2_JSONB_CLEARED_POINTERS` that can never 

447 fire against any real payload-- is a fact about this file's code, 

448 not about any one export, and belongs in a build-time check against 

449 the real key space (the same shape as step 0's generated disposition 

450 table), not in a per-export runtime gate. That check now exists, 

451 shipped the same day as this correction: 

452 `tests/architecture/test_tier2_jsonb_clearances_are_real_keys.py`. 

453 This function no longer stands in for it. 

454 """
