1"""Whether the shipped fleet can actually act, as a value and as a verdict. 

2 

3`_seeded_fleet` made the fleet a value so a gesture could be made over 

4it. This asks the question that value exists for, and says the answer out 

5loud at boot. 

6 

7## Why this is a warning and not a metric 

8 

9A `Defined` Agent is registered and inert: the subscribers' lifecycle 

10gate fires on `Versioned` only, and refuses anything less WITHOUT 

11SAYING SO. On the 2-BM pilot that stranded seventeen agents for three 

12months behind a log that looked clean the entire time. The remedy 

13(`promote_seeded_fleet`) has existed since; what did not exist was 

14anything that told an operator the remedy was needed. 

15 

16So the point here is not the number. It is that a deployment whose fleet 

17cannot act now says so on every boot, at `warning`, naming the members. 

18An absence has to be its own loud verdict, because the alternative is 

19what already happened: a correct system, a clean log, and nothing 

20running. 

21 

22## Four not-ready reasons, kept apart 

23 

24They are not interchangeable and collapsing them would restore the 

25silence at one remove: 

26 

27 `not_ready` `Defined`. Never promoted. THE silent case, and the only 

28 one this treats as a fault, because it is the only one 

29 nobody chose. 

30 `held` `Suspended`. A live operator decision. Reported, never 

31 warned about; a deployment pausing an agent on purpose 

32 should not be nagged for it. 

33 `retired` `Deprecated`. Terminal and deliberate. 

34 `absent` Not in the record at all. Worth seeing on a partly-seeded 

35 deployment rather than counting as a silent zero, the same 

36 reasoning `promote_seeded_fleet` applies to it. 

37"""

58@dataclass(frozen=True) 

59class FleetReadiness: 

60 """Fleet members by whether they can act, named rather than counted. 

61 

62 Names and not ids: this is read by an operator deciding whether to 

63 run a promotion, and `RunTranslator` answers that question where 

64 `01900000-0000-7000-8000-0000aaaa0010` does not. The ids stay 

65 reachable through `SEEDED_FLEET` for anything that needs to act on a 

66 member. 

67 """ 

68 

69 ready: tuple[str, ...] 

70 not_ready: tuple[str, ...] 

71 held: tuple[str, ...] 

72 retired: tuple[str, ...] 

73 absent: tuple[str, ...] 

74 

75 @property 

76 def total(self) -> int: 

77 return ( 

78 len(self.ready) 

79 + len(self.not_ready) 

80 + len(self.held) 

81 + len(self.retired) 

82 + len(self.absent) 

83 ) 

84 

85 @property 

86 def stranded(self) -> bool: 

87 """Whether any member is inert for a reason nobody chose.""" 

88 return bool(self.not_ready)

91def fleet_readiness( 

92 statuses: Mapping[UUID, AgentStatus | None], 

93 fleet: Sequence[SeededAgent] = SEEDED_FLEET, 

94) -> FleetReadiness: 

95 """Sort the fleet by readiness. Pure, so the unit tier drives every 

96 branch without a record. 

97 

98 Ranges over `fleet`, never over `statuses`: a member the caller 

99 failed to look up has to land in `absent`, and a map missing that key 

100 would otherwise drop it from the count entirely. The total is then a 

101 property of the shipped fleet, which is the only thing that makes 

102 "4 of 20" mean anything. 

103 """ 

104 buckets: dict[str, list[str]] = { 

105 "ready": [], 

106 "not_ready": [], 

107 "held": [], 

108 "retired": [], 

109 "absent": [], 

110 } 

111 for member in fleet: 

112 match statuses.get(member.agent_id): 

113 case AgentStatus.VERSIONED: 

114 buckets["ready"].append(member.name) 

115 case AgentStatus.DEFINED: 

116 buckets["not_ready"].append(member.name) 

117 case AgentStatus.SUSPENDED: 

118 buckets["held"].append(member.name) 

119 case AgentStatus.DEPRECATED: 

120 buckets["retired"].append(member.name) 

121 case None: 121 ↛ 111line 121 didn't jump to line 111 because the pattern on line 121 always matched

122 buckets["absent"].append(member.name) 

123 return FleetReadiness( 

124 ready=tuple(buckets["ready"]), 

125 not_ready=tuple(buckets["not_ready"]), 

126 held=tuple(buckets["held"]), 

127 retired=tuple(buckets["retired"]), 

128 absent=tuple(buckets["absent"]), 

129 )

132async def read_fleet_readiness(event_store: EventStore) -> FleetReadiness: 

133 """Load every shipped member and sort it. Read-only.""" 

134 statuses: dict[UUID, AgentStatus | None] = {} 

135 for member in SEEDED_FLEET: 

136 agent = await load_agent(event_store, member.agent_id) 

137 statuses[member.agent_id] = None if agent is None else agent.status 

138 return fleet_readiness(statuses)

141def log_fleet_readiness(readiness: FleetReadiness) -> None: 

142 """Say it out loud, at a level that matches whether anything is wrong. 

143 

144 `warning` when a member is stranded, because that is a deployment 

145 that will quietly do nothing, and the whole reason this function 

146 exists is that the previous behaviour was indistinguishable from a 

147 healthy boot. `info` otherwise: a fleet nobody needs to act on should 

148 not train an operator to scroll past this line. 

149 """ 

150 if readiness.stranded: 

151 _log.warning( 

152 "agent_fleet.stranded", 

153 ready=len(readiness.ready), 

154 total=readiness.total, 

155 not_ready=list(readiness.not_ready), 

156 remedy="promote_seeded_fleet", 

157 ) 

158 else: 

159 _log.info("agent_fleet.ready", ready=len(readiness.ready), total=readiness.total) 

160 if readiness.absent: 

161 _log.warning("agent_fleet.absent", absent=list(readiness.absent))
