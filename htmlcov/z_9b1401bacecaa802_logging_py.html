1"""Structured logging configuration. 

2 

3Uses `structlog` with `contextvars` so context bound at the ASGI boundary 

4and at command-handler entry propagates to every log line emitted inside 

5the decider, repository, and projection. 

6 

7OpenTelemetry trace context (`trace_id`, `span_id`, `trace_flags`) is 

8injected by the `add_trace_context` processor when an active span is 

9present, letting log-aggregator queries pivot from a log line to the 

10corresponding distributed trace. The processor is a no-op when no 

11span is active (default test environment uses the no-op tracer), so 

12unit tests don't need to set up a TracerProvider. 

13 

14The structlog wrapper chain ends with `ProcessorFormatter.wrap_for_formatter` 

15(not a renderer) so the wrapped event_dict reaches the stdlib 

16`ProcessorFormatter`, which runs `JSONRenderer()` exactly once. Terminating 

17the wrapper chain with `JSONRenderer()` would render twice and produce 

18JSON-in-JSON output that log aggregators can't index. 

19 

20Caching nuance — `cache_logger_on_first_use=True` means structlog binds each 

21named logger to the live configuration on first use and keeps that binding. 

22`build_kernel()` calls `configure_logging()` once per `create_app()`, so the 

23function must be safe to call repeatedly, and it splits along that seam: the 

24structlog side configures once per process, the stdlib handler is rebuilt on 

25every call, so log level follows the most recent caller. 

26 

27Configuring structlog more than once is what the guard exists to prevent, and 

28the reason is not cosmetic. A second `structlog.configure()` installs a NEW 

29processor list; loggers already bound and cached against the first list keep 

30using it. In production the two lists are identical so nothing is visibly 

31wrong, but `structlog.testing.capture_logs()` works by swapping the live 

32config's processors, and a cached logger reading the old list never sees the 

33swap. Any test asserting on captured events then collects nothing, while the 

34log line still appears on stdout — a failure that looks like a missing log and 

35is really a stale binding. It surfaces only when a test runs after a second 

36app is built, so it presents as order-dependent flakiness. 

37 

38If per-call reconfiguration is ever genuinely needed, set 

39`cache_logger_on_first_use=False` and accept the per-call binding cost; do not 

40simply drop the guard. 

41 

42## The stdlib handler targets LIVE `sys.stdout`, not a snapshot of it 

43 

44`logging.StreamHandler(sys.stdout)` would capture whatever object `sys.stdout` 

45IS at the moment `configure_logging()` runs, not a live reference to the name 

46`sys.stdout`. That distinction is invisible in production, where `sys.stdout` 

47never changes identity after process start, but it is a real bug under 

48pytest's `capsys` fixture, which monkeypatches `sys.stdout` to a fresh object 

49for every test. A handler built while an EARLIER test's `capsys` patch was 

50active keeps writing to that test's now-torn-down capture buffer; the 

51CURRENT test's own `capsys.readouterr()` then sees nothing, while the JSON 

52line still lands wherever the earlier buffer's underlying stream actually 

53is (typically the real terminal), which is indistinguishable from the log 

54line never having been emitted at all unless someone goes looking for it 

55elsewhere. 

56 

57This bit `test_run_debriefer_seed.py`'s `capsys`-based assertion: it passed 

58in isolation and failed only in a full-suite run, because `configure_logging` 

59is called from `build_kernel` (the production kernel factory) but NOT from 

60`make_inmemory_kernel` (the lighter one most unit tests use), so a test built 

61on `make_inmemory_kernel` inherits whichever handler an EARLIER, unrelated 

62test's `build_kernel()` call last installed in the same worker process -- 

63built under THAT test's `capsys`, not this one's. 

64 

65`_LiveStdout` below is the standard fix for this class of bug: instead of 

66handing the handler an object, hand it a proxy whose `write` / `flush` look 

67up `sys.stdout` FRESH on every call, so the handler is correct regardless of 

68when it was constructed relative to whichever test's `capsys` is currently 

69active. This makes calling `configure_logging()` from every kernel 

70constructor unnecessary, not just unlikely: the handler self-corrects 

71either way. 

72"""

92class _LiveStdout: 

93 """A write target that always resolves to the CURRENT `sys.stdout`. 

94 

95 `logging.StreamHandler` stores whatever object it is given at 

96 construction time and never looks at `sys.stdout` again. Handing it 

97 this proxy instead of `sys.stdout` directly defers that lookup to 

98 every individual write, so the handler stays correct even when 

99 `sys.stdout` is later swapped out from under it (pytest's `capsys` 

100 fixture does exactly that, once per test). See the module 

101 docstring's "live `sys.stdout`" section for the failure this fixes. 

102 """ 

103 

104 def write(self, message: str) -> int: 

105 return sys.stdout.write(message) 

106 

107 def flush(self) -> None: 

108 sys.stdout.flush()

111def configure_logging(level: str = "INFO") -> None: 

112 """Configure structlog and bridge stdlib logging to it. Call once at startup.""" 

113 global _structlog_configured 

114 timestamper = structlog.processors.TimeStamper(fmt="iso", utc=True) 

115 shared_processors: list[Processor] = [ 

116 merge_contextvars, 

117 add_trace_context, 

118 structlog.stdlib.add_log_level, 

119 structlog.stdlib.add_logger_name, 

120 timestamper, 

121 ] 

122 

123 # Configure structlog once per process. `build_kernel()` calls this on every 

124 # `create_app()`, and a second call installs a NEW processor list: loggers 

125 # already bound and cached against the first one keep using it and never see 

126 # the replacement. That is invisible in production, where the two lists are 

127 # identical, but it silently defeats `structlog.testing.capture_logs()` in 

128 # any test that runs after a second app is built, because capture swaps the 

129 # live config's processors and the cached loggers are not reading from it. 

130 # The stdlib handler below is rebuilt every call, so log level still tracks 

131 # the most recent caller. 

132 if not _structlog_configured: 

133 structlog.configure( 

134 processors=[ 

135 *shared_processors, 

136 structlog.processors.StackInfoRenderer(), 

137 structlog.processors.format_exc_info, 

138 # Hand off to the stdlib ProcessorFormatter; do not render here. 

139 structlog.stdlib.ProcessorFormatter.wrap_for_formatter, 

140 ], 

141 wrapper_class=structlog.stdlib.BoundLogger, 

142 context_class=dict, 

143 logger_factory=structlog.stdlib.LoggerFactory(), 

144 cache_logger_on_first_use=True, 

145 ) 

146 _structlog_configured = True 

147 

148 handler = logging.StreamHandler(_LiveStdout()) 

149 handler.setFormatter( 

150 structlog.stdlib.ProcessorFormatter( 

151 processor=structlog.processors.JSONRenderer(), 

152 foreign_pre_chain=shared_processors, 

153 ) 

154 ) 

155 root = logging.getLogger() 

156 root.handlers.clear() 

157 root.addHandler(handler) 

158 root.setLevel(level.upper())
