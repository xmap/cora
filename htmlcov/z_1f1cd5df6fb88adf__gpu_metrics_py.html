1"""GPU-serving observability: the visibility half of the build path. 

2 

3In the locked design in-house serving is metered-free by default, so the 

4GPU time a served call consumes never debits the budget. This module is 

5where a facility nonetheless gets to SEE that cost: it turns each call's 

6occupancy-share GPU-seconds (from the serving meter, 

7`cora.infrastructure.observability.gpu_accounting`) into two OpenTelemetry 

8metrics, tagged by model and device: 

9 

10 - `cora.agent.llm.gpu.seconds` -- per-call GPU-seconds. Summed per 

11 device over wall-time, this is the 

12 capacity signal (how busy the pool 

13 is); it is also the raw input to 

14 the shadow cost. 

15 - `cora.agent.llm.gpu.shadow_cost.usd` -- the same seconds priced at the 

16 facility's amortized GPU-hour rate. 

17 An INDICATIVE figure so a facility 

18 can watch what its metered-free 

19 serving really costs and re-base 

20 the token rate when volume grows. 

21 It is never debited. 

22 

23The `usd_per_gpu_hour` rate is a facility figure supplied at the 

24composition root (it is deliberately NOT on the catalog entry: a 

25single-valued `cost_basis` cannot hold both a debiting token price and the 

26GPU-hour primitive, so the primitive lives here as observability input). 

27`make_gpu_usage_sink` binds it into the `on_measure` callable the `LocalLLM` 

28adapter drives once per call. 

29""" 

30 

31from __future__ import annotations

33import math 

34from typing import TYPE_CHECKING

36from opentelemetry import metrics

38if TYPE_CHECKING:

43_meter = metrics.get_meter("cora.gpu_serving") 

44_gpu_seconds_histogram = _meter.create_histogram( 

45 name="cora.agent.llm.gpu.seconds", 

46 unit="s", 

47 description="Per-call GPU-seconds attributed to a served call by occupancy share",

51 unit="USD", 

52 description="Indicative GPU-hour shadow cost of a served call (visibility, never debited)",

56def gpu_shadow_cost_usd(gpu_seconds: float, usd_per_gpu_hour: float) -> float: 

57 """Price attributed GPU-seconds at a facility GPU-hour rate (visibility only). 

58 

59 The facility's amortized `usd_per_gpu_hour` times the call's 

60 occupancy-share GPU-seconds, in USD. This is what the meter reports so a 

61 facility can see the real cost of its metered-free in-house serving and 

62 decide when to re-base the token rate; it never debits the envelope. 

63 """ 

64 if not math.isfinite(usd_per_gpu_hour) or usd_per_gpu_hour < 0.0: 

65 msg = f"usd_per_gpu_hour must be finite and non-negative, got {usd_per_gpu_hour}" 

66 raise ValueError(msg) 

67 return gpu_seconds / 3600.0 * usd_per_gpu_hour

70def record_gpu_usage(record: GpuUsageRecord, *, usd_per_gpu_hour: float) -> float: 

71 """Emit the GPU-seconds and shadow-cost metrics for one served call. 

72 

73 Returns the computed shadow cost (the histograms are the durable 

74 surface; the return is a convenience for callers that want to log it). 

75 Safe to call when no MeterProvider is installed: the histograms are 

76 no-ops then, exactly like the token-cost histogram in `gen_ai`. 

77 """ 

78 shadow_cost = gpu_shadow_cost_usd(record.gpu_seconds, usd_per_gpu_hour) 

79 attributes = { 

80 "gen_ai.request.model": record.model, 

81 "gpu.device_id": record.device_id, 

82 } 

83 _gpu_seconds_histogram.record(record.gpu_seconds, attributes=attributes) 

84 _gpu_shadow_cost_histogram.record(shadow_cost, attributes=attributes) 

85 return shadow_cost

88def make_gpu_usage_sink(usd_per_gpu_hour: float) -> Callable[[GpuUsageRecord], None]: 

89 """Bind a facility GPU-hour rate into an `on_measure` sink for `LocalLLM`. 

90 

91 The composition root reads the facility's `usd_per_gpu_hour` and hands the 

92 resulting sink to the `LocalLLM` adapter, which calls it once per served 

93 call. The sink emits the GPU-seconds and shadow-cost metrics. 

94 """ 

95 

96 def sink(record: GpuUsageRecord) -> None: 

97 record_gpu_usage(record, usd_per_gpu_hour=usd_per_gpu_hour) 

98 

99 return sink
