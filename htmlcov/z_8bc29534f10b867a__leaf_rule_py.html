1"""The generic recursive leaf rule. 

2 

3Shared by tier-1's `by-value` polymorphic fields (a slot that can hold a 

4scalar or a value object -- `_dispositions.py`'s `by-value` disposition) 

5and tier-2's four recursing jsonb columns (`activities.payload`, 

6`diagnostics.payload`, `outcomes.point`, `outcomes.measurements`). Per 

7`project_record_export_v3.md` F5: numeric and boolean leaves KEEP, 

8UUID-shaped string leaves TOKEN, every other string leaf DROPS unless 

9its `(column, json-pointer)` is on a caller-supplied cleared list. 

10Object KEYS are always published -- in practice they are field names 

11and they carry the structure -- only VALUES are ever redacted. 

12 

13This walker does not know, or need to know, what produced the value it 

14is walking (which `step_kind`, which Recipe step type, ...). That is 

15deliberate: a nested shape nobody enumerated a clearance for simply 

16drops by the same default every other unlisted field drops by, rather 

17than needing a special case per producer. 

18""" 

19

29OMITTED = object() 

30"""Sentinel: the caller must omit this key (dict) or this element (list) 

31entirely, not store `None` in its place."""

33_UUID_RE = re.compile( 

34 r"^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$" 

35)

38def is_uuid_shaped(value: str) -> bool: 

39 return bool(_UUID_RE.match(value))

42def _join_pointer(pointer: str, segment: str) -> str: 

43 return segment if not pointer else f"{pointer}/{segment}"

46def apply_leaf_rule( 

47 value: Any, 

48 *, 

49 token_map: TokenMap, 

50 cleared_pointers: frozenset[str] = frozenset(), 

51 pointer: str = "", 

52 fired_pointers: set[str] | None = None, 

53) -> Any: 

54 """Recursively redact `value`. 

55 

56 Returns a redacted `dict`/`list` (same shape, dropped keys/elements 

57 omitted) for a container, a redacted scalar for a leaf, or `OMITTED` 

58 if the top-level value itself is a leaf that must be omitted by the 

59 caller (dict/list values are never themselves `OMITTED` -- an empty 

60 dict/list is a valid redacted result and is returned as such). 

61 

62 `fired_pointers`, when supplied, collects every `pointer` whose 

63 value was actually kept via `cleared_pointers` (not via the 

64 unconditional number/bool/None/UUID branches) -- the record needed 

65 to check a cleared list ever matched anything, per F5's 

66 unfired-clearance rejection. 

67 """ 

68 if isinstance(value, dict): 

69 result: dict[str, Any] = {} 

70 for key, sub_value in value.items(): 

71 child_pointer = _join_pointer(pointer, str(key)) 

72 redacted = apply_leaf_rule( 

73 sub_value, 

74 token_map=token_map, 

75 cleared_pointers=cleared_pointers, 

76 pointer=child_pointer, 

77 fired_pointers=fired_pointers, 

78 ) 

79 if redacted is not OMITTED: 

80 result[key] = redacted 

81 return result 

82 

83 if isinstance(value, (list, tuple)): 

84 child_pointer = _join_pointer(pointer, "*") 

85 out: list[Any] = [] 

86 for item in value: 

87 redacted = apply_leaf_rule( 

88 item, 

89 token_map=token_map, 

90 cleared_pointers=cleared_pointers, 

91 pointer=child_pointer, 

92 fired_pointers=fired_pointers, 

93 ) 

94 if redacted is not OMITTED: 94 ↛ 86line 94 didn't jump to line 86 because the condition on line 94 was always true

95 out.append(redacted) 

96 return out

98 # Leaf values. 

99 if value is None or isinstance(value, (bool, int, float)): 

100 return value 

101 if isinstance(value, str): 101 ↛ 112line 101 didn't jump to line 112 because the condition on line 101 was always true

102 if is_uuid_shaped(value): 

103 return token_map.token_uuid(value) 

104 if pointer in cleared_pointers: 

105 if fired_pointers is not None: 105 ↛ 107line 105 didn't jump to line 107 because the condition on line 105 was always true

106 fired_pointers.add(pointer) 

107 return value 

108 return OMITTED 

109 

110 # An already-rendered export body (per step 2's F6) never contains 

111 # anything else; fail closed rather than guess at a new type's intent. 

112 return OMITTED
