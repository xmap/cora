1"""CampaignSummaryProjection: folds the Campaign aggregate's events into 

2the `proj_campaign_summary` read model that backs 

3`GET /campaigns`. 

4 

5Subscribed events: 

6 - CampaignRegistered -> INSERT (status='Planned', started_at=NULL, 

7 last_status_*=NULL, run_count=0) 

8 - CampaignStarted -> UPDATE status='Active' + started_at 

9 + last_status_changed_at 

10 - CampaignHeld -> UPDATE status='Held' + last_status_reason 

11 + last_status_changed_at 

12 - CampaignResumed -> UPDATE status='Active' 

13 + last_status_changed_at 

14 (last_status_reason PRESERVED) 

15 - CampaignClosed -> UPDATE status='Closed' 

16 + last_status_changed_at 

17 - CampaignAbandoned -> UPDATE status='Abandoned' 

18 + last_status_reason 

19 + last_status_changed_at 

20 

21## Anti-hooks pinned at the projection layer 

22 

23 - Anti-hook (no auto-notification). The projection is a pull-on-read 

24 read model; nothing else is written when a new campaign lands. 

25 - Anti-hook (no cascade). Campaign state changes never touch member 

26 Run state; the projection has no Run-side coupling. 

27 - The projection does NOT include `external_refs` (lives on the 

28 aggregate stream; per design memo §12 the projection columns include 

29 `external_id` only, not `external_refs`). Reverse-query "Campaigns 

30 by external_ref scheme/value" is a Watch item. 

31 - The projection does NOT include `run_ids` (full set lives on the 

32 aggregate stream; `run_count` is the only denorm). Reverse-query 

33 "Campaigns containing run X" is Watch item #10 (needs Run.campaign_id 

34 indexed scan on proj_run_summary). 

35 

36## SAVEPOINT semantics on CampaignRegistered 

37 

38The INSERT is wrapped in `async with conn.transaction(): ...` so any 

39future cross-stream uniqueness violation (today there is no UNIQUE 

40INDEX; campaign_id PK alone is sufficient and idempotent via 

41`ON CONFLICT (campaign_id) DO NOTHING`) would roll back only the inner 

42SAVEPOINT. The supply projection's pattern is reused verbatim for 

43forward-compat even though the failure mode is narrower here. 

44 

45## Started_at semantics 

46 

47`started_at` is set on the FIRST CampaignStarted only (Planned -> 

48Active transition). Per design memo: CampaignStarted is the 

49single-source genesis transition (decider gates source to Planned). 

50CampaignResumed (Held -> Active) does NOT touch `started_at`: the 

51first-start timestamp is preserved as audit truth for "when did this 

52campaign begin work" and is not the same concept as "resumed". 

53 

54## Last_status_reason preservation on resume 

55 

56CampaignResumed does NOT clear `last_status_reason`. Per design memo: 

57"Keep it -- audit value". The breadcrumb "why was it held before the 

58resume" stays readable after the resume lands. 

59 

60## Run_count 

61 

62`run_count` is denormalized from the Campaign aggregate's 

63`run_ids: frozenset[UUID]` (full set lives on the aggregate stream; 

64`get_campaign` returns the full set from aggregate state). Maintained 

65by two projection arms: 

66 

67 - CampaignRunAdded -> run_count = run_count + 1 

68 - CampaignRunRemoved -> run_count = run_count - 1 

69 

70The events are written by the cross-aggregate `add_run_to_campaign` / 

71`remove_run_from_campaign` slices (and by `start_run` when 

72`StartRun.campaign_id` is provided -- atomic with `RunStarted` via 

73`EventStore.append_streams`). Out-of-order processing is not a concern: 

74the worker processes events in (transaction_id, position) order, so 

75`CampaignRunAdded` cannot arrive before its parent 

76`CampaignRegistered` for the same campaign_id. 

77"""

87_INSERT_CAMPAIGN_SQL = """ 

88INSERT INTO proj_campaign_summary 

89 (campaign_id, name, intent, status, lead_actor_id, subject_id, 

90 description, tags, external_id, run_count, registered_at, 

91 started_at, last_status_changed_at, last_status_reason) 

92VALUES ($1, $2, $3, 'Planned', $4, $5, 

93 $6, $7::text[], $8, 0, $9, 

94 NULL, NULL, NULL) 

95ON CONFLICT (campaign_id) DO NOTHING 

96""" 

97 

98_UPDATE_STARTED_SQL = """ 

99UPDATE proj_campaign_summary 

100SET status = 'Active', 

101 started_at = $2, 

102 last_status_changed_at = $2, 

103 updated_at = now() 

104WHERE campaign_id = $1 

105""" 

106 

107_UPDATE_HELD_SQL = """ 

108UPDATE proj_campaign_summary 

109SET status = 'Held', 

110 last_status_reason = $2, 

111 last_status_changed_at = $3, 

112 updated_at = now() 

113WHERE campaign_id = $1 

114""" 

115 

116_UPDATE_RESUMED_SQL = """ 

117UPDATE proj_campaign_summary 

118SET status = 'Active', 

119 last_status_changed_at = $2, 

120 updated_at = now() 

121WHERE campaign_id = $1 

122"""

124_UPDATE_CLOSED_SQL = """ 

125UPDATE proj_campaign_summary 

126SET status = 'Closed', 

127 last_status_changed_at = $2, 

128 updated_at = now() 

129WHERE campaign_id = $1 

130"""

133UPDATE proj_campaign_summary 

134SET status = 'Abandoned', 

135 last_status_reason = $2, 

136 last_status_changed_at = $3, 

137 updated_at = now() 

138WHERE campaign_id = $1

141_UPDATE_RUN_ADDED_SQL = """ 

142UPDATE proj_campaign_summary 

143SET run_count = run_count + 1, 

144 updated_at = now() 

145WHERE campaign_id = $1 

146""" 

147 

148# N12 gate-review nit: defensive floor guard on the decrement. The 

149# `run_count > 0` predicate in the WHERE clause prevents an underflow 

150# from ever materializing as a negative count even if a duplicate 

151# CampaignRunRemoved arrives (worker re-delivery, replay) or the 

152# event-side decider's idempotency invariant ever slips. A negative 

153# run_count would be a silent integrity bug on the read model; the 

154# guard turns it into a no-op decrement instead. Preferred over a 

155# table-level CHECK (run_count >= 0) constraint because (a) no new 

156# migration is needed and (b) a CHECK violation would block the 

157# projection worker on the bad event forever. 

158_UPDATE_RUN_REMOVED_SQL = """ 

159UPDATE proj_campaign_summary 

160SET run_count = run_count - 1, 

161 updated_at = now() 

162WHERE campaign_id = $1 

163 AND run_count > 0 

164"""

167class CampaignSummaryProjection: 

168 """Maintains the `proj_campaign_summary` read model.""" 

169 

170 name = "proj_campaign_summary" 

171 subscribed_event_types = frozenset( 

172 { 

173 "CampaignRegistered", 

174 "CampaignStarted", 

175 "CampaignHeld", 

176 "CampaignResumed", 

177 "CampaignClosed", 

178 "CampaignAbandoned", 

179 # membership arms maintain run_count denormalization. 

180 "CampaignRunAdded", 

181 "CampaignRunRemoved", 

182 } 

183 ) 

184 

185 async def apply( 

186 self, 

187 event: StoredEvent, 

188 conn: ConnectionLike, 

189 ) -> None: 

190 if event.event_type == "CampaignRegistered": 

191 payload = event.payload 

192 raw_subject = payload.get("subject_id") 

193 raw_external_id = payload.get("external_id") 

194 # Wrap in SAVEPOINT mirroring the supply projection's 

195 # forward-compat pattern; the PK alone makes the INSERT 

196 # idempotent today, but a future cross-stream uniqueness 

197 # constraint would otherwise stall the worker on duplicate. 

198 async with conn.transaction(): 

199 await conn.execute( 

200 _INSERT_CAMPAIGN_SQL, 

201 UUID(payload["campaign_id"]), 

202 payload["name"], 

203 payload["intent"], 

204 UUID(payload["lead_actor_id"]), 

205 UUID(raw_subject) if raw_subject is not None else None, 

206 payload.get("description"), 

207 list(payload.get("tags", [])), 

208 raw_external_id, 

209 datetime.fromisoformat(payload["occurred_at"]), 

210 ) 

211 return 

212 

213 if event.event_type == "CampaignStarted": 

214 await conn.execute( 

215 _UPDATE_STARTED_SQL, 

216 UUID(event.payload["campaign_id"]), 

217 datetime.fromisoformat(event.payload["occurred_at"]), 

218 ) 

219 return 

220 

221 if event.event_type == "CampaignHeld": 

222 await conn.execute( 

223 _UPDATE_HELD_SQL, 

224 UUID(event.payload["campaign_id"]), 

225 event.payload["reason"], 

226 datetime.fromisoformat(event.payload["occurred_at"]), 

227 ) 

228 return 

229 

230 if event.event_type == "CampaignResumed": 

231 await conn.execute( 

232 _UPDATE_RESUMED_SQL, 

233 UUID(event.payload["campaign_id"]), 

234 datetime.fromisoformat(event.payload["occurred_at"]), 

235 ) 

236 return 

237 

238 if event.event_type == "CampaignClosed": 

239 await conn.execute( 

240 _UPDATE_CLOSED_SQL, 

241 UUID(event.payload["campaign_id"]), 

242 datetime.fromisoformat(event.payload["occurred_at"]), 

243 ) 

244 return 

245 

246 if event.event_type == "CampaignAbandoned": 

247 await conn.execute( 

248 _UPDATE_ABANDONED_SQL, 

249 UUID(event.payload["campaign_id"]), 

250 event.payload["reason"], 

251 datetime.fromisoformat(event.payload["occurred_at"]), 

252 ) 

253 return 

254 

255 if event.event_type == "CampaignRunAdded": 

256 # membership add. Increments run_count denorm. 

257 # Status / last_status_reason unchanged (membership mutations 

258 # are NOT status transitions per design memo lock). 

259 await conn.execute( 

260 _UPDATE_RUN_ADDED_SQL, 

261 UUID(event.payload["campaign_id"]), 

262 ) 

263 return 

264 

265 if event.event_type == "CampaignRunRemoved": 

266 # membership remove. Decrements run_count denorm. 

267 # `reason` lives only on the event payload (per-membership 

268 # audit breadcrumb); does NOT update last_status_reason. 

269 await conn.execute( 

270 _UPDATE_RUN_REMOVED_SQL, 

271 UUID(event.payload["campaign_id"]), 

272 ) 

273 return 

274 

275 # Unsubscribed event type (defensive; the worker shouldn't deliver 

276 # foreign event types given subscribed_event_types). 

277 return 

278 

279 

280__all__ = ["CampaignSummaryProjection"]
