1"""LaunchSpec: a Method's vetted, injection-safe compute launch recipe. 

2 

3Per [[project-method-launch-spec-stage0-design]]. A `LaunchSpec` pins 

4HOW a compute Method is launched (the argv) so a conduct caller selects 

5a registered recipe and supplies schema-validated parameters, instead 

6of POSTing a raw command. The command shape lives on the Method, where 

7it is reviewed, versioned, and content-hashed. 

8 

9## Injection safety is a type fact 

10 

11The vocabulary is CLOSED and carries NO interpolation strings: 

12- `base_command` is a tuple of LITERAL argv tokens (argv[0] can never 

13 be a placeholder, because there are no placeholders). 

14- each `LaunchArg` NAMES a key in the Method's `parameters_schema`; it 

15 never CONTAINS a template token. At conduct time the named value is 

16 rendered with `str(...)` into exactly ONE inert argv token. 

17- `ArgStyle` is a closed enum (`value` | `flag_only`). New styles only 

18 via rule-of-three + gate review; the style set is the seam where 

19 interpolation could creep back, so it stays small and typed. 

20 

21There is therefore no template engine to exploit, and the executor 

22(`LocalProcessComputePort`) already runs argv-lists shell-free. A 

23malicious parameter value lands as one argv token, never as a second 

24command. 

25 

26Naming note: the memo sketched the enum as `ArgRepr` with field `repr`; 

27implemented as `ArgStyle` with field `style` to avoid shadowing the 

28`repr` builtin (a lint smell). Same closed two-value vocabulary. 

29 

30## Well-formedness vs cross-checks 

31 

32`validate_launch_spec` here checks ONLY self-contained well-formedness 

33(non-empty literal base_command, flag/position XOR, flag syntax, 

34contiguous positions, flag_only-needs-a-flag, unique names). It does 

35NOT check that a `LaunchArg.name` exists in the owning Method's 

36`parameters_schema` (that cross-check belongs to the 

37`update_method_launch_spec` decider, which has the schema in hand) nor 

38that the schema key is boolean for `FLAG_ONLY` (same place). 

39"""

41from __future__ import annotations 

42 

43import re 

44from dataclasses import dataclass 

45from enum import StrEnum 

46from typing import Any

48# A flag is one short (`-c`) or long (`--num-iter`) option token. No 

49# spaces, no `=`, no value glued on: the value is a SEPARATE argv token 

50# rendered from the named parameter, never spliced into the flag. 

51_FLAG_RE = re.compile(r"^--?[A-Za-z0-9][A-Za-z0-9-]*$")

54class InvalidLaunchSpecError(ValueError): 

55 """A LaunchSpec is malformed independent of any Method / schema. 

56 

57 Raised by `validate_launch_spec` and surfaced as HTTP 422 by the 

58 `update_method_launch_spec` route's validation handler. 

59 """ 

60 

61 def __init__(self, reason: str) -> None: 

62 super().__init__(f"Invalid launch spec: {reason}") 

63 self.reason = reason

66class ArgStyle(StrEnum): 

67 """Closed v1 vocabulary for how a named parameter renders into argv.""" 

68 

69 VALUE = "value" 

70 """Emit the value as a token: `[flag, str(value)]` in flag mode, or 

71 `[str(value)]` in positional mode. The common case (`--num-iter 100`).""" 

72 

73 FLAG_ONLY = "flag_only" 

74 """Boolean switch: emit `[flag]` iff the value is truthy, else nothing 

75 (`--remove-stripe` present or absent). Requires a flag (never 

76 positional) and a boolean schema key (checked at the slice).""" 

77 

78 

79@dataclass(frozen=True) 

80class LaunchArg: 

81 """One binding from a `parameters_schema` key to an argv position. 

82 

83 `name` NAMES a schema key (cross-checked at the slice). Exactly one 

84 of `flag` / `position` is set (flag mode vs positional mode). 

85 `required=True` means the conduct fails if the Run's 

86 effective_parameters lack the key. 

87 """ 

88 

89 name: str 

90 flag: str | None = None 

91 position: int | None = None 

92 required: bool = False 

93 style: ArgStyle = ArgStyle.VALUE 

94 

95 

96@dataclass(frozen=True) 

97class LaunchSpec: 

98 """A Method's full launch recipe: literal base command + ordered bindings. 

99 

100 `base_command` is the literal argv prefix. `args` are order- 

101 significant bindings (flag-mode args render in declared order; 

102 positional-mode args render by `position`). `input_arg` / `output_arg` 

103 are the flags preceding the caller-supplied input / output URIs 

104 (None => append the URI positionally). 

105 """ 

106 

107 base_command: tuple[str, ...] 

108 args: tuple[LaunchArg, ...] = () 

109 input_arg: str | None = None 

110 output_arg: str | None = None 

111

113def validate_launch_spec(spec: LaunchSpec) -> None: 

114 """Raise `InvalidLaunchSpecError` if `spec` is not well-formed. 

115 

116 Self-contained checks only (no schema). See module docstring for the 

117 well-formedness-vs-cross-check split. 

118 """ 

119 if not spec.base_command: 

120 raise InvalidLaunchSpecError("base_command must have at least one token") 

121 if any(not token for token in spec.base_command): 

122 raise InvalidLaunchSpecError("base_command tokens must be non-empty") 

123 

124 seen_names: set[str] = set() 

125 positions: list[int] = [] 

126 for arg in spec.args: 

127 if not arg.name: 127 ↛ 128line 127 didn't jump to line 128 because the condition on line 127 was never true

128 raise InvalidLaunchSpecError("a LaunchArg.name must be non-empty") 

129 if arg.name in seen_names: 

130 raise InvalidLaunchSpecError(f"duplicate LaunchArg name {arg.name!r}") 

131 seen_names.add(arg.name) 

132 

133 has_flag = arg.flag is not None 

134 has_position = arg.position is not None 

135 if has_flag == has_position: 

136 raise InvalidLaunchSpecError( 

137 f"LaunchArg {arg.name!r} must set exactly one of flag / position" 

138 ) 

139 if has_flag and not _FLAG_RE.match(arg.flag or ""): 

140 raise InvalidLaunchSpecError( 

141 f"LaunchArg {arg.name!r} flag {arg.flag!r} is not a valid option" 

142 ) 

143 if arg.style is ArgStyle.FLAG_ONLY and not has_flag: 

144 raise InvalidLaunchSpecError( 

145 f"LaunchArg {arg.name!r} style flag_only requires a flag, not a position" 

146 ) 

147 if has_position: 

148 if (arg.position or 0) < 0: 148 ↛ 149line 148 didn't jump to line 149 because the condition on line 148 was never true

149 raise InvalidLaunchSpecError(f"LaunchArg {arg.name!r} position must be >= 0") 

150 positions.append(arg.position or 0) 

151 

152 # Positional args must be a contiguous 0..n-1 block (no gaps / dups) 

153 # so argv assembly order is unambiguous. 

154 if sorted(positions) != list(range(len(positions))): 

155 raise InvalidLaunchSpecError("positional args must occupy contiguous positions 0..n-1") 

156 

157 for label, flag in (("input_arg", spec.input_arg), ("output_arg", spec.output_arg)): 

158 if flag is not None and not _FLAG_RE.match(flag): 

159 raise InvalidLaunchSpecError(f"{label} {flag!r} is not a valid option") 

160

162def launch_spec_to_dict(spec: LaunchSpec) -> dict[str, Any]: 

163 """JSON-friendly, ORDER-PRESERVING serialization. 

164 

165 `args` are NOT sorted: their order is semantic (it is argv order). 

166 Used both for the event payload and for `Method.content_subset`, so 

167 two recipes differing only in argv order hash differently. 

168 """ 

169 return { 

170 "base_command": list(spec.base_command), 

171 "args": [ 

172 { 

173 "name": arg.name, 

174 "flag": arg.flag, 

175 "position": arg.position, 

176 "required": arg.required, 

177 "style": arg.style.value, 

178 } 

179 for arg in spec.args 

180 ], 

181 "input_arg": spec.input_arg, 

182 "output_arg": spec.output_arg, 

183 } 

184 

185 

186def launch_spec_from_dict(data: dict[str, Any]) -> LaunchSpec: 

187 """Rebuild a `LaunchSpec` from its `launch_spec_to_dict` form. 

188 

189 `Any`-typed like the aggregate `from_stored` codecs: jsonb payloads 

190 arrive untyped and are reconstructed by position/key here. 

191 """ 

192 raw_args: list[dict[str, Any]] = data.get("args", []) 

193 args = tuple( 

194 LaunchArg( 

195 name=a["name"], 

196 flag=a.get("flag"), 

197 position=a.get("position"), 

198 required=a.get("required", False), 

199 style=ArgStyle(a.get("style", ArgStyle.VALUE.value)), 

200 ) 

201 for a in raw_args 

202 ) 

203 return LaunchSpec( 

204 base_command=tuple(data["base_command"]), 

205 args=args, 

206 input_arg=data.get("input_arg"), 

207 output_arg=data.get("output_arg"), 

208 )
