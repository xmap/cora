1"""Coarse post-hoc budget gate for LLM-backed agents. 

2 

3The enforcement seam that turns a declared `AgentBudget` into a real 

4control at the coarse tier of the enforcement ladder: debit after each 

5call (the recorded `cost_usd` on the inference entry) and refuse the 

6NEXT call once a cap is exhausted. Overspend is bounded to about one 

7in-flight call per caller, adequate for the cheap, serialized 

8subscriber agents (RunDebriefer, CautionDrafter). Autonomous or 

9expensive-long-context callers need the per-call pre-estimate tier 

10before this gate alone is safe (one call can exhaust a balance before 

11a post-hoc gate fires). 

12 

13## Windows 

14 

15`monthly_usd_cap` is summed over the UTC calendar month containing 

16`as_of`; `daily_token_cap` over the UTC calendar day. `as_of` is the 

17triggering event's `occurred_at`, NOT wall clock, so a replayed work 

18item gates identically (the non-determinism rule: subscribers decide 

19from event facts, not ambient time). 

20 

21## Failure direction 

22 

23The spend sums undercount (NULL-cost legacy rows, unrecorded cache 

24tokens; see the SpendLookup port docstring), so the gate errs 

25PERMISSIVE. A cap of exactly zero refuses every call: the `AgentBudget` 

26value object already documents zero as recorded no-spend intent, and 

27the gate is what makes that intent real. 

28 

29A SpendLookup ERROR, by contrast, propagates: the subscriber's apply 

30raises, the projection worker's retry loop with backoff is the failure 

31handler, and the bookmark does not advance. The gate never fails OPEN 

32on a lookup error; wrapping it in a permissive except would let a 

33database outage disable enforcement silently. 

34""" 

35 

36from dataclasses import dataclass 

37from datetime import UTC, datetime, timedelta 

38from typing import TYPE_CHECKING 

39 

40if TYPE_CHECKING: 

41 from cora.agent.aggregates.agent import Agent 

42 from cora.infrastructure.ports.spend_lookup import SpendLookup

45@dataclass(frozen=True) 

46class BudgetBreach: 

47 """One exhausted cap: which, by how much, over which window.""" 

48 

49 cap_kind: str # "monthly_usd_cap" | "daily_token_cap" 

50 cap_value: float 

51 spent: float 

52 window_start: datetime 

53 window_end: datetime 

54 

55 def describe(self) -> str: 

56 """One human-readable sentence for a deferred Decision's reasoning.""" 

57 return ( 

58 f"{self.cap_kind} of {self.cap_value:g} reached " 

59 f"(spent {self.spent:g} in the window starting " 

60 f"{self.window_start.isoformat()})" 

61 ) 

62 

63 

64def calendar_month_window(as_of: datetime) -> tuple[datetime, datetime]: 

65 """Half-open UTC calendar month `[first, first-of-next)` containing `as_of`.""" 

66 at = as_of.astimezone(UTC) 

67 start = datetime(at.year, at.month, 1, tzinfo=UTC) 

68 if at.month == 12: 

69 end = datetime(at.year + 1, 1, 1, tzinfo=UTC) 

70 else: 

71 end = datetime(at.year, at.month + 1, 1, tzinfo=UTC) 

72 return start, end 

73 

74 

75def calendar_day_window(as_of: datetime) -> tuple[datetime, datetime]: 

76 """Half-open UTC calendar day `[midnight, next-midnight)` containing `as_of`.""" 

77 at = as_of.astimezone(UTC) 

78 start = datetime(at.year, at.month, at.day, tzinfo=UTC) 

79 return start, start + timedelta(days=1) 

80 

81 

82async def find_budget_breach( 

83 *, 

84 agent: "Agent | None", 

85 spend_lookup: "SpendLookup", 

86 as_of: datetime, 

87) -> BudgetBreach | None: 

88 """Return the first exhausted cap for this agent, or None to permit. 

89 

90 An agent with no declared budget (or no Agent stream at all) is 

91 unlimited: declaration is opt-in, so absence of caps must never 

92 block. Caps are checked in declaration order (monthly USD, then 

93 daily tokens); the first breach wins, one SpendLookup SUM per 

94 declared cap. 

95 """ 

96 if agent is None or agent.budget is None: 

97 return None 

98 budget = agent.budget 

99 

100 if budget.monthly_usd_cap is not None: 

101 window_start, window_end = calendar_month_window(as_of) 

102 spend = await spend_lookup.find_agent_spend( 

103 agent_id=agent.id, 

104 window_start=window_start, 

105 window_end=window_end, 

106 ) 

107 if spend.usd_spent >= budget.monthly_usd_cap: 

108 return BudgetBreach( 

109 cap_kind="monthly_usd_cap", 

110 cap_value=budget.monthly_usd_cap, 

111 spent=spend.usd_spent, 

112 window_start=window_start, 

113 window_end=window_end, 

114 ) 

115 

116 if budget.daily_token_cap is not None: 

117 window_start, window_end = calendar_day_window(as_of) 

118 spend = await spend_lookup.find_agent_spend( 

119 agent_id=agent.id, 

120 window_start=window_start, 

121 window_end=window_end, 

122 ) 

123 if spend.tokens_spent >= budget.daily_token_cap: 

124 return BudgetBreach( 

125 cap_kind="daily_token_cap", 

126 cap_value=float(budget.daily_token_cap), 

127 spent=float(spend.tokens_spent), 

128 window_start=window_start, 

129 window_end=window_end, 

130 ) 

131 

132 return None
