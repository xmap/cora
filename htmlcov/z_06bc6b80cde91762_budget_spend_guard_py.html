1"""BudgetSpendGuard: the Agent BC's production `SpendGuard` adapter. 

2 

3The pre-estimate tier's policy lives here because the Agent BC owns both 

4facts the check needs: the caller's declared `AgentBudget` caps (on the 

5Agent aggregate) and the lifecycle state that says whether the agent may 

6act at all. Recorded spend comes through the same `SpendLookup` the 

7coarse post-hoc gate sums, over the same UTC calendar windows, so the 

8two tiers can never disagree about what has been spent. 

9 

10## Refusal policy 

11 

12 - Agent stream missing: permit. Declaration is opt-in; absence of an 

13 Agent aggregate must never block (mirrors `find_budget_breach`). 

14 - Agent not Versioned: refuse. Suspend means stop on every path, and 

15 the pre-call check is the earliest place the steering brain can be 

16 stopped (the post-hoc record path only notices AFTER spend). 

17 - No declared budget: permit. 

18 - Projected breach: refuse when recorded spend plus the estimated 

19 ceiling would exceed a cap (strictly greater: a call that lands 

20 exactly on the cap is the last one the envelope affords). Monthly 

21 USD is checked first, then daily tokens, one lookup per declared 

22 cap, matching the post-hoc gate's order. 

23 

24A lookup ERROR propagates, per the port's fail-closed stance. 

25""" 

26 

27from __future__ import annotations 

28 

29from typing import TYPE_CHECKING 

30 

31from cora.agent._budget_gate import calendar_day_window, calendar_month_window 

32from cora.agent.aggregates.agent import AgentStatus, load_agent 

33 

34if TYPE_CHECKING: 

35 from datetime import datetime 

36 from uuid import UUID 

37 

38 from cora.infrastructure.ports.event_store import EventStore 

39 from cora.infrastructure.ports.spend_lookup import SpendLookup

42class BudgetSpendGuard: 

43 """`SpendGuard` over the Agent aggregate's caps and the spend ledger.""" 

44 

45 def __init__(self, *, event_store: EventStore, spend_lookup: SpendLookup) -> None: 

46 self._event_store = event_store 

47 self._spend_lookup = spend_lookup 

48 

49 async def refusal_reason( 

50 self, 

51 *, 

52 agent_id: UUID, 

53 estimated_cost_usd: float, 

54 estimated_tokens: int, 

55 as_of: datetime, 

56 ) -> str | None: 

57 agent = await load_agent(self._event_store, agent_id) 

58 if agent is None: 

59 return None 

60 if agent.status is not AgentStatus.VERSIONED: 

61 return f"agent is {agent.status.value}, not Versioned; calls are stopped" 

62 if agent.budget is None: 

63 return None 

64 budget = agent.budget 

65 

66 if budget.monthly_usd_cap is not None: 

67 window_start, window_end = calendar_month_window(as_of) 

68 spend = await self._spend_lookup.find_agent_spend( 

69 agent_id=agent_id, 

70 window_start=window_start, 

71 window_end=window_end, 

72 ) 

73 projected = spend.usd_spent + estimated_cost_usd 

74 if projected > budget.monthly_usd_cap: 

75 return ( 

76 f"monthly_usd_cap of {budget.monthly_usd_cap:g} would be breached: " 

77 f"{spend.usd_spent:g} spent this month plus an estimated " 

78 f"{estimated_cost_usd:g} ceiling for this call" 

79 ) 

80 

81 if budget.daily_token_cap is not None: 

82 window_start, window_end = calendar_day_window(as_of) 

83 spend = await self._spend_lookup.find_agent_spend( 

84 agent_id=agent_id, 

85 window_start=window_start, 

86 window_end=window_end, 

87 ) 

88 projected_tokens = spend.tokens_spent + estimated_tokens 

89 if projected_tokens > budget.daily_token_cap: 

90 return ( 

91 f"daily_token_cap of {budget.daily_token_cap} would be breached: " 

92 f"{spend.tokens_spent} tokens spent today plus an estimated " 

93 f"{estimated_tokens} for this call" 

94 ) 

95 

96 return None 

97
