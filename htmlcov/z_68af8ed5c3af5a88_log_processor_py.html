1"""Structlog processor that injects the active span's trace context. 

2 

3Added to the structlog wrapper chain in `cora.infrastructure.logging` 

4so every log line emitted inside an active span carries `trace_id`, 

5`span_id`, and `trace_flags` — letting log-aggregator queries pivot 

6from a log line to the corresponding distributed trace. 

7 

8When no span is active (no-op tracer in tests, or code paths invoked 

9outside an instrumented entrypoint), the processor is a no-op: it 

10returns the event_dict unchanged. No keys are added rather than 

11`None`-valued keys, which keeps log lines compact and avoids polluting 

12indexed fields with sentinel values. 

13 

14`trace_id` is rendered as a 32-char lowercase hex string (W3C trace 

15context format), matching what OTel exporters write to spans. 

16`span_id` is the 16-char span hex. 

17""" 

18 

19from collections.abc import MutableMapping 

20from typing import Any 

21 

22from opentelemetry import trace 

23from opentelemetry.trace import INVALID_SPAN_ID, INVALID_TRACE_ID 

24 

25# structlog's Processor signature is (WrappedLogger, str, EventDict) 

26# where EventDict is `MutableMapping[str, Any]`. We type-match exactly 

27# rather than depend on `structlog.types.EventDict` so this module 

28# stays free of structlog imports (it's a pure function consumed by 

29# the structlog wrapper chain in `cora.infrastructure.logging`). 

30EventDict = MutableMapping[str, Any] 

31 

32__all__ = ["add_trace_context"] 

33 

34 

35def add_trace_context(_logger: Any, _method_name: str, event_dict: EventDict) -> EventDict: 

36 """Inject trace_id/span_id/trace_flags from the active span, if any.""" 

37 span_context = trace.get_current_span().get_span_context() 

38 if span_context.trace_id == INVALID_TRACE_ID: 

39 return event_dict 

40 event_dict["trace_id"] = format(span_context.trace_id, "032x") 

41 if span_context.span_id != INVALID_SPAN_ID: 41 ↛ 43line 41 didn't jump to line 43 because the condition on line 41 was always true

42 event_dict["span_id"] = format(span_context.span_id, "016x") 

43 event_dict["trace_flags"] = format(span_context.trace_flags, "02x") 

44 return event_dict
