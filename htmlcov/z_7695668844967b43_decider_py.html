1"""Pure decider for the `RegisterDistribution` command. 

2 

3Pure function: given the (always None) Distribution state, a 

4`RegisterDistribution` command, and a pre-loaded 

5`DistributionRegistrationContext`, returns the events to append. 

6No I/O, no awaits, no side effects. 

7 

8`now` and `new_id` are injected by the application handler from 

9the Clock and IdGenerator ports. 

10 

11## Firing order 

12 

13Per [[project-data-distribution-design]] L17 (re-stated here so 

14the order is local-and-greppable on the decider source): 

15 

16 1. Pydantic 422 boundary parse-shape (off-decider). 

17 2. VO field validation: DistributionUri, reused DatasetChecksum, 

18 reused DatasetEncoding raise `Invalid*Error`. 

19 3. Defensive `InvalidAccessProtocolError` re-check (in-process 

20 callers bypassing the boundary). 

21 4. State-is-None genesis guard -> 

22 `DistributionAlreadyExistsError`. Fires BEFORE the 

23 Dataset/Supply context checks because if the stream already 

24 has events, the conflict is unambiguous regardless of other 

25 fields. 

26 5. (handler) Dataset pre-load -> `DatasetNotFoundError` (NOT 

27 here; the context's `dataset` is non-None by handler 

28 contract). 

29 6. `context.dataset.status == DISCARDED` -> 

30 `DistributionCannotRegisterOnDiscardedDatasetError`. 

31 7. (handler) SupplyLookup -> `DistributionSupplyNotFoundError` 

32 (NOT here; the context's `supply` is non-None by handler 

33 contract). 

34 8. `context.supply.kind != "Storage"` -> 

35 `DistributionCannotRegisterOnNonStorageSupplyError`. Per L28 

36 status-agnostic bind: every Supply lifecycle is acceptable 

37 so long as kind is Storage. 

38 9. `command.checksum_algorithm != context.dataset.checksum.algorithm` 

39 -> `DistributionChecksumAlgorithmMismatchError` (checked before the 

40 value, since two 64-hex values under different algorithms are not 

41 comparable). 

42 9b. `command.checksum_value != context.dataset.checksum.value` 

43 -> `DistributionChecksumMismatchError`. 

44 10. `command.byte_size != context.dataset.byte_size` -> 

45 `DistributionByteSizeMismatchError`. 

46 11. Emit `DistributionRegistered`. 

47 12. Writer-side, post-request: projection UNIQUE INDEX collision 

48 handled by the projection writer (L31; NOT decider concern). 

49 

50## VO trim semantics 

51 

52Field VOs (`DistributionUri`, `DatasetChecksum`, `DatasetEncoding`) 

53handle their own trimming + validation in `__post_init__`; the 

54on-the-wire payload carries the trimmed values (so the persisted 

55event payload is canonical). 

56""" 

57 

58from datetime import datetime 

59from uuid import UUID 

60 

61from cora.data.aggregates.dataset import ( 

62 DatasetChecksum, 

63 DatasetEncoding, 

64 DatasetStatus, 

65 InvalidDatasetChecksumError, 

66 InvalidDatasetEncodingError, 

67) 

68from cora.data.aggregates.distribution import ( 

69 STORAGE_SUPPLY_KIND, 

70 AccessProtocol, 

71 Distribution, 

72 DistributionAlreadyExistsError, 

73 DistributionByteSizeMismatchError, 

74 DistributionCannotRegisterOnDiscardedDatasetError, 

75 DistributionCannotRegisterOnNonStorageSupplyError, 

76 DistributionChecksumAlgorithmMismatchError, 

77 DistributionChecksumMismatchError, 

78 DistributionRegistered, 

79 DistributionUri, 

80 InvalidAccessProtocolError, 

81 InvalidDistributionChecksumError, 

82 InvalidDistributionEncodingError, 

83 validate_distribution_byte_size, 

84) 

85from cora.data.features.register_distribution.command import RegisterDistribution 

86from cora.data.features.register_distribution.context import ( 

87 DistributionRegistrationContext, 

88) 

89from cora.shared.identity import ActorId 

90

92def decide( 

93 state: Distribution | None, 

94 command: RegisterDistribution, 

95 *, 

96 context: DistributionRegistrationContext, 

97 now: datetime, 

98 new_id: UUID, 

99 registered_by: ActorId, 

100) -> list[DistributionRegistered]: 

101 """Decide the events produced by registering a new Distribution. 

102 

103 Invariants: 

104 (Firing order per [[project-data-distribution-design]] L17.) 

105 - URI must be valid -> InvalidDistributionUriError 

106 - Checksum shape must be valid -> InvalidDistributionChecksumError 

107 - byte_size must be valid -> InvalidDistributionByteSizeError 

108 - Encoding shape must be valid -> InvalidDistributionEncodingError 

109 - access_protocol must be in closed enum 

110 -> InvalidAccessProtocolError 

111 - State must be None (genesis-only) 

112 -> DistributionAlreadyExistsError 

113 - Dataset must not be Discarded 

114 -> DistributionCannotRegisterOnDiscardedDatasetError 

115 - Supply.kind must be "Storage" 

116 -> DistributionCannotRegisterOnNonStorageSupplyError 

117 - Checksum algorithm equality vs Dataset (checked before value) 

118 -> DistributionChecksumAlgorithmMismatchError 

119 - Checksum value equality vs Dataset 

120 -> DistributionChecksumMismatchError 

121 - byte_size equality vs Dataset 

122 -> DistributionByteSizeMismatchError 

123 """ 

124 # Step 2: VO field validation. Each VO raises its specific error 

125 # class on failure (mapped to HTTP 400). DatasetChecksum and 

126 # DatasetEncoding are reused verbatim per L8; the decider catches 

127 # their Dataset-prefixed exceptions and re-raises with 

128 # Distribution-prefixed context so the operator-facing log says 

129 # "register_distribution" not "register_dataset". 

130 uri = DistributionUri(command.uri) 

131 try: 

132 checksum = DatasetChecksum( 

133 algorithm=command.checksum_algorithm, 

134 value=command.checksum_value, 

135 ) 

136 except InvalidDatasetChecksumError as exc: 

137 raise InvalidDistributionChecksumError( 

138 algorithm=command.checksum_algorithm, 

139 value=command.checksum_value, 

140 reason=str(exc), 

141 ) from exc 

142 byte_size = validate_distribution_byte_size(command.byte_size) 

143 try: 

144 encoding = DatasetEncoding( 

145 media_type=command.media_type, 

146 conforms_to=command.conforms_to, 

147 ) 

148 except InvalidDatasetEncodingError as exc: 

149 raise InvalidDistributionEncodingError(reason=str(exc)) from exc 

150 

151 # Step 3: defensive AccessProtocol re-check. The REST + MCP boundary 

152 # rejects out-of-enum values at 422; this branch fires when an 

153 # in-process caller (saga, atomic cross-BC write, test) bypasses 

154 # the boundary with a bare string. 

155 try: 

156 access_protocol = AccessProtocol(command.access_protocol) 

157 except ValueError as exc: 

158 raise InvalidAccessProtocolError(command.access_protocol) from exc 

159 

160 # Step 4: State-is-None genesis guard. Fires before Dataset / Supply 

161 # context checks per L17. 

162 if state is not None: 

163 raise DistributionAlreadyExistsError(state.id) 

164 

165 # Step 6: Dataset status guard (Discarded means bytes have been 

166 # deleted from storage; cannot bind a new copy). 

167 if context.dataset.status is DatasetStatus.DISCARDED: 

168 raise DistributionCannotRegisterOnDiscardedDatasetError(dataset_id=context.dataset.id) 

169 

170 # Step 8: Supply kind guard. Status-agnostic per L28: every Supply 

171 # lifecycle is acceptable, only kind is gated. 

172 if context.supply.kind != STORAGE_SUPPLY_KIND: 

173 raise DistributionCannotRegisterOnNonStorageSupplyError( 

174 supply_id=context.supply.supply_id, 

175 actual_kind=context.supply.kind, 

176 ) 

177 

178 # Steps 9 + 10: byte-identical-copy invariants. The Dataset is 

179 # already loaded in the context (free O(1) checks). Algorithm is 

180 # compared first: two 64-hex values under different algorithms 

181 # (sha256 vs sha256-tree) are not comparable, so a value match would 

182 # be meaningless without it. 

183 if checksum.algorithm != context.dataset.checksum.algorithm: 

184 raise DistributionChecksumAlgorithmMismatchError( 

185 dataset_id=context.dataset.id, 

186 expected_algorithm=context.dataset.checksum.algorithm, 

187 actual_algorithm=checksum.algorithm, 

188 ) 

189 if checksum.value != context.dataset.checksum.value: 

190 raise DistributionChecksumMismatchError( 

191 dataset_id=context.dataset.id, 

192 expected_checksum=context.dataset.checksum.value, 

193 actual_checksum=checksum.value, 

194 ) 

195 if byte_size != context.dataset.byte_size: 

196 raise DistributionByteSizeMismatchError( 

197 dataset_id=context.dataset.id, 

198 expected_byte_size=context.dataset.byte_size, 

199 actual_byte_size=byte_size, 

200 ) 

201 

202 # Step 11: emit DistributionRegistered. The event payload carries 

203 # primitive types only per CONTRIBUTING.md "Primitives in event 

204 # payloads"; VOs reconstructed by the evolver on fold. 

205 return [ 

206 DistributionRegistered( 

207 distribution_id=new_id, 

208 dataset_id=command.dataset_id, 

209 supply_id=command.supply_id, 

210 uri=uri.value, 

211 checksum_algorithm=checksum.algorithm, 

212 checksum_value=checksum.value, 

213 byte_size=byte_size, 

214 media_type=encoding.media_type, 

215 conforms_to=encoding.conforms_to, 

216 access_protocol=access_protocol.value, 

217 occurred_at=now, 

218 registered_by=registered_by, 

219 ) 

220 ]
