1"""Cross-cutting idempotency decorator for command handlers. 

2 

3`with_idempotency(handler, store, *, command_name, serialize_result, 

4deserialize_result, lock_stale_seconds)` returns a wrapped handler 

5with full Idempotency-Key support: two-phase claim, 4xx error 

6caching, stale-lock recovery. The wrap is applied in each BC's 

7`wire.py` so every create-style command handler gets idempotency 

8through one composition point; slices stay focused on domain logic. 

9 

10Lives at `cora/infrastructure/` (not in any single BC) because it 

11applies uniformly to every BC's command handlers and depends only on 

12the IdempotencyStore port + the cross-BC handler-call convention 

13(`(command, *, principal_id, correlation_id, causation_id) -> TResult`). 

14 

15 

16For each call with a key, the decorator: 

17 

18 1. Computes the canonical hash of the command body. 

19 2. Calls `store.claim()` — a single SQL round-trip (in the happy 

20 path) that either wins the in-flight lock or returns the 

21 existing outcome. 

22 3. Dispatches on the outcome: 

23 - `Claimed` -> run handler. On success, `finalize_success()` 

24 and return. On a cacheable 4xx exception, `finalize_error()` 

25 and re-raise the original. On a 5xx / uncacheable exception, 

26 re-raise without finalizing — the row stays locked and is 

27 recovered on the next retry via stale-lock takeover. 

28 - `CachedSuccess` -> return the deserialized cached result. 

29 - `CachedError` -> raise `CachedHandlerError`; the route layer 

30 reconstructs the original 4xx response. 

31 - `LockedRecent` -> raise `IdempotencyClaimLostError` -> 409 

32 with `Retry-After: 1`. Standard HTTP retry-after semantics. 

33 - `HashConflict` -> raise `IdempotencyConflictError` -> 422. 

34 

35## Hashing 

36 

37Commands MUST be frozen dataclasses (with primitive or nested- 

38dataclass fields). `hash_command(cmd)` serializes via `asdict` + 

39`json.dumps(sort_keys=True, default=str)` + SHA256. 

40 

41`set` and `frozenset` fields on the command get normalized to sorted 

42lists before hashing — Python's set iteration order varies across 

43processes (PYTHONHASHSEED randomizes string hashing) so the same 

44logical set would produce different `repr` and different hashes 

45under multiple workers, manifesting as spurious 422 "Idempotency-Key 

46conflict" responses on legitimate retries. Routes that convert JSON 

47arrays to `frozenset` for command construction (for example permission 

48sets in `DefinePolicy`) rely on this normalization. 

49 

50## Error classification (4xx caching scope) 

51 

52`classify_error_status(exc)` maps an exception to its HTTP status 

53by class-name convention. Returns None for unknown shapes (treated 

54as 5xx -> not cached -> retry-friendly). The convention codifies 

55existing routes.py groupings: 

56 

57 - `Invalid*` -> 400 

58 - `UnauthorizedError` -> 403 

59 - `*NotFoundError` -> 404 

60 - `*AlreadyExistsError`, `*Cannot*Error`, `ConcurrencyError` -> 409 

61 - `IdempotencyConflictError` -> 422 

62 

63Per-class override via `idempotency_http_status: ClassVar[int]` 

64attribute (rare; convention covers the existing 67-class spectrum). 

65 

66## Key length 

67 

68Capped at `_MAX_KEY_LENGTH` (255 chars, matching Stripe's limit). 

69Longer keys raise `ValueError` from the decorator before any store 

70lookup or handler invocation. 

71"""

99def _noop_serialize(_value: None) -> None: 

100 """Serialize codec for None-returning handlers idempotency-wrapped. 

101 

102 `with_idempotency` requires a serialize_result / deserialize_result 

103 pair; for handlers that return None there is no payload to round- 

104 trip. The pair stays symmetric (serializes None to None, deserializes 

105 None to None) so the cache hit replays "success with None". 

106 

107 First use: 6j adjust_run. Other candidates per the design memo: 

108 hold_run / resume_run / Procedure-step appends (all 204-returning). 

109 """ 

110 return None

122def _normalize_for_hash(obj: Any) -> Any: 

123 """Recursively normalize containers so the JSON form is hash-stable. 

124 

125 `set` and `frozenset` are sorted by string form — without this, 

126 PYTHONHASHSEED randomizes set iteration across workers and 

127 breaks idempotency. `dict` and `list`/`tuple` recurse into their 

128 elements. 

129 """ 

130 if isinstance(obj, (set, frozenset)): 

131 items_set = cast("set[Any] | frozenset[Any]", obj) 

132 return sorted((_normalize_for_hash(item) for item in items_set), key=str) 

133 if isinstance(obj, dict): 

134 items_dict = cast("dict[Any, Any]", obj) 

135 return {k: _normalize_for_hash(v) for k, v in items_dict.items()} 

136 if isinstance(obj, (list, tuple)): 

137 items_seq = cast("list[Any] | tuple[Any, ...]", obj) 

138 return [_normalize_for_hash(item) for item in items_seq] 

139 return obj

142def hash_command(command: Any) -> str: 

143 """SHA256 hex digest of canonical JSON of the command's dict form.""" 

144 if not is_dataclass(command) or isinstance(command, type): 

145 msg = ( 

146 f"hash_command requires a dataclass instance, got {type(command).__name__}. " 

147 "Commands across the codebase are frozen dataclasses by convention." 

148 ) 

149 raise TypeError(msg) 

150 normalized = _normalize_for_hash(asdict(command)) 

151 canonical = json.dumps(normalized, sort_keys=True, default=str) 

152 return hashlib.sha256(canonical.encode()).hexdigest()

155def classify_error_status(exc: BaseException) -> int | None: 

156 """Map a domain exception class to its HTTP status by class-name convention. 

157 

158 Returns None when the exception isn't a known cacheable shape 

159 (treated as 5xx -> not cached, retry-friendly). Per-class override 

160 via `idempotency_http_status: ClassVar[int]` attribute on the 

161 exception class. 

162 """ 

163 explicit = getattr(type(exc), "idempotency_http_status", None) 

164 if explicit is not None: 

165 return int(explicit) 

166 name = type(exc).__name__ 

167 if name.startswith("Invalid"): 

168 return 400 

169 if name == "UnauthorizedError": 

170 return 403 

171 if name.endswith("NotFoundError"): 

172 return 404 

173 if name.endswith("AlreadyExistsError"): 

174 return 409 

175 if "Cannot" in name and name.endswith("Error"): 

176 return 409 

177 if name == "ConcurrencyError": 

178 return 409 

179 if name == "IdempotencyConflictError": 

180 return 422 

181 return None

215def with_idempotency[TCommand, TResult]( 

216 handler: _BareHandler[TCommand, TResult], 

217 store: IdempotencyStore, 

218 *, 

219 command_name: str, 

220 serialize_result: Callable[[TResult], Any], 

221 deserialize_result: Callable[[Any], TResult], 

222 lock_stale_seconds: int, 

223) -> _IdempotentHandler[TCommand, TResult]: 

224 """Wrap a bare command handler with two-phase Idempotency-Key support. 

225 

226 `lock_stale_seconds` is sourced from `Settings.idempotency_lock_stale_seconds` 

227 at wire-time. It controls how long a locked row sits before being 

228 re-claimable by another request (worker-crash recovery). 

229 """ 

230 

231 async def wrapped( 

232 command: TCommand, 

233 *, 

234 principal_id: UUID, 

235 correlation_id: UUID, 

236 causation_id: UUID | None = None, 

237 surface_id: UUID = NIL_SENTINEL_ID, 

238 idempotency_key: str | None = None, 

239 ) -> TResult: 

240 if idempotency_key is None: 

241 return await handler( 

242 command, 

243 principal_id=principal_id, 

244 correlation_id=correlation_id, 

245 causation_id=causation_id, 

246 surface_id=surface_id, 

247 ) 

248 

249 if len(idempotency_key) > _MAX_KEY_LENGTH: 

250 msg = f"Idempotency-Key length {len(idempotency_key)} exceeds maximum {_MAX_KEY_LENGTH}" 

251 raise ValueError(msg) 

252 

253 cmd_hash = hash_command(command) 

254 outcome = await store.claim( 

255 principal_id, 

256 idempotency_key, 

257 surface_id, 

258 cmd_hash, 

259 command_name, 

260 lock_stale_seconds=lock_stale_seconds, 

261 ) 

262 

263 log_ctx = { 

264 "command_name": command_name, 

265 "principal_id": str(principal_id), 

266 "correlation_id": str(correlation_id), 

267 "key": idempotency_key, 

268 } 

269 

270 match outcome: 

271 case Claimed(): 

272 _log.info("idempotency.claimed", **log_ctx) 

273 try: 

274 result = await handler( 

275 command, 

276 principal_id=principal_id, 

277 correlation_id=correlation_id, 

278 causation_id=causation_id, 

279 surface_id=surface_id, 

280 ) 

281 except Exception as exc: 

282 status = classify_error_status(exc) 

283 if status is not None and 400 <= status < 500: 

284 await store.finalize_error( 

285 principal_id, 

286 idempotency_key, 

287 surface_id, 

288 error_type=_full_class_name(exc), 

289 error_msg=str(exc), 

290 ) 

291 _log.info( 

292 "idempotency.cached_error", 

293 error_type=_full_class_name(exc), 

294 http_status=status, 

295 **log_ctx, 

296 ) 

297 else: 

298 # 5xx or unknown shape: leave the row locked. 

299 # Stale-lock recovery will let the next retry 

300 # re-claim after `lock_stale_seconds`. 

301 _log.info( 

302 "idempotency.uncached_error", 

303 error_type=_full_class_name(exc), 

304 **log_ctx, 

305 ) 

306 raise 

307 await store.finalize_success( 

308 principal_id, 

309 idempotency_key, 

310 surface_id, 

311 serialize_result(result), 

312 ) 

313 _log.info("idempotency.cached_success", **log_ctx) 

314 return result 

315 case CachedSuccess(result=cached): 

316 _log.info("idempotency.hit_success", **log_ctx) 

317 return deserialize_result(cached) 

318 case CachedError(error_type=error_type, error_msg=error_msg): 

319 _log.info( 

320 "idempotency.hit_error", 

321 cached_error_type=error_type, 

322 **log_ctx, 

323 ) 

324 raise CachedHandlerError(error_type=error_type, error_msg=error_msg) 

325 case LockedRecent(locked_at=locked_at): 

326 _log.info( 

327 "idempotency.claim_lost", 

328 locked_at=locked_at.isoformat(), 

329 **log_ctx, 

330 ) 

331 raise IdempotencyClaimLostError(key=idempotency_key, locked_at=locked_at) 

332 case HashConflict(expected_hash=expected, actual_hash=actual): 332 ↛ exitline 332 didn't return from function 'wrapped' because the pattern on line 332 always matched

333 _log.info( 

334 "idempotency.conflict", 

335 expected_hash=expected, 

336 actual_hash=actual, 

337 **log_ctx, 

338 ) 

339 raise IdempotencyConflictError( 

340 key=idempotency_key, 

341 expected_hash=expected, 

342 actual_hash=actual, 

343 ) 

344 

345 return wrapped
