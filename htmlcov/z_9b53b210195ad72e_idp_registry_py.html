1"""`IdentityProviderRegistry` — process-singleton verifier router. 

2 

3Owns the per-issuer `TokenVerifier` mapping at the composition root. 

4The HTTP/MCP middleware hands the registry a raw bearer token; the 

5registry routes to the right adapter by: 

6 

7 1. **Token shape**: if the token parses as 3 base64url chunks 

8 separated by `.` it's a JWT; otherwise it's opaque. 

9 2. **For JWTs**: peek the unverified header's `iss` claim (no 

10 signature check — that happens inside the verifier) and route 

11 to the matching `JwtTokenVerifier`. 

12 3. **For opaque tokens**: route to the deployment's configured 

13 `IntrospectionTokenVerifier`. Exactly one per registry today; if 

14 deployments ever need multiple opaque-token IdPs, the registry 

15 grows a discriminator (probably the token prefix per the 

16 pattern GitHub `ghp_`/`gho_`/`ghs_` uses). 

17 

18## Why peek-unverified for JWT routing 

19 

20Reading `iss` before signature verification looks scary. It's safe 

21because the verifier we route to STILL checks `iss == self._issuer` 

22strictly — peeking is just routing, not trust. An attacker forging 

23an `iss` claim still has to forge a signature the chosen 

24`JwtTokenVerifier`'s JWKS won't accept. This is the standard pattern 

25(`PyJWT.get_unverified_header` / `decode_complete` with 

26`verify_signature=False` for routing only). 

27 

28## Anti-pattern guards 

29 

30- Empty registry construction is a hard error (you can't have an 

31 `IdpRegistry` with zero IdPs; that's a misconfigured deployment). 

32- Duplicate issuer registration is a hard error (two verifiers for 

33 the same `iss` is ambiguous). 

34- Unknown-issuer token → `InvalidTokenError("wrong_issuer", ...)` 

35 immediately (no fallback to introspection — issuers must be 

36 pre-registered). 

37 

38## Subject mapper note 

39 

40A `SubjectMapper` is required at construction. Today the only 

41implementation is the trivial in-memory `dict` (`StaticSubjectMapper`) 

42used by tests. The production mapper that queries the access 

43projection's `actor_idp_bindings` table is the planned sibling. 

44"""

58_MAX_TOKEN_LENGTH = 8192 

59"""DoS guard (gate-review F11) — reject implausibly large tokens at the 

60boundary before any base64-decode / parse work runs. Real OAuth tokens 

61sit well under this: JWTs at most a few KB even with rich claims; 

62opaque tokens are typically <256 chars. The cap stops an attacker from 

63amplifying a single megabyte-string POST into expensive parse + JWKS 

64fetch CPU."""

81class IdentityProviderRegistry: 

82 """Process-singleton router from inbound token → matching `TokenVerifier`. 

83 

84 Constructed once at lifespan start from `Settings.identity_providers`; 

85 held on the `Kernel` and used by the middleware. 

86 """ 

87 

88 def __init__( 

89 self, 

90 jwt_verifiers: list[JwtTokenVerifier], 

91 introspection_token_verifier: IntrospectionTokenVerifier | None = None, 

92 ) -> None: 

93 """Construct the registry from explicit verifier instances. 

94 

95 `jwt_verifiers` — one per JWT-issuing IdP. May be empty (for example, 

96 deployment that only uses Globus opaque tokens). 

97 `introspection_token_verifier` — at most one per deployment. None 

98 is valid (deployment serves only JWT-issuing IdPs). 

99 

100 At least one verifier (JWT or introspection) MUST be present; 

101 the registry refuses an empty configuration. 

102 """ 

103 if not jwt_verifiers and introspection_token_verifier is None: 

104 msg = ( 

105 "IdentityProviderRegistry must be constructed with at least " 

106 "one verifier (JWT or introspection). An empty registry can " 

107 "never authenticate any request — likely a misconfigured " 

108 "Settings.identity_providers list." 

109 ) 

110 raise ValueError(msg) 

111 

112 by_issuer: dict[str, JwtTokenVerifier] = {} 

113 for v in jwt_verifiers: 

114 if v.issuer in by_issuer: 

115 msg = ( 

116 f"Duplicate JwtTokenVerifier registered for issuer={v.issuer!r}. " 

117 "Each issuer maps to exactly one verifier." 

118 ) 

119 raise ValueError(msg) 

120 by_issuer[v.issuer] = v 

121 self._jwt_by_issuer = by_issuer 

122 self._introspection = introspection_token_verifier 

123 

124 async def verify( 

125 self, 

126 token: str, 

127 *, 

128 expected_audience: UUID, 

129 ) -> VerifiedPrincipal: 

130 """Route the token to the right verifier and return the result. 

131 

132 Raises `InvalidTokenError` if the token shape can't be 

133 classified, the issuer isn't registered, or the chosen 

134 verifier raises. Raises `IntrospectionUnavailableError` only 

135 from the introspection path. 

136 """ 

137 if not token: 

138 raise InvalidTokenError("malformed", "empty token") 

139 if len(token) > _MAX_TOKEN_LENGTH: 

140 raise InvalidTokenError( 

141 "malformed", 

142 f"token length {len(token)} exceeds maximum {_MAX_TOKEN_LENGTH} bytes", 

143 ) 

144 

145 verifier = self._choose_verifier(token) 

146 return await verifier.verify(token, expected_audience=expected_audience) 

147 

148 def _choose_verifier(self, token: str) -> _RegistryEntry: 

149 if _looks_like_jwt(token): 

150 # Peek `iss` from the JWT payload (signature unverified — 

151 # the chosen JwtTokenVerifier still strict-checks `iss` against 

152 # its own configuration, so peeking only routes; trust is 

153 # not taken from this decode). 

154 try: 

155 payload = jwt.decode( 

156 token, 

157 options={"verify_signature": False, "verify_exp": False}, 

158 ) 

159 except jwt.DecodeError as exc: 

160 raise InvalidTokenError("malformed", str(exc)) from exc 

161 iss = payload.get("iss") 

162 if not isinstance(iss, str): 

163 raise InvalidTokenError("malformed", "JWT missing string 'iss' claim") 

164 verifier = self._jwt_by_issuer.get(iss) 

165 if verifier is None: 

166 raise InvalidTokenError( 

167 "wrong_issuer", 

168 f"no JwtTokenVerifier registered for iss={iss!r}", 

169 ) 

170 return verifier 

171 # Opaque token branch. 

172 if self._introspection is None: 

173 raise InvalidTokenError( 

174 "malformed", 

175 "opaque token but no IntrospectionTokenVerifier registered", 

176 ) 

177 return self._introspection 

178 

179 

180def _looks_like_jwt(token: str) -> bool: 

181 """Cheap shape check: a JWT is three `.`-separated base64url chunks. 

182 

183 NOT a security check — opaque tokens that happen to contain two 

184 dots would false-positive. The selected `JwtTokenVerifier` then fails 

185 fast on `bad_signature`/`malformed`. The shape probe just avoids 

186 a pointless introspection call on every clearly-JWT-shaped 

187 token. 

188 """ 

189 return token.count(".") == 2
