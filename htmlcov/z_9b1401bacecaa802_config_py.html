47 # Observability — OpenTelemetry 

48 # `none` keeps the global no-op tracer (used by tests so spans don't 

49 # accumulate across many `create_app()` instances). `console` writes 

50 # spans to stdout (handy for local dev). `otlp` exports to a 

51 # collector via the standard `OTEL_EXPORTER_OTLP_*` env vars 

52 # (we deliberately do NOT shadow them with our own setting so 

53 # existing OTel deployment tooling Just Works). 

54 # Resource attribute `service.name` defaults to `cora-api`; override 

55 # if the same code is deployed under multiple service identities. 

56 # Sampler ratio is only consulted when otel_exporter == "otlp"; the 

57 # console exporter always exports every span (development is loud 

58 # by design). 1.0 = sample everything; lower in high-traffic prod. 

59 otel_exporter: OtelExporter = "none" 

60 otel_service_name: str = "cora-api" 

61 otel_sampler_ratio: float = 1.0

63 # Authorization — Trust BC wiring 

64 # When None, `build_kernel` wires `AllowAllAuthorize` and every 

65 # command is permitted (Phase 1 default; matches dev/test). When 

66 # set to a UUID, `TrustAuthorize` is wired and gates every command 

67 # through that single Policy aggregate. Multi-policy resolution 

68 # via projections lands in a later phase; until then this is one 

69 # policy per deployment. 

70 # 

71 # Phase A (2026-05-18) seeded the System Bootstrap Policy at a 

72 # fixed UUID so production deployments can enable real authz with 

73 # a single env var instead of the old 3-step dance: 

74 # 

75 # TRUST_POLICY_ID=00000000-0000-0000-0000-000000000001 

76 # 

77 # The seed permits SYSTEM_PRINCIPAL_ID to call DefinePolicy + 

78 # RegisterActor on the nil conduit — the minimum needed to 

79 # register a real admin Actor and promote a real admin Policy. 

80 # The default stays None (AllowAllAuthorize) for now because 

81 # ~2400 tests pass arbitrary principal_ids; flipping the default 

82 # is gated on a test-fixture audit (memory: 

83 # project_bootstrap_policy_design.md, WI8). 

84 trust_policy_id: UUID | None = None 

85 

86 # Production deployments behind an auth proxy that sets 

87 # `X-Principal-Id` should set this true: requests without the 

88 # header are then rejected with 401 instead of silently falling 

89 # back to `SYSTEM_PRINCIPAL_ID`. Default false matches the 

90 # Phase 1 dev / test posture where the fallback is convenient. 

91 # The startup check in `create_app()` refuses to boot when 

92 # `app_env in {"prod", "production"}` and this is False, so a 

93 # production deployment cannot accidentally launch with the 

94 # permissive default. 

95 require_authenticated_principal: bool = False 

96 

97 # Projection worker 

98 # `projection_use_listen_notify=True` (default) wires the worker's 

99 # wake-up signal to LISTEN on the `events` channel emitted by the 

100 # AFTER INSERT trigger from migration 20260509120000. Latency from 

101 # event commit to projection write is ~tens of ms under normal load. 

102 # Flip to False to fall back to polling-only when LISTEN/NOTIFY's 

103 # global commit lock causes contention (per Recall.ai July 2025 

104 # incident; trigger documented in `memory/project_deferred.md` under 

105 # the NATS deferred entry). Polling fallback latency is bounded by 

106 # `projection_poll_interval_seconds`. 

107 projection_use_listen_notify: bool = True 

108 # Safety-net poll interval when NOTIFY mode is on (catches missed 

109 # signals from listener disconnect). Becomes the primary signal 

110 # when NOTIFY is off — recommended values are very different 

111 # between the two modes (5s with NOTIFY, 1-2s without). Floor of 

112 # 0.1s prevents accidental tight-loop misconfiguration. 

113 projection_poll_interval_seconds: float = 5.0 

114 

115 # LLM provider — Agent BC wiring (Phase 8f-b iter 2a) 

116 # When None, `build_kernel` wires no LLM and the Kernel 

117 # carries `llm=None`; subscribers that depend on the LLM (the 

118 # 8f-b RunDebrief subscriber) raise / log-and-skip when they 

119 # try to use it. The dev / test default of None matches the 

120 # `AllowAllAuthorize` / `AlwaysCoveredClearanceLookup` test- 

121 # bypass convention: tests don't need real API credentials. 

122 # Production deployments that ship RunDebrief MUST set this; 

123 # the `8f-b iter 2b` wire-up adds a startup gate that refuses 

124 # to register `run_debrief_subscriber` when this is unset (so 

125 # the agent never runs blind). 

126 # 

127 # Read from `ANTHROPIC_API_KEY` env var (case-insensitive per 

128 # pydantic-settings; matches the bare-field-name convention 

129 # `APP_ENV` / `DATABASE_URL` / `TRUST_POLICY_ID` already follow). 

130 # `SecretStr` ensures the key is never serialised by `repr()`, 

131 # `str()`, or `model_dump_json()` (Pydantic redacts it to 

132 # `**********` in all three paths). Production deploys MUST 

133 # access the key via `.get_secret_value()` — only the 

134 # `AnthropicLLMAdapter` factory does this today. A watch-item 

135 # follow-up promotes `database_url` to the same shape so the 

136 # whole Settings surface is repr-safe. 

137 anthropic_api_key: SecretStr | None = None 

138 

139 # Idempotency (Phase 9a) 

140 # `idempotency_ttl_hours` is read by the pruner background task 

141 # which periodically deletes idempotency_keys rows older than this. 

142 # Stripe's industry default is 24h; clients are expected to retry 

143 # within that window or accept that a duplicate request will hit 

144 # a fresh handler invocation. Set to 0 to disable the pruner 

145 # entirely (rows live forever — useful for forensic deployments). 

146 idempotency_ttl_hours: int = 24 

147 # `idempotency_lock_stale_seconds` is the threshold above which an 

148 # in-flight (locked) idempotency row is considered stale and re- 

149 # claimable. Covers the case where a process crashed mid-handler 

150 # and never released its lock. Default 60s is generous over typical 

151 # handler latency (~100ms) but short enough that a crashed worker's 

152 # locked rows recover within a minute. 

153 idempotency_lock_stale_seconds: int = 60 

154 

155 # Edge auth (Phase C) 

156 # `identity_providers` is the list of IdPs CORA accepts tokens 

157 # from. Empty (default) keeps the legacy X-Principal-Id-with- 

158 # SYSTEM-fallback shape; Iter C wires the middleware to use this 

159 # list when populated. Production deployments set this via env 

160 # var as JSON, e.g.: 

161 # 

162 # IDENTITY_PROVIDERS='[{"issuer":"https://idp.example.com", 

163 # "jwks_url":"https://idp.example.com/jwks.json", 

164 # "audiences":{"00000000-0000-0000-0000-000000000020":"https://cora.example/http"}, 

165 # "algorithms_allowed":["RS256"]}]' 

166 # 

167 # pydantic-settings parses the JSON automatically when the env 

168 # value starts with `[`. Schema validation runs at startup so 

169 # malformed config fails fast, not on first auth attempt. 

170 identity_providers: list[IdentityProviderConfig] = []

172 @field_validator("database_url") 

173 @classmethod 

174 def _validate_database_url(cls, value: str) -> str: 

175 """Catch malformed DATABASE_URL at startup, not on first asyncpg call.""" 

176 if not value.startswith(_ALLOWED_DATABASE_SCHEMES): 

177 schemes = " or ".join(_ALLOWED_DATABASE_SCHEMES) 

178 msg = ( 

179 f"DATABASE_URL must start with {schemes} (got: {value[:40]!r}). " 

180 "asyncpg accepts both; SQLAlchemy-style 'postgresql+psycopg2://' " 

181 "URLs are not supported here." 

182 ) 

183 raise ValueError(msg) 

184 return value 

185
