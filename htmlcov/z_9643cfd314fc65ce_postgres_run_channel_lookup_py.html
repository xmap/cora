1"""asyncpg-backed `RunChannelLookup` over `entries_run_observations`. 

2 

3Queries the existing observation table directly: the "range query when a 

4real consumer asks" the entries module reserved, with the closed-loop 

5rules as that consumer. No projection (a projection would cost a 

6permanent fold and read staler than the source on the very freshness 

7signal Rule R depends on). 

8 

9Both queries key on `recorded_at` (the CORA write-time trust anchor) and 

10are channel-scoped, so they ride the 

11`entries_run_observations_run_channel_recorded_idx` 

12`(run_id, channel_name, recorded_at DESC)` btree added alongside this 

13adapter. The pre-existing `sampled_at` indexes do NOT serve these 

14queries (wrong column, no channel_name). 

15""" 

16 

17# pyright: reportUnknownMemberType=false, reportUnknownVariableType=false, reportUnknownArgumentType=false 

18# asyncpg's stubs are loose; suppress at module level for the adapter. 

19 

20from datetime import datetime 

21from uuid import UUID 

22 

23import asyncpg 

24 

25from cora.run.ports.run_channel_lookup import ( 

26 RunChannelLatest, 

27 RunChannelSignal, 

28 RunFeedHealth, 

29) 

30 

31_LATEST_SQL = """ 

32SELECT channel_name, value, units, sampled_at, recorded_at, is_simulated 

33FROM entries_run_observations 

34WHERE run_id = $1 AND channel_name = $2 

35ORDER BY recorded_at DESC 

36LIMIT 1 

37""" 

38 

39_WINDOW_SQL = """ 

40SELECT 

41 count(*) AS count_since, 

42 min(recorded_at) AS first_recorded_at, 

43 max(recorded_at) AS latest_recorded_at, 

44 coalesce(bool_or(is_simulated), false) AS is_simulated_window 

45FROM entries_run_observations 

46WHERE run_id = $1 AND channel_name = $2 AND recorded_at > $3 

47""" 

48 

49_FEED_HEALTH_SQL = """ 

50SELECT max(recorded_at) AS latest_heartbeat_recorded_at 

51FROM entries_run_feed_heartbeats 

52WHERE run_id = $1 

53""" 

54 

55 

56class PostgresRunChannelLookup: 

57 """Production `RunChannelLookup`; reads the observation entry table.""" 

58 

59 def __init__(self, pool: asyncpg.Pool) -> None: 

60 self._pool = pool 

61 

62 async def read_run_channel_latest( 

63 self, *, run_id: UUID, channel_name: str 

64 ) -> RunChannelLatest | None: 

65 async with self._pool.acquire() as conn: 

66 row = await conn.fetchrow(_LATEST_SQL, run_id, channel_name) 

67 if row is None: 

68 return None 

69 return RunChannelLatest( 

70 channel_name=row["channel_name"], 

71 value=row["value"], 

72 units=row["units"], 

73 sampled_at=row["sampled_at"], 

74 recorded_at=row["recorded_at"], 

75 is_simulated=row["is_simulated"], 

76 ) 

77 

78 async def read_run_channel_window( 

79 self, *, run_id: UUID, channel_name: str, since: datetime 

80 ) -> RunChannelSignal: 

81 async with self._pool.acquire() as conn: 

82 row = await conn.fetchrow(_WINDOW_SQL, run_id, channel_name, since) 

83 # count(*) always returns exactly one row, even for an empty window. 

84 assert row is not None 

85 return RunChannelSignal( 

86 channel_name=channel_name, 

87 count_since=row["count_since"], 

88 first_recorded_at=row["first_recorded_at"], 

89 latest_recorded_at=row["latest_recorded_at"], 

90 is_simulated_window=row["is_simulated_window"], 

91 ) 

92 

93 async def read_feed_health(self, *, run_id: UUID) -> RunFeedHealth: 

94 async with self._pool.acquire() as conn: 

95 row = await conn.fetchrow(_FEED_HEALTH_SQL, run_id) 

96 # max() over an empty set returns one row with a NULL aggregate. 

97 latest = row["latest_heartbeat_recorded_at"] if row is not None else None 

98 return RunFeedHealth(latest_heartbeat_recorded_at=latest)
